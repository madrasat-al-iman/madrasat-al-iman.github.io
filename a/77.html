77:8 ⇄ ישעיהו יג, י; יואל ב, י

ישעיהו יג, י: כִּי־כוֹכְבֵי הַשָּׁמַיִם וּכְסִילֵיהֶם לֹא יָהֵלּוּ אוֹרָם חָשַׁךְ הַשֶּׁמֶשׁ בְּצֵאתוֹ וְיָרֵחַ לֹא־יַגִּיהַּ אוֹרוֹ׃ (י) ✦ יואל ב, י: לְפָנָיו רָגְזָה אֶרֶץ רָעֲשׁוּ שָׁמָיִם שֶׁמֶשׁ וְיָרֵחַ קָדָרוּ וְכוֹכָבִים אָסְפוּ נׇגְהָם׃ (י)

إشعياء 13: 10: فَإِنَّ نُجُومَ ٱلسَّمَاوَاتِ وَجَبَابِرَتَهَا لَا تُبْرِزُ نُورَهَا. تُظْلِمُ ٱلشَّمْسُ عِنْدَ طُلُوعِهَا، وَٱلْقَمَرُ لَا يَلْمَعُ بِضَوْئِهِ. (10) ✦ يوئيل 2: 10: قُدَّامَهُ تَرْتَعِدُ ٱلْأَرْضُ وَتَرْجُفُ ٱلسَّمَاءُ. اَلشَّمْسُ وَٱلْقَمَرُ يُظْلِمَانِ، وَٱلنُّجُومُ تَحْجُزُ لَمَعَانَهَا. (10)

״וכאשר הכוכבים יכבו, וכאשר השמיים ייפרצו, וכאשר ההרים ייעקרו״ — זעזועי יום הדין הקוסמיים כבישעיהו יג, י וביואל ב, י.

انطماس النجوم وانفراج السماء ونسف الجبال يوم الفصل يوازي إشعياء 13:10 ويوئيل 2:10.

77:1 ⇄ חנוך א א, ג–ט; עזרא ד׳ ה, ד–ז

פתיחת סורת אלמרסלאת בשבועות רוחות סערה, זעזועי הרים וכוכבים המכבים את אורם לקראת ״יום ההכרעה״ — ז׳אנר אפוקליפטי קרוב לחנוך א פרק א ועזרא ד׳ פרק ה!

الأسلوب الرؤيوي المهيب في سورة المرسلات (الرياح العاصفة، انطماس النجوم، نسف الجبال ليوم الفصل) يوازي أخنوخ الأول 1:3–9 وعزرا الرابع 5!

77:13 ⇄ יואל ד, יד

הֲמוֹנִים הֲמוֹנִים בְּעֵמֶק הֶחָרוּץ כִּי קָרוֹב יוֹם יְהֹוָה בְּעֵמֶק הֶחָרוּץ׃ (יד)

جَمَاهِيرُ جَمَاهِيرُ فِي وَادِي ٱلْقَضَاءِ، لِأَنَّ يَوْمَ ٱلرَّبِّ قَرِيبٌ فِي وَادِي ٱلْقَضَاءِ. (14)

״ליום ההכרעה (יוֹם אלפַצְל)״ — מקביל לכינוי המקראי ביואל ד, יד: ״עמק החרוץ (עמק ההכרעה), כי קרוב יום ה׳ בעמק החרוץ״.

تسمية يوم القيامة «يوم الفصل» توازي «وادي القضاء/الحسم» في يوئيل 3:14.

77:15 ⇄ ישעיהו ה, ח–כב

הוֹי מַגִּיעֵי בַיִת בְּבַיִת שָׂדֶה בְשָׂדֶה יַקְרִיבוּ עַד אֶפֶס מָקוֹם וְהוּשַׁבְתֶּם לְבַדְּכֶם בְּקֶרֶב הָאָרֶץ׃ (ח) בְּאׇזְנָי יְהֹוָה צְבָאוֹת אִם־לֹא בָּתִּים רַבִּים לְשַׁמָּה יִהְיוּ גְּדֹלִים וְטוֹבִים מֵאֵין יוֹשֵׁב׃ (ט) כִּי עֲשֶׂרֶת צִמְדֵּי־כֶרֶם יַעֲשׂוּ בַּת אֶחָת וְזֶרַע חֹמֶר יַעֲשֶׂה אֵיפָה׃ (י) הוֹי מַשְׁכִּימֵי בַבֹּקֶר שֵׁכָר יִרְדֹּפוּ מְאַחֲרֵי בַנֶּשֶׁף יַיִן יַדְלִיקֵם׃ (יא) וְהָיָה כִנּוֹר וָנֶבֶל תֹּף וְחָלִיל וָיַיִן מִשְׁתֵּיהֶם וְאֵת פֹּעַל יְהֹוָה לֹא יַבִּיטוּ וּמַעֲשֵׂה יָדָיו לֹא רָאוּ׃ (יב) לָכֵן גָּלָה עַמִּי מִבְּלִי־דָעַת וּכְבוֹדוֹ מְתֵי רָעָב וַהֲמוֹנוֹ צִחֵה צָמָא׃ (יג) לָכֵן הִרְחִיבָה שְּׁאוֹל נַפְשָׁהּ וּפָעֲרָה פִיהָ לִבְלִי־חֹק וְיָרַד הֲדָרָהּ וַהֲמוֹנָהּ וּשְׁאוֹנָהּ וְעָלֵז בָּהּ׃ (יד) וַיִּשַּׁח אָדָם וַיִּשְׁפַּל־אִישׁ וְעֵינֵי גְבֹהִים תִּשְׁפַּלְנָה׃ (טו) וַיִּגְבַּהּ יְהֹוָה צְבָאוֹת בַּמִּשְׁפָּט וְהָאֵל הַקָּדוֹשׁ נִקְדָּשׁ בִּצְדָקָה׃ (טז) וְרָעוּ כְבָשִׂים כְּדׇבְרָם וְחׇרְבוֹת מֵחִים גָּרִים יֹאכֵלוּ׃ (יז) הוֹי מֹשְׁכֵי הֶעָוֺן בְּחַבְלֵי הַשָּׁוְא וְכַעֲבוֹת הָעֲגָלָה חַטָּאָה׃ (יח) הָאֹמְרִים יְמַהֵר יָחִישָׁה מַעֲשֵׂהוּ לְמַעַן נִרְאֶה וְתִקְרַב וְתָבוֹאָה עֲצַת קְדוֹשׁ יִשְׂרָאֵל וְנֵדָעָה׃ (יט) הוֹי הָאֹמְרִים לָרַע טוֹב וְלַטּוֹב רָע שָׂמִים חֹשֶׁךְ לְאוֹר וְאוֹר לְחֹשֶׁךְ שָׂמִים מַר לְמָתוֹק וּמָתוֹק לְמָר׃ (כ) הוֹי חֲכָמִים בְּעֵינֵיהֶם וְנֶגֶד פְּנֵיהֶם נְבֹנִים׃ (כא) הוֹי גִּבּוֹרִים לִשְׁתּוֹת יָיִן וְאַנְשֵׁי־חַיִל לִמְסֹךְ שֵׁכָר׃ (כב)

وَيْلٌ لِلَّذِينَ يَصِلُونَ بَيْتًا بِبَيْتٍ، وَيَقْرِنُونَ حَقْلًا بِحَقْلٍ، حَتَّى لَمْ يَبْقَ مَوْضِعٌ. فَصِرْتُمْ تَسْكُنُونَ وَحْدَكُمْ فِي وَسَطِ ٱلْأَرْضِ. (8) فِي أُذُنَيَّ قَالَ رَبُّ ٱلْجُنُودِ: «أَلَا إِنَّ بُيُوتًا كَثِيرَةً تَصِيرُ خَرَابًا. بُيُوتًا كَبِيرَةً وَحَسَنَةً بِلَا سَاكِنٍ. (9) لِأَنَّ عَشْرَةَ فَدَادِينِ كَرْمٍ تَصْنَعُ بَثًّا وَاحِدًا، وَحُومَرَ بِذَارٍ يَصْنَعُ إِيفَةً». (10) وَيْلٌ لِلْمُبَكِّرِينَ صَبَاحًا يَتْبَعُونَ ٱلْمُسْكِرَ، لِلْمُتَأَخِّرِينَ فِي ٱلْعَتَمَةِ تُلْهِبُهُمُ ٱلْخَمْرُ. (11) وَصَارَ ٱلْعُودُ وَٱلرَّبَابُ وَٱلدُّفُّ وَٱلنَّايُ وَٱلْخَمْرُ وَلَائِمَهُمْ، وَإِلَى فَعْلِ ٱلرَّبِّ لَا يَنْظُرُونَ، وَعَمَلَ يَدَيْهِ لَا يَرَوْنَ. (12) لِذَلِكَ سُبِيَ شَعْبِي لِعَدَمِ ٱلْمَعْرِفَةِ، وَتَصِيرُ شُرَفَاؤُهُ رِجَالَ جُوعٍ، وَعَامَّتُهُ يَابِسِينَ مِنَ ٱلْعَطَشِ. (13) لِذَلِكَ وَسَّعَتِ ٱلْهَاوِيَةُ نَفْسَهَا، وَفَغَرَتْ فَاهَا بِلَا حَدٍّ، فَيَنْزِلُ بَهَاؤُهَا وَجُمْهُورُهَا وَضَجِيجُهَا وَٱلْمُبْتَهِجُ فِيهَا! (14) وَيُذَلُّ ٱلْإِنْسَانُ وَيُحَطُّ ٱلرَّجُلُ، وَعُيُونُ ٱلْمُسْتَعْلِينَ تُوضَعُ. (15) وَيَتَعَالَى رَبُّ ٱلْجُنُودِ بِٱلْعَدْلِ، وَيَتَقَدَّسُ ٱلإِلَهُ ٱلْقُدُّوسُ بِٱلْبِرِّ. (16) وَتَرْعَى ٱلْخِرْفَانُ حَيْثُمَا تُسَاقُ، وَخِرَبُ ٱلسِّمَانِ تَأْكُلُهَا ٱلْغُرَبَاءُ. (17) وَيْلٌ لِلْجَاذِبِينَ ٱلْإِثْمَ بِحِبَالِ ٱلْبُطْلِ، وَٱلْخَطِيَّةَ كَأَنَّهُ بِرُبُطِ ٱلْعَجَلَةِ، (18) ٱلْقَائِلِينَ: «لِيُسْرِعْ، لِيُعَجِّلْ عَمَلَهُ لِكَيْ نَرَى، وَلْيَقْرُبْ وَيَأْتِ مَقْصَدُ قُدُّوسِ إِسْرَائِيلَ لِنَعْلَمَ». (19) وَيْلٌ لِلْقَائِلِينَ لِلشَّرِّ خَيْرًا وَلِلْخَيْرِ شَرًّا، ٱلْجَاعِلِينَ ٱلظَّلَامَ نُورًا وَٱلنُّورَ ظَلَامًا، ٱلْجَاعِلِينَ ٱلْمُرَّ حُلْوًا وَٱلْحُلْوَ مُرًّا. (20) وَيْلٌ لِلْحُكَمَاءِ فِي أَعْيُنِ أَنْفُسِهِمْ، وَٱلْفُهَمَاءِ عِنْدَ ذَوَاتِهِمْ. (21) وَيْلٌ لِلْأَبْطَالِ عَلَى شُرْبِ ٱلْخَمْرِ، وَلِذَوِي ٱلْقُدْرَةِ عَلَى مَزْجِ ٱلْمُسْكِرِ. (22)

הפזמון החוזר ״אוי ביום ההוא למכחישים״ — תבנית נבואית־רטורית של קריאות ״הוי״ חוזרות, כבישעיהו ה, ח–כב.

תנו רבנן: שלשה שותפין יש באדם — הקדוש ברוך הוא, ואביו, ואמו. אביו מזריע הלובן, שממנו עצמות, וגידים, וצפרנים, ומוח שבראשו, ולובן שבעין. אמו מזרעת אודם, שממנו עור, ובשר, ושערות, ושחור שבעין. והקדוש ברוך הוא נותן בו רוח, ונשמה, וקלסתר פנים, וראיית העין, ושמיעת האוזן, ודבור פה, והלוך רגלים, ובינה, והשכל.

علّم الحكماء: ثلاثة شركاء في الإنسان: القدّوس تبارك هو، وأبوه، وأمّه. أبوه يزرع البياض الذي منه العظام والأعصاب والأظافر والمخّ الذي في رأسه وبياض العين؛ وأمّه تزرع الحُمرة التي منها الجلد واللحم والشعر وسواد العين؛ والقدّوس تبارك هو يعطي فيه روحًا ونَفَسًا، ونضارة الوجه، ورؤية العين، وسمع الأذن، ونطق الفم، ومشي الرجلين، والفهم، والإدراك.

77:48 ⇄ ישעיהו מה, כג; תהלים צה, ו

ישעיהו מה, כג: בִּי נִשְׁבַּעְתִּי יָצָא מִפִּי צְדָקָה דָּבָר וְלֹא יָשׁוּב כִּי־לִי תִּכְרַע כׇּל־בֶּרֶךְ תִּשָּׁבַע כׇּל־לָשׁוֹן׃ (כג) ✦ תהלים צה, ו: בֹּאוּ נִשְׁתַּחֲוֶה וְנִכְרָעָה נִבְרְכָה לִפְנֵי־יְהֹוָה עֹשֵׂנוּ׃ (ו)

إشعياء 45: 23: بِذَاتِي أَقْسَمْتُ، خَرَجَ مِنْ فَمِي ٱلصِّدْقُ كَلِمَةٌ لَا تَرْجِعُ: إِنَّهُ لِي تَجْثُو كُلُّ رُكْبَةٍ، يَحْلِفُ كُلُّ لِسَانٍ. (23) ✦ المزامير 95: 6: هَلُمَّ نَسْجُدُ وَنَرْكَعُ وَنَجْثُو أَمَامَ ٱلرَّبِّ خَالِقِنَا، (6)

״וכאשר נאמר להם כרעו – אינם כורעים״ — מול ״כי לי תכרע כל ברך״ (ישעיהו מה, כג).

رفض المستكبرين للركوع لله مقابل إشعياء 45:23 («إن لي تجثو كل ركبة»).
