פרשת יתרוبرشا يِترو

לתצוגה המלאה באתר (חיפוש, השמעה, שיתוף)إلى العرض الكامل في الموقع

18:1וַיִּשְׁמַע יִתְרוֹ כֹהֵן מִדְיָן חֹתֵן מֹשֶׁה אֵת כׇּל־אֲשֶׁר עָשָׂה אֱלֹהִים לְמֹשֶׁה וּלְיִשְׂרָאֵל עַמּוֹ כִּי־הוֹצִיא יְהֹוָה אֶת־יִשְׂרָאֵל מִמִּצְרָיִם׃
فَسَمِعَ يَثْرُونُ كَاهِنُ مِدْيَانَ، حَمُو مُوسَى، كُلَّ مَا صَنَعَ ٱللهُ إِلَى مُوسَى وَإِلَى إِسْرَائِيلَ شَعْبِهِ: أَنَّ ٱلرَّبَّ أَخْرَجَ إِسْرَائِيلَ مِنْ مِصْرَ.
18:2וַיִּקַּח יִתְרוֹ חֹתֵן מֹשֶׁה אֶת־צִפֹּרָה אֵשֶׁת מֹשֶׁה אַחַר שִׁלּוּחֶיהָ׃
فَأَخَذَ يَثْرُونُ حَمُو مُوسَى صِفُّورَةَ ٱمْرَأَةَ مُوسَى بَعْدَ صَرْفِهَا
18:3וְאֵת שְׁנֵי בָנֶיהָ אֲשֶׁר שֵׁם הָאֶחָד גֵּרְשֹׁם כִּי אָמַר גֵּר הָיִיתִי בְּאֶרֶץ נׇכְרִיָּה׃
وَٱبْنَيْهَا، ٱللَّذَيْنِ ٱسْمُ أَحَدِهِمَا جِرْشُومُ، لِأَنَّهُ قَالَ: «كُنْتُ نَزِيلًا فِي أَرْضٍ غَرِيبَةٍ».
18:4וְשֵׁם הָאֶחָד אֱלִיעֶזֶר כִּי־אֱלֹהֵי אָבִי בְּעֶזְרִי וַיַּצִּלֵנִי מֵחֶרֶב פַּרְעֹה׃
وَٱسْمُ ٱلْآخَرِ أَلِيعَازَرُ، لِأَنَّهُ قَالَ: «إِلَهُ أَبِي كَانَ عَوْنِي وَأَنْقَذَنِي مِنْ سَيْفِ فِرْعَوْنَ».
18:5וַיָּבֹא יִתְרוֹ חֹתֵן מֹשֶׁה וּבָנָיו וְאִשְׁתּוֹ אֶל־מֹשֶׁה אֶל־הַמִּדְבָּר אֲשֶׁר־הוּא חֹנֶה שָׁם הַר הָאֱלֹהִים׃
وَأَتَى يَثْرُونُ حَمُو مُوسَى وَٱبْنَاهُ وَٱمْرَأَتُهُ إِلَى مُوسَى إِلَى ٱلْبَرِّيَّةِ حَيْثُ كَانَ نَازِلًا عِنْدَ جَبَلِ ٱللهِ.
18:6וַיֹּאמֶר אֶל־מֹשֶׁה אֲנִי חֹתֶנְךָ יִתְרוֹ בָּא אֵלֶיךָ וְאִשְׁתְּךָ וּשְׁנֵי בָנֶיהָ עִמָּהּ׃
فَقَالَ لِمُوسَى: «أَنَا حَمُوكَ يَثْرُونُ، آتٍ إِلَيْكَ وَٱمْرَأَتُكَ وَٱبْنَاهَا مَعَهَا».
18:7וַיֵּצֵא מֹשֶׁה לִקְרַאת חֹתְנוֹ וַיִּשְׁתַּחוּ וַיִּשַּׁק־לוֹ וַיִּשְׁאֲלוּ אִישׁ־לְרֵעֵהוּ לְשָׁלוֹם וַיָּבֹאוּ הָאֹהֱלָה׃
فَخَرَجَ مُوسَى لِٱسْتِقْبَالِ حَمِيهِ وَسَجَدَ وَقَبَّلَهُ. وَسَأَلَ كُلُّ وَاحِدٍ صَاحِبَهُ عَنْ سَلَامَتِهِ، ثُمَّ دَخَلَا إِلَى ٱلْخَيْمَةِ.

שמות יח, ז ⇄ תרמד׳י 2728

תרמד׳י 2728: אמר אדם: ״שליח אללה, אחד מאיתנו פוגש את אחיו או את חברו – האם יקוד לו?״ אמר: ״לא״. אמר: ״האם יחבקהו וינשקהו?״ אמר: ״לא״. אמר: ״האם ייטול את ידו ויילחץ אותה?״ אמר: ״כן״.

سنن الترمذي 2728: قال رجل يا رسول الله الرجل منا يلقى أخاه أو صديقه أينحني له قال " لا " . قال أفيلتزمه ويقبله قال " لا " . قال أفيأخذ بيده ويصافحه قال " نعم " .

״ויצא משה לקראת חותנו וישתחו ויישק לו וישאלו איש לרעהו לשלום״ — יציאה לקראת האורח ושאלת שלום חמה; ובתרמד׳י 2728 נשאל הנביא אם מותר לאדם להשתחוות לחברו ולחבקו ולנשקו בפגישה, והשיב שלא – אלא ילחץ את ידו. מנהגי הכבוד בפגישה – קרבה וניגוד.

خروج موسى لاستقبال صهره وتقبيله والسؤال عن السلام يوازي أدب استقبال الضيف والسؤال عن السلام؛ وفي الترمذي 2728 سُئل النبي: أينحني الرجل لأخيه أو يلتزمه ويقبّله؟ فقال: لا، بل يأخذ بيده ويصافحه – تقارب وتباين في آداب اللقاء.

18:8וַיְסַפֵּר מֹשֶׁה לְחֹתְנוֹ אֵת כׇּל־אֲשֶׁר עָשָׂה יְהֹוָה לְפַרְעֹה וּלְמִצְרַיִם עַל אוֹדֹת יִשְׂרָאֵל אֵת כׇּל־הַתְּלָאָה אֲשֶׁר מְצָאָתַם בַּדֶּרֶךְ וַיַּצִּלֵם יְהֹוָה׃
فَقَصَّ مُوسَى عَلَى حَمِيهِ كُلَّ مَا صَنَعَ ٱلرَّبُّ بِفِرْعَوْنَ وَٱلْمِصْرِيِّينَ مِنْ أَجْلِ إِسْرَائِيلَ، وَكُلَّ ٱلْمَشَقَّةِ ٱلَّتِي أَصَابَتْهُمْ فِي ٱلطَّرِيقِ فَخَلَّصَهُمُ ٱلرَّبُّ.
18:9וַיִּחַדְּ יִתְרוֹ עַל כׇּל־הַטּוֹבָה אֲשֶׁר־עָשָׂה יְהֹוָה לְיִשְׂרָאֵל אֲשֶׁר הִצִּילוֹ מִיַּד מִצְרָיִם׃
فَفَرِحَ يَثْرُونُ بِجَمِيعِ ٱلْخَيْرِ ٱلَّذِي صَنَعَهُ إِلَى إِسْرَائِيلَ ٱلرَّبُّ، ٱلَّذِي أَنْقَذَهُ مِنْ أَيْدِي ٱلْمِصْرِيِّينَ.
18:10וַיֹּאמֶר יִתְרוֹ בָּרוּךְ יְהֹוָה אֲשֶׁר הִצִּיל אֶתְכֶם מִיַּד מִצְרַיִם וּמִיַּד פַּרְעֹה אֲשֶׁר הִצִּיל אֶת־הָעָם מִתַּחַת יַד־מִצְרָיִם׃
وَقَالَ يَثْرُونُ: «مُبَارَكٌ ٱلرَّبُّ ٱلَّذِي أَنْقَذَكُمْ مِنْ أَيْدِي ٱلْمِصْرِيِّينَ وَمِنْ يَدِ فِرْعَوْنَ. اَلَّذِي أَنْقَذَ ٱلشَّعْبَ مِنْ تَحْتِ أَيْدِي ٱلْمِصْرِيِّينَ.
18:11עַתָּה יָדַעְתִּי כִּי־גָדוֹל יְהֹוָה מִכׇּל־הָאֱלֹהִים כִּי בַדָּבָר אֲשֶׁר זָדוּ עֲלֵיהֶם׃
ٱلْآنَ عَلِمْتُ أَنَّ ٱلرَّبَّ أَعْظَمُ مِنْ جَمِيعِ ٱلْآلِهَةِ، لِأَنَّهُ فِي ٱلشَّيْءِ ٱلَّذِي بَغَوْا بِهِ كَانَ عَلَيْهِمْ».
18:12וַיִּקַּח יִתְרוֹ חֹתֵן מֹשֶׁה עֹלָה וּזְבָחִים לֵאלֹהִים וַיָּבֹא אַהֲרֹן וְכֹל זִקְנֵי יִשְׂרָאֵל לֶאֱכׇל־לֶחֶם עִם־חֹתֵן מֹשֶׁה לִפְנֵי הָאֱלֹהִים׃
فَأَخَذَ يَثْرُونُ حَمُو مُوسَى مُحْرَقَةً وَذَبَائِحَ لِلهِ. وَجَاءَ هَارُونُ وَجَمِيعُ شُيُوخِ إِسْرَائِيلَ لِيَأْكُلُوا طَعَامًا مَعَ حَمِي مُوسَى أَمَامَ ٱللهِ.
18:13וַיְהִי מִמׇּחֳרָת וַיֵּשֶׁב מֹשֶׁה לִשְׁפֹּט אֶת־הָעָם וַיַּעֲמֹד הָעָם עַל־מֹשֶׁה מִן־הַבֹּקֶר עַד־הָעָרֶב׃
وَحَدَثَ فِي ٱلْغَدِ أَنَّ مُوسَى جَلَسَ لِيَقْضِيَ لِلشَّعْبِ. فَوَقَفَ ٱلشَّعْبُ عِنْدَ مُوسَى مِنَ ٱلصَّبَاحِ إِلَى ٱلْمَسَاءِ.
18:14וַיַּרְא חֹתֵן מֹשֶׁה אֵת כׇּל־אֲשֶׁר־הוּא עֹשֶׂה לָעָם וַיֹּאמֶר מָה־הַדָּבָר הַזֶּה אֲשֶׁר אַתָּה עֹשֶׂה לָעָם מַדּוּעַ אַתָּה יוֹשֵׁב לְבַדֶּךָ וְכׇל־הָעָם נִצָּב עָלֶיךָ מִן־בֹּקֶר עַד־עָרֶב׃
فَلَمَّا رَأَى حَمُو مُوسَى كُلَّ مَا هُوَ صَانِعٌ لِلشَّعْبِ، قَالَ: «مَا هَذَا ٱلْأَمْرُ ٱلَّذِي أَنْتَ صَانِعٌ لِلشَّعْبِ؟ مَا بَالُكَ جَالِسًا وَحْدَكَ وَجَمِيعُ ٱلشَّعْبِ وَاقِفٌ عِنْدَكَ مِنَ ٱلصَّبَاحِ إِلَى ٱلْمَسَاءِ؟»
18:15וַיֹּאמֶר מֹשֶׁה לְחֹתְנוֹ כִּי־יָבֹא אֵלַי הָעָם לִדְרֹשׁ אֱלֹהִים׃
فَقَالَ مُوسَى لِحَمِيهِ: «إِنَّ ٱلشَّعْبَ يَأْتِي إِلَيَّ لِيَسْأَلَ ٱللهَ.
18:16כִּי־יִהְיֶה לָהֶם דָּבָר בָּא אֵלַי וְשָׁפַטְתִּי בֵּין אִישׁ וּבֵין רֵעֵהוּ וְהוֹדַעְתִּי אֶת־חֻקֵּי הָאֱלֹהִים וְאֶת־תּוֹרֹתָיו׃
إِذَا كَانَ لَهُمْ دَعْوَى يَأْتُونَ إِلَيَّ فَأَقْضِي بَيْنَ ٱلرَّجُلِ وَصَاحِبِهِ، وَأُعَرِّفُهُمْ فَرَائِضَ ٱللهِ وَشَرَائِعَهُ».
18:17וַיֹּאמֶר חֹתֵן מֹשֶׁה אֵלָיו לֹא־טוֹב הַדָּבָר אֲשֶׁר אַתָּה עֹשֶׂה׃
فَقَالَ حَمُو مُوسَى لَهُ: «لَيْسَ جَيِّدًا ٱلْأَمْرُ ٱلَّذِي أَنْتَ صَانِعٌ.

שמות יח, יז–כד ⇄ בוח׳ארי 7369; אבו דאוד 3592 (התייעצות המנהיג וקבלת עצה חכמה מן הזולת!)

בוח׳ארי 7369: (עאאשה סיפרה, בפרשת העלילה עליה:) קרא שליח אללה לעלי בן אבי טאלב ולאֻסַאמַה בן זַיְד, כשההתגלות התעכבה, כדי לשאול אותם; והוא מתייעץ עמהם על פרידה מאשתו. אֻסַאמַה יעץ על פי מה שידע על חפותה של אשתו. ועלי אמר: ״אללה לא הצר את צעדיך, ונשים מלבדה רבות; ושאל את השפחה – והיא תאמר לך את האמת״. אמר (הנביא לשפחה): ״האם ראית דבר המעורר בך חשד?״ אמרה: ״לא ראיתי דבר מלבד זאת שהיא נערה צעירה, הנרדמת ליד הבצק של בני ביתה, ובאה העז הביתית ואוכלת אותו״. קם (הנביא) על הדוכן ואמר: ״הוי מוסלמים, מי יצדיק אותי (בהענשת) אדם שפגיעתו בבני ביתי הגיעה אליי? חי אללה, לא ידעתי על בני ביתי אלא טוב״. ✦ אבו דאוד 3592: שליח אללה, כאשר רצה לשלוח את מֻעאד׳ לתימן, אמר: ״כיצד תשפוט כאשר יובא לפניך משפט?״ אמר: ״אשפוט לפי ספר אללה״. אמר: ״ואם לא תמצא בספר אללה?״ אמר: ״אז לפי הסֻנַּה של שליח אללה״. אמר: ״ואם לא תמצא בסֻנַּת שליח אללה ולא בספר אללה?״ אמר: ״אפעיל את שיקול דעתי ולא אחסוך מאמץ״. טפח שליח אללה על חזהו ואמר: ״השבח לאללה, שכיוון את שליחו של שליח אללה למה שמרצה את שליח אללה״.

صحيح البخاري 7369: ودعا رسول الله صلى الله عليه وسلم علي بن أبي طالب وأسامة بن زيد حين استلبث الوحى يسألهما، وهو يستشيرهما في فراق أهله، فأما أسامة فأشار بالذي يعلم من براءة أهله، وأما علي فقال لم يضيق الله عليك، والنساء سواها كثير، وسل الجارية تصدقك. فقال " هل رأيت من شىء يريبك ". قالت ما رأيت أمرا أكثر من أنها جارية حديثة السن تنام عن عجين أهلها فتأتي الداجن فتأكله. فقام على المنبر فقال " يا معشر المسلمين من يعذرني من رجل بلغني أذاه في أهلي، والله ما علمت على أهلي إلا خيرا ". ✦ سنن أبي داود 3592: أن رسول الله صلى الله عليه وسلم لما أراد أن يبعث معاذا إلى اليمن قال " كيف تقضي إذا عرض لك قضاء " . قال أقضي بكتاب الله . قال " فإن لم تجد في كتاب الله " . قال فبسنة رسول الله صلى الله عليه وسلم . قال " فإن لم تجد في سنة رسول الله صلى الله عليه وسلم ولا في كتاب الله " . قال أجتهد رأيي ولا آلو . فضرب رسول الله صلى الله عليه وسلم صدره وقال " الحمد لله الذي وفق رسول رسول الله لما يرضي رسول الله " .

עצת יתרו הארגונית למשה להקים מערכת שפיטה היררכית (שרי אלפים, מאות, חמישים ועשרות) ומשה מקבל את עצת חותנו בענווה! — כדוגמת הנביא המקבל את עצת סלמאן הפרסי לחפור חפיר (ח׳נדק) וממנה שופטים בפרובינציות (מעאד׳ לתימן, אבו דאוד 3592)!

ٱرۡحَمۡهُمَا كَمَا رَبَّيَانِي صَغِيرٗا (24) رَّبُّكُمۡ أَعۡلَمُ بِمَا فِي نُفُوسِكُمۡۚ إِن تَكُونُواْ صَٰلِحِينَ فَإِنَّهُۥ كَانَ لِلۡأَوَّـٰبِينَ غَفُورٗا (25) وَءَاتِ ذَا ٱلۡقُرۡبَىٰ حَقَّهُۥ وَٱلۡمِسۡكِينَ وَٱبۡنَ ٱلسَّبِيلِ وَلَا تُبَذِّرۡ تَبۡذِيرًا (26) إِنَّ ٱلۡمُبَذِّرِينَ كَانُوٓاْ إِخۡوَٰنَ ٱلشَّيَٰطِينِۖ وَكَانَ ٱلشَّيۡطَٰنُ لِرَبِّهِۦ كَفُورٗا (27) وَإِمَّا تُعۡرِضَنَّ عَنۡهُمُ ٱبۡتِغَآءَ رَحۡمَةٖ مِّن رَّبِّكَ تَرۡجُوهَا فَقُل لَّهُمۡ قَوۡلٗا مَّيۡسُورٗا (28) وَلَا تَجۡعَلۡ يَدَكَ مَغۡلُولَةً إِلَىٰ عُنُقِكَ وَلَا تَبۡسُطۡهَا كُلَّ ٱلۡبَسۡطِ فَتَقۡعُدَ مَلُومٗا مَّحۡسُورًا (29) إِنَّ رَبَّكَ يَبۡسُطُ ٱلرِّزۡقَ لِمَن يَشَآءُ وَيَقۡدِرُۚ إِنَّهُۥ كَانَ بِعِبَادِهِۦ خَبِيرَۢا بَصِيرٗا (30) وَلَا تَقۡتُلُوٓاْ أَوۡلَٰدَكُمۡ خَشۡيَةَ إِمۡلَٰقٖۖ نَّحۡنُ نَرۡزُقُهُمۡ وَإِيَّاكُمۡۚ إِنَّ قَتۡلَهُمۡ كَانَ خِطۡـٔٗا كَبِيرٗا (31) وَلَا تَقۡرَبُواْ ٱلزِّنَىٰٓۖ إِنَّهُۥ كَانَ فَٰحِشَةٗ وَسَآءَ سَبِيلٗا (32) وَلَا تَقۡتُلُواْ ٱلنَّفۡسَ ٱلَّتِي حَرَّمَ ٱللَّهُ إِلَّا بِٱلۡحَقِّۗ وَمَن قُتِلَ مَظۡلُومٗا فَقَدۡ جَعَلۡنَا لِوَلِيِّهِۦ سُلۡطَٰنٗا فَلَا يُسۡرِف فِّي ٱلۡقَتۡلِۖ إِنَّهُۥ كَانَ مَنصُورٗا (33) وَلَا تَقۡرَبُواْ مَالَ ٱلۡيَتِيمِ إِلَّا بِٱلَّتِي هِيَ أَحۡسَنُ حَتَّىٰ يَبۡلُغَ أَشُدَّهُۥۚ وَأَوۡفُواْ بِٱلۡعَهۡدِۖ إِنَّ ٱلۡعَهۡدَ كَانَ مَسۡـُٔولٗا (34) وَأَوۡفُواْ ٱلۡكَيۡلَ إِذَا كِلۡتُمۡ وَزِنُواْ بِٱلۡقِسۡطَاسِ ٱلۡمُسۡتَقِيمِۚ ذَٰلِكَ خَيۡرٞ وَأَحۡسَنُ تَأۡوِيلٗا (35) وَلَا تَقۡفُ مَا لَيۡسَ لَكَ بِهِۦ عِلۡمٌۚ إِنَّ ٱلسَّمۡعَ وَٱلۡبَصَرَ وَٱلۡفُؤَادَ كُلُّ أُوْلَـٰٓئِكَ كَانَ عَنۡهُ مَسۡـُٔولٗا (36) وَلَا تَمۡشِ فِي ٱلۡأَرۡضِ مَرَحًاۖ إِنَّكَ لَن تَخۡرِقَ ٱلۡأَرۡضَ وَلَن تَبۡلُغَ ٱلۡجِبَالَ طُولٗا (37) كُلُّ ذَٰلِكَ كَانَ سَيِّئُهُۥ عِندَ رَبِّكَ مَكۡرُوهٗا (38) ذَٰلِكَ مِمَّآ أَوۡحَىٰٓ إِلَيۡكَ رَبُّكَ مِنَ ٱلۡحِكۡمَةِۗ وَلَا تَجۡعَلۡ مَعَ ٱللَّهِ إِلَٰهًا ءَاخَرَ فَتُلۡقَىٰ فِي جَهَنَّمَ مَلُومٗا مَّدۡحُورًا (39)

الوصايا العشر في القرآن – مقابلة موضوعية

20:2אָנֹכִי יְהֹוָה אֱלֹהֶיךָ אֲשֶׁר הוֹצֵאתִיךָ מֵאֶרֶץ מִצְרַיִם מִבֵּית עֲבָדִים׃
«أَنَا ٱلرَّبُّ إِلَهُكَ ٱلَّذِي أَخْرَجَكَ مِنْ أَرْضِ مِصْرَ مِنْ بَيْتِ ٱلْعُبُودِيَّةِ.

שמות כ, ב–יד ⇄ 6:151–153

אמור: ״בואו ואקרא את אשר אסר עליכם ריבונכם: שלא תשתפו עמו דבר; ולהורים תיטיבו; ואל תהרגו את ילדיכם מפני עוני – אנחנו נפרנס אתכם ואותם; ואל תקרבו אל התועבות, הגלויות והנסתרות; ואל תהרגו את הנפש אשר אסר אללה, אלא כדין. את זאת ציווה אתכם, למען תבינו״. (151) ואל תקרבו אל הון היתום אלא בדרך הטובה ביותר, עד אשר יגיע לבגרותו; ומלאו את המידה ואת המשקל בצדק – לא נטיל על נפש אלא כיכולתה; וכאשר תדברו – היו צודקים, אף אם הוא קרוב; וקיימו את ברית אללה. את זאת ציווה אתכם, למען תיזכרו. (152) וזו דרכי, ישרה; לכו בה, ואל תלכו בדרכים (אחרות), פן יפרידו אתכם מדרכו. את זאת ציווה אתכם, למען תיראו. (153)

۞قُلۡ تَعَالَوۡاْ أَتۡلُ مَا حَرَّمَ رَبُّكُمۡ عَلَيۡكُمۡۖ أَلَّا تُشۡرِكُواْ بِهِۦ شَيۡـٔٗاۖ وَبِٱلۡوَٰلِدَيۡنِ إِحۡسَٰنٗاۖ وَلَا تَقۡتُلُوٓاْ أَوۡلَٰدَكُم مِّنۡ إِمۡلَٰقٖ نَّحۡنُ نَرۡزُقُكُمۡ وَإِيَّاهُمۡۖ وَلَا تَقۡرَبُواْ ٱلۡفَوَٰحِشَ مَا ظَهَرَ مِنۡهَا وَمَا بَطَنَۖ وَلَا تَقۡتُلُواْ ٱلنَّفۡسَ ٱلَّتِي حَرَّمَ ٱللَّهُ إِلَّا بِٱلۡحَقِّۚ ذَٰلِكُمۡ وَصَّىٰكُم بِهِۦ لَعَلَّكُمۡ تَعۡقِلُونَ (151) وَلَا تَقۡرَبُواْ مَالَ ٱلۡيَتِيمِ إِلَّا بِٱلَّتِي هِيَ أَحۡسَنُ حَتَّىٰ يَبۡلُغَ أَشُدَّهُۥۚ وَأَوۡفُواْ ٱلۡكَيۡلَ وَٱلۡمِيزَانَ بِٱلۡقِسۡطِۖ لَا نُكَلِّفُ نَفۡسًا إِلَّا وُسۡعَهَاۖ وَإِذَا قُلۡتُمۡ فَٱعۡدِلُواْ وَلَوۡ كَانَ ذَا قُرۡبَىٰۖ وَبِعَهۡدِ ٱللَّهِ أَوۡفُواْۚ ذَٰلِكُمۡ وَصَّىٰكُم بِهِۦ لَعَلَّكُمۡ تَذَكَّرُونَ (152) وَأَنَّ هَٰذَا صِرَٰطِي مُسۡتَقِيمٗا فَٱتَّبِعُوهُۖ وَلَا تَتَّبِعُواْ ٱلسُّبُلَ فَتَفَرَّقَ بِكُمۡ عَن سَبِيلِهِۦۚ ذَٰلِكُمۡ وَصَّىٰكُم بِهِۦ لَعَلَّكُمۡ تَتَّقُونَ (153)

״בואו ואקרא את אשר אסר עליכם ריבונכם״ – רשימה מקבילה לעשרת הדיברות.

«قل تعالوا أتل ما حرّم ربكم عليكم» – قائمة مقابلة للوصايا العشر.

שמות כ, ב–יד ⇄ 17:22–39

אל תעמיד עם אללה אלוה אחר, פן תשב מגונה ועזוב. (22) וגזר ריבונך שלא תעבדו אלא אותו, ולהורים תיטיבו. אם יגיע אחד מהם או שניהם אצלך לזקנה – אל תאמר להם ״אוף״ ואל תגער בהם, ודבר אליהם דברים נכבדים. (23) והשפל לפניהם את כנף הענווה מתוך רחמים, ואמור: ״ריבוני, רחם עליהם כפי שגידלוני בקטנותי״. (24) ריבונכם יודע היטב את אשר בנפשכם. אם תהיו ישרים – הוא סולח לשבים אליו. (25) ותן לקרוב את המגיע לו, וכן לעני ולהולך בדרך, ואל תבזבז בזבוז. (26) המבזבזים הם אחי השטנים, והשטן כפוי טובה לריבונו. (27) ואם תתרחק מהם, בבקשך רחמים מאת ריבונך שאתה מקווה להם – דבר אליהם דברים רכים. (28) אל תעשה את ידך כבולה אל צווארך, ואל תפשטנה עד תום, פן תשב מגונה ומרושש. (29) ריבונך מרחיב את הפרנסה למי שירצה ומצמצם; הוא יודע את עבדיו ורואה אותם. (30) ואל תהרגו את ילדיכם מפחד עוני; אנחנו נפרנס אותם ואתכם. הריגתם חטא גדול. (31) ואל תקרבו אל הניאוף; תועבה הוא ודרך רעה. (32) ואל תהרגו את הנפש אשר אסר אללה, אלא כדין. ומי שנהרג בעוול – נתנו לגואלו סמכות; אך אל יפריז בהריגה, כי הוא נעזר. (33) ואל תקרבו אל הון היתום אלא בדרך הטובה ביותר, עד אשר יגיע לבגרותו. וקיימו את הברית, כי על הברית תישאלו. (34) ומלאו את המידה כאשר תמודו, ושקלו במאזניים ישרים; זה טוב יותר ומיטיב באחריתו. (35) ואל תלך אחר דבר שאין לך בו דעת; השמיעה, הראייה והלב – על כל אלה ישאלו. (36) ואל תתהלך בארץ ביהירות; לא תבקע את הארץ ולא תגיע לגובה ההרים. (37) רעתם של כל אלה שנואה בעיני ריבונך. (38) זה מן החכמה אשר השרה אליך ריבונך. ואל תעמיד עם אללה אלוה אחר, פן תושלך אל גיהינום מגונה ונדחה. (39)

لَّا تَجۡعَلۡ مَعَ ٱللَّهِ إِلَٰهًا ءَاخَرَ فَتَقۡعُدَ مَذۡمُومٗا مَّخۡذُولٗا (22) ۞وَقَضَىٰ رَبُّكَ أَلَّا تَعۡبُدُوٓاْ إِلَّآ إِيَّاهُ وَبِٱلۡوَٰلِدَيۡنِ إِحۡسَٰنًاۚ إِمَّا يَبۡلُغَنَّ عِندَكَ ٱلۡكِبَرَ أَحَدُهُمَآ أَوۡ كِلَاهُمَا فَلَا تَقُل لَّهُمَآ أُفّٖ وَلَا تَنۡهَرۡهُمَا وَقُل لَّهُمَا قَوۡلٗا كَرِيمٗا (23) وَٱخۡفِضۡ لَهُمَا جَنَاحَ ٱلذُّلِّ مِنَ ٱلرَّحۡمَةِ وَقُل رَّبِّ ٱرۡحَمۡهُمَا كَمَا رَبَّيَانِي صَغِيرٗا (24) رَّبُّكُمۡ أَعۡلَمُ بِمَا فِي نُفُوسِكُمۡۚ إِن تَكُونُواْ صَٰلِحِينَ فَإِنَّهُۥ كَانَ لِلۡأَوَّـٰبِينَ غَفُورٗا (25) وَءَاتِ ذَا ٱلۡقُرۡبَىٰ حَقَّهُۥ وَٱلۡمِسۡكِينَ وَٱبۡنَ ٱلسَّبِيلِ وَلَا تُبَذِّرۡ تَبۡذِيرًا (26) إِنَّ ٱلۡمُبَذِّرِينَ كَانُوٓاْ إِخۡوَٰنَ ٱلشَّيَٰطِينِۖ وَكَانَ ٱلشَّيۡطَٰنُ لِرَبِّهِۦ كَفُورٗا (27) وَإِمَّا تُعۡرِضَنَّ عَنۡهُمُ ٱبۡتِغَآءَ رَحۡمَةٖ مِّن رَّبِّكَ تَرۡجُوهَا فَقُل لَّهُمۡ قَوۡلٗا مَّيۡسُورٗا (28) وَلَا تَجۡعَلۡ يَدَكَ مَغۡلُولَةً إِلَىٰ عُنُقِكَ وَلَا تَبۡسُطۡهَا كُلَّ ٱلۡبَسۡطِ فَتَقۡعُدَ مَلُومٗا مَّحۡسُورًا (29) إِنَّ رَبَّكَ يَبۡسُطُ ٱلرِّزۡقَ لِمَن يَشَآءُ وَيَقۡدِرُۚ إِنَّهُۥ كَانَ بِعِبَادِهِۦ خَبِيرَۢا بَصِيرٗا (30) وَلَا تَقۡتُلُوٓاْ أَوۡلَٰدَكُمۡ خَشۡيَةَ إِمۡلَٰقٖۖ نَّحۡنُ نَرۡزُقُهُمۡ وَإِيَّاكُمۡۚ إِنَّ قَتۡلَهُمۡ كَانَ خِطۡـٔٗا كَبِيرٗا (31) وَلَا تَقۡرَبُواْ ٱلزِّنَىٰٓۖ إِنَّهُۥ كَانَ فَٰحِشَةٗ وَسَآءَ سَبِيلٗا (32) وَلَا تَقۡتُلُواْ ٱلنَّفۡسَ ٱلَّتِي حَرَّمَ ٱللَّهُ إِلَّا بِٱلۡحَقِّۗ وَمَن قُتِلَ مَظۡلُومٗا فَقَدۡ جَعَلۡنَا لِوَلِيِّهِۦ سُلۡطَٰنٗا فَلَا يُسۡرِف فِّي ٱلۡقَتۡلِۖ إِنَّهُۥ كَانَ مَنصُورٗا (33) وَلَا تَقۡرَبُواْ مَالَ ٱلۡيَتِيمِ إِلَّا بِٱلَّتِي هِيَ أَحۡسَنُ حَتَّىٰ يَبۡلُغَ أَشُدَّهُۥۚ وَأَوۡفُواْ بِٱلۡعَهۡدِۖ إِنَّ ٱلۡعَهۡدَ كَانَ مَسۡـُٔولٗا (34) وَأَوۡفُواْ ٱلۡكَيۡلَ إِذَا كِلۡتُمۡ وَزِنُواْ بِٱلۡقِسۡطَاسِ ٱلۡمُسۡتَقِيمِۚ ذَٰلِكَ خَيۡرٞ وَأَحۡسَنُ تَأۡوِيلٗا (35) وَلَا تَقۡفُ مَا لَيۡسَ لَكَ بِهِۦ عِلۡمٌۚ إِنَّ ٱلسَّمۡعَ وَٱلۡبَصَرَ وَٱلۡفُؤَادَ كُلُّ أُوْلَـٰٓئِكَ كَانَ عَنۡهُ مَسۡـُٔولٗا (36) وَلَا تَمۡشِ فِي ٱلۡأَرۡضِ مَرَحًاۖ إِنَّكَ لَن تَخۡرِقَ ٱلۡأَرۡضَ وَلَن تَبۡلُغَ ٱلۡجِبَالَ طُولٗا (37) كُلُّ ذَٰلِكَ كَانَ سَيِّئُهُۥ عِندَ رَبِّكَ مَكۡرُوهٗا (38) ذَٰلِكَ مِمَّآ أَوۡحَىٰٓ إِلَيۡكَ رَبُّكَ مِنَ ٱلۡحِكۡمَةِۗ وَلَا تَجۡعَلۡ مَعَ ٱللَّهِ إِلَٰهًا ءَاخَرَ فَتُلۡقَىٰ فِي جَهَنَّمَ مَلُومٗا مَّدۡحُورًا (39)

שמות כ, ב–ג ⇄ בוח׳ארי 8; בוח׳ארי 2856 (זכות אללה על העבדים – תוחיד!)

בוח׳ארי 8: מפי אבן עמר: אמר שליח אללה: ״האסלאם נבנה על חמישה: העדות שאין אל מלבד אללה ושמוחמד הוא שליח אללה, קיום התפילה, נתינת הזכאת, העלייה לרגל וצום הרמדאן״. ✦ בוח׳ארי 2856: (מֻעאד׳ סיפר:) רכבתי מאחורי הנביא על חמור ששמו עֻפַיְר, ואמר: ״מֻעאד׳, היודע אתה מהי זכותו של אללה על עבדיו, ומהי זכות העבדים על אללה?״ אמרתי: ״אללה ושליחו יודעים יותר״. אמר: ״זכותו של אללה על העבדים היא שיעבדוהו ולא ישתפו עמו דבר; וזכות העבדים על אללה היא שלא יעניש את מי שאינו משתף עמו דבר״. אמרתי: ״שליח אללה, הלא אבשר זאת לאנשים?״ אמר: ״אל תבשרם, פן יסמכו על כך (ויתרשלו במעשים)״.

صحيح البخاري 8: حَدَّثَنَا عُبَيْدُ اللَّهِ بْنُ مُوسَى، قَالَ أَخْبَرَنَا حَنْظَلَةُ بْنُ أَبِي سُفْيَانَ، عَنْ عِكْرِمَةَ بْنِ خَالِدٍ، عَنِ ابْنِ عُمَرَ ـ رضى الله عنهما ـ قَالَ قَالَ رَسُولُ اللَّهِ صلى الله عليه وسلم ‏ "‏ بُنِيَ الإِسْلاَمُ عَلَى خَمْسٍ شَهَادَةِ أَنْ لاَ إِلَهَ إِلاَّ اللَّهُ وَأَنَّ مُحَمَّدًا رَسُولُ اللَّهِ، وَإِقَامِ الصَّلاَةِ، وَإِيتَاءِ الزَّكَاةِ، وَالْحَجِّ، وَصَوْمِ رَمَضَانَ ‏"‏‏.‏ ✦ صحيح البخاري 2856: كنت ردف النبي صلى الله عليه وسلم على حمار يقال له عفير، فقال " يا معاذ، هل تدري حق الله على عباده وما حق العباد على الله ". قلت الله ورسوله أعلم. قال " فإن حق الله على العباد أن يعبدوه ولا يشركوا به شيئا، وحق العباد على الله أن لا يعذب من لا يشرك به شيئا ". فقلت يا رسول الله، أفلا أبشر به الناس قال " لا تبشرهم فيتكلوا ".

הדיבר הראשון והשני (״אנוכי ה׳ אלוהיך... לא יהיה לך אלוהים אחרים על פניי״) — זהה לחדית׳ מֻעאד׳ בבוח׳ארי 2856: ״זכות אללה על עבדיו היא שיעבדו אותו ולא ישתפו עמו דבר!״

الوصية الأولى والثانية بالتوحيد الخالص ونبذ الشرك تطابقان البخاري 2856: «حق الله على العباد أن يعبدوه ولا يشركوا به شيئًا»!

שמות כ, ג ⇄ 17:22–39

אל תעמיד עם אללה אלוה אחר, פן תשב מגונה ועזוב. (22) וגזר ריבונך שלא תעבדו אלא אותו, ולהורים תיטיבו. אם יגיע אחד מהם או שניהם אצלך לזקנה – אל תאמר להם ״אוף״ ואל תגער בהם, ודבר אליהם דברים נכבדים. (23) והשפל לפניהם את כנף הענווה מתוך רחמים, ואמור: ״ריבוני, רחם עליהם כפי שגידלוני בקטנותי״. (24) ריבונכם יודע היטב את אשר בנפשכם. אם תהיו ישרים – הוא סולח לשבים אליו. (25) ותן לקרוב את המגיע לו, וכן לעני ולהולך בדרך, ואל תבזבז בזבוז. (26) המבזבזים הם אחי השטנים, והשטן כפוי טובה לריבונו. (27) ואם תתרחק מהם, בבקשך רחמים מאת ריבונך שאתה מקווה להם – דבר אליהם דברים רכים. (28) אל תעשה את ידך כבולה אל צווארך, ואל תפשטנה עד תום, פן תשב מגונה ומרושש. (29) ריבונך מרחיב את הפרנסה למי שירצה ומצמצם; הוא יודע את עבדיו ורואה אותם. (30) ואל תהרגו את ילדיכם מפחד עוני; אנחנו נפרנס אותם ואתכם. הריגתם חטא גדול. (31) ואל תקרבו אל הניאוף; תועבה הוא ודרך רעה. (32) ואל תהרגו את הנפש אשר אסר אללה, אלא כדין. ומי שנהרג בעוול – נתנו לגואלו סמכות; אך אל יפריז בהריגה, כי הוא נעזר. (33) ואל תקרבו אל הון היתום אלא בדרך הטובה ביותר, עד אשר יגיע לבגרותו. וקיימו את הברית, כי על הברית תישאלו. (34) ומלאו את המידה כאשר תמודו, ושקלו במאזניים ישרים; זה טוב יותר ומיטיב באחריתו. (35) ואל תלך אחר דבר שאין לך בו דעת; השמיעה, הראייה והלב – על כל אלה ישאלו. (36) ואל תתהלך בארץ ביהירות; לא תבקע את הארץ ולא תגיע לגובה ההרים. (37) רעתם של כל אלה שנואה בעיני ריבונך. (38) זה מן החכמה אשר השרה אליך ריבונך. ואל תעמיד עם אללה אלוה אחר, פן תושלך אל גיהינום מגונה ונדחה. (39)

يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓاْ إِذَا نُودِيَ لِلصَّلَوٰةِ مِن يَوۡمِ ٱلۡجُمُعَةِ فَٱسۡعَوۡاْ إِلَىٰ ذِكۡرِ ٱللَّهِ وَذَرُواْ ٱلۡبَيۡعَۚ ذَٰلِكُمۡ خَيۡرٞ لَّكُمۡ إِن كُنتُمۡ تَعۡلَمُونَ (9) فَإِذَا قُضِيَتِ ٱلصَّلَوٰةُ فَٱنتَشِرُواْ فِي ٱلۡأَرۡضِ وَٱبۡتَغُواْ مِن فَضۡلِ ٱللَّهِ وَٱذۡكُرُواْ ٱللَّهَ كَثِيرٗا لَّعَلَّكُمۡ تُفۡلِحُونَ (10)

יום השישי: עזבו המסחר לתפילה, ואחריה התפזרו – מול השבת.

يوم الجمعة: ذروا البيع إلى الصلاة، وبعدها انتشروا – مقابل السبت.

שמות כ, ח–יא ⇄ בוח׳ארי 876

״אנחנו האחרונים והראשונים ביום התקומה, אף שלהם (לאנשי הספר) ניתן הספר לפנינו. זה (יום השישי) יומם אשר הוטל עליהם, ונחלקו בו, ואללה הדריך אותנו אליו; ובני האדם הולכים אחרינו בו – היהודים מחר, והנוצרים מחרתיים״.

نحن الآخرون السابقون يوم القيامة، بيد أنهم أوتوا الكتاب من قبلنا، ثم هذا يومهم الذي فرض عليهم فاختلفوا فيه، فهدانا الله، فالناس لنا فيه تبع، اليهود غدا والنصارى بعد غد ".

יום השישי – ״היהודים מחר והנוצרים מחרתיים״; הם נחלקו בו.

يوم الجمعة – «اليهود غدا والنصارى بعد غد»؛ اختلفوا فيه.

אל תעמיד עם אללה אלוה אחר, פן תשב מגונה ועזוב. (22) וגזר ריבונך שלא תעבדו אלא אותו, ולהורים תיטיבו. אם יגיע אחד מהם או שניהם אצלך לזקנה – אל תאמר להם ״אוף״ ואל תגער בהם, ודבר אליהם דברים נכבדים. (23) והשפל לפניהם את כנף הענווה מתוך רחמים, ואמור: ״ריבוני, רחם עליהם כפי שגידלוני בקטנותי״. (24) ריבונכם יודע היטב את אשר בנפשכם. אם תהיו ישרים – הוא סולח לשבים אליו. (25) ותן לקרוב את המגיע לו, וכן לעני ולהולך בדרך, ואל תבזבז בזבוז. (26) המבזבזים הם אחי השטנים, והשטן כפוי טובה לריבונו. (27) ואם תתרחק מהם, בבקשך רחמים מאת ריבונך שאתה מקווה להם – דבר אליהם דברים רכים. (28) אל תעשה את ידך כבולה אל צווארך, ואל תפשטנה עד תום, פן תשב מגונה ומרושש. (29) ריבונך מרחיב את הפרנסה למי שירצה ומצמצם; הוא יודע את עבדיו ורואה אותם. (30) ואל תהרגו את ילדיכם מפחד עוני; אנחנו נפרנס אותם ואתכם. הריגתם חטא גדול. (31) ואל תקרבו אל הניאוף; תועבה הוא ודרך רעה. (32) ואל תהרגו את הנפש אשר אסר אללה, אלא כדין. ומי שנהרג בעוול – נתנו לגואלו סמכות; אך אל יפריז בהריגה, כי הוא נעזר. (33) ואל תקרבו אל הון היתום אלא בדרך הטובה ביותר, עד אשר יגיע לבגרותו. וקיימו את הברית, כי על הברית תישאלו. (34) ומלאו את המידה כאשר תמודו, ושקלו במאזניים ישרים; זה טוב יותר ומיטיב באחריתו. (35) ואל תלך אחר דבר שאין לך בו דעת; השמיעה, הראייה והלב – על כל אלה ישאלו. (36) ואל תתהלך בארץ ביהירות; לא תבקע את הארץ ולא תגיע לגובה ההרים. (37) רעתם של כל אלה שנואה בעיני ריבונך. (38) זה מן החכמה אשר השרה אליך ריבונך. ואל תעמיד עם אללה אלוה אחר, פן תושלך אל גיהינום מגונה ונדחה. (39)

(33) ואל תקרבו אל הון היתום אלא בדרך הטובה ביותר, עד אשר יגיע לבגרותו. וקיימו את הברית, כי על הברית תישאלו. (34) ומלאו את המידה כאשר תמודו, ושקלו במאזניים ישרים; זה טוב יותר ומיטיב באחריתו. (35) ואל תלך אחר דבר שאין לך בו דעת; השמיעה, הראייה והלב – על כל אלה ישאלו. (36) ואל תתהלך בארץ ביהירות; לא תבקע את הארץ ולא תגיע לגובה ההרים. (37) רעתם של כל אלה שנואה בעיני ריבונך. (38) זה מן החכמה אשר השרה אליך ריבונך. ואל תעמיד עם אללה אלוה אחר, פן תושלך אל גיהינום מגונה ונדחה. (39)

صحيح البخاري 446: رسول الله صلى الله عليه وسلم مبنيا باللبن، وسقفه الجريد، وعمده خشب النخل، فلم يزد فيه أبو بكر شيئا، وزاد فيه عمر وبناه على بنيانه في عهد رسول الله صلى الله عليه وسلم باللبن والجريد، وأعاد عمده خشبا، ثم غيره عثمان، فزاد فيه زيادة كثيرة، وبنى جداره بالحجارة المنقوشة والقصة، وجعل عمده من حجارة منقوشة، وسقفه بالساج ✦ سنن أبي داود 448: قال رسول الله صلى الله عليه وسلم " ما أمرت بتشييد المساجد " . قال ابن عباس لتزخرفنها كما زخرفت اليهود والنصارى
