30:11וַיְדַבֵּר יְהֹוָה אֶל־מֹשֶׁה לֵּאמֹר׃
وَكَلَّمَ ٱلرَّبُّ مُوسَى قَائِلًا:
30:12כִּי תִשָּׂא אֶת־רֹאשׁ בְּנֵי־יִשְׂרָאֵל לִפְקֻדֵיהֶם וְנָתְנוּ אִישׁ כֹּפֶר נַפְשׁוֹ לַיהֹוָה בִּפְקֹד אֹתָם וְלֹא־יִהְיֶה בָהֶם נֶגֶף בִּפְקֹד אֹתָם׃
«إِذَا أَخَذْتَ كَمِّيَّةَ بَنِي إِسْرَائِيلَ بِحَسَبِ ٱلْمَعْدُودِينَ مِنْهُمْ، يُعْطُونَ كُلُّ وَاحِدٍ فِدْيَةَ نَفْسِهِ لِلرَّبِّ عِنْدَمَا تَعُدُّهُمْ، لِئَلَّا يَصِيرَ فِيهِمْ وَبَأٌ عِنْدَمَا تَعُدُّهُمْ.

שמות ל, יב–טו ⇄ בוח׳ארי 1503; מוסלם 984

בוח׳ארי 1503: ״צאע תמרים או צאע שעורים – על העבד ועל החופשי, על הזכר ועל הנקבה, על הקטן ועל הגדול״. ✦ מוסלם 984: מפי אבן עמר: שליח אללה הטיל את זכאת הפִטְר של הרמדאן על בני האדם – צאע של תמרים או צאע של שעורה – על כל בן חורין או עבד, זכר או נקבה, מן המוסלמים.

صحيح البخاري 1503: صَاعًا مِنْ تَمْرٍ أَوْ صَاعًا مِنْ شَعِيرٍ عَلَى الْعَبْدِ وَالْحُرِّ وَالذَّكَرِ وَالأُنْثَى وَالصَّغِيرِ وَالْكَبِيرِ ✦ صحيح مسلم 984: حَدَّثَنَا عَبْدُ اللَّهِ بْنُ مَسْلَمَةَ بْنِ قَعْنَبٍ، وَقُتَيْبَةُ بْنُ سَعِيدٍ، قَالاَ حَدَّثَنَا مَالِكٌ، ح وَحَدَّثَنَا يَحْيَى بْنُ يَحْيَى، - وَاللَّفْظُ لَهُ - قَالَ قَرَأْتُ عَلَى مَالِكٍ عَنْ نَافِعٍ عَنِ ابْنِ عُمَرَ أَنَّ رَسُولَ اللَّهِ صلى الله عليه وسلم فَرَضَ زَكَاةَ الْفِطْرِ مِنْ رَمَضَانَ عَلَى النَّاسِ صَاعًا مِنْ تَمْرٍ أَوْ صَاعًا مِنْ شَعِيرٍ عَلَى كُلِّ حُرٍّ أَوْ عَبْدٍ ذَكَرٍ أَوْ أُنْثَى مِنَ الْمُسْلِمِينَ ‏.‏

מצוות מחצית השקל (״העשיר לא ירבה והדל לא ימעיט״ – מס גולגולת שוויוני אחיד לכל נפש!) — מקבילה הלכתית מדויקת לזכאת אלפטר (שיעור אחיד קבוע של צאע אחד לכל נפש, עשיר כעני, בוח׳ארי 1503; מוסלם 984)!

فريضة «نصف الشاقل» بمقدار واحد متساوٍ للغني والفقير («الغني لا يكثر والفقير لا يقلل») تطابق تمامًا «زكاة الفطر» (صاع واحد عن كل نفس بالتساوي، البخاري 1503)!

30:13זֶה יִתְּנוּ כׇּל־הָעֹבֵר עַל־הַפְּקֻדִים מַחֲצִית הַשֶּׁקֶל בְּשֶׁקֶל הַקֹּדֶשׁ עֶשְׂרִים גֵּרָה הַשֶּׁקֶל מַחֲצִית הַשֶּׁקֶל תְּרוּמָה לַיהֹוָה׃
هَذَا مَا يُعْطِيهِ كُلُّ مَنِ ٱجْتَازَ إِلَى ٱلْمَعْدُودِينَ: نِصْفُ ٱلشَّاقِلِ بِشَاقِلِ ٱلْقُدْسِ. ٱلشَّاقِلُ هُوَ عِشْرُونَ جِيرَةً. نِصْفُ ٱلشَّاقِلِ تَقْدِمَةً لِلرَّبِّ.
30:14כֹּל הָעֹבֵר עַל־הַפְּקֻדִים מִבֶּן עֶשְׂרִים שָׁנָה וָמָעְלָה יִתֵּן תְּרוּמַת יְהֹוָה׃
كُلُّ مَنِ ٱجْتَازَ إِلَى ٱلْمَعْدُودِينَ مِنِ ٱبْنِ عِشْرِينَ سَنَةً فَصَاعِدًا يُعْطِي تَقْدِمَةً لِلرَّبِّ.
30:15הֶעָשִׁיר לֹא־יַרְבֶּה וְהַדַּל לֹא יַמְעִיט מִמַּחֲצִית הַשָּׁקֶל לָתֵת אֶת־תְּרוּמַת יְהֹוָה לְכַפֵּר עַל־נַפְשֹׁתֵיכֶם׃
اَلْغَنِيُّ لَا يُكَثِّرُ وَٱلْفَقِيرُ لَا يُقَلِّلُ عَنْ نِصْفِ ٱلشَّاقِلِ حِينَ تُعْطُونَ تَقْدِمَةَ ٱلرَّبِّ لِلتَّكْفِيرِ عَنْ نُفُوسِكُمْ.
30:16וְלָקַחְתָּ אֶת־כֶּסֶף הַכִּפֻּרִים מֵאֵת בְּנֵי יִשְׂרָאֵל וְנָתַתָּ אֹתוֹ עַל־עֲבֹדַת אֹהֶל מוֹעֵד וְהָיָה לִבְנֵי יִשְׂרָאֵל לְזִכָּרוֹן לִפְנֵי יְהֹוָה לְכַפֵּר עַל־נַפְשֹׁתֵיכֶם׃
وَتَأْخُذُ فِضَّةَ ٱلْكَفَّارَةِ مِنْ بَنِي إِسْرَائِيلَ، وَتَجْعَلُهَا لِخِدْمَةِ خَيْمَةِ ٱلِٱجْتِمَاعِ. فَتَكُونُ لِبَنِي إِسْرَائِيلَ تَذْكَارًا أَمَامَ ٱلرَّبِّ لِلتَّكْفِيرِ عَنْ نُفُوسِكُمْ».
30:17וַיְדַבֵּר יְהֹוָה אֶל־מֹשֶׁה לֵּאמֹר׃
وَكَلَّمْ ٱلرَّبُّ مُوسَى قَائِلًا:
30:18וְעָשִׂיתָ כִּיּוֹר נְחֹשֶׁת וְכַנּוֹ נְחֹשֶׁת לְרׇחְצָה וְנָתַתָּ אֹתוֹ בֵּין־אֹהֶל מוֹעֵד וּבֵין הַמִּזְבֵּחַ וְנָתַתָּ שָׁמָּה מָיִם׃
«وَتَصْنَعُ مِرْحَضَةً مِنْ نُحَاسٍ، وَقَاعِدَتَهَا مِنْ نُحَاسٍ، لِلِٱغْتِسَالِ. وَتَجْعَلُهَا بَيْنَ خَيْمَةِ ٱلِٱجْتِمَاعِ وَٱلْمَذْبَحِ، وَتَجْعَلُ فِيهَا مَاءً.

שמות ל, יח–כא ⇄ בוח׳ארי 135; בוח׳ארי 159; מוסלם 224

בוח׳ארי 135: מפי אבו הֻרַיְרַה: אמר שליח אללה: ״לא תתקבל תפילתו של מי שנטמא (בחַדַת׳) עד שיעשה וֻצ׳וּא״. אמר אדם מחַצְ׳רַמַוְת: ״מהו החדת׳, אבו הֻרַיְרַה?״ אמר: ״הפחה או נפיחה״. ✦ בוח׳ארי 159: (חֻמְרַאן סיפר) שראה את עֻת׳מאן בן עַפַּאן מבקש כלי (מים). שפך על כפיו שלוש פעמים ורחץ אותן; אחר כך הכניס את ימינו לכלי, ושטף את פיו והעלה מים לאפו; אחר כך רחץ את פניו שלוש פעמים, ואת ידיו עד המרפקים שלוש פעמים; אחר כך העביר יד על ראשו; אחר כך רחץ את רגליו שלוש פעמים עד הקרסוליים. אחר כך אמר: אמר שליח אללה: ״מי שנוטל ידיים כנטילתי זו, ואחר כך מתפלל שתי השתחוויות (רַכְּעות) מבלי שיסיח בהן את דעתו – נמחלים לו חטאיו הקודמים״. ✦ מוסלם 224: מפי מֻצְעַבּ בן סעד: עבדאללה בן עמר נכנס אל אבן עאמר לבקרו בחוליו. אמר (אבן עאמר): ״האם לא תתפלל לאללה בעדי, אבן עמר?״ אמר: ״שמעתי את שליח אללה אומר: לא תתקבל תפילה בלא טהרה, ולא צדקה מממון גזול״ – ואתה היית (מושל) על בצרה.

صحيح البخاري 135: حَدَّثَنَا إِسْحَاقُ بْنُ إِبْرَاهِيمَ الْحَنْظَلِيُّ، قَالَ أَخْبَرَنَا عَبْدُ الرَّزَّاقِ، قَالَ أَخْبَرَنَا مَعْمَرٌ، عَنْ هَمَّامِ بْنِ مُنَبِّهٍ، أَنَّهُ سَمِعَ أَبَا هُرَيْرَةَ، يَقُولُ قَالَ رَسُولُ اللَّهِ صلى الله عليه وسلم ‏ "‏ لاَ تُقْبَلُ صَلاَةُ مَنْ أَحْدَثَ حَتَّى يَتَوَضَّأَ ‏"‏‏.‏ قَالَ رَجُلٌ مِنْ حَضْرَمَوْتَ مَا الْحَدَثُ يَا أَبَا هُرَيْرَةَ قَالَ فُسَاءٌ أَوْ ضُرَاطٌ‏.‏ ✦ صحيح البخاري 159: رأى عثمان بن عفان دعا بإناء، فأفرغ على كفيه ثلاث مرار فغسلهما، ثم أدخل يمينه في الإناء فمضمض، واستنشق، ثم غسل وجهه ثلاثا، ويديه إلى المرفقين ثلاث مرار، ثم مسح برأسه، ثم غسل رجليه ثلاث مرار إلى الكعبين، ثم قال قال رسول الله صلى الله عليه وسلم " من توضأ نحو وضوئي هذا، ثم صلى ركعتين، لا يحدث فيهما نفسه، غفر له ما تقدم من ذنبه ". ✦ صحيح مسلم 224: حَدَّثَنَا سَعِيدُ بْنُ مَنْصُورٍ، وَقُتَيْبَةُ بْنُ سَعِيدٍ، وَأَبُو كَامِلٍ الْجَحْدَرِيُّ - وَاللَّفْظُ لِسَعِيدٍ - قَالُوا حَدَّثَنَا أَبُو عَوَانَةَ، عَنْ سِمَاكِ بْنِ حَرْبٍ، عَنْ مُصْعَبِ بْنِ سَعْدٍ، قَالَ دَخَلَ عَبْدُ اللَّهِ بْنُ عُمَرَ عَلَى ابْنِ عَامِرٍ يَعُودُهُ وَهُوَ مَرِيضٌ فَقَالَ أَلاَ تَدْعُو اللَّهَ لِي يَا ابْنَ عُمَرَ ‏.‏ قَالَ إِنِّي سَمِعْتُ رَسُولَ اللَّهِ صلى الله عليه وسلم يَقُولُ ‏ "‏ لاَ تُقْبَلُ صَلاَةٌ بِغَيْرِ طُهُورٍ وَلاَ صَدَقَةٌ مِنْ غُلُولٍ ‏"‏ ‏.‏ وَكُنْتَ عَلَى الْبَصْرَةِ ‏.‏

כיור הנחושת לרחיצת ידיים ורגליים לפני כל כניסה לעבודה — מקביל להלכות אלוֻצ׳וא (בוח׳ארי 159: תיאור רחיצת הידיים והרגליים של הנביא לפני התפילה).

غسل الأيدي والأرجل عند المرحضة قبل الدخول للعبادة يوازي تمامًا أحكام «الوضوء» قبل الصلاة (البخاري 135 و159).

30:19וְרָחֲצוּ אַהֲרֹן וּבָנָיו מִמֶּנּוּ אֶת־יְדֵיהֶם וְאֶת־רַגְלֵיהֶם׃
فَيَغْسِلُ هَارُونُ وَبَنُوهُ أَيْدِيَهُمْ وَأَرْجُلَهُمْ مِنْهَا.

שמות ל, יט–כא ⇄ 5:6

הוי המאמינים, כאשר תקומו לתפילה – רחצו את פניכם ואת ידיכם עד המרפקים, ונגבו את ראשיכם, ואת רגליכם עד הקרסוליים. ואם טמאים אתם (ג׳ונוב) – היטהרו. ואם חולים אתם או בדרך, או בא אחד מכם מבית הכיסא, או נגעתם בנשים, ולא מצאתם מים – בקשו עפר נקי ונגבו ממנו את פניכם וידיכם. אללה אינו חפץ להטיל עליכם קושי, אלא חפץ לטהרכם ולמלא עליכם את חסדו, למען תודו. (6)

يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓاْ إِذَا قُمۡتُمۡ إِلَى ٱلصَّلَوٰةِ فَٱغۡسِلُواْ وُجُوهَكُمۡ وَأَيۡدِيَكُمۡ إِلَى ٱلۡمَرَافِقِ وَٱمۡسَحُواْ بِرُءُوسِكُمۡ وَأَرۡجُلَكُمۡ إِلَى ٱلۡكَعۡبَيۡنِۚ وَإِن كُنتُمۡ جُنُبٗا فَٱطَّهَّرُواْۚ وَإِن كُنتُم مَّرۡضَىٰٓ أَوۡ عَلَىٰ سَفَرٍ أَوۡ جَآءَ أَحَدٞ مِّنكُم مِّنَ ٱلۡغَآئِطِ أَوۡ لَٰمَسۡتُمُ ٱلنِّسَآءَ فَلَمۡ تَجِدُواْ مَآءٗ فَتَيَمَّمُواْ صَعِيدٗا طَيِّبٗا فَٱمۡسَحُواْ بِوُجُوهِكُمۡ وَأَيۡدِيكُم مِّنۡهُۚ مَا يُرِيدُ ٱللَّهُ لِيَجۡعَلَ عَلَيۡكُم مِّنۡ حَرَجٖ وَلَٰكِن يُرِيدُ لِيُطَهِّرَكُمۡ وَلِيُتِمَّ نِعۡمَتَهُۥ عَلَيۡكُمۡ لَعَلَّكُمۡ تَشۡكُرُونَ (6)

عِنْدَ دُخُولِهِمْ إِلَى خَيْمَةِ ٱلِٱجْتِمَاعِ يَغْسِلُونَ بِمَاءٍ لِئَلَّا يَمُوتُوا، أَوْ عِنْدَ ٱقْتِرَابِهِمْ إِلَى ٱلْمَذْبَحِ لِلْخِدْمَةِ لِيُوقِدُوا وَقُودًا لِلرَّبِّ.
30:21וְרָחֲצוּ יְדֵיהֶם וְרַגְלֵיהֶם וְלֹא יָמֻתוּ וְהָיְתָה לָהֶם חׇק־עוֹלָם לוֹ וּלְזַרְעוֹ לְדֹרֹתָם׃
يَغْسِلُونَ أَيْدِيَهُمْ وَأَرْجُلَهُمْ لِئَلَّا يَمُوتُوا. وَيَكُونُ لَهُمْ فَرِيضَةً أَبَدِيَّةً لَهُ وَلِنَسْلِهِ فِي أَجْيَالِهِمْ».
30:22וַיְדַבֵּר יְהֹוָה אֶל־מֹשֶׁה לֵּאמֹר׃
وَكَلَّمَ ٱلرَّبُّ مُوسَى قَائِلًا:
30:23וְאַתָּה קַח־לְךָ בְּשָׂמִים רֹאשׁ מׇר־דְּרוֹר חֲמֵשׁ מֵאוֹת וְקִנְּמׇן־בֶּשֶׂם מַחֲצִיתוֹ חֲמִשִּׁים וּמָאתָיִם וּקְנֵה־בֹשֶׂם חֲמִשִּׁים וּמָאתָיִם׃
«وَأَنْتَ تَأْخُذُ لَكَ أَفْخَرَ ٱلْأَطْيَابِ: مُرًّا قَاطِرًا خَمْسَ مِئَةِ شَاقِلٍ، وَقِرْفَةً عَطِرَةً نِصْفَ ذَلِكَ: مِئَتَيْنِ وَخَمْسِينَ، وَقَصَبَ ٱلذَّرِيرَةِ مِئَتَيْنِ وَخَمْسِينَ،
30:24וְקִדָּה חֲמֵשׁ מֵאוֹת בְּשֶׁקֶל הַקֹּדֶשׁ וְשֶׁמֶן זַיִת הִין׃
وَسَلِيخَةً خَمْسَ مِئَةٍ بِشَاقِلِ ٱلْقُدْسِ، وَمِنْ زَيْتِ ٱلزَّيْتُونِ هِينًا.
30:25וְעָשִׂיתָ אֹתוֹ שֶׁמֶן מִשְׁחַת־קֹדֶשׁ רֹקַח מִרְקַחַת מַעֲשֵׂה רֹקֵחַ שֶׁמֶן מִשְׁחַת־קֹדֶשׁ יִהְיֶה׃
وَتَصْنَعُهُ دُهْنًا مُقَدَّسًا لِلْمَسْحَةِ. عِطْرَ عِطَارَةٍ صَنْعَةَ ٱلْعَطَّارِ. دُهْنًا مُقَدَّسًا لِلْمَسْحَةِ يَكُونُ.
30:26וּמָשַׁחְתָּ בוֹ אֶת־אֹהֶל מוֹעֵד וְאֵת אֲרוֹן הָעֵדֻת׃
وَتَمْسَحُ بِهِ خَيْمَةَ ٱلِٱجْتِمَاعِ، وَتَابُوتَ ٱلشَّهَادَةِ،
30:27וְאֶת־הַשֻּׁלְחָן וְאֶת־כׇּל־כֵּלָיו וְאֶת־הַמְּנֹרָה וְאֶת־כֵּלֶיהָ וְאֵת מִזְבַּח הַקְּטֹרֶת׃
وَٱلْمَائِدَةَ وَكُلَّ آنِيَتِهَا، وَٱلْمَنَارَةَ وَآنِيَتَهَا، وَمَذْبَحَ ٱلْبَخُورِ،
30:28וְאֶת־מִזְבַּח הָעֹלָה וְאֶת־כׇּל־כֵּלָיו וְאֶת־הַכִּיֹּר וְאֶת־כַּנּוֹ׃
وَمَذْبَحَ ٱلْمُحْرَقَةِ وَكُلَّ آنِيَتِهِ، وَٱلْمِرْحَضَةَ وَقَاعِدَتَهَا.
30:29וְקִדַּשְׁתָּ אֹתָם וְהָיוּ קֹדֶשׁ קׇדָשִׁים כׇּל־הַנֹּגֵעַ בָּהֶם יִקְדָּשׁ׃
وَتُقَدِّسُهَا فَتَكُونُ قُدْسَ أَقْدَاسٍ. كُلُّ مَا مَسَّهَا يَكُونُ مُقَدَّسًا.
30:30וְאֶת־אַהֲרֹן וְאֶת־בָּנָיו תִּמְשָׁח וְקִדַּשְׁתָּ אֹתָם לְכַהֵן לִי׃
وَتَمْسَحُ هَارُونَ وَبَنِيهِ وَتُقَدِّسُهُمْ لِيَكْهَنُوا لِي.
30:31וְאֶל־בְּנֵי יִשְׂרָאֵל תְּדַבֵּר לֵאמֹר שֶׁמֶן מִשְׁחַת־קֹדֶשׁ יִהְיֶה זֶה לִי לְדֹרֹתֵיכֶם׃
وَتُكَلِّمُ بَنِي إِسْرَائِيلَ قَائِلًا: يَكُونُ هَذَا لِي دُهْنًا مُقَدَّسًا لِلْمَسْحَةِ فِي أَجْيَالِكُمْ.
30:32עַל־בְּשַׂר אָדָם לֹא יִיסָךְ וּבְמַתְכֻּנְתּוֹ לֹא תַעֲשׂוּ כָּמֹהוּ קֹדֶשׁ הוּא קֹדֶשׁ יִהְיֶה לָכֶם׃
عَلَى جَسَدِ إِنْسَانٍ لَا يُسْكَبُ، وَعَلَى مَقَادِيرِهِ لَا تَصْنَعُوا مِثْلَهُ. مُقَدَّسٌ هُوَ، وَيَكُونُ مُقَدَّسًا عِنْدَكُمْ.
30:33אִישׁ אֲשֶׁר יִרְקַח כָּמֹהוּ וַאֲשֶׁר יִתֵּן מִמֶּנּוּ עַל־זָר וְנִכְרַת מֵעַמָּיו׃
كُلُّ مَنْ رَكَّبَ مِثْلَهُ وَمَنْ جَعَلَ مِنْهُ عَلَى أَجْنَبِيٍّ يُقْطَعُ مِنْ شَعْبِهِ».
30:34וַיֹּאמֶר יְהֹוָה אֶל־מֹשֶׁה קַח־לְךָ סַמִּים נָטָף וּשְׁחֵלֶת וְחֶלְבְּנָה סַמִּים וּלְבֹנָה זַכָּה בַּד בְּבַד יִהְיֶה׃
وَقَالَ ٱلرَّبُّ لِمُوسَى: «خُذْ لَكَ أَعْطَارًا: مَيْعَةً وَأَظْفَارًا وَقِنَّةً عَطِرَةً وَلُبَانًا نَقِيًّا. تَكُونُ أَجْزَاءً مُتَسَاوِيَةً،
30:35וְעָשִׂיתָ אֹתָהּ קְטֹרֶת רֹקַח מַעֲשֵׂה רוֹקֵחַ מְמֻלָּח טָהוֹר קֹדֶשׁ׃
فَتَصْنَعُهَا بَخُورًا عَطِرًا صَنْعَةَ ٱلْعَطَّارِ، مُمَلَّحًا نَقِيًّا مُقَدَّسًا.
30:36וְשָׁחַקְתָּ מִמֶּנָּה הָדֵק וְנָתַתָּה מִמֶּנָּה לִפְנֵי הָעֵדֻת בְּאֹהֶל מוֹעֵד אֲשֶׁר אִוָּעֵד לְךָ שָׁמָּה קֹדֶשׁ קׇדָשִׁים תִּהְיֶה לָכֶם׃
وَتَسْحَقُ مِنْهُ نَاعِمًا، وَتَجْعَلُ مِنْهُ قُدَّامَ ٱلشَّهَادَةِ فِي خَيْمَةِ ٱلِٱجْتِمَاعِ حَيْثُ أَجْتَمِعُ بِكَ. قُدْسَ أَقْدَاسٍ يَكُونُ عِنْدَكُمْ.
30:37וְהַקְּטֹרֶת אֲשֶׁר תַּעֲשֶׂה בְּמַתְכֻּנְתָּהּ לֹא תַעֲשׂוּ לָכֶם קֹדֶשׁ תִּהְיֶה לְךָ לַיהֹוָה׃
وَٱلْبَخُورُ ٱلَّذِي تَصْنَعُهُ عَلَى مَقَادِيرِهِ لَا تَصْنَعُوا لِأَنْفُسِكُمْ. يَكُونُ عِنْدَكَ مُقَدَّسًا لِلرَّبِّ.
30:38אִישׁ אֲשֶׁר־יַעֲשֶׂה כָמוֹהָ לְהָרִיחַ בָּהּ וְנִכְרַת מֵעַמָּיו׃
كُلُّ مَنْ صَنَعَ مِثْلَهُ لِيَشُمَّهُ يُقْطَعُ مِنْ شَعْبِهِ».
31:1וַיְדַבֵּר יְהֹוָה אֶל־מֹשֶׁה לֵּאמֹר׃
وَكَلَّمَ ٱلرَّبُّ مُوسَى قَائِلًا:
31:2רְאֵה קָרָאתִי בְשֵׁם בְּצַלְאֵל בֶּן־אוּרִי בֶן־חוּר לְמַטֵּה יְהוּדָה׃
«اُنْظُرْ. قَدْ دَعَوْتُ بَصَلْئِيلَ بْنَ أُورِي بْنَ حُورَ مِنْ سِبْطِ يَهُوذَا بِٱسْمِهِ،
31:3וָאֲמַלֵּא אֹתוֹ רוּחַ אֱלֹהִים בְּחׇכְמָה וּבִתְבוּנָה וּבְדַעַת וּבְכׇל־מְלָאכָה׃
وَمَلَأْتُهُ مِنْ رُوحِ ٱللهِ بِٱلْحِكْمَةِ وَٱلْفَهْمِ وَٱلْمَعْرِفَةِ وَكُلِّ صَنْعَةٍ،
31:4לַחְשֹׁב מַחֲשָׁבֹת לַעֲשׂוֹת בַּזָּהָב וּבַכֶּסֶף וּבַנְּחֹשֶׁת׃
لِٱخْتِرَاعِ مُخْتَرَعَاتٍ لِيَعْمَلَ فِي ٱلذَّهَبِ وَٱلْفِضَّةِ وَٱلنُّحَاسِ،
31:5וּבַחֲרֹשֶׁת אֶבֶן לְמַלֹּאת וּבַחֲרֹשֶׁת עֵץ לַעֲשׂוֹת בְּכׇל־מְלָאכָה׃
وَنَقْشِ حِجَارَةٍ لِلتَّرْصِيعِ، وَنِجَارَةِ ٱلْخَشَبِ، لِيَعْمَلَ فِي كُلِّ صَنْعَةٍ.
31:6וַאֲנִי הִנֵּה נָתַתִּי אִתּוֹ אֵת אׇהֳלִיאָב בֶּן־אֲחִיסָמָךְ לְמַטֵּה־דָן וּבְלֵב כׇּל־חֲכַם־לֵב נָתַתִּי חׇכְמָה וְעָשׂוּ אֵת כׇּל־אֲשֶׁר צִוִּיתִךָ׃
وَهَا أَنَا قَدْ جَعَلْتُ مَعَهُ أُهُولِيآبَ بْنَ أَخِيسَامَاكَ مِنْ سِبْطِ دَانَ. وَفِي قَلْبِ كُلِّ حَكِيمِ ٱلْقَلْبِ جَعَلْتُ حِكْمَةً، لِيَصْنَعُوا كُلَّ مَا أَمَرْتُكَ:
31:7אֵת אֹהֶל מוֹעֵד וְאֶת־הָאָרֹן לָעֵדֻת וְאֶת־הַכַּפֹּרֶת אֲשֶׁר עָלָיו וְאֵת כׇּל־כְּלֵי הָאֹהֶל׃
خَيْمَةَ ٱلِٱجْتِمَاعِ، وَتَابُوتَ ٱلشَّهَادَةِ، وَٱلْغِطَاءَ ٱلَّذِي عَلَيْهِ، وَكُلَّ آنِيَةِ ٱلْخَيْمَةِ،
31:8וְאֶת־הַשֻּׁלְחָן וְאֶת־כֵּלָיו וְאֶת־הַמְּנֹרָה הַטְּהֹרָה וְאֶת־כׇּל־כֵּלֶיהָ וְאֵת מִזְבַּח הַקְּטֹרֶת׃
وَٱلْمَائِدَةَ وَآنِيَتَهَا، وَٱلْمَنَارَةَ ٱلطَّاهِرَةَ وَكُلَّ آنِيَتِهَا، وَمَذْبَحَ ٱلْبَخُورِ،
31:9וְאֶת־מִזְבַּח הָעֹלָה וְאֶת־כׇּל־כֵּלָיו וְאֶת־הַכִּיּוֹר וְאֶת־כַּנּוֹ׃
وَمَذْبَحَ ٱلْمُحْرَقَةِ وَكُلَّ آنِيَتِهِ، وَٱلْمِرْحَضَةَ وَقَاعِدَتَهَا،
31:10וְאֵת בִּגְדֵי הַשְּׂרָד וְאֶת־בִּגְדֵי הַקֹּדֶשׁ לְאַהֲרֹן הַכֹּהֵן וְאֶת־בִּגְדֵי בָנָיו לְכַהֵן׃
وَٱلثِّيَابَ ٱلْمَنْسُوجَةَ، وَٱلثِّيَابَ ٱلْمُقَدَّسَةَ لِهَارُونَ ٱلْكَاهِنِ وَثِيَابَ بَنِيهِ لِلْكَهَانَةِ،
31:11וְאֵת שֶׁמֶן הַמִּשְׁחָה וְאֶת־קְטֹרֶת הַסַּמִּים לַקֹּדֶשׁ כְּכֹל אֲשֶׁר־צִוִּיתִךָ יַעֲשׂוּ׃
وَدُهْنَ ٱلْمَسْحَةِ وَٱلْبَخُورَ ٱلْعَطِرَ لِلْقُدْسِ. حَسَبَ كُلِّ مَا أَمَرْتُكَ بِهِ يَصْنَعُونَ».
31:12וַיֹּאמֶר יְהֹוָה אֶל־מֹשֶׁה לֵּאמֹר׃
وَكَلَّمَ ٱلرَّبُّ مُوسَى قَائِلًا:
31:13וְאַתָּה דַּבֵּר אֶל־בְּנֵי יִשְׂרָאֵל לֵאמֹר אַךְ אֶת־שַׁבְּתֹתַי תִּשְׁמֹרוּ כִּי אוֹת הִוא בֵּינִי וּבֵינֵיכֶם לְדֹרֹתֵיכֶם לָדַעַת כִּי אֲנִי יְהֹוָה מְקַדִּשְׁכֶם׃
«وَأَنْتَ تُكَلِّمُ بَنِي إِسْرَائِيلَ قَائِلًا: سُبُوتِي تَحْفَظُونَهَا، لِأَنَّهُ عَلَامَةٌ بَيْنِي وَبَيْنَكُمْ فِي أَجْيَالِكُمْ لِتَعْلَمُوا أَنِّي أَنَا ٱلرَّبُّ ٱلَّذِي يُقَدِّسُكُمْ،

בוח׳ארי 6304: שליח אללה אמר: ״לכל נביא תפילה אחת שנענית, שבה הוא מתפלל; ואני רוצה להצפין את תפילתי כמליצה בעד אומתי בעולם הבא״. ✦ מוסלם 199: אמר שליח אללה: ״לכל נביא תפילה אחת שנענית; כל נביא הקדים (ומימש) את תפילתו, ואני הצפנתי את תפילתי כמליצה בעד אומתי ביום התקומה – והיא תגיע, אם ירצה אללה, לכל מי מאומתי שמת מבלי לשתף דבר עם אללה״.

صحيح البخاري 6304: أن رسول الله صلى الله عليه وسلم قال " لكل نبي دعوة مستجابة يدعو بها، وأريد أن أختبئ دعوتي شفاعة لأمتي في الآخرة ". ✦ صحيح مسلم 199: قال رسول الله صلى الله عليه وسلم " لكل نبي دعوة مستجابة فتعجل كل نبي دعوته وإني اختبأت دعوتي شفاعة لأمتي يوم القيامة فهي نائلة إن شاء الله من مات من أمتي لا يشرك بالله شيئا " .

מסירות הנפש של משה למען עמו (״ואם אין מחני נא מספרך אשר כתבת!״) — מול תפילתו המיוחדת של כל נביא שהנביא מוחמד שמר אותה כשפאעה להצלת אומתו ביום הדין (בוח׳ארי 6304).

وَيُكَلِّمُ ٱلرَّبُّ مُوسَى وَجْهًا لِوَجْهٍ، كَمَا يُكَلِّمُ ٱلرَّجُلُ صَاحِبَهُ. وَإِذَا رَجَعَ مُوسَى إِلَى ٱلْمَحَلَّةِ كَانَ خَادِمُهُ يَشُوعُ بْنُ نُونَ ٱلْغُلَامُ، لَا يَبْرَحُ مِنْ دَاخِلِ ٱلْخَيْمَةِ.

בוח׳ארי 3408: ״אל תעדיפוני על משה״. ✦ מוסלם 2373: (אבו הֻרַיְרַה סיפר:) בעוד יהודי מציע סחורה שלו, והוצע לו בה מחיר שלא מצא חן בעיניו, אמר: ״לא, חי אשר בחר במשה על פני בני האדם!״ שמעו אדם מן האנצאר וסטר על פניו, ואמר: ״אתה אומר ׳חי אשר בחר במשה על פני בני האדם׳, ושליח אללה בינינו?!״ הלך היהודי אל שליח אללה ואמר: ״אבו אלקאסם, יש לי חסות וברית, ופלוני סטר על פניי״. אמר שליח אללה (לאנצארי): ״מדוע סטרת על פניו?״ אמר: ״שליח אללה, (הוא אמר:) ׳חי אשר בחר במשה על פני בני האדם׳ – ואתה בינינו!״ כעס שליח אללה עד שניכר הכעס בפניו, ואחר כך אמר: ״אל תעדיפו בין נביאי אללה; כי יתקעו בשופר, ויתעלף כל אשר בשמיים ובארץ, מלבד מי שירצה אללה; אחר כך יתקעו בו פעם נוספת, ואהיה הראשון שיקום (לתחייה) – או בין הראשונים שיקומו – והנה משה אוחז בכס. ואיני יודע אם נזקפה לו התעלפותו ביום ההר (סיני), או שקם לפניי. ואיני אומר שיש מי שמעולה מיונס בן מַתַּי״.

صحيح البخاري 3408: لاَ تُخَيِّرُونِي عَلَى مُوسَى ✦ صحيح مسلم 2373: بينما يهودي يعرض سلعة له أعطي بها شيئا كرهه أو لم يرضه - شك عبد العزيز - قال لا والذي اصطفى موسى عليه السلام على البشر . قال فسمعه رجل من الأنصار فلطم وجهه - قال - تقول والذي اصطفى موسى عليه السلام على البشر ورسول الله صلى الله عليه وسلم بين أظهرنا قال فذهب اليهودي إلى رسول الله صلى الله عليه وسلم فقال يا أبا القاسم إن لي ذمة وعهدا . وقال فلان لطم وجهي . فقال رسول الله صلى الله عليه وسلم " لم لطمت وجهه " . قال . قال يا رسول الله والذي اصطفى موسى عليه السلام على البشر وأنت بين أظهرنا . قال فغضب رسول الله صلى الله عليه وسلم حتى عرف الغضب في وجهه ثم قال " لا تفضلوا بين أنبياء الله فإنه ينفخ في الصور فيصعق من في السموات ومن في الأرض إلا من شاء الله - قال - ثم ينفخ فيه أخرى فأكون أول من بعث أو في أول من بعث فإذا موسى عليه السلام آخذ بالعرش فلا أدري أحوسب بصعقته يوم الطور أو بعث قبلي ولا أقول إن أحدا أفضل من يونس بن متى عليه السلام " .

בוח׳ארי 1965: (אבו הֻרַיְרַה סיפר:) אסר שליח אללה על הצום הרצוף (בלי לאכול בלילה). אמר לו אחד מן המוסלמים: ״הרי אתה צם ברצף, שליח אללה!״ אמר: ״ומי מכם כמוני? אני לן – וריבוני מאכיל ומשקה אותי״. כשסירבו לחדול מן הצום הרצוף, צם עמם ברצף יום ועוד יום; אחר כך ראו את הירח החדש, ואמר: ״אילו התאחר – הייתי מוסיף לכם״ – כמו עונש להם, על שסירבו לחדול. ✦ מוסלם 1102: הנביא אסר על הצום הרצוף. אמרו: ״הרי אתה צם ברצף!״ אמר: ״אינני כמותכם; אני מואכל ומושקה״.

صحيح البخاري 1965: نهى رسول الله صلى الله عليه وسلم عن الوصال في الصوم فقال له رجل من المسلمين إنك تواصل يا رسول الله قال " وأيكم مثلي إني أبيت يطعمني ربي ويسقين ". فلما أبوا أن ينتهوا عن الوصال واصل بهم يوما ثم يوما، ثم رأوا الهلال، فقال " لو تأخر لزدتكم ". كالتنكيل لهم، حين أبوا أن ينتهوا. ✦ صحيح مسلم 1102: أن النبي صلى الله عليه وسلم نهى عن الوصال قالوا إنك تواصل . قال " إني لست كهيئتكم إني أطعم وأسقى " .

צום ארבעים היום והלילה של משה על הר סיני ללא לחם ומים — מול תיאור צום ההתמסרות המיוחד של הנביא (בוח׳ארי 1965: ״אינכם כמותי – אני לן וריבוני מאכילני ומשקני״) תוך איסורו על המאמינים להחמיר בכך על עצמם!

صوم موسى المتواصل أربعين يومًا وليلة في الجبل يوازي خصوصية «الوصال» للنبي ﷺ («إني أبيت يطعمني ربي ويسقيني»، البخاري 1965) مع النهي عن التشديد على عامة الناس!

34:29וַיְהִי בְּרֶדֶת מֹשֶׁה מֵהַר סִינַי וּשְׁנֵי לֻחֹת הָעֵדֻת בְּיַד־מֹשֶׁה בְּרִדְתּוֹ מִן־הָהָר וּמֹשֶׁה לֹא־יָדַע כִּי קָרַן עוֹר פָּנָיו בְּדַבְּרוֹ אִתּוֹ׃
وَكَانَ لَمَّا نَزَلَ مُوسَى مِنْ جَبَلِ سِينَاءَ وَلَوْحَا ٱلشَّهَادَةِ فِي يَدِ مُوسَى، عِنْدَ نُزُولِهِ مِنَ ٱلْجَبَلِ، أَنَّ مُوسَى لَمْ يَعْلَمْ أَنَّ جِلْدَ وَجْهِهِ صَارَ يَلْمَعُ فِي كَلَامِهِ مَعَهُ.

שמות לד, כט–ל ⇄ בוח׳ארי 136; מוסלם 246 (אור הפנים מאור ההתגלות והטהרה!)

בוח׳ארי 136: (נֻעַיְם אלמֻג׳ְמִר סיפר:) עליתי עם אבו הֻרַיְרַה על גג המסגד; הוא נטל ידיים ואמר: שמעתי את הנביא אומר: ״אומתי תיקרא ביום התקומה בפנים ובגפיים זוהרים מעקבות נטילת הידיים; מי מכם שיכול להאריך את זוהרו – יעשה זאת״. ✦ מוסלם 246: (נֻעַיְם בן עבד אללה אלמֻג׳ְמִר סיפר:) ראיתי את אבו הֻרַיְרַה נוטל ידיים: רחץ את פניו והשלים את הנטילה; אחר כך רחץ את ידו הימנית עד שהגיע אל הזרוע, ואחר כך את ידו השמאלית עד שהגיע אל הזרוע; אחר כך העביר יד על ראשו; אחר כך רחץ את רגלו הימנית עד שהגיע אל השוק, ואחר כך את רגלו השמאלית עד שהגיע אל השוק. אחר כך אמר: ״כך ראיתי את שליח אללה נוטל ידיים״. ואמר: אמר שליח אללה: ״אתם בעלי הפנים והגפיים הזוהרים ביום התקומה, מהשלמת נטילת הידיים; מי מכם שיכול – יאריך את זוהר פניו וגפיו״.

صحيح البخاري 136: رقيت مع أبي هريرة على ظهر المسجد، فتوضأ فقال إني سمعت النبي صلى الله عليه وسلم يقول " إن أمتي يدعون يوم القيامة غرا محجلين من آثار الوضوء، فمن استطاع منكم أن يطيل غرته فليفعل ". ✦ صحيح مسلم 246: رأيت أبا هريرة يتوضأ فغسل وجهه فأسبغ الوضوء ثم غسل يده اليمنى حتى أشرع في العضد ثم يده اليسرى حتى أشرع في العضد ثم مسح رأسه ثم غسل رجله اليمنى حتى أشرع في الساق ثم غسل رجله اليسرى حتى أشرع في الساق ثم قال هكذا رأيت رسول الله صلى الله عليه وسلم يتوضأ . وقال قال رسول الله صلى الله عليه وسلم " أنتم الغر المحجلون يوم القيامة من إسباغ الوضوء فمن استطاع منكم فليطل غرته وتحجيله " .

״כי קרן עור פניו בדברו איתו״ — קרינת אור הפנים מן הקרבה לאל, בדומה לאור הפנים והגפיים של המאמינים ביום הדין (ע׳ֻרַּן מֻחַגַּ׳לין, בוח׳ארי 136) וזוהר המלאכים.

إشراق وجه موسى بنور المناجاة الإلهية يوازي إشراق وجوه المؤمنين «غُرًّا محجّلين» من أثر الطهارة والعبادة (البخاري 136).

34:30וַיַּרְא אַהֲרֹן וְכׇל־בְּנֵי יִשְׂרָאֵל אֶת־מֹשֶׁה וְהִנֵּה קָרַן עוֹר פָּנָיו וַיִּירְאוּ מִגֶּשֶׁת אֵלָיו׃
فَنَظَرَ هَارُونُ وَجَمِيعُ بَنِي إِسْرَائِيلَ مُوسَى وَإِذَا جِلْدُ وَجْهِهِ يَلْمَعُ، فَخَافُوا أَنْ يَقْتَرِبُوا إِلَيْهِ.
34:31וַיִּקְרָא אֲלֵהֶם מֹשֶׁה וַיָּשֻׁבוּ אֵלָיו אַהֲרֹן וְכׇל־הַנְּשִׂאִים בָּעֵדָה וַיְדַבֵּר מֹשֶׁה אֲלֵהֶם׃
فَدَعَاهُمْ مُوسَى. فَرَجَعَ إِلَيْهِ هَارُونُ وَجَمِيعُ ٱلرُّؤَسَاءِ فِي ٱلْجَمَاعَةِ، فَكَلَّمَهُمْ مُوسَى.
34:32וְאַחֲרֵי־כֵן נִגְּשׁוּ כׇּל־בְּנֵי יִשְׂרָאֵל וַיְצַוֵּם אֵת כׇּל־אֲשֶׁר דִּבֶּר יְהֹוָה אִתּוֹ בְּהַר סִינָי׃
وَبَعْدَ ذَلِكَ ٱقْتَرَبَ جَمِيعُ بَنِي إِسْرَائِيلَ، فَأَوْصَاهُمْ بِكُلِّ مَا تَكَلَّمَ بِهِ ٱلرَّبُّ مَعَهُ فِي جَبَلِ سِينَاءَ.
34:33וַיְכַל מֹשֶׁה מִדַּבֵּר אִתָּם וַיִּתֵּן עַל־פָּנָיו מַסְוֶה׃
وَلَمَّا فَرَغَ مُوسَى مِنَ ٱلْكَلَامِ مَعَهُمْ، جَعَلَ عَلَى وَجْهِهِ بُرْقُعًا.
34:34וּבְבֹא מֹשֶׁה לִפְנֵי יְהֹוָה לְדַבֵּר אִתּוֹ יָסִיר אֶת־הַמַּסְוֶה עַד־צֵאתוֹ וְיָצָא וְדִבֶּר אֶל־בְּנֵי יִשְׂרָאֵל אֵת אֲשֶׁר יְצֻוֶּה׃
وَكَانَ مُوسَى عِنْدَ دُخُولِهِ أَمَامَ ٱلرَّبِّ لِيَتَكَلَّمَ مَعَهُ يَنْزِعُ ٱلْبُرْقُعَ حَتَّى يَخْرُجَ، ثُمَّ يَخْرُجُ وَيُكَلِّمُ بَنِي إِسْرَائِيلَ بِمَا يُوصَى.
34:35וְרָאוּ בְנֵי־יִשְׂרָאֵל אֶת־פְּנֵי מֹשֶׁה כִּי קָרַן עוֹר פְּנֵי מֹשֶׁה וְהֵשִׁיב מֹשֶׁה אֶת־הַמַּסְוֶה עַל־פָּנָיו עַד־בֹּאוֹ לְדַבֵּר אִתּוֹ׃
فَإِذَا رَأَى بَنُو إِسْرَائِيلَ وَجْهَ مُوسَى أَنَّ جِلْدَهُ يَلْمَعُ كَانَ مُوسَى يَرُدُّ ٱلْبُرْقُعَ عَلَى وَجْهِهِ حَتَّى يَدْخُلَ لِيَتَكَلَّمَ مَعَهُ.
