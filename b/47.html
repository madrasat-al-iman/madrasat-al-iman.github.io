11:27אֶת־הַבְּרָכָה אֲשֶׁר תִּשְׁמְעוּ אֶל־מִצְוֺת יְהֹוָה אֱלֹהֵיכֶם אֲשֶׁר אָנֹכִי מְצַוֶּה אֶתְכֶם הַיּוֹם׃
ٱلْبَرَكَةُ إِذَا سَمِعْتُمْ لِوَصَايَا ٱلرَّبِّ إِلَهِكُمُ ٱلَّتِي أَنَا أُوصِيكُمْ بِهَا ٱلْيَوْمَ.
11:28וְהַקְּלָלָה אִם־לֹא תִשְׁמְעוּ אֶל־מִצְוֺת יְהֹוָה אֱלֹהֵיכֶם וְסַרְתֶּם מִן־הַדֶּרֶךְ אֲשֶׁר אָנֹכִי מְצַוֶּה אֶתְכֶם הַיּוֹם לָלֶכֶת אַחֲרֵי אֱלֹהִים אֲחֵרִים אֲשֶׁר לֹא־יְדַעְתֶּם׃
وَٱللَّعْنَةُ إِذَا لَمْ تَسْمَعُوا لِوَصَايَا ٱلرَّبِّ إِلَهِكُمْ، وَزُغْتُمْ عَنِ ٱلطَّرِيقِ ٱلَّتِي أَنَا أُوصِيكُمْ بِهَا ٱلْيَوْمَ لِتَذْهَبُوا وَرَاءَ آلِهَةٍ أُخْرَى لَمْ تَعْرِفُوهَا.
11:29וְהָיָה כִּי יְבִיאֲךָ יְהֹוָה אֱלֹהֶיךָ אֶל־הָאָרֶץ אֲשֶׁר־אַתָּה בָא־שָׁמָּה לְרִשְׁתָּהּ וְנָתַתָּה אֶת־הַבְּרָכָה עַל־הַר גְּרִזִים וְאֶת־הַקְּלָלָה עַל־הַר עֵיבָל׃
وَإِذَا جَاءَ بِكَ ٱلرَّبُّ إِلَهُكَ إِلَى ٱلْأَرْضِ ٱلَّتِي أَنْتَ دَاخِلٌ إِلَيْهَا لِكِيْ تَمْتَلِكَهَا، فَٱجْعَلِ ٱلْبَرَكَةَ عَلَى جَبَلِ جِرِزِّيمَ، وَٱللَّعْنَةَ عَلَى جَبَلِ عِيبَالَ.
11:30הֲלֹא־הֵמָּה בְּעֵבֶר הַיַּרְדֵּן אַחֲרֵי דֶּרֶךְ מְבוֹא הַשֶּׁמֶשׁ בְּאֶרֶץ הַכְּנַעֲנִי הַיֹּשֵׁב בָּעֲרָבָה מוּל הַגִּלְגָּל אֵצֶל אֵלוֹנֵי מֹרֶה׃
أَمَا هُمَا فِي عَبْرِ ٱلْأُرْدُنِّ، وَرَاءَ طَرِيقِ غُرُوبِ ٱلشَّمْسِ فِي أَرْضِ ٱلْكَنْعَانِيِّينَ ٱلسَّاكِنِينَ فِي ٱلْعَرَبَةِ، مُقَابِلَ ٱلْجِلْجَالِ، بِجَانِبِ بَلُّوطَاتِ مُورَةَ؟
11:31כִּי אַתֶּם עֹבְרִים אֶת־הַיַּרְדֵּן לָבֹא לָרֶשֶׁת אֶת־הָאָרֶץ אֲשֶׁר־יְהֹוָה אֱלֹהֵיכֶם נֹתֵן לָכֶם וִירִשְׁתֶּם אֹתָהּ וִישַׁבְתֶּם־בָּהּ׃
لِأَنَّكُمْ عَابِرُونَ ٱلْأُرْدُنَّ لِتَدْخُلُوا وَتَمْتَلِكُوا ٱلْأَرْضَ ٱلَّتِي ٱلرَّبُّ إِلَهُكُمْ يُعْطِيكُمْ. تَمْتَلِكُونَهَا وَتَسْكُنُونَهَا.
11:32וּשְׁמַרְתֶּם לַעֲשׂוֹת אֵת כׇּל־הַחֻקִּים וְאֶת־הַמִּשְׁפָּטִים אֲשֶׁר אָנֹכִי נֹתֵן לִפְנֵיכֶם הַיּוֹם׃
فَٱحْفَظُوا جَمِيعَ ٱلْفَرَائِضِ وَٱلْأَحْكَامِ ٱلَّتِي أَنَا وَاضِعٌ أَمَامَكُمُ ٱلْيَوْمَ لِتَعْمَلُوهَا.
12:1אֵלֶּה הַחֻקִּים וְהַמִּשְׁפָּטִים אֲשֶׁר תִּשְׁמְרוּן לַעֲשׂוֹת בָּאָרֶץ אֲשֶׁר נָתַן יְהֹוָה אֱלֹהֵי אֲבֹתֶיךָ לְךָ לְרִשְׁתָּהּ כׇּל־הַיָּמִים אֲשֶׁר־אַתֶּם חַיִּים עַל־הָאֲדָמָה׃
«هَذِهِ هِيَ ٱلْفَرَائِضُ وَٱلْأَحْكَامُ ٱلَّتِي تَحْفَظُونَ لِتَعْمَلُوهَا فِي ٱلْأَرْضِ ٱلَّتِي أَعْطَاكَ ٱلرَّبُّ إِلَهُ آبَائِكَ لِتَمْتَلِكَهَا، كُلَّ ٱلْأَيَّامِ ٱلَّتِي تَحْيَوْنَ عَلَى ٱلْأَرْضِ:
12:2אַבֵּד תְּאַבְּדוּן אֶת־כׇּל־הַמְּקֹמוֹת אֲשֶׁר עָבְדוּ־שָׁם הַגּוֹיִם אֲשֶׁר אַתֶּם יֹרְשִׁים אֹתָם אֶת־אֱלֹהֵיהֶם עַל־הֶהָרִים הָרָמִים וְעַל־הַגְּבָעוֹת וְתַחַת כׇּל־עֵץ רַעֲנָן׃
تُخْرِبُونَ جَمِيعَ ٱلْأَمَاكِنِ حَيْثُ عَبَدَتِ ٱلْأُمَمُ ٱلَّتِي تَرِثُونَهَا آلِهَتَهَا عَلَى ٱلْجِبَالِ ٱلشَّامِخَةِ، وَعَلَى ٱلتِّلَالِ، وَتَحْتَ كُلِّ شَجَرَةٍ خَضْرَاءَ.
12:3וְנִתַּצְתֶּם אֶת־מִזְבְּחֹתָם וְשִׁבַּרְתֶּם אֶת־מַצֵּבֹתָם וַאֲשֵׁרֵיהֶם תִּשְׂרְפוּן בָּאֵשׁ וּפְסִילֵי אֱלֹהֵיהֶם תְּגַדֵּעוּן וְאִבַּדְתֶּם אֶת־שְׁמָם מִן־הַמָּקוֹם הַהוּא׃
وَتَهْدِمُونَ مَذَابِحَهُمْ، وَتُكَسِّرُونَ أَنْصَابَهُمْ، وَتُحْرِقُونَ سَوَارِيَهُمْ بِٱلنَّارِ، وَتُقَطِّعُونَ تَمَاثِيلَ آلِهَتِهِمْ، وَتَمْحُونَ ٱسْمَهُمْ مِنْ ذَلِكَ ٱلْمَكَانِ.
12:4לֹא־תַעֲשׂוּן כֵּן לַיהֹוָה אֱלֹהֵיכֶם׃
لَا تَفْعَلُوا هَكَذَا لِلرَّبِّ إِلَهِكُمْ.
12:5כִּי אִם־אֶל־הַמָּקוֹם אֲשֶׁר־יִבְחַר יְהֹוָה אֱלֹהֵיכֶם מִכׇּל־שִׁבְטֵיכֶם לָשׂוּם אֶת־שְׁמוֹ שָׁם לְשִׁכְנוֹ תִדְרְשׁוּ וּבָאתָ שָּׁמָּה׃
بَلِ ٱلْمَكَانُ ٱلَّذِي يَخْتَارُهُ ٱلرَّبُّ إِلَهُكُمْ مِنْ جَمِيعِ أَسْبَاطِكُمْ لِيَضَعَ ٱسْمَهُ فِيهِ، سُكْنَاهُ تَطْلُبُونَ وَإِلَى هُنَاكَ تَأْتُونَ،

דברים יב, ה ⇄ 2:124

וכאשר ניסה ריבונו את אברהם בדברים, וקיימם, אמר: ״הנני עושה אותך מנהיג לבני האדם״. אמר: ״וגם מזרעי?״ אמר: ״בריתי לא תגיע לעושי העוול״. (124)

۞وَإِذِ ٱبۡتَلَىٰٓ إِبۡرَٰهِـۧمَ رَبُّهُۥ بِكَلِمَٰتٖ فَأَتَمَّهُنَّۖ قَالَ إِنِّي جَاعِلُكَ لِلنَّاسِ إِمَامٗاۖ قَالَ وَمِن ذُرِّيَّتِيۖ قَالَ لَا يَنَالُ عَهۡدِي ٱلظَّـٰلِمِينَ (124)

״המקום אשר יבחר״ מול הבית ומקום אברהם (2:124–125).

«المكان الذي يختاره» مقابل البيت ومقام إبراهيم (2:124–125).

12:6וַהֲבֵאתֶם שָׁמָּה עֹלֹתֵיכֶם וְזִבְחֵיכֶם וְאֵת מַעְשְׂרֹתֵיכֶם וְאֵת תְּרוּמַת יֶדְכֶם וְנִדְרֵיכֶם וְנִדְבֹתֵיכֶם וּבְכֹרֹת בְּקַרְכֶם וְצֹאנְכֶם׃
وَتُقَدِّمُونَ إِلَى هُنَاكَ: مُحْرَقَاتِكُمْ وَذَبَائِحَكُمْ وَعُشُورَكُمْ وَرَفَائِعَ أَيْدِيكُمْ وَنُذُورَكُمْ وَنَوَافِلَكُمْ وَأَبْكَارَ بَقَرِكُمْ وَغَنَمِكُمْ،
12:7וַאֲכַלְתֶּם־שָׁם לִפְנֵי יְהֹוָה אֱלֹהֵיכֶם וּשְׂמַחְתֶּם בְּכֹל מִשְׁלַח יֶדְכֶם אַתֶּם וּבָתֵּיכֶם אֲשֶׁר בֵּרַכְךָ יְהֹוָה אֱלֹהֶיךָ׃
وَتَأْكُلُونَ هُنَاكَ أَمَامَ ٱلرَّبِّ إِلَهِكُمْ، وَتَفْرَحُونَ بِكُلِّ مَا تَمْتَدُّ إِلَيْهِ أَيْدِيكُمْ أَنْتُمْ وَبُيُوتُكُمْ كَمَا بَارَكَكُمُ ٱلرَّبُّ إِلَهُكُمْ.

דברים יב, ז ⇄ 2:58–59

וכאשר אמרנו: ״היכנסו אל העיר הזאת ואכלו ממנה בשפע מכל אשר תרצו, והיכנסו בשער בהשתחוויה, ואמרו ׳חִטָּה׳ (מחילה), ונסלח לכם על חטאיכם, ונוסיף למיטיבים״. (58) והחליפו עושי העוול דיבור אחר מזה שנאמר להם, והורדנו על עושי העוול פגע מן השמים, על אשר פשעו. (59)

وَإِذۡ قُلۡنَا ٱدۡخُلُواْ هَٰذِهِ ٱلۡقَرۡيَةَ فَكُلُواْ مِنۡهَا حَيۡثُ شِئۡتُمۡ رَغَدٗا وَٱدۡخُلُواْ ٱلۡبَابَ سُجَّدٗا وَقُولُواْ حِطَّةٞ نَّغۡفِرۡ لَكُمۡ خَطَٰيَٰكُمۡۚ وَسَنَزِيدُ ٱلۡمُحۡسِنِينَ (58) فَبَدَّلَ ٱلَّذِينَ ظَلَمُواْ قَوۡلًا غَيۡرَ ٱلَّذِي قِيلَ لَهُمۡ فَأَنزَلۡنَا عَلَى ٱلَّذِينَ ظَلَمُواْ رِجۡزٗا مِّنَ ٱلسَّمَآءِ بِمَا كَانُواْ يَفۡسُقُونَ (59)

12:8לֹא תַעֲשׂוּן כְּכֹל אֲשֶׁר אֲנַחְנוּ עֹשִׂים פֹּה הַיּוֹם אִישׁ כׇּל־הַיָּשָׁר בְּעֵינָיו׃
«لَا تَعْمَلُوا حَسَبَ كُلِّ مَا نَحْنُ عَامِلُونَ هُنَا ٱلْيَوْمَ، أَيْ كُلُّ إِنْسَانٍ مَهْمَا صَلَحَ فِي عَيْنَيْهِ.
12:9כִּי לֹא־בָאתֶם עַד־עָתָּה אֶל־הַמְּנוּחָה וְאֶל־הַנַּחֲלָה אֲשֶׁר־יְהֹוָה אֱלֹהֶיךָ נֹתֵן לָךְ׃
لِأَنَّكُمْ لَمْ تَدْخُلُوا حَتَّى ٱلْآنَ إِلَى ٱلْمَقَرِّ وَٱلنَّصِيبِ ٱللَّذَيْنِ يُعْطِيكُمُ ٱلرَّبُّ إِلَهُكُمْ.

בוח׳ארי 3609: מפי הנביא: ״לא תקום השעה עד שיילחמו שתי קבוצות, ויהיה ביניהן הרג גדול, וקריאתן (טענתן) אחת; ולא תקום השעה עד שיישלחו דַג׳ַּאלִים שקרנים, כשלושים, כל אחד מהם טוען שהוא שליח אללה״. ✦ בוח׳ארי 7121: שליח אללה אמר: ״לא תקום השעה עד שיילחמו שתי קבוצות גדולות, ויהיה ביניהן הרג גדול, וקריאתן אחת; ועד שיישלחו דַג׳ַּאלִים שקרנים, כשלושים, כל אחד מהם טוען שהוא שליח אללה; ועד שתיאסף הדעת, וירבו רעידות האדמה, ויתקצר הזמן, ויתגלו המהומות, וירבה ההרג (אלהַרְג׳ – הוא ההריגה); ועד שירבה בכם הממון ויגלוש, עד שבעל הממון יהיה טרוד במי שיקבל את צדקתו, ועד שיציענה והאדם שהוצעה לו יאמר: ׳אין לי בה צורך׳; ועד שיתחרו בני האדם בבנייה לגובה; ועד שיעבור אדם ליד קבר אדם ויאמר: ׳הלוואי הייתי במקומו׳; ועד שתזרח השמש ממערבה – וכאשר תזרח ויראו אותה בני האדם, יאמינו כולם; וזו העת שבה ׳לא תועיל לנפש אמונתה אם לא האמינה קודם לכן או לא רכשה טוב באמונתה׳ (קוראן 6:158). ותקום השעה כששני אנשים פרשו את בגדם ביניהם – ולא יספיקו למכרו ולא לקפלו; ותקום השעה כשאדם יצא עם חלב נאקתו – ולא יספיק לטעום ממנו; ותקום השעה כשהוא מטייח את שוקתו – ולא יספיק להשקות בה; ותקום השעה כשהרים את פתו אל פיו – ולא יספיק לאכלה״.

صحيح البخاري 3609: عن النبي صلى الله عليه وسلم قال " لا تقوم الساعة حتى يقتتل فئتان، فيكون بينهما مقتلة عظيمة، دعواهما واحدة، ولا تقوم الساعة حتى يبعث دجالون كذابون قريبا من ثلاثين، كلهم يزعم أنه رسول الله ". ✦ صحيح البخاري 7121: أن رسول الله صلى الله عليه وسلم قال " لا تقوم الساعة حتى تقتتل فئتان عظيمتان، يكون بينهما مقتلة عظيمة، دعوتهما واحدة، وحتى يبعث دجالون كذابون، قريب من ثلاثين، كلهم يزعم أنه رسول الله، وحتى يقبض العلم، وتكثر الزلازل، ويتقارب الزمان، وتظهر الفتن، ويكثر الهرج وهو القتل، وحتى يكثر فيكم المال فيفيض، حتى يهم رب المال من يقبل صدقته، وحتى يعرضه فيقول الذي يعرضه عليه لا أرب لي به. وحتى يتطاول الناس في البنيان، وحتى يمر الرجل بقبر الرجل فيقول يا ليتني مكانه. وحتى تطلع الشمس من مغربها، فإذا طلعت ورآها الناس يعني آمنوا أجمعون، فذلك حين لا ينفع نفسا إيمانها لم تكن آمنت من قبل، أو كسبت في إيمانها خيرا، ولتقومن الساعة وقد نشر الرجلان ثوبهما بينهما، فلا يتبايعانه ولا يطويانه، ولتقومن الساعة وقد انصرف الرجل بلبن لقحته فلا يطعمه، ولتقومن الساعة وهو يليط حوضه فلا يسقي فيه، ولتقومن الساعة وقد رفع أكلته إلى فيه فلا يطعمها ".

صحيح البخاري 1429: حَدَّثَنَا أَبُو النُّعْمَانِ، قَالَ حَدَّثَنَا حَمَّادُ بْنُ زَيْدٍ، عَنْ أَيُّوبَ، عَنْ نَافِعٍ، عَنِ ابْنِ عُمَرَ ـ رضى الله عنهما ـ قَالَ سَمِعْتُ النَّبِيَّ صلى الله عليه وسلم ح‏.‏ وَحَدَّثَنَا عَبْدُ اللَّهِ بْنُ مَسْلَمَةَ، عَنْ مَالِكٍ، عَنْ نَافِعٍ، عَنْ عَبْدِ اللَّهِ بْنِ عُمَرَ ـ رضى الله عنهما ـ أَنَّ رَسُولَ اللَّهِ صلى الله عليه وسلم قَالَ وَهُوَ عَلَى الْمِنْبَرِ، وَذَكَرَ الصَّدَقَةَ وَالتَّعَفُّفَ وَالْمَسْأَلَةَ ‏ "‏ الْيَدُ الْعُلْيَا خَيْرٌ مِنَ الْيَدِ السُّفْلَى، فَالْيَدُ الْعُلْيَا هِيَ الْمُنْفِقَةُ، وَالسُّفْلَى هِيَ السَّائِلَةُ ‏"‏‏.‏ ✦ صحيح البخاري 1417: اتَّقُوا النَّارَ وَلَوْ بِشِقِّ تَمْرَةٍ

״כי לא יחדל אביון מקרב הארץ על כן אנוכי מצווך לאמור פתוח תפתח את ידך״ — מציאות העוני כמבחן מתמיד של ערבות הדדית ונדיבות (בוח׳ארי 1429: ״היד העליונה הנותנת טובה מן התחתונה המקבלת״).

استمرار وجود المحتاجين كاختبار دائم للتكافل («تفتح يدك لأخيك المسكين») مقابل «اليد العليا خير من اليد السفلى» (البخاري 1429).

דברים טו, יג–יד ⇄ בוח׳ארי 2517

בוח׳ארי 2517: מפי אבו הֻרַיְרַה: אמר הנביא: ״כל אדם ששחרר אדם מוסלמי – אללה יציל בכל איבר ממנו איבר ממנו מן האש״. אמר סעיד אבן מַרְג׳ַאנַה: הלכתי אל עלי בן חֻסַיְן (וסיפרתי לו), ועלי בן חסין פנה אל עבד שלו, שעבדאללה בן ג׳עפר הציע לו בעדו עשרת אלפים דרהם (או אלף דינר), ושחרר אותו.

صحيح البخاري 2517: حَدَّثَنَا أَحْمَدُ بْنُ يُونُسَ، حَدَّثَنَا عَاصِمُ بْنُ مُحَمَّدٍ، قَالَ حَدَّثَنِي وَاقِدُ بْنُ مُحَمَّدٍ، قَالَ حَدَّثَنِي سَعِيدٌ ابْنُ مَرْجَانَةَ، صَاحِبُ عَلِيِّ بْنِ حُسَيْنٍ قَالَ لِي أَبُو هُرَيْرَةَ ـ رضى الله عنه ـ قَالَ النَّبِيُّ صلى الله عليه وسلم ‏ "‏ أَيُّمَا رَجُلٍ أَعْتَقَ امْرَأً مُسْلِمًا اسْتَنْقَذَ اللَّهُ بِكُلِّ عُضْوٍ مِنْهُ عُضْوًا مِنْهُ مِنَ النَّارِ ‏"‏‏.‏ قَالَ سَعِيدٌ ابْنُ مَرْجَانَةَ فَانْطَلَقْتُ إِلَى عَلِيِّ بْنِ حُسَيْنٍ فَعَمَدَ عَلِيُّ بْنُ حُسَيْنٍ ـ رضى الله عنهما ـ إِلَى عَبْدٍ لَهُ قَدْ أَعْطَاهُ بِهِ عَبْدُ اللَّهِ بْنُ جَعْفَرٍ عَشَرَةَ آلاَفِ دِرْهَمٍ ـ أَوْ أَلْفَ دِينَارٍ ـ فَأَعْتَقَهُ‏.‏

מצוות הענקה לעבד משתחרר (״לא תשלחנו ריקם, הענק תעניק לו״ – כדי שיוכל לפתוח חיים חדשים בכבוד!) — ברוח עידוד המֻכַּאתַבַּה והמתנה למשתחרר באסלאם (24:33; בוח׳ארי 2517).

إعطاء العبد المحرَّر مالاً لبدء حياة كريمة («لا تطلقه فارغًا بل زوّده بسخاء») يوازي «وآتوهم من مال الله الذي آتاكم» (النور 24:33) والبخاري 2517.

لَا يَحِلُّ لَكَ أَنْ تَذْبَحَ ٱلْفِصْحَ فِي أَحَدِ أَبْوَابِكَ ٱلَّتِي يُعْطِيكَ ٱلرَّبُّ إِلَهُكَ،
16:6כִּי אִם־אֶל־הַמָּקוֹם אֲשֶׁר־יִבְחַר יְהֹוָה אֱלֹהֶיךָ לְשַׁכֵּן שְׁמוֹ שָׁם תִּזְבַּח אֶת־הַפֶּסַח בָּעָרֶב כְּבוֹא הַשֶּׁמֶשׁ מוֹעֵד צֵאתְךָ מִמִּצְרָיִם׃
بَلْ فِي ٱلْمَكَانِ ٱلَّذِي يَخْتَارُهُ ٱلرَّبُّ إِلَهُكَ لِيُحِلَّ ٱسْمَهُ فِيهِ. هُنَاكَ تَذْبَحُ ٱلْفِصْحَ مَسَاءً نَحْوَ غُرُوبِ ٱلشَّمْسِ فِي مِيعَادِ خُرُوجِكَ مِنْ مِصْرَ،
16:7וּבִשַּׁלְתָּ וְאָכַלְתָּ בַּמָּקוֹם אֲשֶׁר יִבְחַר יְהֹוָה אֱלֹהֶיךָ בּוֹ וּפָנִיתָ בַבֹּקֶר וְהָלַכְתָּ לְאֹהָלֶיךָ׃
وَتَطْبُخُ وَتَأْكُلُ فِي ٱلْمَكَانِ ٱلَّذِي يَخْتَارُهُ ٱلرَّبُّ إِلَهُكَ، ثُمَّ تَنْصَرِفُ فِي ٱلْغَدِ وَتَذْهَبُ إِلَى خِيَامِكَ.
16:8שֵׁשֶׁת יָמִים תֹּאכַל מַצּוֹת וּבַיּוֹם הַשְּׁבִיעִי עֲצֶרֶת לַיהֹוָה אֱלֹהֶיךָ לֹא תַעֲשֶׂה מְלָאכָה׃
سِتَّةَ أَيَّامٍ تَأْكُلُ فَطِيرًا، وَفِي ٱلْيَوْمِ ٱلسَّابِعِ ٱعْتِكَافٌ لِلرَّبِّ إِلَهِكَ. لَا تَعْمَلْ فِيهِ عَمَلًا.
16:9שִׁבְעָה שָׁבֻעֹת תִּסְפׇּר־לָךְ מֵהָחֵל חֶרְמֵשׁ בַּקָּמָה תָּחֵל לִסְפֹּר שִׁבְעָה שָׁבֻעוֹת׃
«سَبْعَةَ أَسَابِيعَ تَحْسُبُ لَكَ. مِنِ ٱبْتِدَاءِ ٱلْمِنْجَلِ فِي ٱلزَّرْعِ، تَبْتَدِئُ أَنْ تَحْسُبَ سَبْعَةَ أَسَابِيعَ.
16:10וְעָשִׂיתָ חַג שָׁבֻעוֹת לַיהֹוָה אֱלֹהֶיךָ מִסַּת נִדְבַת יָדְךָ אֲשֶׁר תִּתֵּן כַּאֲשֶׁר יְבָרֶכְךָ יְהֹוָה אֱלֹהֶיךָ׃
وَتَعْمَلُ عِيدَ أَسَابِيعَ لِلرَّبِّ إِلَهِكَ عَلَى قَدْرِ مَا تَسْمَحُ يَدُكَ أَنْ تُعْطِيَ، كَمَا يُبَارِكُكَ ٱلرَّبُّ إِلَهُكَ.
16:11וְשָׂמַחְתָּ לִפְנֵי יְהֹוָה אֱלֹהֶיךָ אַתָּה וּבִנְךָ וּבִתֶּךָ וְעַבְדְּךָ וַאֲמָתֶךָ וְהַלֵּוִי אֲשֶׁר בִּשְׁעָרֶיךָ וְהַגֵּר וְהַיָּתוֹם וְהָאַלְמָנָה אֲשֶׁר בְּקִרְבֶּךָ בַּמָּקוֹם אֲשֶׁר יִבְחַר יְהֹוָה אֱלֹהֶיךָ לְשַׁכֵּן שְׁמוֹ שָׁם׃
وَتَفْرَحُ أَمَامَ ٱلرَّبِّ إِلَهِكَ أَنْتَ وَٱبْنُكَ وَٱبْنَتُكَ وَعَبْدُكَ وَأَمَتُكَ وَٱللَّاوِيُّ ٱلَّذِي فِي أَبْوَابِكَ، وَٱلْغَرِيبُ وَٱلْيَتِيمُ وَٱلْأَرْمَلَةُ ٱلَّذِينَ فِي وَسَطِكَ فِي ٱلْمَكَانِ ٱلَّذِي يَخْتَارُهُ ٱلرَّبُّ إِلَهُكَ لِيُحِلَّ ٱسْمَهُ فِيهِ.

דברים טז, יא, יד ⇄ בוח׳ארי 1503; אבו דאוד 1609

בוח׳ארי 1503: ״צאע תמרים או צאע שעורים – על העבד ועל החופשי, על הזכר ועל הנקבה, על הקטן ועל הגדול״. ✦ אבו דאוד 1609: מפי אבן עבאס: שליח אללה הטיל את זכאת הפִטְר כטהרה לצם מדברי הבל ומניבול פה, וכמזון לעניים; מי שנתן אותה לפני התפילה – זו זכאת מקובלת, ומי שנתן אותה אחרי התפילה – זו צדקה מן הצדקות.

صحيح البخاري 1503: صَاعًا مِنْ تَمْرٍ أَوْ صَاعًا مِنْ شَعِيرٍ عَلَى الْعَبْدِ وَالْحُرِّ وَالذَّكَرِ وَالأُنْثَى وَالصَّغِيرِ وَالْكَبِيرِ ✦ سنن أبي داود 1609: حَدَّثَنَا مَحْمُودُ بْنُ خَالِدٍ الدِّمَشْقِيُّ، وَعَبْدُ اللَّهِ بْنُ عَبْدِ الرَّحْمَنِ السَّمَرْقَنْدِيُّ، قَالاَ حَدَّثَنَا مَرْوَانُ، قَالَ عَبْدُ اللَّهِ حَدَّثَنَا أَبُو يَزِيدَ الْخَوْلاَنِيُّ، - وَكَانَ شَيْخَ صِدْقٍ وَكَانَ ابْنُ وَهْبٍ يَرْوِي عَنْهُ - حَدَّثَنَا سَيَّارُ بْنُ عَبْدِ الرَّحْمَنِ، - قَالَ مَحْمُودٌ الصَّدَفِيُّ - عَنْ عِكْرِمَةَ، عَنِ ابْنِ عَبَّاسٍ، قَالَ فَرَضَ رَسُولُ اللَّهِ صلى الله عليه وسلم زَكَاةَ الْفِطْرِ طُهْرَةً لِلصَّائِمِ مِنَ اللَّغْوِ وَالرَّفَثِ وَطُعْمَةً لِلْمَسَاكِينِ مَنْ أَدَّاهَا قَبْلَ الصَّلاَةِ فَهِيَ زَكَاةٌ مَقْبُولَةٌ وَمَنْ أَدَّاهَا بَعْدَ الصَّلاَةِ فَهِيَ صَدَقَةٌ مِنَ الصَّدَقَاتِ ‏.‏

״ושמחת בחגך אתה ובנך ובתך... והלוי והגר והיתום והאלמנה״ — אין שמחת חג בלי דאגה ליתום, לאלמנה ולגר, כמטרת זכאת אלפטר (״טעמה ללמסאכין – מזון לעניים״, אבו דאוד 1609).

لا تكتمل بهجة العيد إلا بإشراك اليتيم والأرملة والغريب («وتفرح في عيدك... والغريب واليتيم والأرملة»)، كما في زكاة الفطر «طعمة للمساكين» (أبو داود 1609).

16:12וְזָכַרְתָּ כִּי־עֶבֶד הָיִיתָ בְּמִצְרָיִם וְשָׁמַרְתָּ וְעָשִׂיתָ אֶת־הַחֻקִּים הָאֵלֶּה׃
وَتَذْكُرُ أَنَّكَ كُنْتَ عَبْدًا فِي مِصْرَ وَتَحْفَظُ، وَتَعْمَلُ هَذِهِ ٱلْفَرَائِضَ.
16:13חַג הַסֻּכֹּת תַּעֲשֶׂה לְךָ שִׁבְעַת יָמִים בְּאׇסְפְּךָ מִגׇּרְנְךָ וּמִיִּקְבֶךָ׃
«تَعْمَلُ لِنَفْسِكَ عِيدَ ٱلْمَظَالِّ سَبْعَةَ أَيَّامٍ عِنْدَمَا تَجْمَعُ مِنْ بَيْدَرِكَ وَمِنْ مِعْصَرَتِكَ.
16:14וְשָׂמַחְתָּ בְּחַגֶּךָ אַתָּה וּבִנְךָ וּבִתֶּךָ וְעַבְדְּךָ וַאֲמָתֶךָ וְהַלֵּוִי וְהַגֵּר וְהַיָּתוֹם וְהָאַלְמָנָה אֲשֶׁר בִּשְׁעָרֶיךָ׃
وَتَفْرَحُ فِي عِيدِكَ أَنْتَ وَٱبْنُكَ وَٱبْنَتُكَ وَعَبْدُكَ وَأَمَتُكَ وَاللَّاوِيُّ وَٱلْغَرِيبُ وَٱلْيَتِيمُ وَٱلْأَرْمَلَةُ ٱلَّذِينَ فِي أَبْوَابِكَ.
16:15שִׁבְעַת יָמִים תָּחֹג לַיהֹוָה אֱלֹהֶיךָ בַּמָּקוֹם אֲשֶׁר־יִבְחַר יְהֹוָה כִּי יְבָרֶכְךָ יְהֹוָה אֱלֹהֶיךָ בְּכֹל תְּבוּאָתְךָ וּבְכֹל מַעֲשֵׂה יָדֶיךָ וְהָיִיתָ אַךְ שָׂמֵחַ׃
سَبْعَةَ أَيَّامٍ تُعَيِّدُ لِلرَّبِّ إِلَهِكَ فِي ٱلْمَكَانِ ٱلَّذِي يَخْتَارُهُ ٱلرَّبُّ، لِأَنَّ ٱلرَّبَّ إِلَهَكَ يُبَارِكُكَ فِي كُلِّ مَحْصُولِكَ وَفِي كُلِّ عَمَلِ يَدَيْكَ، فَلَا تَكُونُ إِلَّا فَرِحًا.
16:16שָׁלוֹשׁ פְּעָמִים בַּשָּׁנָה יֵרָאֶה כׇל־זְכוּרְךָ אֶת־פְּנֵי יְהֹוָה אֱלֹהֶיךָ בַּמָּקוֹם אֲשֶׁר יִבְחָר בְּחַג הַמַּצּוֹת וּבְחַג הַשָּׁבֻעוֹת וּבְחַג הַסֻּכּוֹת וְלֹא יֵרָאֶה אֶת־פְּנֵי יְהֹוָה רֵיקָם׃
«ثَلَاثَ مَرَّاتٍ فِي ٱلسَّنَةِ يَحْضُرُ جَمِيعُ ذُكُورِكَ أَمَامَ ٱلرَّبِّ إِلَهِكَ فِي ٱلْمَكَانِ ٱلَّذِي يَخْتَارُهُ، فِي عِيدِ ٱلْفَطِيرِ وَعِيدِ ٱلْأَسَابِيعِ وَعِيدِ ٱلْمَظَالِّ. وَلَا يَحْضُرُوا أَمَامَ ٱلرَّبِّ فَارِغِينَ.

דברים טז, טז–יז ⇄ בוח׳ארי 1519; מוסלם 1349

בוח׳ארי 1519: נשאל הנביא: ״איזה מן המעשים הוא המעולה?״ אמר: ״אמונה באללה ובשליחו״. נאמר: ״ואחר כך מה?״ אמר: ״ג׳יהאד בדרך אללה״. נאמר: ״ואחר כך מה?״ אמר: ״עלייה לרגל מקובלת (חג׳ מַבְּרוּר)״. ✦ מוסלם 1349: שליח אללה אמר: ״העֻמְרַה (העלייה הקטנה) עד העֻמְרַה הבאה – כפרה למה שביניהן; והחג׳ המקובל – אין לו גמול מלבד גן העדן״.

صحيح البخاري 1519: سئل النبي صلى الله عليه وسلم أى الأعمال أفضل قال " إيمان بالله ورسوله ". قيل ثم ماذا قال " جهاد في سبيل الله ". قيل ثم ماذا قال " حج مبرور ". ✦ صحيح مسلم 1349: أن رسول الله صلى الله عليه وسلم قال " العمرة إلى العمرة كفارة لما بينهما والحج المبرور ليس له جزاء إلا الجنة " .

עלייה לרגל בשלוש רגלים אל המקום המקודש והבאת מתנה ״כמתנת ידו כברכת ה׳״ — מקביל לשבח אלחג׳ אלמבּרוּר והבאת ההַדְי לפי יכולת העולה לרגל (בוח׳ארי 1519).

الحج إلى البيت المقدس وتقديم الهدي بحسب الاستطاعة يوازي فضل «الحج المبرور» في البخاري 1519.
