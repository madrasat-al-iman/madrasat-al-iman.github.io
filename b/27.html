ויקרא יב, א–ב ⇄ קוראן 31:14 (״אמו נשאה אותו חולשה על חולשה״)

31:14: וציווינו את האדם על הוריו – אמו נשאה אותו חולשה על חולשה, וגמילתו בשנתיים – ״הודה לי ולהוריך; אליי השיבה. (14)

وَوَصَّيۡنَا ٱلۡإِنسَٰنَ بِوَٰلِدَيۡهِ حَمَلَتۡهُ أُمُّهُۥ وَهۡنًا عَلَىٰ وَهۡنٖ وَفِصَٰلُهُۥ فِي عَامَيۡنِ أَنِ ٱشۡكُرۡ لِي وَلِوَٰلِدَيۡكَ إِلَيَّ ٱلۡمَصِيرُ (14)

״אשה כי תזריע וילדה״ – פרשת היולדת. בקוראן: ״אמו נשאה אותו חולשה על חולשה, וגמילתו בשנתיים – הודה לי ולהוריך״.

أحكام الوالدة، مقابل «حملته أمه وهنًا على وهن وفصاله في عامين».

12:2דַּבֵּר אֶל־בְּנֵי יִשְׂרָאֵל לֵאמֹר אִשָּׁה כִּי תַזְרִיעַ וְיָלְדָה זָכָר וְטָמְאָה שִׁבְעַת יָמִים כִּימֵי נִדַּת דְּוֺתָהּ תִּטְמָא׃
«كَلِّمْ بَنِي إِسْرَائِيلَ قَائِلًا: إِذَا حَبِلَتِ ٱمْرَأَةٌ وَوَلَدَتْ ذَكَرًا، تَكُونُ نَجِسَةً سَبْعَةَ أَيَّامٍ. كَمَا فِي أَيَّامِ طَمْثِ عِلَّتِهَا تَكُونُ نَجِسَةً.

ויקרא יב, ב–ה ⇄ אבו דאוד 311

אמרה אם סלמה: בימי שליח אללה הייתה היולדת יושבת אחרי לידתה ארבעים יום, או ארבעים לילה; והיינו מורחות על פנינו וַרְס (צמח צבע) – כלומר נגד כתמי הפנים.

كانت النفساء على عهد رسول الله صلى الله عليه وسلم تقعد بعد نفاسها أربعين يوما أو أربعين ليلة وكنا نطلي على وجوهنا الورس يعني من الكلف

היולדת יושבת ארבעים יום – כמו ״שבעת ימים… ושלשים יום ושלשת ימים״ ליולדת זכר (40 יום).

النفساء تقعد أربعين يوما – مثل «سبعة أيام… ثم ثلاثة وثلاثين يوما» لمن ولدت ذكرا (أربعون يوما).

ויקרא יב, ב ⇄ קוראן 3:36 (״ריבוני, ילדתי בת… ואין הזכר כנקבה״)

3:36: וכאשר ילדה אותה, אמרה: ״ריבוני, ילדתי בת״ – ואללה ידע את אשר ילדה, ואין הזכר כנקבה – ״וקראתי את שמה מרים, ואני מבקשת לה ולזרעה את מחסך מפני השטן המגורש״. (36)

فَلَمَّا وَضَعَتۡهَا قَالَتۡ رَبِّ إِنِّي وَضَعۡتُهَآ أُنثَىٰ وَٱللَّهُ أَعۡلَمُ بِمَا وَضَعَتۡ وَلَيۡسَ ٱلذَّكَرُ كَٱلۡأُنثَىٰۖ وَإِنِّي سَمَّيۡتُهَا مَرۡيَمَ وَإِنِّيٓ أُعِيذُهَا بِكَ وَذُرِّيَّتَهَا مِنَ ٱلشَّيۡطَٰنِ ٱلرَّجِيمِ (36)

״וילדה זכר… ואם נקבה תלד״ – הבחנה בין לידת זכר לנקבה. בקוראן אשת עמרן ילדה את מרים ואמרה: ״ריבוני, ילדתי בת… ואין הזכר כנקבה״ – והאל ״קיבלה בקבלה טובה״.

التمييز بين ولادة الذكر والأنثى، مقابل «رب إني وضعتها أنثى… وليس الذكر كالأنثى».

ויקרא יב, ב ⇄ קוראן 23:12–14 (״ברא את האדם מתמצית טיט… טיפה… קריש…״)

23:12–14: ובראנו את האדם מתמצית של טיט, (12) אחר כך עשינוהו טיפה במקום מבטחים, (13) אחר כך בראנו את הטיפה קריש דם, ובראנו את הקריש גוש בשר, ובראנו את הגוש עצמות, והלבשנו את העצמות בשר; אחר כך הקמנוהו בריאה אחרת. יתברך אללה, הטוב שביוצרים! (14)

وَلَقَدۡ خَلَقۡنَا ٱلۡإِنسَٰنَ مِن سُلَٰلَةٖ مِّن طِينٖ (12) ثُمَّ جَعَلۡنَٰهُ نُطۡفَةٗ فِي قَرَارٖ مَّكِينٖ (13) ثُمَّ خَلَقۡنَا ٱلنُّطۡفَةَ عَلَقَةٗ فَخَلَقۡنَا ٱلۡعَلَقَةَ مُضۡغَةٗ فَخَلَقۡنَا ٱلۡمُضۡغَةَ عِظَٰمٗا فَكَسَوۡنَا ٱلۡعِظَٰمَ لَحۡمٗا ثُمَّ أَنشَأۡنَٰهُ خَلۡقًا ءَاخَرَۚ فَتَبَارَكَ ٱللَّهُ أَحۡسَنُ ٱلۡخَٰلِقِينَ (14)

״אשה כי תזריע וילדה״ – פלא הלידה (״הוא יצרך בבטן״). בקוראן שלבי התפתחות העובר: ״עשינו אותו טיפה במקום מבטחים, ואחר כך את הטיפה קריש…״.

«إذا حبلت امرأة وولدت»، مقابل أطوار الخلق: «ثم جعلناه نطفة في قرار مكين ثم خلقنا النطفة علقة».

ויקרא יב, ב–ה ⇄ תרמד׳י 139; אבן מאג׳ה 648

תרמד׳י 139: היולדות היו יושבות (בטומאתן) בימי שליח אללה ארבעים יום, והיינו מורחות את פנינו בצמח הוַרְס מפני הכתמים. ✦ אבן מאג׳ה 648: (היולדת הייתה, בימי) שליח אללה, יושבת ארבעים יום, והיינו מורחות את פנינו בצמח הוַרְס מפני הכתמים.

سنن الترمذي 139: كانت النفساء تجلس على عهد رسول الله صلى الله عليه وسلم أربعين يوما فكنا نطلي وجوهنا بالورس من الكلف ✦ سنن ابن ماجه 648: رسول الله صلى الله عليه وسلم تجلس أربعين يوما وكنا نطلي وجوهنا بالورس من الكلف

פרק זמן טהרת היולדת (7+33 = 40 יום ליולדת זכר בתורה!) — זהה במדויק לשיעור ימי הנִפַאס המקובל בפקה האסלאמי (40 יום מן הלידה, תרמד׳י 139 – מלבד אבו דאוד 311 שכבר במאגר)!

مدة النفاس في التوراة لوالدة الذكر (7 + 33 = 40 يومًا) تطابق تمامًا المدة القصوى للنفاس في الفقه الإسلامي (أربعون يومًا، الترمذي 139).

ויקרא יב, ב ⇄ 31:14; 46:15

31:14: וציווינו את האדם על הוריו – אמו נשאה אותו חולשה על חולשה, וגמילתו בשנתיים – ״הודה לי ולהוריך; אליי השיבה. (14) ✦ 46:15: וציווינו את האדם להיטיב להוריו; אמו נשאה אותו בקושי וילדה אותו בקושי, והריונו וגמילתו – שלושים חודש. עד שכאשר הגיע לבגרותו והגיע לגיל ארבעים – אמר: ״ריבוני, השרה בי שאודה על חסדך אשר העניקות לי ולהוריי, ושאעשה טוב שתרצה בו, ותקן לי בצאצאיי; אני שב אליך, ואני מן המוסלמים (המתמסרים)״. (15)

وَوَصَّيۡنَا ٱلۡإِنسَٰنَ بِوَٰلِدَيۡهِ حَمَلَتۡهُ أُمُّهُۥ وَهۡنًا عَلَىٰ وَهۡنٖ وَفِصَٰلُهُۥ فِي عَامَيۡنِ أَنِ ٱشۡكُرۡ لِي وَلِوَٰلِدَيۡكَ إِلَيَّ ٱلۡمَصِيرُ (14) ✦ وَوَصَّيۡنَا ٱلۡإِنسَٰنَ بِوَٰلِدَيۡهِ إِحۡسَٰنًاۖ حَمَلَتۡهُ أُمُّهُۥ كُرۡهٗا وَوَضَعَتۡهُ كُرۡهٗاۖ وَحَمۡلُهُۥ وَفِصَٰلُهُۥ ثَلَٰثُونَ شَهۡرًاۚ حَتَّىٰٓ إِذَا بَلَغَ أَشُدَّهُۥ وَبَلَغَ أَرۡبَعِينَ سَنَةٗ قَالَ رَبِّ أَوۡزِعۡنِيٓ أَنۡ أَشۡكُرَ نِعۡمَتَكَ ٱلَّتِيٓ أَنۡعَمۡتَ عَلَيَّ وَعَلَىٰ وَٰلِدَيَّ وَأَنۡ أَعۡمَلَ صَٰلِحٗا تَرۡضَىٰهُ وَأَصۡلِحۡ لِي فِي ذُرِّيَّتِيٓۖ إِنِّي تُبۡتُ إِلَيۡكَ وَإِنِّي مِنَ ٱلۡمُسۡلِمِينَ (15)

קושי ההיריון והלידה (״אישה כי תזריע וילדה״) והגנת היולדת — בקוראן 31:14 ו־46:15 מנומק כיבוד האם במפורש בסבלה: ״נשאה אותו אמו בחולשה על חולשה... וילדה אותו בקושי״.

تقدير مشقة الحمل والولادة وحماية النفساء، وهو الأساس الذي يبني عليه القرآن برّ الأم («حملته أمه كرهًا ووضعته كرهًا»، الأحقاف 46:15).

وَفِي ٱلْيَوْمِ ٱلثَّامِنِ يُخْتَنُ لَحْمُ غُرْلَتِهِ.

ויקרא יב, ג ⇄ בוח׳ארי 5891 (״הפִטְרַה – חמש: המילה…״)

בוח׳ארי 5891: (אבו הוריירה:) שמעתי את הנביא אומר: ״הפִטְרַה (מנהגי הטבע הראשוני) – חמישה: המילה, גילוח שער הערווה, קיצוץ השפם, גזיזת הציפורניים ומריטת שער בתי השחי״.

صحيح البخاري 5891: سمعت النبي صلى الله عليه وسلم يقول " الفطرة خمس الختان، والاستحداد، وقص الشارب، وتقليم الأظفار، ونتف الآباط "

״וביום השמיני ימול בשר ערלתו״ – מצוות המילה. בהלכה האסלאמית המילה (ח׳ִתַאן) היא מ״מנהגי הפִטְרַה״: ״הפטרה – חמש: המילה, גילוח הערווה, קיצוץ השפם, גזיזת הציפורניים ומריטת השחי״.

الختان في اليوم الثامن، مقابل «الفطرة خمس: الختان…».

ויקרא יב, ג ⇄ בוח׳ארי 5889; מוסלם 257

בוח׳ארי 5889: מפי אבו הֻרַיְרַה (ממסורת הנביא): ״הפִטְרַה (הטבע המקורי) – חמישה (או: חמישה מן הפטרה): המילה, גילוח שיער הערווה, מריטת שיער בית השחי, גזיזת הציפורניים וקיצוץ השפם״. ✦ מוסלם 257: מפי אבו הֻרַיְרַה: הנביא אמר: ״הפִטְרַה – חמישה (או: חמישה מן הפטרה): המילה, גילוח שיער הערווה, גזיזת הציפורניים, מריטת שיער בית השחי וקיצוץ השפם״.

صحيح البخاري 5889: حَدَّثَنَا عَلِيٌّ، حَدَّثَنَا سُفْيَانُ، قَالَ الزُّهْرِيُّ حَدَّثَنَا عَنْ سَعِيدِ بْنِ الْمُسَيَّبِ، عَنْ أَبِي هُرَيْرَةَ، رِوَايَةً ‏ "‏ الْفِطْرَةُ خَمْسٌ ـ أَوْ خَمْسٌ مِنَ الْفِطْرَةِ ـ الْخِتَانُ، وَالاِسْتِحْدَادُ، وَنَتْفُ الإِبْطِ، وَتَقْلِيمُ الأَظْفَارِ، وَقَصُّ الشَّارِبِ ‏"‏‏.‏ ✦ صحيح مسلم 257: حَدَّثَنَا أَبُو بَكْرِ بْنُ أَبِي شَيْبَةَ، وَعَمْرٌو النَّاقِدُ، وَزُهَيْرُ بْنُ حَرْبٍ، جَمِيعًا عَنْ سُفْيَانَ، - قَالَ أَبُو بَكْرٍ حَدَّثَنَا ابْنُ عُيَيْنَةَ، - عَنِ الزُّهْرِيِّ، عَنْ سَعِيدِ بْنِ الْمُسَيَّبِ، عَنْ أَبِي هُرَيْرَةَ، عَنِ النَّبِيِّ صلى الله عليه وسلم قَالَ ‏ "‏ الْفِطْرَةُ خَمْسٌ - أَوْ خَمْسٌ مِنَ الْفِطْرَةِ - الْخِتَانُ وَالاِسْتِحْدَادُ وَتَقْلِيمُ الأَظْفَارِ وَنَتْفُ الإِبْطِ وَقَصُّ الشَّارِبِ ‏"‏ ‏.‏

״וביום השמיני ימול בשר ערלתו״ — מצוות המילה לכל בן זכר, מול היות המילה אחת מחמש פעולות אלפטרה (הטבע המתוקן) בסונה (בוח׳ארי 5889).

ختان المولود في اليوم الثامن في التوراة مقابل كون الختان من خصال الفطرة الخمس في السنة (البخاري 5889).

ثُمَّ تُقِيمُ ثَلَاثَةً وَثَلَاثِينَ يَوْمًا فِي دَمِ تَطْهِيرِهَا. كُلَّ شَيْءٍ مُقَدَّسٍ لَا تَمَسَّ، وَإِلَى ٱلْمَقْدِسِ لَا تَجِئْ حَتَّى تَكْمُلَ أَيَّامُ تَطْهِيرِهَا.

ויקרא יב, ד ⇄ קוראן 56:79 (״לא ייגעו בו אלא הטהורים״)

56:79: שלא ייגעו בו אלא הטהורים, (79)

لَّا يَمَسُّهُۥٓ إِلَّا ٱلۡمُطَهَّرُونَ (79)

״בכל קדש לא תגע ואל המקדש לא תבא עד מלאת ימי טהרה״. בהלכה האסלאמית אישה בימי לידה או נידה אינה נוגעת בקוראן ואינה שוהה במסגד – על פי ״לא ייגעו בו אלא הטהורים״.

«كل شيء مقدّس لا تمسّ وإلى المقدس لا تجيء»، مقابل «لا يمسّه إلا المطهَّرون».

فَإِنْ رَأَى ٱلْكَاهِنُ بَعْدَ غَسْلِ ٱلْمَضْرُوبِ وَإِذَا ٱلضَّرْبَةُ لَمْ تُغَيِّرْ مَنْظَرَهَا، وَلَا ٱمْتَدَّتِ ٱلضَّرْبَةُ، فَهُوَ نَجِسٌ. بِٱلنَّارِ تُحْرِقُهُ. إِنَّهَا نُخْرُوبٌ فِي جُرْدَةِ بَاطِنِهِ أَوْ ظَاهِرِهِ.
13:56וְאִם רָאָה הַכֹּהֵן וְהִנֵּה כֵּהָה הַנֶּגַע אַחֲרֵי הֻכַּבֵּס אֹתוֹ וְקָרַע אֹתוֹ מִן־הַבֶּגֶד אוֹ מִן־הָעוֹר אוֹ מִן־הַשְּׁתִי אוֹ מִן־הָעֵרֶב׃
لَكِنْ إِنْ رَأَى ٱلْكَاهِنُ وَإِذَا ٱلضَّرْبَةُ كَامِدَةُ ٱللَّوْنِ بَعْدَ غَسْلِهِ، يُمَزِّقُهَا مِنَ ٱلثَّوْبِ أَوِ ٱلْجِلْدِ مِنَ ٱلسَّدَى أَوِ ٱللُّحْمَةِ.
13:57וְאִם־תֵּרָאֶה עוֹד בַּבֶּגֶד אוֹ־בַשְּׁתִי אוֹ־בָעֵרֶב אוֹ בְכׇל־כְּלִי־עוֹר פֹּרַחַת הִוא בָּאֵשׁ תִּשְׂרְפֶנּוּ אֵת אֲשֶׁר־בּוֹ הַנָּגַע׃
ثُمَّ إِنْ ظَهَرَتْ أَيْضًا فِي ٱلثَّوْبِ فِي ٱلسَّدَى أَوِ ٱللُّحْمَةِ أَوْ فِي مَتَاعِ ٱلْجِلْدِ فَهِيَ مُفْرِخَةٌ. بِٱلنَّارِ تُحْرِقُ مَا فِيهِ ٱلضَّرْبَةُ.
13:58וְהַבֶּגֶד אוֹ־הַשְּׁתִי אוֹ־הָעֵרֶב אוֹ־כׇל־כְּלִי הָעוֹר אֲשֶׁר תְּכַבֵּס וְסָר מֵהֶם הַנָּגַע וְכֻבַּס שֵׁנִית וְטָהֵר׃
وَأَمَّا ٱلثَّوْبُ، ٱلسَّدَى أَوِ ٱللُّحْمَةُ أَوْ مَتَاعُ ٱلْجِلْدِ ٱلَّذِي تَغْسِلُهُ وَتَزُولُ مِنْهُ ٱلضَّرْبَةُ، فَيُغْسَلُ ثَانِيَةً فَيَطْهُرُ.
13:59זֹאת תּוֹרַת נֶגַע־צָרַעַת בֶּגֶד הַצֶּמֶר אוֹ הַפִּשְׁתִּים אוֹ הַשְּׁתִי אוֹ הָעֵרֶב אוֹ כׇּל־כְּלִי־עוֹר לְטַהֲרוֹ אוֹ לְטַמְּאוֹ׃
«هَذِهِ شَرِيعَةُ ضَرْبَةِ ٱلْبَرَصِ فِي ٱلصُّوفِ أَوِ ٱلْكَتَّانِ، فِي ٱلسَّدَى أَوِ ٱللُّحْمَةِ أَوْ فِي كُلِّ مَتَاعٍ مِنْ جِلْدٍ، لِلْحُكْمِ بِطَهَارَتِهِ أَوْ نَجَاسَتِهِ».
