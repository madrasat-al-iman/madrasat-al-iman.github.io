35:1וַיַּקְהֵל מֹשֶׁה אֶת־כׇּל־עֲדַת בְּנֵי יִשְׂרָאֵל וַיֹּאמֶר אֲלֵהֶם אֵלֶּה הַדְּבָרִים אֲשֶׁר־צִוָּה יְהֹוָה לַעֲשֹׂת אֹתָם׃
وَجَمَعَ مُوسَى كُلَّ جَمَاعَةِ بَنِي إِسْرَائِيلَ وَقَالَ لَهُمْ: «هَذِهِ هِيَ ٱلْكَلِمَاتُ ٱلَّتِي أَمَرَ ٱلرَّبُّ أَنْ تُصْنَعَ:

שמות לה, א–ג ⇄ 16:124

השבת הוטלה רק על אלה שנחלקו בה; וריבונך ישפוט ביניהם ביום התקומה בדבר שנחלקו בו. (124)

إِنَّمَا جُعِلَ ٱلسَّبۡتُ عَلَى ٱلَّذِينَ ٱخۡتَلَفُواْ فِيهِۚ وَإِنَّ رَبَّكَ لَيَحۡكُمُ بَيۡنَهُمۡ يَوۡمَ ٱلۡقِيَٰمَةِ فِيمَا كَانُواْ فِيهِ يَخۡتَلِفُونَ (124)

השבת: ״לא תבערו אש בכל משבתיכם״ – ״השבת הוטלה רק על אלה שנחלקו בה״.

السبت: «لا تشعلوا نارا في جميع مساكنكم يوم السبت» – «إنما جُعل السبت على الذين اختلفوا فيه».

שמות לה, א ⇄ קוראן 62:9 (״כאשר יקראו לתפילה ביום ההתכנסות – מהרו לזכר אללה״)

62:9: הוי המאמינים, כאשר יקראו לתפילה ביום הקהל (יום השישי) – מהרו אל זכר אללה ועזבו את המסחר; זה טוב לכם, לו ידעתם. (9)

يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓاْ إِذَا نُودِيَ لِلصَّلَوٰةِ مِن يَوۡمِ ٱلۡجُمُعَةِ فَٱسۡعَوۡاْ إِلَىٰ ذِكۡرِ ٱللَّهِ وَذَرُواْ ٱلۡبَيۡعَۚ ذَٰلِكُمۡ خَيۡرٞ لَّكُمۡ إِن كُنتُمۡ تَعۡلَمُونَ (9)

״ויקהל משה את כל עדת בני ישראל״ – הקהלה לשם דבר ה׳. היום השישי באסלאם נקרא ״יום ההתכנסות״ (אלג׳ֻמְעַה): ״כאשר יקראו לתפילה ביום ההתכנסות – מהרו לזכר אללה והניחו את המסחר״.

«وجمع موسى كل جماعة بني إسرائيل»، مقابل «إذا نودي للصلاة من يوم الجمعة فاسعوا إلى ذكر الله».

שמות לה, א ⇄ 62:9; 24:62

62:9: הוי המאמינים, כאשר יקראו לתפילה ביום הקהל (יום השישי) – מהרו אל זכר אללה ועזבו את המסחר; זה טוב לכם, לו ידעתם. (9) ✦ 24:62: המאמינים הם רק אלה שהאמינו באללה ובשליחו, וכאשר הם עמו בעניין משותף – אינם הולכים עד שיבקשו ממנו רשות. אלה המבקשים ממך רשות – הם המאמינים באללה ובשליחו. וכאשר יבקשו ממך רשות לצורך עניין שלהם – תן רשות למי שתרצה מהם, ובקש להם מחילה מאללה; אללה סולח ורחום. (62)

يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓاْ إِذَا نُودِيَ لِلصَّلَوٰةِ مِن يَوۡمِ ٱلۡجُمُعَةِ فَٱسۡعَوۡاْ إِلَىٰ ذِكۡرِ ٱللَّهِ وَذَرُواْ ٱلۡبَيۡعَۚ ذَٰلِكُمۡ خَيۡرٞ لَّكُمۡ إِن كُنتُمۡ تَعۡلَمُونَ (9) ✦ إِنَّمَا ٱلۡمُؤۡمِنُونَ ٱلَّذِينَ ءَامَنُواْ بِٱللَّهِ وَرَسُولِهِۦ وَإِذَا كَانُواْ مَعَهُۥ عَلَىٰٓ أَمۡرٖ جَامِعٖ لَّمۡ يَذۡهَبُواْ حَتَّىٰ يَسۡتَـٔۡذِنُوهُۚ إِنَّ ٱلَّذِينَ يَسۡتَـٔۡذِنُونَكَ أُوْلَـٰٓئِكَ ٱلَّذِينَ يُؤۡمِنُونَ بِٱللَّهِ وَرَسُولِهِۦۚ فَإِذَا ٱسۡتَـٔۡذَنُوكَ لِبَعۡضِ شَأۡنِهِمۡ فَأۡذَن لِّمَن شِئۡتَ مِنۡهُمۡ وَٱسۡتَغۡفِرۡ لَهُمُ ٱللَّهَۚ إِنَّ ٱللَّهَ غَفُورٞ رَّحِيمٞ (62)

״ויקהל משה את כל עדת בני ישראל״ — כינוס הקהילה לשמיעת דברי תורה וציוויי הכלל, ככינוס האומה לשמיעת דרשת יום השישי (ח׳טבת אלג׳ומעה, 62:9).

جمع موسى لكافة الجماعة لسماع التوجيهات العامة يوازي اجتماع الجماعة للذكر والخطبة في سورة الجمعة 62:9 والنور 24:62.

שמות לה, א, כא ⇄ 7:153; 3:103

7:153: ואלה שעשו רעות ואחר כך שבו אחריהן והאמינו – ריבונך אחריהן סולח, רחום. (153) ✦ 3:103: והיאחזו כולכם בחבל אללה, ואל תתפלגו; וזכרו את חסד אללה עליכם, כאשר הייתם אויבים והוא חיבר בין לבבותיכם, והייתם בחסדו לאחים; והייתם על שפת בור של אש, והוא הצילכם ממנו. כך מבהיר לכם אללה את אותותיו, למען תודרכו. (103)

وَٱلَّذِينَ عَمِلُواْ ٱلسَّيِّـَٔاتِ ثُمَّ تَابُواْ مِنۢ بَعۡدِهَا وَءَامَنُوٓاْ إِنَّ رَبَّكَ مِنۢ بَعۡدِهَا لَغَفُورٞ رَّحِيمٞ (153) ✦ وَٱعۡتَصِمُواْ بِحَبۡلِ ٱللَّهِ جَمِيعٗا وَلَا تَفَرَّقُواْۚ وَٱذۡكُرُواْ نِعۡمَتَ ٱللَّهِ عَلَيۡكُمۡ إِذۡ كُنتُمۡ أَعۡدَآءٗ فَأَلَّفَ بَيۡنَ قُلُوبِكُمۡ فَأَصۡبَحۡتُم بِنِعۡمَتِهِۦٓ إِخۡوَٰنٗا وَكُنتُمۡ عَلَىٰ شَفَا حُفۡرَةٖ مِّنَ ٱلنَّارِ فَأَنقَذَكُم مِّنۡهَاۗ كَذَٰلِكَ يُبَيِّنُ ٱللَّهُ لَكُمۡ ءَايَٰتِهِۦ لَعَلَّكُمۡ تَهۡتَدُونَ (103)

מעבר מחטא העגל אל התגייסות מלאה בבניית המשכן — כוחה של תשובה קהילתית המיתרגמת למעשה בנייה מאחד (7:153: ״ואלה שעשו רעות ואחר כך שבו והאמינו – ריבונך אחריה סולח ורחום״).

النهوض بعد فتنة العجل بتوبة جماعية تتحول إلى بناء وتوحيد للصفوف (الأعراف 7:153 وآل عمران 3:103).

35:2שֵׁשֶׁת יָמִים תֵּעָשֶׂה מְלָאכָה וּבַיּוֹם הַשְּׁבִיעִי יִהְיֶה לָכֶם קֹדֶשׁ שַׁבַּת שַׁבָּתוֹן לַיהֹוָה כׇּל־הָעֹשֶׂה בוֹ מְלָאכָה יוּמָת׃
سِتَّةَ أَيَّامٍ يُعْمَلُ عَمَلٌ، وَأَمَّا ٱلْيَوْمُ ٱلْسَّابِعُ فَفِيهِ يَكُونُ لَكُمْ سَبْتُ عُطْلَةٍ مُقَدَّسٌ لِلرَّبِّ. كُلُّ مَنْ يَعْمَلُ فِيهِ عَمَلًا يُقْتَلُ.

שמות לה, ב–ג ⇄ קוראן 7:163 (״עדת הדייגים… כשהם עוברים על השבת״); בוח׳ארי 876 (״אנו האחרונים והראשונים״)

7:163: ושאל אותם על העיר אשר הייתה על חוף הים, כאשר עברו על השבת: כאשר באו אליהם דגיהם ביום שבתם, צפים על פני המים, וביום שאינם שובתים לא באו אליהם. כך ניסינו אותם, על אשר פשעו. (163) ✦ בוח׳ארי 876: ״אנו האחרונים (בזמן) והראשונים (ביום הדין)״ – ובהמשך: היום השישי, השבת ויום ראשון.

وَسۡـَٔلۡهُمۡ عَنِ ٱلۡقَرۡيَةِ ٱلَّتِي كَانَتۡ حَاضِرَةَ ٱلۡبَحۡرِ إِذۡ يَعۡدُونَ فِي ٱلسَّبۡتِ إِذۡ تَأۡتِيهِمۡ حِيتَانُهُمۡ يَوۡمَ سَبۡتِهِمۡ شُرَّعٗا وَيَوۡمَ لَا يَسۡبِتُونَ لَا تَأۡتِيهِمۡۚ كَذَٰلِكَ نَبۡلُوهُم بِمَا كَانُواْ يَفۡسُقُونَ (163) ✦ صحيح البخاري 876: نَحْنُ الآخِرُونَ السَّابِقُونَ

״ששת ימים תעשה מלאכה וביום השביעי יהיה לכם קדש… לא תבערו אש״. בקוראן סיפור העיר שעברה על השבת (7:163); ובחדית׳ השבת והיום השישי: ״זה היום שנכתב עליהם (על אנשי הספר) ונחלקו בו, ואללה הנחה אותנו אליו… היהודים מחר והנוצרים מחרתיים״.

أحكام السبت ومنع إيقاد النار، مقابل أصحاب السبت و«هذا يومهم الذي فُرض عليهم فاختلفوا فيه… اليهود غدًا والنصارى بعد غد».

שמות לה, ב ⇄ קוראן 2:65 (״וכבר ידעתם את אלה מכם אשר עברו על השבת״)

2:65: והלא ידעתם את אלה מכם אשר עברו על השבת, ואמרנו להם: ״היו קופים נבזים״. (65)

وَلَقَدۡ عَلِمۡتُمُ ٱلَّذِينَ ٱعۡتَدَوۡاْ مِنكُمۡ فِي ٱلسَّبۡتِ فَقُلۡنَا لَهُمۡ كُونُواْ قِرَدَةً خَٰسِـِٔينَ (65)

״כל העשה בו מלאכה יומת״ – חומרת השבת. בקוראן סיפור עוברי השבת שנענשו: ״היו לקופים נבזים״.

«كل من يعمل فيه عملًا يُقتل»، مقابل «ولقد علمتم الذين اعتدوا منكم في السبت».

62:10: וכאשר תסתיים התפילה – התפזרו בארץ ובקשו מחסד אללה, וזכרו את אללה הרבה, למען תצליחו. (10)

فَإِذَا قُضِيَتِ ٱلصَّلَوٰةُ فَٱنتَشِرُواْ فِي ٱلۡأَرۡضِ وَٱبۡتَغُواْ مِن فَضۡلِ ٱللَّهِ وَٱذۡكُرُواْ ٱللَّهَ كَثِيرٗا لَّعَلَّكُمۡ تُفۡلِحُونَ (10)

״לא תבערו אש בכל משבתיכם ביום השבת״ – שביתה מוחלטת ממלאכה. ביום השישי באסלאם האיסור על מסחר הוא רק בשעת התפילה: ״וכאשר תסתיים התפילה – התפזרו בארץ ובקשו מחסד אללה״. שאלה לדיון: יום מנוחה מול יום התכנסות.

«لا تشعلوا نارًا في جميع مساكنكم يوم السبت»، مقابل الجمعة: «فإذا قُضيت الصلاة فانتشروا في الأرض».

2:274: אלה הנותנים את ממונם בלילה וביום, בסתר ובגלוי – שכרם אצל ריבונם, ואין עליהם פחד, ולא יתעצבו. (274)

ٱلَّذِينَ يُنفِقُونَ أَمۡوَٰلَهُم بِٱلَّيۡلِ وَٱلنَّهَارِ سِرّٗا وَعَلَانِيَةٗ فَلَهُمۡ أَجۡرُهُمۡ عِندَ رَبِّهِمۡ وَلَا خَوۡفٌ عَلَيۡهِمۡ وَلَا هُمۡ يَحۡزَنُونَ (274)

״קחו מאתכם תרומה לה׳ כל נדיב לבו יביאה״. בקוראן: ״הנותנים את ממונם בלילה וביום, בסתר ובגלוי – שכרם אצל ריבונם״.

«كل من قلبه سموح فليأت بها تقدمة للرب»، مقابل «الذين ينفقون أموالهم بالليل والنهار سرًّا وعلانية».

38:15וְלַכָּתֵף הַשֵּׁנִית מִזֶּה וּמִזֶּה לְשַׁעַר הֶחָצֵר קְלָעִים חֲמֵשׁ עֶשְׂרֵה אַמָּה עַמֻּדֵיהֶם שְׁלֹשָׁה וְאַדְנֵיהֶם שְׁלֹשָׁה׃
وَلِلْجَانِبِ ٱلثَّانِي مِنْ بَابِ ٱلدَّارِ إِلَى هُنَا وَإِلَى هُنَا أَسْتَارٌ خَمْسَ عَشَرَةَ ذِرَاعًا، أَعْمِدَتُهَا ثَلَاثَةٌ وَقَوَاعِدُهَا ثَلَاثٌ.
38:16כׇּל־קַלְעֵי הֶחָצֵר סָבִיב שֵׁשׁ מׇשְׁזָר׃
جَمِيعُ أَسْتَارِ ٱلدَّارِ حَوَالَيْهَا مِنْ بُوصٍ مَبْرُومٍ،
38:17וְהָאֲדָנִים לָעַמֻּדִים נְחֹשֶׁת וָוֵי הָעַמּוּדִים וַחֲשׁוּקֵיהֶם כֶּסֶף וְצִפּוּי רָאשֵׁיהֶם כָּסֶף וְהֵם מְחֻשָּׁקִים כֶּסֶף כֹּל עַמֻּדֵי הֶחָצֵר׃
وَقَوَاعِدُ ٱلْأَعْمِدَةِ مِنْ نُحَاسٍ. رُزَزُ ٱلْأَعْمِدَةِ وَقُضْبَانُهَا مِنْ فِضَّةٍ وَتَغْشِيَةُ رُؤُوسِهَا مِنْ فِضَّةٍ وَجَمِيعُ أَعْمِدَةِ ٱلدَّارِ مَوْصُولَةٌ بِقُضْبَانٍ مِنْ فِضَّةٍ.
38:18וּמָסַךְ שַׁעַר הֶחָצֵר מַעֲשֵׂה רֹקֵם תְּכֵלֶת וְאַרְגָּמָן וְתוֹלַעַת שָׁנִי וְשֵׁשׁ מׇשְׁזָר וְעֶשְׂרִים אַמָּה אֹרֶךְ וְקוֹמָה בְרֹחַב חָמֵשׁ אַמּוֹת לְעֻמַּת קַלְעֵי הֶחָצֵר׃
وَسَجْفُ بَابِ ٱلدَّارِ صَنْعَةَ ٱلطَّرَّازِ مِنْ أَسْمَانْجُونِيٍّ وَأُرْجُوَانٍ وَقِرْمِزٍ وَبُوصٍ مَبْرُومٍ، وَطُولُهُ عِشْرُونَ ذِرَاعًا، وَٱرْتِفَاعُهُ بِٱلْعَرْضِ خَمْسُ أَذْرُعٍ بِسَوِيَّةِ أَسْتَارِ ٱلدَّارِ،
38:19וְעַמֻּדֵיהֶם אַרְבָּעָה וְאַדְנֵיהֶם אַרְבָּעָה נְחֹשֶׁת וָוֵיהֶם כֶּסֶף וְצִפּוּי רָאשֵׁיהֶם וַחֲשֻׁקֵיהֶם כָּסֶף׃
وَأَعْمِدَتُهَا أَرْبَعَةٌ، وَقَوَاعِدُهَا أَرْبَعٌ مِنْ نُحَاسٍ. رُزَزُهَا مِنْ فِضَّةٍ، وَتَغْشِيَةُ رُؤُوسِهَا وَقُضْبَانِهَا مِنْ فِضَّةٍ.
38:20וְכׇל־הַיְתֵדֹת לַמִּשְׁכָּן וְלֶחָצֵר סָבִיב נְחֹשֶׁת׃
وَجَمِيعُ أَوْتَادِ ٱلْمَسْكَنِ وَٱلدَّارِ حَوَالَيْهَا مِنْ نُحَاسٍ.
