ויקרא יט, ג ⇄ 31:14–15

וציווינו את האדם על הוריו – אמו נשאה אותו חולשה על חולשה, וגמילתו בשנתיים – ״הודה לי ולהוריך; אליי השיבה. (14) ואם יתאמצו להביאך לשתף עמי את אשר אין לך בו דעת – אל תשמע להם, אבל התלווה אליהם בעולם הזה כראוי, ולך בדרך מי ששב אליי. אחר כך אליי שיבתכם, ואגיד לכם את אשר עשיתם״. (15)

وَوَصَّيۡنَا ٱلۡإِنسَٰنَ بِوَٰلِدَيۡهِ حَمَلَتۡهُ أُمُّهُۥ وَهۡنًا عَلَىٰ وَهۡنٖ وَفِصَٰلُهُۥ فِي عَامَيۡنِ أَنِ ٱشۡكُرۡ لِي وَلِوَٰلِدَيۡكَ إِلَيَّ ٱلۡمَصِيرُ (14) وَإِن جَٰهَدَاكَ عَلَىٰٓ أَن تُشۡرِكَ بِي مَا لَيۡسَ لَكَ بِهِۦ عِلۡمٞ فَلَا تُطِعۡهُمَاۖ وَصَاحِبۡهُمَا فِي ٱلدُّنۡيَا مَعۡرُوفٗاۖ وَٱتَّبِعۡ سَبِيلَ مَنۡ أَنَابَ إِلَيَّۚ ثُمَّ إِلَيَّ مَرۡجِعُكُمۡ فَأُنَبِّئُكُم بِمَا كُنتُمۡ تَعۡمَلُونَ (15)

כבד את הוריך – אבל אל תשמע להם אם יאלצוך לשתף.

أكرم والديك – ولكن لا تطعهما إن جاهداك على أن تشرك.

ויקרא יט, ג, ל ⇄ 22:32; 62:9

22:32: זאת; ומי שמכבד את סמלי אללה (שַׁעַאאִר אללה) – הרי זה מיראת הלבבות. (32) ✦ 62:9: הוי המאמינים, כאשר יקראו לתפילה ביום הקהל (יום השישי) – מהרו אל זכר אללה ועזבו את המסחר; זה טוב לכם, לו ידעתם. (9)

ذَٰلِكَۖ وَمَن يُعَظِّمۡ شَعَـٰٓئِرَ ٱللَّهِ فَإِنَّهَا مِن تَقۡوَى ٱلۡقُلُوبِ (32) ✦ يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓاْ إِذَا نُودِيَ لِلصَّلَوٰةِ مِن يَوۡمِ ٱلۡجُمُعَةِ فَٱسۡعَوۡاْ إِلَىٰ ذِكۡرِ ٱللَّهِ وَذَرُواْ ٱلۡبَيۡعَۚ ذَٰلِكُمۡ خَيۡرٞ لَّكُمۡ إِن كُنتُمۡ تَعۡلَمُونَ (9)

״את שבתותיי תשמורו ומקדשי תיראו״ — חיבור כבוד יום הקודש עם יראת כבוד כלפי בית התפילה וסמלי האל (22:32; 62:9).

الجمع بين حرمة يوم العبادة وتعظيم بيت الله وشعائره (الحج 22:32 والجمعة 62:9).

ויקרא יט, ד ⇄ 22:30–31; 16:36

22:30–31: כך הוא. ומי שמכבד את קודשי אללה – טוב לו אצל ריבונו. והותרו לכם הבהמות, מלבד מה שנקרא לפניכם. התרחקו מטומאת האלילים, והתרחקו מדבר שקר. (30) חניפים לאללה, שאינם משתפים לו; ומי שמשתף לאללה – כאילו נפל מן השמיים, ותחטוף אותו הציפור, או תשליך אותו הרוח אל מקום רחוק. (31) ✦ 16:36: וכבר שלחנו בכל אומה שליח: ״עבדו את אללה והתרחקו מן הטאע׳וּת (האלילים)״. ויש מהם שאללה הדריך, ויש מהם שנגזרה עליו התעייה. לכו אפוא בארץ וראו מה הייתה אחרית המכחישים. (36)

ذَٰلِكَۖ وَمَن يُعَظِّمۡ حُرُمَٰتِ ٱللَّهِ فَهُوَ خَيۡرٞ لَّهُۥ عِندَ رَبِّهِۦۗ وَأُحِلَّتۡ لَكُمُ ٱلۡأَنۡعَٰمُ إِلَّا مَا يُتۡلَىٰ عَلَيۡكُمۡۖ فَٱجۡتَنِبُواْ ٱلرِّجۡسَ مِنَ ٱلۡأَوۡثَٰنِ وَٱجۡتَنِبُواْ قَوۡلَ ٱلزُّورِ (30) حُنَفَآءَ لِلَّهِ غَيۡرَ مُشۡرِكِينَ بِهِۦۚ وَمَن يُشۡرِكۡ بِٱللَّهِ فَكَأَنَّمَا خَرَّ مِنَ ٱلسَّمَآءِ فَتَخۡطَفُهُ ٱلطَّيۡرُ أَوۡ تَهۡوِي بِهِ ٱلرِّيحُ فِي مَكَانٖ سَحِيقٖ (31) ✦ وَلَقَدۡ بَعَثۡنَا فِي كُلِّ أُمَّةٖ رَّسُولًا أَنِ ٱعۡبُدُواْ ٱللَّهَ وَٱجۡتَنِبُواْ ٱلطَّـٰغُوتَۖ فَمِنۡهُم مَّنۡ هَدَى ٱللَّهُ وَمِنۡهُم مَّنۡ حَقَّتۡ عَلَيۡهِ ٱلضَّلَٰلَةُۚ فَسِيرُواْ فِي ٱلۡأَرۡضِ فَٱنظُرُواْ كَيۡفَ كَانَ عَٰقِبَةُ ٱلۡمُكَذِّبِينَ (36)

״אל תפנו אל האלילים ואלוהי מסכה לא תעשו לכם״ — קריאת הייחוד הטהורה ושלילת האלילים (״התרחקו מטומאת האלילים״, 22:30).

«لا تلتفتوا إلى الأوثان وآلهة مسبوكة لا تصنعوا لأنفسكم» يطابق «فاجتنبوا الرجس من الأوثان» (الحج 22:30).

ויקרא יט, ט–י ⇄ 6:141

והוא אשר הצמיח גנים מודלים ושאינם מודלים, ואת הדקל ואת התבואה – שונים בפרים – ואת הזית ואת הרימון, דומים ושאינם דומים. אכלו מפריו כאשר יניב, ותנו את חלקו (המגיע) ביום קצירו; ואל תבזבזו, כי הוא אינו אוהב את המבזבזים. (141)

۞وَهُوَ ٱلَّذِيٓ أَنشَأَ جَنَّـٰتٖ مَّعۡرُوشَٰتٖ وَغَيۡرَ مَعۡرُوشَٰتٖ وَٱلنَّخۡلَ وَٱلزَّرۡعَ مُخۡتَلِفًا أُكُلُهُۥ وَٱلزَّيۡتُونَ وَٱلرُّمَّانَ مُتَشَٰبِهٗا وَغَيۡرَ مُتَشَٰبِهٖۚ كُلُواْ مِن ثَمَرِهِۦٓ إِذَآ أَثۡمَرَ وَءَاتُواْ حَقَّهُۥ يَوۡمَ حَصَادِهِۦۖ وَلَا تُسۡرِفُوٓاْۚ إِنَّهُۥ لَا يُحِبُّ ٱلۡمُسۡرِفِينَ (141)

״ותנו את חקו ביום קצירו״ – פאה, לקט ושכחה.

«وآتوا حقه يوم حصاده» – زاوية الحقل واللقاط والمنسيّ.

אל תעמיד עם אללה אלוה אחר, פן תשב מגונה ועזוב. (22) וגזר ריבונך שלא תעבדו אלא אותו, ולהורים תיטיבו. אם יגיע אחד מהם או שניהם אצלך לזקנה – אל תאמר להם ״אוף״ ואל תגער בהם, ודבר אליהם דברים נכבדים. (23) והשפל לפניהם את כנף הענווה מתוך רחמים, ואמור: ״ריבוני, רחם עליהם כפי שגידלוני בקטנותי״. (24) ריבונכם יודע היטב את אשר בנפשכם. אם תהיו ישרים – הוא סולח לשבים אליו. (25) ותן לקרוב את המגיע לו, וכן לעני ולהולך בדרך, ואל תבזבז בזבוז. (26) המבזבזים הם אחי השטנים, והשטן כפוי טובה לריבונו. (27) ואם תתרחק מהם, בבקשך רחמים מאת ריבונך שאתה מקווה להם – דבר אליהם דברים רכים. (28) אל תעשה את ידך כבולה אל צווארך, ואל תפשטנה עד תום, פן תשב מגונה ומרושש. (29) ריבונך מרחיב את הפרנסה למי שירצה ומצמצם; הוא יודע את עבדיו ורואה אותם. (30) ואל תהרגו את ילדיכם מפחד עוני; אנחנו נפרנס אותם ואתכם. הריגתם חטא גדול. (31) ואל תקרבו אל הניאוף; תועבה הוא ודרך רעה. (32) ואל תהרגו את הנפש אשר אסר אללה, אלא כדין. ומי שנהרג בעוול – נתנו לגואלו סמכות; אך אל יפריז בהריגה, כי הוא נעזר. (33) ואל תקרבו אל הון היתום אלא בדרך הטובה ביותר, עד אשר יגיע לבגרותו. וקיימו את הברית, כי על הברית תישאלו. (34) ומלאו את המידה כאשר תמודו, ושקלו במאזניים ישרים; זה טוב יותר ומיטיב באחריתו. (35) ואל תלך אחר דבר שאין לך בו דעת; השמיעה, הראייה והלב – על כל אלה ישאלו. (36) ואל תתהלך בארץ ביהירות; לא תבקע את הארץ ולא תגיע לגובה ההרים. (37) רעתם של כל אלה שנואה בעיני ריבונך. (38) זה מן החכמה אשר השרה אליך ריבונך. ואל תעמיד עם אללה אלוה אחר, פן תושלך אל גיהינום מגונה ונדחה. (39)

20:4וְאִם הַעְלֵם יַעְלִימוּ עַם הָאָרֶץ אֶת־עֵינֵיהֶם מִן־הָאִישׁ הַהוּא בְּתִתּוֹ מִזַּרְעוֹ לַמֹּלֶךְ לְבִלְתִּי הָמִית אֹתוֹ׃
وَإِنْ غَمَّضَ شَعْبُ ٱلْأَرْضِ أَعْيُنَهُمْ عَنْ ذَلِكَ ٱلْإِنْسَانِ عِنْدَمَا يُعْطِي مِنْ زَرْعِهِ لِمُولَكَ، فَلَمْ يَقْتُلُوهُ،
20:5וְשַׂמְתִּי אֲנִי אֶת־פָּנַי בָּאִישׁ הַהוּא וּבְמִשְׁפַּחְתּוֹ וְהִכְרַתִּי אֹתוֹ וְאֵת כׇּל־הַזֹּנִים אַחֲרָיו לִזְנוֹת אַחֲרֵי הַמֹּלֶךְ מִקֶּרֶב עַמָּם׃

20:6וְהַנֶּפֶשׁ אֲשֶׁר תִּפְנֶה אֶל־הָאֹבֹת וְאֶל־הַיִּדְּעֹנִים לִזְנֹת אַחֲרֵיהֶם וְנָתַתִּי אֶת־פָּנַי בַּנֶּפֶשׁ הַהִוא וְהִכְרַתִּי אֹתוֹ מִקֶּרֶב עַמּוֹ׃
وَٱلنَّفْسُ ٱلَّتِي تَلْتَفِتُ إِلَى ٱلْجَانِّ، وَإِلَى ٱلتَّوَابِعِ لِتَزْنِيَ وَرَاءَهُمْ، أَجْعَلُ وَجْهِي ضِدَّ تِلْكَ ٱلنَّفْسِ وَأَقْطَعُهَا مِنْ شَعْبِهَا.
20:7וְהִתְקַדִּשְׁתֶּם וִהְיִיתֶם קְדֹשִׁים כִּי אֲנִי יְהֹוָה אֱלֹהֵיכֶם׃
فَتَتَقَدَّسُونَ وَتَكُونُونَ قِدِّيسِينَ، لِأَنِّي أَنَا ٱلرَّبُّ إِلَهُكُمْ.
20:8וּשְׁמַרְתֶּם אֶת־חֻקֹּתַי וַעֲשִׂיתֶם אֹתָם אֲנִי יְהֹוָה מְקַדִּשְׁכֶם׃
وَتَحْفَظُونَ فَرَائِضِي وَتَعْمَلُونَهَا. أَنَا ٱلرَّبُّ مُقَدِّسُكُمْ.

«كُلُّ إِنْسَانٍ سَبَّ أَبَاهُ أَوْ أُمَّهُ فَإِنَّهُ يُقْتَلُ. قَدْ سَبَّ أَبَاهُ أَوْ أُمَّهُ. دَمُهُ عَلَيْهِ.
20:10וְאִישׁ אֲשֶׁר יִנְאַף אֶת־אֵשֶׁת אִישׁ אֲשֶׁר יִנְאַף אֶת־אֵשֶׁת רֵעֵהוּ מוֹת־יוּמַת הַנֹּאֵף וְהַנֹּאָפֶת׃

ויקרא כ, י ⇄ בוח׳ארי 3635

היהודים באו אל שליח אללה וסיפרו לו שאיש ואישה מהם נאפו. אמר להם שליח אללה: ״מה אתם מוצאים בתורה בעניין הרגימה?״ אמרו: ״נבייש אותם, והם מולקים״. אמר עבדאללה בן סלאם: ״שיקרתם! יש בה רגימה״. והביאו את התורה ופתחוה, ואחד מהם הניח את ידו על פסוק הרגימה, וקרא את שלפניו ואת שאחריו. אמר לו עבדאללה בן סלאם: ״הרם את ידך״. והרים את ידו, והנה בה פסוק הרגימה. אמרו: ״אמת אמר, מוחמד; יש בה פסוק הרגימה״. וציווה שליח אללה, ונרגמו.

أن اليهود، جاءوا إلى رسول الله صلى الله عليه وسلم فذكروا له أن رجلا منهم وامرأة زنيا فقال لهم رسول الله صلى الله عليه وسلم " ما تجدون في التوراة في شأن الرجم ". فقالوا نفضحهم ويجلدون. فقال عبد الله بن سلام كذبتم، إن فيها الرجم. فأتوا بالتوراة فنشروها، فوضع أحدهم يده على آية الرجم، فقرأ ما قبلها وما بعدها. فقال له عبد الله بن سلام ارفع يدك. فرفع يده فإذا فيها آية الرجم. فقالوا صدق يا محمد، فيها آية الرجم. فأمر بهما رسول الله صلى الله عليه وسلم فرجما

היהודים מביאים זוג נואפים; ״פסוק הרגימה״ בתורה שהסתירו ביד.

اليهود يأتون بزانيين؛ «آية الرجم» في التوراة التي ستروها باليد.

ואל תישאו את הנשים אשר נשאו אבותיכם – מלבד מה שכבר היה; כי תועבה היא ושיקוץ, ודרך רעה. 20:17וְאִישׁ אֲשֶׁר־יִקַּח אֶת־אֲחֹתוֹ בַּת־אָבִיו אוֹ בַת־אִמּוֹ וְרָאָה אֶת־עֶרְוָתָהּ וְהִיא־תִרְאֶה אֶת־עֶרְוָתוֹ חֶסֶד הוּא וְנִכְרְתוּ לְעֵינֵי בְּנֵי עַמָּם עֶרְוַת אֲחֹתוֹ גִּלָּה עֲוֺנוֹ יִשָּׂא׃
وَإِذَا أَخَذَ رَجُلٌ أُخْتَهُ بِنْتَ أَبِيهِ أَوْ بِنْتَ أُمِّهِ، وَرَأَى عَوْرَتَهَا وَرَأَتْ هِيَ عَوْرَتَهُ، فَذَلِكَ عَارٌ. يُقْطَعَانِ أَمَامَ أَعْيُنِ بَنِي شَعْبِهِمَا. قَدْ كَشَفَ عَوْرَةَ أُخْتِهِ. يَحْمِلُ ذَنْبَهُ.
20:18וְאִישׁ אֲשֶׁר־יִשְׁכַּב אֶת־אִשָּׁה דָּוָה וְגִלָּה אֶת־עֶרְוָתָהּ אֶת־מְקֹרָהּ הֶעֱרָה וְהִוא גִּלְּתָה אֶת־מְקוֹר דָּמֶיהָ וְנִכְרְתוּ שְׁנֵיהֶם מִקֶּרֶב עַמָּם׃
وَإِذَا ٱضْطَجَعَ رَجُلٌ مَعَ ٱمْرَأَةٍ طَامِثٍ وَكَشَفَ عَوْرَتَهَا، عَرَّى يَنْبُوعَهَا وَكَشَفَتْ هِيَ يَنْبُوعَ دَمِهَا، يُقْطَعَانِ كِلَاهُمَا مِنْ شَعِبْهِمَا.
20:19וְעֶרְוַת אֲחוֹת אִמְּךָ וַאֲחוֹת אָבִיךָ לֹא תְגַלֵּה כִּי אֶת־שְׁאֵרוֹ הֶעֱרָה עֲוֺנָם יִשָּׂאוּ׃
عَوْرَةَ أُخْتِ أُمِّكَ، أَوْ أُخْتِ أَبِيكَ لَا تَكْشِفْ. إِنَّهُ قَدْ عَرَّى قَرِيبَتَهُ. يَحْمِلَانِ ذَنْبَهُمَا.
20:20וְאִישׁ אֲשֶׁר יִשְׁכַּב אֶת־דֹּדָתוֹ עֶרְוַת דֹּדוֹ גִּלָּה חֶטְאָם יִשָּׂאוּ עֲרִירִים יָמֻתוּ׃
وَإِذَا ٱضْطَجَعَ رَجُلٌ مَعَ ٱمْرَأَةِ عَمِّهِ فَقَدْ كَشَفَ عَوْرَةَ عَمِّهِ. يَحْمِلَانِ ذَنْبَهُمَا. يَمُوتَانِ عَقِيمَيْنِ.
20:21וְאִישׁ אֲשֶׁר יִקַּח אֶת־אֵשֶׁת אָחִיו נִדָּה הִוא עֶרְוַת אָחִיו גִּלָּה עֲרִירִים יִהְיוּ׃
وَإِذَا أَخَذَ رَجُلٌ ٱمْرَأَةَ أَخِيهِ، فَذَلِكَ نَجَاسَةٌ. قَدْ كَشَفَ عَوْرَةَ أَخِيهِ. يَكُونَانِ عَقِيمَيْنِ.
20:22וּשְׁמַרְתֶּם אֶת־כׇּל־חֻקֹּתַי וְאֶת־כׇּל־מִשְׁפָּטַי וַעֲשִׂיתֶם אֹתָם וְלֹא־תָקִיא אֶתְכֶם הָאָרֶץ אֲשֶׁר אֲנִי מֵבִיא אֶתְכֶם שָׁמָּה לָשֶׁבֶת בָּהּ׃
«فَتَحْفَظُونَ جَمِيعَ فَرَائِضِي وَجَمِيعَ أَحْكَامِي، وَتَعْمَلُونَهَا لِكَيْ لَا تَقْذِفَكُمُ ٱلْأَرْضُ ٱلَّتِي أَنَا آتٍ بِكُمْ إِلَيْهَا لِتَسْكُنُوا فِيهَا.
20:23וְלֹא תֵלְכוּ בְּחֻקֹּת הַגּוֹי אֲשֶׁר־אֲנִי מְשַׁלֵּחַ מִפְּנֵיכֶם כִּי אֶת־כׇּל־אֵלֶּה עָשׂוּ וָאָקֻץ בָּם׃
وَلَا تَسْلُكُونَ فِي رُسُومِ ٱلشُّعُوبِ ٱلَّذِينَ أَنَا طَارِدُهُمْ مِنْ أَمَامِكُمْ. لِأَنَّهُمْ قَدْ فَعَلُوا كُلَّ هَذِهِ، فَكَرِهْتُهُمْ.
20:24וָאֹמַר לָכֶם אַתֶּם תִּירְשׁוּ אֶת־אַדְמָתָם וַאֲנִי אֶתְּנֶנָּה לָכֶם לָרֶשֶׁת אֹתָהּ אֶרֶץ זָבַת חָלָב וּדְבָשׁ אֲנִי יְהֹוָה אֱלֹהֵיכֶם אֲשֶׁר־הִבְדַּלְתִּי אֶתְכֶם מִן־הָעַמִּים׃
وَقُلْتُ لَكُمْ: تَرِثُونَ أَنْتُمْ أَرْضَهُمْ، وَأَنَا أُعْطِيكُمْ إِيَّاهَا لِتَرِثُوهَا، أَرْضًا تَفِيضُ لَبَنًا وَعَسَلًا. أَنَا ٱلرَّبُّ إِلَهُكُمُ ٱلَّذِي مَيَّزَكُمْ مِنَ ٱلشُّعُوبِ.
20:25וְהִבְדַּלְתֶּם בֵּין־הַבְּהֵמָה הַטְּהֹרָה לַטְּמֵאָה וּבֵין־הָעוֹף הַטָּמֵא לַטָּהֹר וְלֹא־תְשַׁקְּצוּ אֶת־נַפְשֹׁתֵיכֶם בַּבְּהֵמָה וּבָעוֹף וּבְכֹל אֲשֶׁר תִּרְמֹשׂ הָאֲדָמָה אֲשֶׁר־הִבְדַּלְתִּי לָכֶם לְטַמֵּא׃
فَتُمَيِّزُونَ بَيْنَ ٱلْبَهَائِمِ ٱلطَّاهِرَةِ وَٱلنَّجِسَةِ، وَبَيْنَ ٱلطُّيُورِ ٱلنَّجِسَةِ وَٱلطَّاهِرَةِ. فَلَا تُدَنِّسُوا نُفُوسَكُمْ بِٱلْبَهَائِمِ وَٱلطُّيُورِ، وَلَا بِكُلِّ مَا يَدِبُّ عَلَى ٱلْأَرْضِ مِمَّا مَيَّزْتُهُ لَكُمْ لِيَكُونَ نَجِسًا.
20:26וִהְיִיתֶם לִי קְדֹשִׁים כִּי קָדוֹשׁ אֲנִי יְהֹוָה וָאַבְדִּל אֶתְכֶם מִן־הָעַמִּים לִהְיוֹת לִי׃
وَتَكُونُونَ لِي قِدِّيسِينَ لِأَنِّي قُدُّوسٌ أَنَا ٱلرَّبُّ، وَقَدْ مَيَّزْتُكُمْ مِنَ ٱلشُّعُوبِ لِتَكُونُوا لِي.
20:27וְאִישׁ אוֹ־אִשָּׁה כִּי־יִהְיֶה בָהֶם אוֹב אוֹ יִדְּעֹנִי מוֹת יוּמָתוּ בָּאֶבֶן יִרְגְּמוּ אֹתָם דְּמֵיהֶם בָּם׃
«وَإِذَا كَانَ فِي رَجُلٍ أَوِ ٱمْرَأَةٍ جَانٌّ أَوْ تَابِعَةٌ فَإِنَّهُ يُقْتَلُ. بِٱلْحِجَارَةِ يَرْجُمُونَهُ. دَمُهُ عَلَيْهِ».
