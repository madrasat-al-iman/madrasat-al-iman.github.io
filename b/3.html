בראשית יב, א–ג ⇄ 3:65–68

אנשי הספר, מדוע תתווכחו על אברהם, והרי התורה והאינג׳יל לא הורדו אלא אחריו? הלא תבינו? (65) הנה אתם – התווכחתם על דבר שיש לכם בו דעת; מדוע תתווכחו על דבר שאין לכם בו דעת? ואללה יודע, ואתם אינכם יודעים. (66) אברהם לא היה יהודי ולא נוצרי, אלא חניף מתמסר (מוסלם), ולא היה מן המשתפים. (67) הקרובים ביותר לאברהם הם אלה שהלכו אחריו, והנביא הזה והמאמינים; ואללה מגן המאמינים. (68)

يَـٰٓأَهۡلَ ٱلۡكِتَٰبِ لِمَ تُحَآجُّونَ فِيٓ إِبۡرَٰهِيمَ وَمَآ أُنزِلَتِ ٱلتَّوۡرَىٰةُ وَٱلۡإِنجِيلُ إِلَّا مِنۢ بَعۡدِهِۦٓۚ أَفَلَا تَعۡقِلُونَ (65) هَـٰٓأَنتُمۡ هَـٰٓؤُلَآءِ حَٰجَجۡتُمۡ فِيمَا لَكُم بِهِۦ عِلۡمٞ فَلِمَ تُحَآجُّونَ فِيمَا لَيۡسَ لَكُم بِهِۦ عِلۡمٞۚ وَٱللَّهُ يَعۡلَمُ وَأَنتُمۡ لَا تَعۡلَمُونَ (66) مَا كَانَ إِبۡرَٰهِيمُ يَهُودِيّٗا وَلَا نَصۡرَانِيّٗا وَلَٰكِن كَانَ حَنِيفٗا مُّسۡلِمٗا وَمَا كَانَ مِنَ ٱلۡمُشۡرِكِينَ (67) إِنَّ أَوۡلَى ٱلنَّاسِ بِإِبۡرَٰهِيمَ لَلَّذِينَ ٱتَّبَعُوهُ وَهَٰذَا ٱلنَّبِيُّ وَٱلَّذِينَ ءَامَنُواْۗ وَٱللَّهُ وَلِيُّ ٱلۡمُؤۡمِنِينَ (68)

בראשית יב, א–ה ⇄ 26:69–89

וספר להם את מעשה אברהם, (69) כאשר אמר לאביו ולעמו: ״מה אתם עובדים?״ (70) אמרו: ״אנו עובדים פסלים, ונוסיף לדבוק בהם״. (71) אמר: ״האם הם שומעים אתכם כאשר אתם קוראים, (72) או מועילים לכם או מזיקים?״ (73) אמרו: ״לא, אבל מצאנו את אבותינו עושים כך״. (74) אמר: ״הראיתם את אשר עבדתם, (75) אתם ואבותיכם הקדמונים? (76) הם אויבים לי – מלבד ריבון העולמים, (77) אשר בראני, והוא מדריכני, (78) ואשר הוא מאכילני ומשקני, (79) וכאשר אחלה – הוא מרפאני, (80) ואשר ימיתני ואחר כך יחייני, (81) ואשר אני מקווה כי יסלח לי על חטאי ביום הדין. (82) ריבוני, תן לי חוכמה (משפט) וצרפני אל הצדיקים, (83) ועשה לי לשון אמת (שם טוב) באחרונים, (84) ועשני מיורשי גן העונג, (85) וסלח לאבי, כי היה מן התועים, (86) ואל תביישני ביום שבו יוקמו – (87) יום שבו לא יועילו הון ובנים, (88) אלא מי שיבוא אל אללה בלב שלם״. (89)

وَٱتۡلُ عَلَيۡهِمۡ نَبَأَ إِبۡرَٰهِيمَ (69) إِذۡ قَالَ لِأَبِيهِ وَقَوۡمِهِۦ مَا تَعۡبُدُونَ (70) قَالُواْ نَعۡبُدُ أَصۡنَامٗا فَنَظَلُّ لَهَا عَٰكِفِينَ (71) قَالَ هَلۡ يَسۡمَعُونَكُمۡ إِذۡ تَدۡعُونَ (72) أَوۡ يَنفَعُونَكُمۡ أَوۡ يَضُرُّونَ (73) قَالُواْ بَلۡ وَجَدۡنَآ ءَابَآءَنَا كَذَٰلِكَ يَفۡعَلُونَ (74) قَالَ أَفَرَءَيۡتُم مَّا كُنتُمۡ تَعۡبُدُونَ (75) أَنتُمۡ وَءَابَآؤُكُمُ ٱلۡأَقۡدَمُونَ (76) فَإِنَّهُمۡ عَدُوّٞ لِّيٓ إِلَّا رَبَّ ٱلۡعَٰلَمِينَ (77) ٱلَّذِي خَلَقَنِي فَهُوَ يَهۡدِينِ (78) وَٱلَّذِي هُوَ يُطۡعِمُنِي وَيَسۡقِينِ (79) وَإِذَا مَرِضۡتُ فَهُوَ يَشۡفِينِ (80) وَٱلَّذِي يُمِيتُنِي ثُمَّ يُحۡيِينِ (81) وَٱلَّذِيٓ أَطۡمَعُ أَن يَغۡفِرَ لِي خَطِيٓـَٔتِي يَوۡمَ ٱلدِّينِ (82) رَبِّ هَبۡ لِي حُكۡمٗا وَأَلۡحِقۡنِي بِٱلصَّـٰلِحِينَ (83) وَٱجۡعَل لِّي لِسَانَ صِدۡقٖ فِي ٱلۡأٓخِرِينَ (84) وَٱجۡعَلۡنِي مِن وَرَثَةِ جَنَّةِ ٱلنَّعِيمِ (85) وَٱغۡفِرۡ لِأَبِيٓ إِنَّهُۥ كَانَ مِنَ ٱلضَّآلِّينَ (86) وَلَا تُخۡزِنِي يَوۡمَ يُبۡعَثُونَ (87) يَوۡمَ لَا يَنفَعُ مَالٞ وَلَا بَنُونَ (88) إِلَّا مَنۡ أَتَى ٱللَّهَ بِقَلۡبٖ سَلِيمٖ (89)

إبراهيم وقومه – الجدال في الأصنام – خلفية / تأثير

בראשית יב, א–ה ⇄ 29:16–27

ואברהם – כאשר אמר לעמו: ״עבדו את אללה ויראו אותו; זה טוב לכם, אם יודעים אתם. (16) אתם עובדים מלבד אללה אלילים, ובודים שקר. אלה שאתם עובדים מלבד אללה אין בידם לפרנס אתכם; לכן בקשו את הפרנסה אצל אללה, ועבדוהו והודו לו; אליו תושבו. (17) ואם תכחישו – כבר הכחישו אומות לפניכם; ואין על השליח אלא להודיע הודעה ברורה״. (18) הלא ראו איך אללה מתחיל את הבריאה ואחר כך משיבה? זה קל על אללה. (19) אמור: ״לכו בארץ וראו איך התחיל את הבריאה; ואחר כך אללה יקים את הבריאה האחרונה; אללה כל יכול. (20) מעניש את מי שירצה ומרחם על מי שירצה, ואליו תושבו. (21) ולא תוכלו להימלט בארץ ולא בשמים, ואין לכם מלבד אללה מגן ולא עוזר״. (22) ואלה שכפרו באותות אללה ובפגישה עמו – הם התייאשו מרחמיי, ולהם עונש כואב. (23) ולא הייתה תשובת עמו (של אברהם) אלא שאמרו: ״הרגוהו או שרפוהו!״ והצילו אללה מן האש. בזאת אותות לעם המאמין. (24) ואמר: ״עשיתם לכם מלבד אללה אלילים מתוך אהבה ביניכם בחיי העולם הזה; ואחר כך, ביום התקומה, יכפור זה בזה ויקלל זה את זה, ומשכנכם האש, ואין לכם עוזרים״. (25) והאמין לו לוט, ואמר (אברהם): ״אני מהגר אל ריבוני; הוא האדיר, החכם״. (26) ונתנו לו את יצחק ואת יעקב, ושמנו בזרעו את הנבואה ואת הספר, ונתנו לו את שכרו בעולם הזה, ובעולם הבא הוא מן הצדיקים. (27)

وَإِبۡرَٰهِيمَ إِذۡ قَالَ لِقَوۡمِهِ ٱعۡبُدُواْ ٱللَّهَ وَٱتَّقُوهُۖ ذَٰلِكُمۡ خَيۡرٞ لَّكُمۡ إِن كُنتُمۡ تَعۡلَمُونَ (16) إِنَّمَا تَعۡبُدُونَ مِن دُونِ ٱللَّهِ أَوۡثَٰنٗا وَتَخۡلُقُونَ إِفۡكًاۚ إِنَّ ٱلَّذِينَ تَعۡبُدُونَ مِن دُونِ ٱللَّهِ لَا يَمۡلِكُونَ لَكُمۡ رِزۡقٗا فَٱبۡتَغُواْ عِندَ ٱللَّهِ ٱلرِّزۡقَ وَٱعۡبُدُوهُ وَٱشۡكُرُواْ لَهُۥٓۖ إِلَيۡهِ تُرۡجَعُونَ (17) وَإِن تُكَذِّبُواْ فَقَدۡ كَذَّبَ أُمَمٞ مِّن قَبۡلِكُمۡۖ وَمَا عَلَى ٱلرَّسُولِ إِلَّا ٱلۡبَلَٰغُ ٱلۡمُبِينُ (18) أَوَلَمۡ يَرَوۡاْ كَيۡفَ يُبۡدِئُ ٱللَّهُ ٱلۡخَلۡقَ ثُمَّ يُعِيدُهُۥٓۚ إِنَّ ذَٰلِكَ عَلَى ٱللَّهِ يَسِيرٞ (19) قُلۡ سِيرُواْ فِي ٱلۡأَرۡضِ فَٱنظُرُواْ كَيۡفَ بَدَأَ ٱلۡخَلۡقَۚ ثُمَّ ٱللَّهُ يُنشِئُ ٱلنَّشۡأَةَ ٱلۡأٓخِرَةَۚ إِنَّ ٱللَّهَ عَلَىٰ كُلِّ شَيۡءٖ قَدِيرٞ (20) يُعَذِّبُ مَن يَشَآءُ وَيَرۡحَمُ مَن يَشَآءُۖ وَإِلَيۡهِ تُقۡلَبُونَ (21) وَمَآ أَنتُم بِمُعۡجِزِينَ فِي ٱلۡأَرۡضِ وَلَا فِي ٱلسَّمَآءِۖ وَمَا لَكُم مِّن دُونِ ٱللَّهِ مِن وَلِيّٖ وَلَا نَصِيرٖ (22) وَٱلَّذِينَ كَفَرُواْ بِـَٔايَٰتِ ٱللَّهِ وَلِقَآئِهِۦٓ أُوْلَـٰٓئِكَ يَئِسُواْ مِن رَّحۡمَتِي وَأُوْلَـٰٓئِكَ لَهُمۡ عَذَابٌ أَلِيمٞ (23) فَمَا كَانَ جَوَابَ قَوۡمِهِۦٓ إِلَّآ أَن قَالُواْ ٱقۡتُلُوهُ أَوۡ حَرِّقُوهُ فَأَنجَىٰهُ ٱللَّهُ مِنَ ٱلنَّارِۚ إِنَّ فِي ذَٰلِكَ لَأٓيَٰتٖ لِّقَوۡمٖ يُؤۡمِنُونَ (24) وَقَالَ إِنَّمَا ٱتَّخَذۡتُم مِّن دُونِ ٱللَّهِ أَوۡثَٰنٗا مَّوَدَّةَ بَيۡنِكُمۡ فِي ٱلۡحَيَوٰةِ ٱلدُّنۡيَاۖ ثُمَّ يَوۡمَ ٱلۡقِيَٰمَةِ يَكۡفُرُ بَعۡضُكُم بِبَعۡضٖ وَيَلۡعَنُ بَعۡضُكُم بَعۡضٗا وَمَأۡوَىٰكُمُ ٱلنَّارُ وَمَا لَكُم مِّن نَّـٰصِرِينَ (25) ۞فَـَٔامَنَ لَهُۥ لُوطٞۘ وَقَالَ إِنِّي مُهَاجِرٌ إِلَىٰ رَبِّيٓۖ إِنَّهُۥ هُوَ ٱلۡعَزِيزُ ٱلۡحَكِيمُ (26) وَوَهَبۡنَا لَهُۥٓ إِسۡحَٰقَ وَيَعۡقُوبَ وَجَعَلۡنَا فِي ذُرِّيَّتِهِ ٱلنُّبُوَّةَ وَٱلۡكِتَٰبَ وَءَاتَيۡنَٰهُ أَجۡرَهُۥ فِي ٱلدُّنۡيَاۖ وَإِنَّهُۥ فِي ٱلۡأٓخِرَةِ لَمِنَ ٱلصَّـٰلِحِينَ (27)

בראשית יב, א–ה ⇄ 37:83–98

ומסיעתו (של נח) – אברהם, (83) כאשר בא אל ריבונו בלב שלם, (84) כאשר אמר לאביו ולעמו: ״מה אתם עובדים? (85) האם שקר – אלים מלבד אללה – אתם רוצים? (86) ומה סברתכם על ריבון העולמים?״ (87) והביט מבט בכוכבים, (88) ואמר: ״אני חולה״. (89) ופנו ממנו והלכו. (90) ופנה בחשאי אל אליהם ואמר: ״הלא תאכלו? 12:11וַיְהִי כַּאֲשֶׁר הִקְרִיב לָבוֹא מִצְרָיְמָה וַיֹּאמֶר אֶל־שָׂרַי אִשְׁתּוֹ הִנֵּה־נָא יָדַעְתִּי כִּי אִשָּׁה יְפַת־מַרְאֶה אָתְּ׃
وَحَدَثَ لَمَّا قَرُبَ أَنْ يَدْخُلَ مِصْرَ أَنَّهُ قَالَ لِسَارَايَ ٱمْرَأَتِهِ: «إِنِّي قَدْ عَلِمْتُ أَنَّكِ ٱمْرَأَةٌ حَسَنَةُ ٱلْمَنْظَرِ.
12:12וְהָיָה כִּי־יִרְאוּ אֹתָךְ הַמִּצְרִים וְאָמְרוּ אִשְׁתּוֹ זֹאת וְהָרְגוּ אֹתִי וְאֹתָךְ יְחַיּוּ׃
فَيَكُونُ إِذَا رَآكِ ٱلْمِصْرِيُّونَ أَنَّهُمْ يَقُولُونَ: هَذِهِ ٱمْرَأَتُهُ. فَيَقْتُلُونَنِي وَيَسْتَبْقُونَكِ.
12:13אִמְרִי־נָא אֲחֹתִי אָתְּ לְמַעַן יִיטַב־לִי בַעֲבוּרֵךְ וְחָיְתָה נַפְשִׁי בִּגְלָלֵךְ׃
قُولِي إِنَّكِ أُخْتِي، لِيَكُونَ لِي خَيْرٌ بِسَبَبِكِ وَتَحْيَا نَفْسِي مِنْ أَجْلِكِ».
12:14וַיְהִי כְּבוֹא אַבְרָם מִצְרָיְמָה וַיִּרְאוּ הַמִּצְרִים אֶת־הָאִשָּׁה כִּי־יָפָה הִוא מְאֹד׃
فَحَدَثَ لَمَّا دَخَلَ أَبْرَامُ إِلَى مِصْرَ أَنَّ ٱلْمِصْرِيِّينَ رَأَوْا ٱلْمَرْأَةَ أَنَّهَا حَسَنَةٌ جِدًّا.
12:15וַיִּרְאוּ אֹתָהּ שָׂרֵי פַרְעֹה וַיְהַלְלוּ אֹתָהּ אֶל־פַּרְעֹה וַתֻּקַּח הָאִשָּׁה בֵּית פַּרְעֹה׃
وَرَآهَا رُؤَسَاءُ فِرْعَوْنَ وَمَدَحُوهَا لَدَى فِرْعَوْنَ، فَأُخِذَتِ ٱلْمَرْأَةُ إِلَى بَيْتِ فِرْعَوْنَ،
12:16וּלְאַבְרָם הֵיטִיב בַּעֲבוּרָהּ וַיְהִי־לוֹ צֹאן־וּבָקָר וַחֲמֹרִים וַעֲבָדִים וּשְׁפָחֹת וַאֲתֹנֹת וּגְמַלִּים׃
فَصَنَعَ إِلَى أَبْرَامَ خَيْرًا بِسَبَبِهَا، وَصَارَ لَهُ غَنَمٌ وَبَقَرٌ وَحَمِيرٌ وَعَبِيدٌ وَإِمَاءٌ وَأُتُنٌ وَجِمَالٌ.
12:17וַיְנַגַּע יְהֹוָה אֶת־פַּרְעֹה נְגָעִים גְּדֹלִים וְאֶת־בֵּיתוֹ עַל־דְּבַר שָׂרַי אֵשֶׁת אַבְרָם׃
فَضَرَبَ ٱلرَّبُّ فِرْعَوْنَ وَبَيْتَهُ ضَرَبَاتٍ عَظِيمَةً بِسَبَبِ سَارَايَ ٱمْرَأَةِ أَبْرَامَ.
12:18וַיִּקְרָא פַרְעֹה לְאַבְרָם וַיֹּאמֶר מַה־זֹּאת עָשִׂיתָ לִּי לָמָּה לֹא־הִגַּדְתָּ לִּי כִּי אִשְׁתְּךָ הִוא׃
فَدَعَا فِرْعَوْنُ أَبْرَامَ وَقَالَ: «مَا هَذَا ٱلَّذِي صَنَعْتَ بِي؟ لِمَاذَا لَمْ تُخْبِرْنِي أَنَّهَا ٱمْرَأَتُكَ؟
12:19לָמָה אָמַרְתָּ אֲחֹתִי הִוא וָאֶקַּח אֹתָהּ לִי לְאִשָּׁה וְעַתָּה הִנֵּה אִשְׁתְּךָ קַח וָלֵךְ׃
لِمَاذَا قُلْتَ: هِيَ أُخْتِي، حَتَّى أَخَذْتُهَا لِي لِتَكُونَ زَوْجَتِي؟ وَٱلْآنَ هُوَذَا ٱمْرَأَتُكَ! خُذْهَا وَٱذْهَبْ!».
12:20וַיְצַו עָלָיו פַּרְעֹה אֲנָשִׁים וַיְשַׁלְּחוּ אֹתוֹ וְאֶת־אִשְׁתּוֹ וְאֶת־כׇּל־אֲשֶׁר־לוֹ׃
فَأَوْصَى عَلَيْهِ فِرْعَوْنُ رِجَالًا فَشَيَّعُوهُ وَٱمْرَأَتَهُ وَكُلَّ مَا كَانَ لَهُ.
13:1וַיַּעַל אַבְרָם מִמִּצְרַיִם הוּא וְאִשְׁתּוֹ וְכׇל־אֲשֶׁר־לוֹ וְלוֹט עִמּוֹ הַנֶּגְבָּה׃
فَصَعِدَ أَبْرَامُ مِنْ مِصْرَ هُوَ وَٱمْرَأَتُهُ وَكُلُّ مَا كَانَ لَهُ، وَلُوطٌ مَعَهُ إِلَى ٱلْجَنُوبِ.
13:2וְאַבְרָם כָּבֵד מְאֹד בַּמִּקְנֶה בַּכֶּסֶף וּבַזָּהָב׃
وَكَانَ أَبْرَامُ غَنِيًّا جِدًّا فِي ٱلْمَوَاشِي وَٱلْفِضَّةِ وَٱلذَّهَبِ.
13:3וַיֵּלֶךְ לְמַסָּעָיו מִנֶּגֶב וְעַד־בֵּית־אֵל עַד־הַמָּקוֹם אֲשֶׁר־הָיָה שָׁם אׇהֳלֹה בַּתְּחִלָּה בֵּין בֵּית־אֵל וּבֵין הָעָי׃
وَسَارَ فِي رِحْلَاتِهِ مِنَ ٱلْجَنُوبِ إِلَى بَيْتِ إِيلَ، إِلَى ٱلْمَكَانِ ٱلَّذِي كَانَتْ خَيْمَتُهُ فِيهِ فِي ٱلْبَدَاءَةِ، بَيْنَ بَيْتِ إِيلَ وَعَايَ،
13:4אֶל־מְקוֹם הַמִּזְבֵּחַ אֲשֶׁר־עָשָׂה שָׁם בָּרִאשֹׁנָה וַיִּקְרָא שָׁם אַבְרָם בְּשֵׁם יְהֹוָה׃
إِلَى مَكَانِ ٱلْمَذْبَحِ ٱلَّذِي عَمِلَهُ هُنَاكَ أَوَّلًا. وَدَعَا هُنَاكَ أَبْرَامُ بِٱسْمِ ٱلرَّبِّ.
13:5וְגַם־לְלוֹט הַהֹלֵךְ אֶת־אַבְרָם הָיָה צֹאן־וּבָקָר וְאֹהָלִים׃
وَلُوطٌ ٱلسَّائِرُ مَعَ أَبْرَامَ، كَانَ لَهُ أَيْضًا غَنَمٌ وَبَقَرٌ وَخِيَامٌ.
13:6וְלֹא־נָשָׂא אֹתָם הָאָרֶץ לָשֶׁבֶת יַחְדָּו כִּי־הָיָה רְכוּשָׁם רָב וְלֹא יָכְלוּ לָשֶׁבֶת יַחְדָּו׃
وَلَمْ تَحْتَمِلْهُمَا ٱلْأَرْضُ أَنْ يَسْكُنَا مَعًا، إِذْ كَانَتْ أَمْلَاكُهُمَا كَثِيرَةً، فَلَمْ يَقْدِرَا أَنْ يَسْكُنَا مَعًا.
13:7וַיְהִי־רִיב בֵּין רֹעֵי מִקְנֵה־אַבְרָם וּבֵין רֹעֵי מִקְנֵה־לוֹט וְהַכְּנַעֲנִי וְהַפְּרִזִּי אָז יֹשֵׁב בָּאָרֶץ׃
فَحَدَثَتْ مُخَاصَمَةٌ بَيْنَ رُعَاةِ مَوَاشِي أَبْرَامَ وَرُعَاةِ مَوَاشِي لُوطٍ. وَكَانَ ٱلْكَنْعَانِيُّونَ وَٱلْفَرِزِّيُّونَ حِينَئِذٍ سَاكِنِينَ فِي ٱلْأَرْضِ.
13:8וַיֹּאמֶר אַבְרָם אֶל־לוֹט אַל־נָא תְהִי מְרִיבָה בֵּינִי וּבֵינֶךָ וּבֵין רֹעַי וּבֵין רֹעֶיךָ כִּי־אֲנָשִׁים אַחִים אֲנָחְנוּ׃
فَقَالَ أَبْرَامُ لِلُوطٍ: «لَا تَكُنْ مُخَاصَمَةٌ بَيْنِي وَبَيْنَكَ، وَبَيْنَ رُعَاتِي وَرُعَاتِكَ، لِأَنَّنَا نَحْنُ أَخَوَانِ.

בראשית יג, ח–ט ⇄ 49:9–10; אבו דאוד 4800

49:9–10: ואם שתי קבוצות מן המאמינים נלחמות זו בזו – השכינו שלום ביניהן; ואם אחת מהן מתנכלת לאחרת – הילחמו במתנכלת עד שתשוב אל דבר אללה; ואם שבה – השכינו ביניהן שלום בצדק, ונהגו ביושר; אללה אוהב את הנוהגים ביושר. (9) המאמינים אינם אלא אחים; השכינו אפוא שלום בין שני אחיכם, ויראו את אללה – למען ירוחמו עליכם. (10) ✦ אבו דאוד 4800: ״אני ערב לבית בפאתי גן העדן למי שנמנע מוויכוח גם כשהוא צודק״.

وَإِن طَآئِفَتَانِ مِنَ ٱلۡمُؤۡمِنِينَ ٱقۡتَتَلُواْ فَأَصۡلِحُواْ بَيۡنَهُمَاۖ فَإِنۢ بَغَتۡ إِحۡدَىٰهُمَا عَلَى ٱلۡأُخۡرَىٰ فَقَٰتِلُواْ ٱلَّتِي تَبۡغِي حَتَّىٰ تَفِيٓءَ إِلَىٰٓ أَمۡرِ ٱللَّهِۚ فَإِن فَآءَتۡ فَأَصۡلِحُواْ بَيۡنَهُمَا بِٱلۡعَدۡلِ وَأَقۡسِطُوٓاْۖ إِنَّ ٱللَّهَ يُحِبُّ ٱلۡمُقۡسِطِينَ (9) إِنَّمَا ٱلۡمُؤۡمِنُونَ إِخۡوَةٞ فَأَصۡلِحُواْ بَيۡنَ أَخَوَيۡكُمۡۚ وَٱتَّقُواْ ٱللَّهَ لَعَلَّكُمۡ تُرۡحَمُونَ (10) ✦ سنن أبي داود 4800: أَنَا زَعِيمٌ بِبَيْتٍ فِي رَبَضِ الْجَنَّةِ لِمَنْ تَرَكَ الْمِرَاءَ وَإِنْ كَانَ مُحِقًّا

ויתור אברהם ללוט למניעת מריבה בין אחים (״אל נא תהי מריבה ביני ובינך״) — מול הבטחת בית בגן העדן למי שמוותר על מריבה גם כשהצדק עמו (אבו דאוד 4800).

تنازل إبراهيم للوط دفعًا للخصومة («لا تكن مخاصمة بيني وبينك») مقابل فضل ترك المراء ولو كان محقًا (أبو داود 4800).

13:9הֲלֹא כׇל־הָאָרֶץ לְפָנֶיךָ הִפָּרֶד נָא מֵעָלָי אִם־הַשְּׂמֹאל וְאֵימִנָה וְאִם־הַיָּמִין וְאַשְׂמְאִילָה׃
أَلَيْسَتْ كُلُّ ٱلْأَرْضِ أَمَامَكَ؟ ٱعْتَزِلْ عَنِّي. إِنْ ذَهَبْتَ شِمَالًا فَأَنَا يَمِينًا، وَإِنْ يَمِينًا فَأَنَا شِمَالًا».
13:10וַיִּשָּׂא־לוֹט אֶת־עֵינָיו וַיַּרְא אֶת־כׇּל־כִּכַּר הַיַּרְדֵּן כִּי כֻלָּהּ מַשְׁקֶה לִפְנֵי שַׁחֵת יְהֹוָה אֶת־סְדֹם וְאֶת־עֲמֹרָה כְּגַן־יְהֹוָה כְּאֶרֶץ מִצְרַיִם בֹּאֲכָה צֹעַר׃
فَرَفَعَ لُوطٌ عَيْنَيْهِ وَرَأَى كُلَّ دَائِرَةِ ٱلْأُرْدُنِّ أَنَّ جَمِيعَهَا سَقْيٌ، قَبْلَمَا أَخْرَبَ ٱلرَّبُّ سَدُومَ وَعَمُورَةَ، كَجَنَّةِ ٱلرَّبِّ، كَأَرْضِ مِصْرَ. حِينَمَا تَجِيءُ إِلَى صُوغَرَ.
13:11וַיִּבְחַר־לוֹ לוֹט אֵת כׇּל־כִּכַּר הַיַּרְדֵּן וַיִּסַּע לוֹט מִקֶּדֶם וַיִּפָּרְדוּ אִישׁ מֵעַל אָחִיו׃
فَٱخْتَارَ لُوطٌ لِنَفْسِهِ كُلَّ دَائِرَةِ ٱلْأُرْدُنِّ، وَٱرْتَحَلَ لُوطٌ شَرْقًا. فَٱعْتَزَلَ ٱلْوَاحِدُ عَنِ ٱلْآخَرِ.
13:12אַבְרָם יָשַׁב בְּאֶרֶץ־כְּנָעַן וְלוֹט יָשַׁב בְּעָרֵי הַכִּכָּר וַיֶּאֱהַל עַד־סְדֹם׃
أَبْرَامُ سَكَنَ فِي أَرْضِ كَنْعَانَ، وَلُوطٌ سَكَنَ فِي مُدُنِ ٱلدَّائِرَةِ، وَنَقَلَ خِيَامَهُ إِلَى سَدُومَ.
13:13וְאַנְשֵׁי סְדֹם רָעִים וְחַטָּאִים לַיהֹוָה מְאֹד׃
وَكَانَ أَهْلُ سَدُومَ أَشْرَارًا وَخُطَاةً لَدَى ٱلرَّبِّ جِدًّا.
13:14וַיהֹוָה אָמַר אֶל־אַבְרָם אַחֲרֵי הִפָּרֶד־לוֹט מֵעִמּוֹ שָׂא נָא עֵינֶיךָ וּרְאֵה מִן־הַמָּקוֹם אֲשֶׁר־אַתָּה שָׁם צָפֹנָה וָנֶגְבָּה וָקֵדְמָה וָיָמָּה׃

13:15כִּי אֶת־כׇּל־הָאָרֶץ אֲשֶׁר־אַתָּה רֹאֶה לְךָ אֶתְּנֶנָּה וּלְזַרְעֲךָ עַד־עוֹלָם׃

13:16וְשַׂמְתִּי אֶת־זַרְעֲךָ כַּעֲפַר הָאָרֶץ אֲשֶׁר אִם־יוּכַל אִישׁ לִמְנוֹת אֶת־עֲפַר הָאָרֶץ גַּם־זַרְעֲךָ יִמָּנֶה׃

13:17קוּם הִתְהַלֵּךְ בָּאָרֶץ לְאׇרְכָּהּ וּלְרׇחְבָּהּ כִּי לְךָ אֶתְּנֶנָּה׃

13:18וַיֶּאֱהַל אַבְרָם וַיָּבֹא וַיֵּשֶׁב בְּאֵלֹנֵי מַמְרֵא אֲשֶׁר בְּחֶבְרוֹן וַיִּבֶן־שָׁם מִזְבֵּחַ לַיהֹוָה׃

בראשית יג, יח ⇄ 2:124

בראשית יג, יח ⇄ 14:35–41

וכאשר אמר אברהם: ״ריבוני, עשה את העיר הזאת מבטחים, והרחק אותי ואת בניי מעבודת הפסלים. (35) ריבוני, הם התעו רבים מבני האדם. מי שהולך אחריי – הוא ממני, ומי שממרה אותי – אתה סולח, רחום. (36) ״ריבוננו, הושבתי מזרעי בעמק אשר אין בו זרע, ליד ביתך המקודש, ריבוננו, למען יקיימו את התפילה. לכן הטה אליהם לבבות מבני האדם, ופרנסם מן הפירות, למען יודו״. (37) ריבוננו, אתה יודע את אשר נסתיר ואת אשר נגלה; ואין נסתר מאללה דבר בארץ ובשמים. (38) השבח לאללה אשר נתן לי לעת זקנה את ישמעאל ואת יצחק; ריבוני שומע תפילה. (39) ריבוני, עשני מקיים את התפילה, וגם מזרעי; ריבוננו, וקבל את תפילתי. (40) ריבוננו, סלח לי ולהוריי ולמאמינים ביום שבו יקום החשבון״. (41)

وَإِذۡ قَالَ إِبۡرَٰهِيمُ رَبِّ ٱجۡعَلۡ هَٰذَا ٱلۡبَلَدَ ءَامِنٗا وَٱجۡنُبۡنِي وَبَنِيَّ أَن نَّعۡبُدَ ٱلۡأَصۡنَامَ (35) رَبِّ إِنَّهُنَّ أَضۡلَلۡنَ كَثِيرٗا مِّنَ ٱلنَّاسِۖ فَمَن تَبِعَنِي فَإِنَّهُۥ مِنِّيۖ وَمَنۡ عَصَانِي فَإِنَّكَ غَفُورٞ رَّحِيمٞ (36) رَّبَّنَآ إِنِّيٓ أَسۡكَنتُ مِن ذُرِّيَّتِي بِوَادٍ غَيۡرِ ذِي زَرۡعٍ عِندَ بَيۡتِكَ ٱلۡمُحَرَّمِ رَبَّنَا لِيُقِيمُواْ ٱلصَّلَوٰةَ فَٱجۡعَلۡ أَفۡـِٔدَةٗ مِّنَ ٱلنَّاسِ تَهۡوِيٓ إِلَيۡهِمۡ وَٱرۡزُقۡهُم مِّنَ ٱلثَّمَرَٰتِ لَعَلَّهُمۡ يَشۡكُرُونَ (37) رَبَّنَآ إِنَّكَ تَعۡلَمُ مَا نُخۡفِي وَمَا نُعۡلِنُۗ وَمَا يَخۡفَىٰ عَلَى ٱللَّهِ مِن شَيۡءٖ فِي ٱلۡأَرۡضِ وَلَا فِي ٱلسَّمَآءِ (38) ٱلۡحَمۡدُ لِلَّهِ ٱلَّذِي وَهَبَ لِي عَلَى ٱلۡكِبَرِ إِسۡمَٰعِيلَ وَإِسۡحَٰقَۚ إِنَّ رَبِّي لَسَمِيعُ ٱلدُّعَآءِ (39) رَبِّ ٱجۡعَلۡنِي مُقِيمَ ٱلصَّلَوٰةِ وَمِن ذُرِّيَّتِيۚ رَبَّنَا وَتَقَبَّلۡ دُعَآءِ (40) رَبَّنَا ٱغۡفِرۡ لِي وَلِوَٰلِدَيَّ وَلِلۡمُؤۡمِنِينَ يَوۡمَ يَقُومُ ٱلۡحِسَابُ (41)

בראשית יג, יח ⇄ 22:26–29

וכאשר קבענו לאברהם את מקום הבית: ״אל תשתף עמי דבר, וטהר את ביתי לסובבים, לעומדים, לכורעים ולמשתחווים. (26) והכרז בבני האדם על העלייה לרגל; יבואו אליך רגלים ועל כל גמל כחוש, יבואו מכל דרך עמוקה, (27) למען יהיו עדים לתועלת להם ויזכירו את שם אללה בימים ידועים על הבהמות אשר פרנסם; אכלו מהן והאכילו את המסכן העני. (28) אחר כך יסירו את זוהמתם, ויקיימו את נדריהם, ויסובו סביב הבית העתיק״. (29)

وَإِذۡ بَوَّأۡنَا لِإِبۡرَٰهِيمَ مَكَانَ ٱلۡبَيۡتِ أَن لَّا تُشۡرِكۡ بِي شَيۡـٔٗا وَطَهِّرۡ بَيۡتِيَ لِلطَّآئِفِينَ وَٱلۡقَآئِمِينَ وَٱلرُّكَّعِ ٱلسُّجُودِ (26) وَأَذِّن فِي ٱلنَّاسِ بِٱلۡحَجِّ يَأۡتُوكَ رِجَالٗا وَعَلَىٰ كُلِّ ضَامِرٖ يَأۡتِينَ مِن كُلِّ فَجٍّ عَمِيقٖ (27) لِّيَشۡهَدُواْ مَنَٰفِعَ لَهُمۡ وَيَذۡكُرُواْ ٱسۡمَ ٱللَّهِ فِيٓ أَيَّامٖ مَّعۡلُومَٰتٍ عَلَىٰ مَا رَزَقَهُم مِّنۢ بَهِيمَةِ ٱلۡأَنۡعَٰمِۖ فَكُلُواْ مِنۡهَا وَأَطۡعِمُواْ ٱلۡبَآئِسَ ٱلۡفَقِيرَ (28) ثُمَّ لۡيَقۡضُواْ تَفَثَهُمۡ وَلۡيُوفُواْ نُذُورَهُمۡ وَلۡيَطَّوَّفُواْ بِٱلۡبَيۡتِ ٱلۡعَتِيقِ (29)

14:1וַיְהִי בִּימֵי אַמְרָפֶל מֶלֶךְ־שִׁנְעָר אַרְיוֹךְ מֶלֶךְ אֶלָּסָר כְּדׇרְלָעֹמֶר מֶלֶךְ עֵילָם וְתִדְעָל מֶלֶךְ גּוֹיִם׃
وَحَدَثَ فِي أَيَّامِ أَمْرَافَلَ مَلِكِ شِنْعَارَ، وَأَرْيُوكَ مَلِكِ أَلَّاسَارَ، وَكَدَرْلَعَوْمَرَ مَلِكِ عِيلَامَ، وَتِدْعَالَ مَلِكِ جُويِيمَ،
14:2עָשׂוּ מִלְחָמָה אֶת־בֶּרַע מֶלֶךְ סְדֹם וְאֶת־בִּרְשַׁע מֶלֶךְ עֲמֹרָה שִׁנְאָב מֶלֶךְ אַדְמָה וְשֶׁמְאֵבֶר מֶלֶךְ (צביים) [צְבוֹיִם] וּמֶלֶךְ בֶּלַע הִיא־צֹעַר׃
أَنَّ هَؤُلَاءِ صَنَعُوا حَرْبًا مَعَ بَارَعَ مَلِكِ سَدُومَ، وَبِرْشَاعَ مَلِكِ عَمُورَةَ، وَشِنْآبَ مَلِكِ أَدْمَةَ، وَشِمْئِيبَرَ مَلِكِ صَبُويِيمَ، وَمَلِكِ بَالَعَ ٱلَّتِي هِيَ صُوغَرُ.
14:3כׇּל־אֵלֶּה חָבְרוּ אֶל־עֵמֶק הַשִּׂדִּים הוּא יָם הַמֶּלַח׃
جَمِيعُ هَؤُلَاءِ ٱجْتَمَعُوا مُتَعَاهِدِينَ إِلَى عُمْقِ ٱلسِّدِّيمِ ٱلَّذِي هُوَ بَحْرُ ٱلْمِلْحِ.
14:4שְׁתֵּים עֶשְׂרֵה שָׁנָה עָבְדוּ אֶת־כְּדׇרְלָעֹמֶר וּשְׁלֹשׁ־עֶשְׂרֵה שָׁנָה מָרָדוּ׃
اِثْنَتَيْ عَشَرَةَ سَنَةً ٱسْتُعْبِدُوا لِكَدَرْلَعَوْمَرَ، وَٱلسَّنَةَ ٱلثَّالِثَةَ عَشَرَةَ عَصَوْا عَلَيْهِ.
14:5וּבְאַרְבַּע עֶשְׂרֵה שָׁנָה בָּא כְדׇרְלָעֹמֶר וְהַמְּלָכִים אֲשֶׁר אִתּוֹ וַיַּכּוּ אֶת־רְפָאִים בְּעַשְׁתְּרֹת קַרְנַיִם וְאֶת־הַזּוּזִים בְּהָם וְאֵת הָאֵימִים בְּשָׁוֵה קִרְיָתָיִם׃
وَفِي ٱلسَّنَةِ ٱلرَّابِعَةَ عَشَرْةَ أَتَى كَدَرْلَعَوْمَرُ وَٱلْمُلُوكُ ٱلَّذِينَ مَعَهُ وَضَرَبُوا ٱلرَّفَائِيِّينَ فِي عَشْتَارُوثَ قَرْنَايِمَ، وَٱلزُّوزِيِّينَ فِي هَامَ، وَٱلْإِيمِيِّينَ فِي شَوَى قَرْيَتَايِمَ،
14:6וְאֶת־הַחֹרִי בְּהַרְרָם שֵׂעִיר עַד אֵיל פָּארָן אֲשֶׁר עַל־הַמִּדְבָּר׃
وَٱلْحُورِيِّينَ فِي جَبَلِهِمْ سَعِيرَ إِلَى بُطْمَةِ فَارَانَ ٱلَّتِي عِنْدَ ٱلْبَرِّيَّةِ.
14:7וַיָּשֻׁבוּ וַיָּבֹאוּ אֶל־עֵין מִשְׁפָּט הִוא קָדֵשׁ וַיַּכּוּ אֶת־כׇּל־שְׂדֵה הָעֲמָלֵקִי וְגַם אֶת־הָאֱמֹרִי הַיֹּשֵׁב בְּחַצְצֹן תָּמָר׃
ثُمَّ رَجَعُوا وَجَاءُوا إِلَى عَيْنِ مِشْفَاطَ ٱلَّتِي هِيَ قَادِشُ. وَضَرَبُوا كُلَّ بِلَادِ ٱلْعَمَالِقَةِ، وَأَيْضًا ٱلْأَمُورِيِّينَ ٱلسَّاكِنِينَ فِي حَصُّونَ تَامَارَ.
14:8וַיֵּצֵא מֶלֶךְ־סְדֹם וּמֶלֶךְ עֲמֹרָה וּמֶלֶךְ אַדְמָה וּמֶלֶךְ (צביים) [צְבוֹיִם] וּמֶלֶךְ בֶּלַע הִוא־צֹעַר וַיַּעַרְכוּ אִתָּם מִלְחָמָה בְּעֵמֶק הַשִּׂדִּים׃
فَخَرَجَ مَلِكُ سَدُومَ، وَمَلِكُ عَمُورَةَ، وَمَلِكُ أَدْمَةَ، وَمَلِكُ صَبُويِيمَ، وَمَلِكُ بَالَعَ، ٱلَّتِي هِيَ صُوغَرُ، وَنَظَمُوا حَرْبًا مَعَهُمْ فِي عُمْقِ ٱلسِّدِّيمِ.
14:9אֵת כְּדׇרְלָעֹמֶר מֶלֶךְ עֵילָם וְתִדְעָל מֶלֶךְ גּוֹיִם וְאַמְרָפֶל מֶלֶךְ שִׁנְעָר וְאַרְיוֹךְ מֶלֶךְ אֶלָּסָר אַרְבָּעָה מְלָכִים אֶת־הַחֲמִשָּׁה׃
مَعَ كَدَرْلَعَوْمَرَ مَلِكِ عِيلَامَ، وَتِدْعَالَ مَلِكِ جُويِيمَ، وَأَمْرَافَلَ مَلِكِ شِنْعَارَ، وَأَرْيُوكَ مَلِكِ أَلَّاسَارَ. أَرْبَعَةُ مُلُوكٍ مَعَ خَمْسَةٍ.

וכאשר ניסה ריבונו את אברהם בדברים, וקיימם, אמר: ״הנני עושה אותך מנהיג לבני האדם״. אמר: ״וגם מזרעי?״ אמר: ״בריתי לא תגיע לעושי העוול״. (124) וכאשר עשינו את הבית מקום מועד לבני האדם ומבטחים: ״ועשו לכם ממקום עמידתו של אברהם מקום תפילה״. וציווינו את אברהם ואת ישמעאל: ״טהרו את ביתי לסובבים, למתבודדים, לכורעים ולמשתחווים״. (125) וכאשר אמר אברהם: ״ריבוני, עשה את זו עיר מבטחים, ופרנס את יושביה מן הפירות – את מי מהם שמאמין באללה וביום האחרון״. אמר: ״ומי שכופר – אניח לו ליהנות מעט, ואחר כך אכפה אותו אל עונש האש; ורעה השיבה״. (126) וכאשר הרים אברהם את יסודות הבית, וישמעאל עמו: ״ריבוננו, קבל ממנו, כי אתה השומע, היודע. (127) ריבוננו, עשנו מתמסרים לך, ומזרענו אומה מתמסרת לך, והראנו את מנהגי פולחננו, ושוב אלינו ברחמים, כי אתה המקבל בתשובה, הרחום״. (128) ״ריבוננו, שלח בהם שליח מקרבם, אשר יקרא לפניהם את אותותיך וילמדם את הספר ואת החכמה ויטהרם; כי אתה האדיר, החכם״. (129)

إبراهيم وإسماعيل عند البيت؛ الدعاء للذرية – مقابلة موضوعية

בראשית יז, א–ח ⇄ 14:35–41

וכאשר אמר אברהם: ״ריבוני, עשה את העיר הזאת מבטחים, והרחק אותי ואת בניי מעבודת הפסלים. (35) ריבוני, הם התעו רבים מבני האדם. מי שהולך אחריי – הוא ממני, ומי שממרה אותי – אתה סולח, רחום. (36) ״ריבוננו, הושבתי מזרעי בעמק אשר אין בו זרע, ליד ביתך המקודש, ריבוננו, למען יקיימו את התפילה. לכן הטה אליהם לבבות מבני האדם, ופרנסם מן הפירות, למען יודו״. (37) ריבוננו, אתה יודע את אשר נסתיר ואת אשר נגלה; ואין נסתר מאללה דבר בארץ ובשמים. (38) השבח לאללה אשר נתן לי לעת זקנה את ישמעאל ואת יצחק; ריבוני שומע תפילה. (39) ריבוני, עשני מקיים את התפילה, וגם מזרעי; ריבוננו, וקבל את תפילתי. (40) ריבוננו, סלח לי ולהוריי ולמאמינים ביום שבו יקום החשבון״. (41)

وَإِذۡ قَالَ إِبۡرَٰهِيمُ رَبِّ ٱجۡعَلۡ هَٰذَا ٱلۡبَلَدَ ءَامِنٗا وَٱجۡنُبۡنِي وَبَنِيَّ أَن نَّعۡبُدَ ٱلۡأَصۡنَامَ (35) رَبِّ إِنَّهُنَّ أَضۡلَلۡنَ كَثِيرٗا مِّنَ ٱلنَّاسِۖ فَمَن تَبِعَنِي فَإِنَّهُۥ مِنِّيۖ وَمَنۡ عَصَانِي فَإِنَّكَ غَفُورٞ رَّحِيمٞ (36) رَّبَّنَآ إِنِّيٓ أَسۡكَنتُ مِن ذُرِّيَّتِي بِوَادٍ غَيۡرِ ذِي زَرۡعٍ عِندَ بَيۡتِكَ ٱلۡمُحَرَّمِ رَبَّنَا لِيُقِيمُواْ ٱلصَّلَوٰةَ فَٱجۡعَلۡ أَفۡـِٔدَةٗ مِّنَ ٱلنَّاسِ تَهۡوِيٓ إِلَيۡهِمۡ وَٱرۡزُقۡهُم مِّنَ ٱلثَّمَرَٰتِ لَعَلَّهُمۡ يَشۡكُرُونَ (37) رَبَّنَآ إِنَّكَ تَعۡلَمُ مَا نُخۡفِي وَمَا نُعۡلِنُۗ وَمَا يَخۡفَىٰ عَلَى ٱللَّهِ مِن شَيۡءٖ فِي ٱلۡأَرۡضِ وَلَا فِي ٱلسَّمَآءِ (38) ٱلۡحَمۡدُ لِلَّهِ ٱلَّذِي وَهَبَ لِي عَلَى ٱلۡكِبَرِ إِسۡمَٰعِيلَ وَإِسۡحَٰقَۚ إِنَّ رَبِّي لَسَمِيعُ ٱلدُّعَآءِ (39) رَبِّ ٱجۡعَلۡنِي مُقِيمَ ٱلصَّلَوٰةِ وَمِن ذُرِّيَّتِيۚ رَبَّنَا وَتَقَبَّلۡ دُعَآءِ (40) رَبَّنَا ٱغۡفِرۡ لِي وَلِوَٰلِدَيَّ وَلِلۡمُؤۡمِنِينَ يَوۡمَ يَقُومُ ٱلۡحِسَابُ (41)

17:16וּבֵרַכְתִּי אֹתָהּ וְגַם נָתַתִּי מִמֶּנָּה לְךָ בֵּן וּבֵרַכְתִּיהָ וְהָיְתָה לְגוֹיִם מַלְכֵי עַמִּים מִמֶּנָּה יִהְיוּ׃
وَأُبَارِكُهَا وَأُعْطِيكَ أَيْضًا مِنْهَا ٱبْنًا. أُبَارِكُهَا فَتَكُونُ أُمَمًا، وَمُلُوكُ شُعُوبٍ مِنْهَا يَكُونُونَ».
17:17וַיִּפֹּל אַבְרָהָם עַל־פָּנָיו וַיִּצְחָק וַיֹּאמֶר בְּלִבּוֹ הַלְּבֶן מֵאָה־שָׁנָה יִוָּלֵד וְאִם־שָׂרָה הֲבַת־תִּשְׁעִים שָׁנָה תֵּלֵד׃
فَسَقَطَ إِبْرَاهِيمُ عَلَى وَجْهِهِ وَضَحِكَ، وَقَالَ فِي قَلْبِهِ: «هَلْ يُولَدُ لِٱبْنِ مِئَةِ سَنَةٍ؟ وَهَلْ تَلِدُ سَارَةُ وَهِيَ بِنْتُ تِسْعِينَ سَنَةً؟».
17:18וַיֹּאמֶר אַבְרָהָם אֶל־הָאֱלֹהִים לוּ יִשְׁמָעֵאל יִחְיֶה לְפָנֶיךָ׃
وَقَالَ إِبْرَاهِيمُ لِلهِ: «لَيْتَ إِسْمَاعِيلَ يَعِيشُ أَمَامَكَ!».
17:19וַיֹּאמֶר אֱלֹהִים אֲבָל שָׂרָה אִשְׁתְּךָ יֹלֶדֶת לְךָ בֵּן וְקָרָאתָ אֶת־שְׁמוֹ יִצְחָק וַהֲקִמֹתִי אֶת־בְּרִיתִי אִתּוֹ לִבְרִית עוֹלָם לְזַרְעוֹ אַחֲרָיו׃
فَقَالَ ٱللهُ: «بَلْ سَارَةُ ٱمْرَأَتُكَ تَلِدُ لَكَ ٱبْنًا وَتَدْعُو ٱسْمَهُ إِسْحَاقَ. وَأُقِيمُ عَهْدِي مَعَهُ عَهْدًا أَبَدِيًّا لِنَسْلِهِ مِنْ بَعْدِهِ.
17:20וּלְיִשְׁמָעֵאל שְׁמַעְתִּיךָ הִנֵּה בֵּרַכְתִּי אֹתוֹ וְהִפְרֵיתִי אֹתוֹ וְהִרְבֵּיתִי אֹתוֹ בִּמְאֹד מְאֹד שְׁנֵים־עָשָׂר נְשִׂיאִם יוֹלִיד וּנְתַתִּיו לְגוֹי גָּדוֹל׃
وَأَمَّا إِسْمَاعِيلُ فَقَدْ سَمِعْتُ لَكَ فِيهِ. هَا أَنَا أُبَارِكُهُ وَأُثْمِرُهُ وَأُكَثِّرُهُ كَثِيرًا جِدًّا. اِثْنَيْ عَشَرَ رَئِيسًا يَلِدُ، وَأَجْعَلُهُ أُمَّةً كَبِيرَةً.
17:21וְאֶת־בְּרִיתִי אָקִים אֶת־יִצְחָק אֲשֶׁר תֵּלֵד לְךָ שָׂרָה לַמּוֹעֵד הַזֶּה בַּשָּׁנָה הָאַחֶרֶת׃

17:22וַיְכַל לְדַבֵּר אִתּוֹ וַיַּעַל אֱלֹהִים מֵעַל אַבְרָהָם׃

17:23וַיִּקַּח אַבְרָהָם אֶת־יִשְׁמָעֵאל בְּנוֹ וְאֵת כׇּל־יְלִידֵי בֵיתוֹ וְאֵת כׇּל־מִקְנַת כַּסְפּוֹ כׇּל־זָכָר בְּאַנְשֵׁי בֵּית אַבְרָהָם וַיָּמׇל אֶת־בְּשַׂר עׇרְלָתָם בְּעֶצֶם הַיּוֹם הַזֶּה כַּאֲשֶׁר דִּבֶּר אִתּוֹ אֱלֹהִים׃

17:25וְיִשְׁמָעֵאל בְּנוֹ בֶּן־שְׁלֹשׁ עֶשְׂרֵה שָׁנָה בְּהִמֹּלוֹ אֵת בְּשַׂר עׇרְלָתוֹ׃
وَكَانَ إِسْمَاعِيلُ ٱبْنُهُ ٱبْنَ ثَلَاثَ عَشَرَةَ سَنَةً حِينَ خُتِنَ فِي لَحْمِ غُرْلَتِهِ.
17:26בְּעֶצֶם הַיּוֹם הַזֶּה נִמּוֹל אַבְרָהָם וְיִשְׁמָעֵאל בְּנוֹ׃
فِي ذَلِكَ ٱلْيَوْمِ عَيْنِهِ خُتِنَ إِبْرَاهِيمُ وَإِسْمَاعِيلُ ٱبْنُهُ.
17:27וְכׇל־אַנְשֵׁי בֵיתוֹ יְלִיד בָּיִת וּמִקְנַת־כֶּסֶף מֵאֵת בֶּן־נֵכָר נִמֹּלוּ אִתּוֹ׃
وَكُلُّ رِجَالِ بَيْتِهِ وِلْدَانِ ٱلْبَيْتِ وَٱلْمُبْتَاعِينَ بِٱلْفِضَّةِ مِنِ ٱبْنِ ٱلْغَرِيبِ خُتِنُوا مَعَهُ.
