41:1וַיְהִי מִקֵּץ שְׁנָתַיִם יָמִים וּפַרְעֹה חֹלֵם וְהִנֵּה עֹמֵד עַל־הַיְאֹר׃
وَحَدَثَ مِنْ بَعْدِ سَنَتَيْنِ مِنَ ٱلزَّمَانِ أَنَّ فِرْعَوْنَ رَأَى حُلْمًا: وَإِذَا هُوَ وَاقِفٌ عِنْدَ ٱلنَّهْرِ،

בראשית מא, א–לו ⇄ 12:43–49

ואמר המלך: ״רואה אני שבע פרות בריאות שאוכלות אותן שבע כחושות, ושבע שיבולים ירוקות ואחרות יבשות. השרים, חוו דעתכם על חזוני, אם פותרי חזיונות אתם.״ (43) אמרו: ״בליל חלומות! ואנו איננו יודעים בפתרון חלומות.״ (44) ואמר זה מהם שניצל, ונזכר לאחר זמן: ״אני אגיד לכם את פתרונו; שלחוני.״ (45) ״יוסף, הצדיק, חווה לנו דעה על שבע פרות בריאות שאוכלות אותן שבע כחושות, ושבע שיבולים ירוקות ואחרות יבשות, למען אשוב אל האנשים, למען ידעו.״ (46) אמר: ״תזרעו שבע שנים כדרככם; ואת אשר תקצרו – השאירוהו בשיבולו, מלבד מעט ממה שתאכלו. (47) ואחר כך יבואו שבע שנים קשות, שיאכלו את אשר הכנתם להן, מלבד מעט ממה שתצפינו. (48) ואחר כך תבוא שנה שבה יוושעו האנשים, ובה יסחטו.״ (49)

وَقَالَ ٱلۡمَلِكُ إِنِّيٓ أَرَىٰ سَبۡعَ بَقَرَٰتٖ سِمَانٖ يَأۡكُلُهُنَّ سَبۡعٌ عِجَافٞ وَسَبۡعَ سُنۢبُلَٰتٍ خُضۡرٖ وَأُخَرَ يَابِسَٰتٖۖ يَـٰٓأَيُّهَا ٱلۡمَلَأُ أَفۡتُونِي فِي رُءۡيَٰيَ إِن كُنتُمۡ لِلرُّءۡيَا تَعۡبُرُونَ (43) قَالُوٓاْ أَضۡغَٰثُ أَحۡلَٰمٖۖ وَمَا نَحۡنُ بِتَأۡوِيلِ ٱلۡأَحۡلَٰمِ بِعَٰلِمِينَ (44) وَقَالَ ٱلَّذِي نَجَا مِنۡهُمَا وَٱدَّكَرَ بَعۡدَ أُمَّةٍ أَنَا۠ أُنَبِّئُكُم بِتَأۡوِيلِهِۦ فَأَرۡسِلُونِ (45) يُوسُفُ أَيُّهَا ٱلصِّدِّيقُ أَفۡتِنَا فِي سَبۡعِ بَقَرَٰتٖ سِمَانٖ يَأۡكُلُهُنَّ سَبۡعٌ عِجَافٞ وَسَبۡعِ سُنۢبُلَٰتٍ خُضۡرٖ وَأُخَرَ يَابِسَٰتٖ لَّعَلِّيٓ أَرۡجِعُ إِلَى ٱلنَّاسِ لَعَلَّهُمۡ يَعۡلَمُونَ (46) قَالَ تَزۡرَعُونَ سَبۡعَ سِنِينَ دَأَبٗا فَمَا حَصَدتُّمۡ فَذَرُوهُ فِي سُنۢبُلِهِۦٓ إِلَّا قَلِيلٗا مِّمَّا تَأۡكُلُونَ (47) ثُمَّ يَأۡتِي مِنۢ بَعۡدِ ذَٰلِكَ سَبۡعٞ شِدَادٞ يَأۡكُلۡنَ مَا قَدَّمۡتُمۡ لَهُنَّ إِلَّا قَلِيلٗا مِّمَّا تُحۡصِنُونَ (48) ثُمَّ يَأۡتِي مِنۢ بَعۡدِ ذَٰلِكَ عَامٞ فِيهِ يُغَاثُ ٱلنَّاسُ وَفِيهِ يَعۡصِرُونَ (49)

أحلام الملك – مقابلة مباشرة

41:2וְהִנֵּה מִן־הַיְאֹר עֹלֹת שֶׁבַע פָּרוֹת יְפוֹת מַרְאֶה וּבְרִיאֹת בָּשָׂר וַתִּרְעֶינָה בָּאָחוּ׃
وَهُوَذَا سَبْعُ بَقَرَاتٍ طَالِعَةٍ مِنَ ٱلنَّهْرِ حَسَنَةِ ٱلْمَنْظَرِ وَسَمِينَةِ ٱللَّحْمِ، فَٱرْتَعَتْ فِي رَوْضَةٍ.
41:3וְהִנֵּה שֶׁבַע פָּרוֹת אֲחֵרוֹת עֹלוֹת אַחֲרֵיהֶן מִן־הַיְאֹר רָעוֹת מַרְאֶה וְדַקּוֹת בָּשָׂר וַתַּעֲמֹדְנָה אֵצֶל הַפָּרוֹת עַל־שְׂפַת הַיְאֹר׃
ثُمَّ هُوَذَا سَبْعُ بَقَرَاتٍ أُخْرَى طَالِعَةٍ وَرَاءَهَا مِنَ ٱلنَّهْرِ قَبِيحَةِ ٱلْمَنْظَرِ وَرَقِيقَةِ ٱللَّحْمِ، فَوَقَفَتْ بِجَانِبِ ٱلْبَقَرَاتِ ٱلْأُولَى عَلَى شَاطِئِ ٱلنَّهْرِ،
41:4וַתֹּאכַלְנָה הַפָּרוֹת רָעוֹת הַמַּרְאֶה וְדַקֹּת הַבָּשָׂר אֵת שֶׁבַע הַפָּרוֹת יְפֹת הַמַּרְאֶה וְהַבְּרִיאֹת וַיִּיקַץ פַּרְעֹה׃
فَأَكَلَتِ ٱلْبَقَرَاتُ ٱلْقَبِيحَةُ ٱلْمَنْظَرِ وَٱلرَّقِيقَةُ ٱللَّحْمِ ٱلْبَقَرَاتِ ٱلسَّبْعَ ٱلْحَسَنَةَ ٱلْمَنْظَرِ وَٱلسَّمِينَةَ. وَٱسْتَيْقَظَ فِرْعَوْنُ.
41:5וַיִּישָׁן וַיַּחֲלֹם שֵׁנִית וְהִנֵּה שֶׁבַע שִׁבֳּלִים עֹלוֹת בְּקָנֶה אֶחָד בְּרִיאוֹת וְטֹבוֹת׃
ثُمَّ نَامَ فَحَلُمَ ثَانِيَةً: وَهُوَذَا سَبْعُ سَنَابِلَ طَالِعَةٍ فِي سَاقٍ وَاحِدٍ سَمِينَةٍ وَحَسَنَةٍ.
41:6וְהִנֵּה שֶׁבַע שִׁבֳּלִים דַּקּוֹת וּשְׁדוּפֹת קָדִים צֹמְחוֹת אַחֲרֵיהֶן׃
ثُمَّ هُوَذَا سَبْعُ سَنَابِلَ رَقِيقَةٍ وَمَلْفُوحَةٍ بِٱلرِّيحِ ٱلشَّرْقِيَّةِ نَابِتَةٍ وَرَاءَهَا.
41:7וַתִּבְלַעְנָה הַשִּׁבֳּלִים הַדַּקּוֹת אֵת שֶׁבַע הַשִּׁבֳּלִים הַבְּרִיאוֹת וְהַמְּלֵאוֹת וַיִּיקַץ פַּרְעֹה וְהִנֵּה חֲלוֹם׃
فَٱبْتَلَعَتِ ٱلسَّنَابِلُ ٱلرَّقِيقَةُ ٱلسَّنَابِلَ ٱلسَّبْعَ ٱلسَّمِينَةَ ٱلْمُمْتَلِئَةَ. وَٱسْتَيْقَظَ فِرْعَوْنُ، وَإِذَا هُوَ حُلْمٌ.
41:8וַיְהִי בַבֹּקֶר וַתִּפָּעֶם רוּחוֹ וַיִּשְׁלַח וַיִּקְרָא אֶת־כׇּל־חַרְטֻמֵּי מִצְרַיִם וְאֶת־כׇּל־חֲכָמֶיהָ וַיְסַפֵּר פַּרְעֹה לָהֶם אֶת־חֲלֹמוֹ וְאֵין־פּוֹתֵר אוֹתָם לְפַרְעֹה׃
وَكَانَ فِي ٱلصَّبَاحِ أَنَّ نَفْسَهُ ٱنْزَعَجَتْ، فَأَرْسَلَ وَدَعَا جَمِيعَ سَحَرَةِ مِصْرَ وَجَمِيعَ حُكَمَائِهَا. وَقَصَّ عَلَيْهِمْ فِرْعَوْنُ حُلْمَهُ، فَلَمْ يَكُنْ مَنْ يُعَبِّرُهُ لِفِرْعَوْنَ.
41:9וַיְדַבֵּר שַׂר הַמַּשְׁקִים אֶת־פַּרְעֹה לֵאמֹר אֶת־חֲטָאַי אֲנִי מַזְכִּיר הַיּוֹם׃
ثُمَّ كَلَّمَ رَئِيسُ ٱلسُّقَاةِ فِرْعَوْنَ قَائِلًا: «أَنَا أَتَذَكَّرُ ٱلْيَوْمَ خَطَايَايَ.
41:10פַּרְעֹה קָצַף עַל־עֲבָדָיו וַיִּתֵּן אֹתִי בְּמִשְׁמַר בֵּית שַׂר הַטַּבָּחִים אֹתִי וְאֵת שַׂר הָאֹפִים׃
فِرْعَوْنُ سَخَطَ عَلَى عَبْدَيْهِ، فَجَعَلَنِي فِي حَبْسِ بَيْتِ رَئِيسِ ٱلشُّرَطِ أَنَا وَرَئِيسَ ٱلْخَبَّازِينَ.
41:11וַנַּחַלְמָה חֲלוֹם בְּלַיְלָה אֶחָד אֲנִי וָהוּא אִישׁ כְּפִתְרוֹן חֲלֹמוֹ חָלָמְנוּ׃
فَحَلُمْنَا حُلْمًا فِي لَيْلَةٍ وَاحِدَةٍ أَنَا وَهُوَ. حَلُمْنَا كُلُّ وَاحِدٍ بِحَسَبِ تَعْبِيرِ حُلْمِهِ.
41:12וְשָׁם אִתָּנוּ נַעַר עִבְרִי עֶבֶד לְשַׂר הַטַּבָּחִים וַנְּסַפֶּר־לוֹ וַיִּפְתׇּר־לָנוּ אֶת־חֲלֹמֹתֵינוּ אִישׁ כַּחֲלֹמוֹ פָּתָר׃
وَكَانَ هُنَاكَ مَعَنَا غُلَامٌ عِبْرَانِيٌّ عَبْدٌ لِرَئِيسِ ٱلشُّرَطِ، فَقَصَصْنَا عَلَيْهِ، فَعَبَّرَ لَنَا حُلْمَيْنَا. عَبَّرَ لِكُلِّ وَاحِدٍ بِحَسَبِ حُلْمِهِ.
41:13וַיְהִי כַּאֲשֶׁר פָּתַר־לָנוּ כֵּן הָיָה אֹתִי הֵשִׁיב עַל־כַּנִּי וְאֹתוֹ תָלָה׃
وَكَمَا عَبَّرَ لَنَا هَكَذَا حَدَثَ. رَدَّنِي أَنَا إِلَى مَقَامِي، وَأَمَّا هُوَ فَعَلَّقَهُ».
41:14וַיִּשְׁלַח פַּרְעֹה וַיִּקְרָא אֶת־יוֹסֵף וַיְרִיצֻהוּ מִן־הַבּוֹר וַיְגַלַּח וַיְחַלֵּף שִׂמְלֹתָיו וַיָּבֹא אֶל־פַּרְעֹה׃
فَأَرْسَلَ فِرْعَوْنُ وَدَعَا يُوسُفَ، فَأَسْرَعُوا بِهِ مِنَ ٱلسِّجْنِ. فَحَلَقَ وَأَبْدَلَ ثِيَابَهُ وَدَخَلَ عَلَى فِرْعَوْنَ.

בראשית מא, יד–טז ⇄ 12:43–49

ואמר המלך: ״רואה אני שבע פרות בריאות שאוכלות אותן שבע כחושות, ושבע שיבולים ירוקות ואחרות יבשות. השרים, חוו דעתכם על חזוני, אם פותרי חזיונות אתם.״ (43) אמרו: ״בליל חלומות! ואנו איננו יודעים בפתרון חלומות.״ (44) ואמר זה מהם שניצל, ונזכר לאחר זמן: ״אני אגיד לכם את פתרונו; שלחוני.״ (45) ״יוסף, הצדיק, חווה לנו דעה על שבע פרות בריאות שאוכלות אותן שבע כחושות, ושבע שיבולים ירוקות ואחרות יבשות, למען אשוב אל האנשים, למען ידעו.״ (46) אמר: ״תזרעו שבע שנים כדרככם; ואת אשר תקצרו – השאירוהו בשיבולו, מלבד מעט ממה שתאכלו. (47) ואחר כך יבואו שבע שנים קשות, שיאכלו את אשר הכנתם להן, מלבד מעט ממה שתצפינו. (48) ואחר כך תבוא שנה שבה יוושעו האנשים, ובה יסחטו.״ (49)

قَالُواْ فَمَا جَزَـٰٓؤُهُۥٓ إِن كُنتُمۡ كَٰذِبِينَ (74) قَالُواْ جَزَـٰٓؤُهُۥ مَن وُجِدَ فِي رَحۡلِهِۦ فَهُوَ جَزَـٰٓؤُهُۥۚ كَذَٰلِكَ نَجۡزِي ٱلظَّـٰلِمِينَ (75)

44:10וַיֹּאמֶר גַּם־עַתָּה כְדִבְרֵיכֶם כֶּן־הוּא אֲשֶׁר יִמָּצֵא אִתּוֹ יִהְיֶה־לִּי עָבֶד וְאַתֶּם תִּהְיוּ נְקִיִּם׃
فَقَالَ: «نَعَمِ، ٱلْآنَ بِحَسَبِ كَلَامِكُمْ هَكَذَا يَكُونُ. ٱلَّذِي يُوجَدُ مَعَهُ يَكُونُ لِي عَبْدًا، وَأَمَّا أَنْتُمْ فَتَكُونُونَ أَبْرِيَاءَ».

בראשית מד, י ⇄ 12:74–75

אמרו: ״ומה עונשו, אם שקרנים אתם?״ (74) אמרו: ״עונשו – מי שיימצא באמתחתו, הוא עצמו יהיה עונשו. כך אנו גומלים לעושי העוול.״ (75)

قَالُواْ فَمَا جَزَـٰٓؤُهُۥٓ إِن كُنتُمۡ كَٰذِبِينَ (74) قَالُواْ جَزَـٰٓؤُهُۥ مَن وُجِدَ فِي رَحۡلِهِۦ فَهُوَ جَزَـٰٓؤُهُۥۚ كَذَٰلِكَ نَجۡزِي ٱلظَّـٰلِمِينَ (75)

44:11וַיְמַהֲרוּ וַיּוֹרִדוּ אִישׁ אֶת־אַמְתַּחְתּוֹ אָרְצָה וַיִּפְתְּחוּ אִישׁ אַמְתַּחְתּוֹ׃
فَٱسْتَعْجَلُوا وَأَنْزَلُوا كُلُّ وَاحِدٍ عِدْلَهُ إِلَى ٱلْأَرْضِ، وَفَتَحُوا كُلُّ وَاحِدٍ عِدْلَهُ.
44:12וַיְחַפֵּשׂ בַּגָּדוֹל הֵחֵל וּבַקָּטֹן כִּלָּה וַיִּמָּצֵא הַגָּבִיעַ בְּאַמְתַּחַת בִּנְיָמִן׃
فَفَتَّشَ مُبْتَدِئًا مِنَ ٱلْكَبِيرِ حَتَّى ٱنْتَهَى إِلَى ٱلصَّغِيرِ، فَوُجِدَ ٱلطَّاسُ فِي عِدْلِ بَنْيَامِينَ.
44:13וַיִּקְרְעוּ שִׂמְלֹתָם וַיַּעֲמֹס אִישׁ עַל־חֲמֹרוֹ וַיָּשֻׁבוּ הָעִירָה׃
فَمَزَّقُوا ثِيَابَهُمْ وَحَمَّلَ كُلُّ وَاحِدٍ عَلَى حِمَارِهِ وَرَجَعُوا إِلَى ٱلْمَدِينَةِ.
44:14וַיָּבֹא יְהוּדָה וְאֶחָיו בֵּיתָה יוֹסֵף וְהוּא עוֹדֶנּוּ שָׁם וַיִּפְּלוּ לְפָנָיו אָרְצָה׃
فَدَخَلَ يَهُوذَا وَإِخْوَتُهُ إِلَى بَيْتِ يُوسُفَ وَهُوَ بَعْدُ هُنَاكَ، وَوَقَعُوا أَمَامَهُ عَلَى ٱلْأَرْضِ.
44:15וַיֹּאמֶר לָהֶם יוֹסֵף מָה־הַמַּעֲשֶׂה הַזֶּה אֲשֶׁר עֲשִׂיתֶם הֲלוֹא יְדַעְתֶּם כִּי־נַחֵשׁ יְנַחֵשׁ אִישׁ אֲשֶׁר כָּמֹנִי׃
فَقَالَ لَهُمْ يُوسُفُ: «مَا هَذَا ٱلْفِعْلُ ٱلَّذِي فَعَلْتُمْ؟ أَلَمْ تَعْلَمُوا أَنَّ رَجُلًا مِثْلِي يَتَفَاءَلُ؟»
44:16וַיֹּאמֶר יְהוּדָה מַה־נֹּאמַר לַאדֹנִי מַה־נְּדַבֵּר וּמַה־נִּצְטַדָּק הָאֱלֹהִים מָצָא אֶת־עֲוֺן עֲבָדֶיךָ הִנֶּנּוּ עֲבָדִים לַאדֹנִי גַּם־אֲנַחְנוּ גַּם אֲשֶׁר־נִמְצָא הַגָּבִיעַ בְּיָדוֹ׃
فَقَالَ يَهُوذَا: «مَاذَا نَقُولُ لِسَيِّدِي؟ مَاذَا نَتَكَلَّمُ؟ وَبِمَاذَا نَتَبَرَّرُ؟ ٱللهُ قَدْ وَجَدَ إِثْمَ عَبِيدِكَ. هَا نَحْنُ عَبِيدٌ لِسَيِّدِي، نَحْنُ وَٱلَّذِي وُجِدَ ٱلطَّاسُ فِي يَدِهِ جَمِيعًا».

בראשית מד, טז ⇄ 42:30; 4:79

42:30: וכל אסון שפוגע בכם – בעבור מה שעשו ידיכם; והוא מוחל על הרבה. (30) ✦ 4:79: כל טוב שמגיע אליך – מאללה הוא, וכל רעה שפוגעת בך – מעצמך היא. ושלחנוך לבני האדם כשליח; ודי באללה כעד. (79)

وَمَآ أَصَٰبَكُم مِّن مُّصِيبَةٖ فَبِمَا كَسَبَتۡ أَيۡدِيكُمۡ وَيَعۡفُواْ عَن كَثِيرٖ (30) ✦ مَّآ أَصَابَكَ مِنۡ حَسَنَةٖ فَمِنَ ٱللَّهِۖ وَمَآ أَصَابَكَ مِن سَيِّئَةٖ فَمِن نَّفۡسِكَۚ وَأَرۡسَلۡنَٰكَ لِلنَّاسِ رَسُولٗاۚ وَكَفَىٰ بِٱللَّهِ شَهِيدٗا (79)

״מה נאמר לאדוני... האלוהים מצא את עוון עבדיך״ — הודאת יהודה באשמה הישנה (מכירת יוסף) בעת צרה, ברוח ההתבוננות העצמית ב־42:30 (״וכל צרה הפוקדת אתכם – בגלל אשר עשו ידיכם, ועל רב הוא מוחל״).

«الله قد وجد إثم عبيدك» — اعتراف يهوذا بالذنب القديم عند الشدة، على غرار الشورى 42:30 «وما أصابكم من مصيبة فبما كسبت أيديكم ويعفو عن كثير».
