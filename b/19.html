لَن تَنَالُواْ ٱلۡبِرَّ حَتَّىٰ تُنفِقُواْ مِمَّا تُحِبُّونَۚ وَمَا تُنفِقُواْ مِن شَيۡءٖ فَإِنَّ ٱللَّهَ بِهِۦ عَلِيمٞ (92) ✦ صحيح البخاري 1: إِنَّمَا الأَعْمَالُ بِالنِّيَّاتِ

״ויקחו לי תרומה מאת כל איש אשר ידבנו לבו״ – הנדבה נמדדת בלב. בחדית׳ הפותח את צחיח אלבוח׳ארי: ״המעשים – לפי הכוונות״; ובקוראן: ״לא תשיגו את הצדקה עד שתתנו ממה שאתם אוהבים״.

«من كل من يحثه قلبه تأخذون تقدمتي»، مقابل «إنما الأعمال بالنيات» و«لن تنالوا البر حتى تنفقوا مما تحبون».

שמות כה, א–ב ⇄ 2:272; 9:99; 57:10

2:272: לא עליך להדריכם, אלא אללה מדריך את מי שירצה. וכל טוב שתתנו – לנפשכם הוא; ואינכם נותנים אלא כדי לבקש את פני אללה. וכל טוב שתתנו – יושלם לכם, ולא ייעשה לכם עוול; (272) ✦ 9:99: ויש מן הבדווים מי שמאמין באללה וביום האחרון, ורואה את מה שהוא נותן כקרבה (קֻרֻבַּאת) אצל אללה ובתפילות השליח (ברכותיו). אכן, היא קרבה להם; יכניסם אללה ברחמיו; אללה סולח ורחום. (99) ✦ 57:10: ומה לכם שלא תתנו בדרך אללה, ולאללה ירושת השמים והארץ? אינו שווה מי מכם שנתן ונלחם לפני הכיבוש – אלה גדולים במעלה מאלה שנתנו ונלחמו אחר כך; ולכולם הבטיח אללה את הטוב. ואללה יודע את מעשיכם. (10)

۞لَّيۡسَ عَلَيۡكَ هُدَىٰهُمۡ وَلَٰكِنَّ ٱللَّهَ يَهۡدِي مَن يَشَآءُۗ وَمَا تُنفِقُواْ مِنۡ خَيۡرٖ فَلِأَنفُسِكُمۡۚ وَمَا تُنفِقُونَ إِلَّا ٱبۡتِغَآءَ وَجۡهِ ٱللَّهِۚ وَمَا تُنفِقُواْ مِنۡ خَيۡرٖ يُوَفَّ إِلَيۡكُمۡ وَأَنتُمۡ لَا تُظۡلَمُونَ (272) ✦ وَمِنَ ٱلۡأَعۡرَابِ مَن يُؤۡمِنُ بِٱللَّهِ وَٱلۡيَوۡمِ ٱلۡأٓخِرِ وَيَتَّخِذُ مَا يُنفِقُ قُرُبَٰتٍ عِندَ ٱللَّهِ وَصَلَوَٰتِ ٱلرَّسُولِۚ أَلَآ إِنَّهَا قُرۡبَةٞ لَّهُمۡۚ سَيُدۡخِلُهُمُ ٱللَّهُ فِي رَحۡمَتِهِۦٓۚ إِنَّ ٱللَّهَ غَفُورٞ رَّحِيمٞ (99) ✦ وَمَا لَكُمۡ أَلَّا تُنفِقُواْ فِي سَبِيلِ ٱللَّهِ وَلِلَّهِ مِيرَٰثُ ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِۚ لَا يَسۡتَوِي مِنكُم مَّنۡ أَنفَقَ مِن قَبۡلِ ٱلۡفَتۡحِ وَقَٰتَلَۚ أُوْلَـٰٓئِكَ أَعۡظَمُ دَرَجَةٗ مِّنَ ٱلَّذِينَ أَنفَقُواْ مِنۢ بَعۡدُ وَقَٰتَلُواْۚ وَكُلّٗا وَعَدَ ٱللَّهُ ٱلۡحُسۡنَىٰۚ وَٱللَّهُ بِمَا تَعۡمَلُونَ خَبِيرٞ (10)

״מאת כל איש אשר ידבנו לבו תיקחו את תרומתי״ — נתינה מתוך רצון חופשי וטהרת הלב לשם האל בלבד (״אין אתם מוציאים אלא לבקש את פני אללה״, 2:272).

«من كل من يحثّه قلبه تأخذون تقدمتي» — الإنفاق طواعيةً وابتغاء وجه الله لا إكراهًا (البقرة 2:272 والتوبة 9:99).

שמות כה, ב ⇄ בוח׳ארי 1417 (״היזהרו מן האש, ולו בחצי תמר״)

בוח׳ארי 1417: ״היזהרו מן האש, ולו בחצי תמר״.

صحيح البخاري 1417: اتَّقُوا النَّارَ وَلَوْ بِشِقِّ تَمْرَةٍ

התרומה לכל אחד לפי לבו – גם מעט. בחדית׳: ״היזהרו מן האש ולו בחצי תמר״ – גם תרומה זעירה נחשבת.

تقدمة كل واحد بحسب قلبه ولو قليلة، مقابل «اتقوا النار ولو بشق تمرة».

שמות כה, ב ⇄ מוסלם 2588 (״לא חסרה צדקה מן הממון״)

מוסלם 2588: ״לא חסרה צדקה מן הממון״.

صحيح مسلم 2588: مَا نَقَصَتْ صَدَقَةٌ مِنْ مَالٍ

התרומה למשכן אינה גורעת. בחדית׳: ״לא חסרה צדקה מן הממון, ואללה מוסיף לעבד המוחל כבוד, ואין אדם מתענו לאללה אלא שאללה מרוממו״.

التقدمة للمسكن لا تُنقص، مقابل «ما نقصت صدقة من مال».

שמות כה, ב ⇄ בוח׳ארי 1; מוסלם 1907

בוח׳ארי 1: ״המעשים – לפי הכוונות״. ✦ מוסלם 1907: אמר שליח אללה: ״המעשים אינם אלא לפי הכוונה, ולכל אדם רק מה שהתכוון אליו. מי שהגירתו הייתה אל אללה ושליחו – הגירתו אל אללה ושליחו; ומי שהגירתו הייתה כדי להשיג דבר מן העולם הזה או כדי לשאת אישה – הגירתו אל מה שהיגר אליו״.

صحيح البخاري 1: إِنَّمَا الأَعْمَالُ بِالنِّيَّاتِ ✦ صحيح مسلم 1907: قال رسول الله صلى الله عليه وسلم " إنما الأعمال بالنية وإنما لامرئ ما نوى فمن كانت هجرته إلى الله ورسوله فهجرته إلى الله ورسوله ومن كانت هجرته لدنيا يصيبها أو امرأة يتزوجها فهجرته إلى ما هاجر إليه "

״אשר ידבנו לבו״ — הלב והכוונה הם שמעניקים ערך לכל נדבה ומצווה (״אין המעשים אלא לפי הכוונות״, בוח׳ארי 1).

«من يحثّه قلبه» — النية الخالصة أساس قبول كل عبادة ونفقة («إنما الأعمال بالنيات»، البخاري 1).

57:7: האמינו באללה ובשליחו, ותנו ממה שעשה אתכם ממלאי מקום (נאמנים) עליו; ואלה מכם שהאמינו ונתנו – להם שכר גדול. (7)

ءَامِنُواْ بِٱللَّهِ وَرَسُولِهِۦ وَأَنفِقُواْ مِمَّا جَعَلَكُم مُّسۡتَخۡلَفِينَ فِيهِۖ فَٱلَّذِينَ ءَامَنُواْ مِنكُمۡ وَأَنفَقُواْ لَهُمۡ أَجۡرٞ كَبِيرٞ (7)

مواد المسكن من أموال الشعب، مقابل «وأنفقوا مما جعلكم مستخلفين فيه».

שמות כה, ג–ז ⇄ 3:92; 2:267

3:92: לא תשיגו את הצדקה עד שתתנו ממה שאתם אוהבים; וכל דבר שתתנו – אללה יודעו. (92) ✦ 2:267: הוי המאמינים, תנו מן הטוב שהרווחתם וממה שהוצאנו לכם מן הארץ, ואל תבחרו בפגום כדי לתת ממנו, דבר שאתם עצמכם לא הייתם לוקחים אלא בעצימת עין. ודעו כי אללה אינו נצרך, משובח. (267)

لَن تَنَالُواْ ٱلۡبِرَّ حَتَّىٰ تُنفِقُواْ مِمَّا تُحِبُّونَۚ وَمَا تُنفِقُواْ مِن شَيۡءٖ فَإِنَّ ٱللَّهَ بِهِۦ عَلِيمٞ (92) ✦ يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓاْ أَنفِقُواْ مِن طَيِّبَٰتِ مَا كَسَبۡتُمۡ وَمِمَّآ أَخۡرَجۡنَا لَكُم مِّنَ ٱلۡأَرۡضِۖ وَلَا تَيَمَّمُواْ ٱلۡخَبِيثَ مِنۡهُ تُنفِقُونَ وَلَسۡتُم بِـَٔاخِذِيهِ إِلَّآ أَن تُغۡمِضُواْ فِيهِۚ وَٱعۡلَمُوٓاْ أَنَّ ٱللَّهَ غَنِيٌّ حَمِيدٌ (267)

תרומת זהב, כסף, אבני שוהם ומיטב הרכוש למשכן — מול הציווי להפריש מן המובחר והאהוב (״לא תשיגו את הצדקה עד שתוציאו ממה שאתם אוהבים״, 3:92; 2:267).

تقديم أطيب الأموال والنفائس لبيت الله يوازي «لن تنالوا البرّ حتى تنفقوا مما تحبون» (آل عمران 3:92 والبقرة 2:267).

שמות כה, ג–ז ⇄ אבו דאוד 448; נסאאי 689; בוח׳ארי 446

אבו דאוד 448: אמר שליח אללה: ״לא נצטוויתי להגביה ולפאר את המסגדים״. אמר אבן עבאס: ״אתם עוד תקשטו אותם כפי שקישטו היהודים והנוצרים״. ✦ נסאאי 689: אמר הנביא: ״מאותות השעה (קץ הימים) – שבני האדם יתגאו זה על זה במסגדים״. ✦ בוח׳ארי 446: (בימי) שליח אללה (היה המסגד) בנוי לבנים, תקרתו מכפות תמרים ועמודיו מגזעי דקל. אבו בכר לא הוסיף עליו דבר; עמר הרחיב אותו ובנה אותו על יסודותיו כפי שהיה בימי שליח אללה – בלבנים ובכפות תמרים – והחליף את עמודיו בעץ. אחר כך שינה אותו עת׳מאן והרחיבו הרבה: בנה את קירו מאבנים מגולפות ומטיח, עשה את עמודיו מאבנים מגולפות וקירה אותו בעץ טיק.

سنن أبي داود 448: قال رسول الله صلى الله عليه وسلم " ما أمرت بتشييد المساجد " . قال ابن عباس لتزخرفنها كما زخرفت اليهود والنصارى ✦ سنن النسائي 689: أن النبي صلى الله عليه وسلم قال " من أشراط الساعة أن يتباهى الناس في المساجد " ✦ صحيح البخاري 446: رسول الله صلى الله عليه وسلم مبنيا باللبن، وسقفه الجريد، وعمده خشب النخل، فلم يزد فيه أبو بكر شيئا، وزاد فيه عمر وبناه على بنيانه في عهد رسول الله صلى الله عليه وسلم باللبن والجريد، وأعاد عمده خشبا، ثم غيره عثمان، فزاد فيه زيادة كثيرة، وبنى جداره بالحجارة المنقوشة والقصة، وجعل عمده من حجارة منقوشة، وسقفه بالساج

הדר ופאר המשכן בזהב ותכלת (״זה אלי ואנווהו״) — נקודת השוואה מרתקת מול פשטות המסגד הראשון באלמדינה והאזהרה מפני התהדרות מופרזת בבניין מסגדים (אבו דאוד 448).

الزينة والذهب في بناء المسكن مقابل بساطة مسجد النبي ﷺ الأول في المدينة والنهي عن المباهاة بزخرفة المساجد (أبو داود 448 والبخاري 446).

״ועשו לי מקדש ושכנתי בתוכם״ – ״בתוכו לא נאמר אלא בתוכם״. בקוראן: ״בבתים אשר התיר אללה לרוממם ולהזכיר בהם את שמו״; ובחדית׳: ״מי שבונה מסגד לאללה – יבנה לו אללה כמותו בגן העדן״.

«فيصنعون لي مقدسًا لأسكن في وسطهم»، مقابل «في بيوت أذن الله أن تُرفع» و«من بنى مسجدًا… بنى الله له مثله في الجنة».

בוח׳ארי 450: ״מי שבונה מסגד (לאללה – יבנה לו אללה כמותו בגן העדן)״. ✦ מוסלם 533: שמעתי את שליח אללה אומר: ״מי שבנה מסגד לאללה – אמר (המוסר): נדמה לי שאמר: בבקשו בכך את פני אללה – יבנה לו אללה בית בגן העדן״.

صحيح البخاري 450: مَنْ بَنَى مَسْجِدًا ✦ صحيح مسلم 533: سمعت رسول الله صلى الله عليه وسلم يقول " من بنى مسجدا لله تعالى - قال بكير حسبت أنه قال - يبتغي به وجه الله - بنى الله له بيتا في الجنة "

מצוות הקמת בית תפילה וקדושה (״ועשו לי מקדש״) — מקבילה לחדית׳ המפורסם בבוח׳ארי 450: ״מי שבונה מסגד למען פני אללה – בונה לו אללה בית כמותו בגן העדן״.

فريضة بناء بيت لله («فيصنعون لي مقدسًا») توازي الحديث الصحيح: «من بنى مسجدًا يبتغي به وجه الله بنى الله له مثله في الجنة» (البخاري 450).

22:40: אלה שגורשו ממשכנותיהם שלא כדין, רק משום שאמרו: ״ריבוננו אללה״. ולולא הדף אללה את בני האדם אלה באלה – היו נהרסים מנזרים וכנסיות ובתי תפילה (צַלַוַאת) ומסגדים, שבהם נזכר שם אללה הרבה. ואללה ודאי יעזור למי שעוזר לו; אללה חזק ועזוז; (40) ✦ 24:36: בבתים שהתיר אללה שיורמו ויוזכר בהם שמו – משבחים לו בהם בבוקר ובערב (36) ✦ 72:18: ושהמסגדים לאללה – אל תקראו אפוא עם אללה איש; (18)

בתי תפילה המקודשים לזכירת שם האל (״בתים שאללה התיר לרומם ולהזכיר בהם את שמו״, 24:36; והגנת בתי כנסת ומסגדים ב־22:40; ״המסגדים לאללה הם״, 72:18).

قدسية بيوت الله التي يُذكر فيها اسمه (النور 24:36 والجن 72:18)، وذكر القرآن لحماية الصلوات (الكُنس) والمساجد معًا في الحج 22:40.

שמות כה, ח ⇄ 9:18

9:18: רק מי שמאמין באללה וביום האחרון, ומקיים את התפילה ונותן את הזכאת, ואינו ירא אלא מאללה – הוא שיתחזק את מסגדי אללה; ואולי יהיו אלה מן המודרכים. (18)

إِنَّمَا يَعۡمُرُ مَسَٰجِدَ ٱللَّهِ مَنۡ ءَامَنَ بِٱللَّهِ وَٱلۡيَوۡمِ ٱلۡأٓخِرِ وَأَقَامَ ٱلصَّلَوٰةَ وَءَاتَى ٱلزَّكَوٰةَ وَلَمۡ يَخۡشَ إِلَّا ٱللَّهَۖ فَعَسَىٰٓ أُوْلَـٰٓئِكَ أَن يَكُونُواْ مِنَ ٱلۡمُهۡتَدِينَ (18)

בניית המקדש ואחזקתו כעדות לאמונה פעילה — עקרון עִמארת אלמסאג׳ד (״אין מקימים ומתחזקים את מסגדי אללה אלא מי שמאמין באללה וביום האחרון״, 9:18).

عمارة بيوت الله كعلامة على صدق الإيمان («إنما يعمر مساجد الله من آمن بالله واليوم الآخر»، التوبة 9:18).

בוח׳ארי 438: מפי ג׳אבר בן עבדאללה: אמר שליח אללה: ״ניתנו לי חמישה דברים שלא ניתנו לאף אחד מן הנביאים לפניי: נעזרתי באימה – מהלך חודש; הארץ נעשתה לי מסגד ואמצעי טהרה – וכל אדם מאומתי שתשיג אותו (שעת) התפילה, יתפלל (במקומו); הותר לי השלל; כל נביא היה נשלח לעמו בלבד, ואני נשלחתי אל בני האדם כולם; וניתנה לי (זכות) ההשתדלות (השַׁפַאעַה)״. ✦ מוסלם 521: אמר שליח אללה: ״ניתנו לי חמישה דברים שלא ניתנו לאיש לפניי: כל נביא נשלח לעמו בלבד, ואני נשלחתי לכל אדום ושחור; הותר לי השלל, ולא הותר לאיש לפניי; הארץ נעשתה לי טובה, מטהרת ומקום סגידה – כל אדם שהגיעה שעת התפילה יתפלל במקום שבו הוא נמצא; נעזרתי באימה (שהוטלה באויב) למרחק מהלך חודש; וניתנה לי ההשתדלות (השפאעה)״.

המשכן הנייד שהתלווה לעם במדבר — וההרחבה האוניברסלית באסלאם (בוח׳ארי 438): ״נעשתה לי הארץ כולה מקום סגידה וטהרה״, לצד משכן/מקדש מרכזי.

فكرة المسكن المتنقل مع الجماعة في التيه، والانفتاح الكوني في الحديث: «جُعلت لي الأرض مسجدًا وطهورًا» (صحيح البخاري 438).

35:27–28: הלא ראית כי אללה הוריד מן השמיים מים, והוצאנו בהם פירות בצבעים שונים? ומן ההרים – רצועות לבנות ואדומות בצבעים שונים, ושחורות כעורב. (27) ומבני האדם ומבעלי החיים ומן הבהמות – בצבעים שונים, כך גם הם. רק היודעים (אל־עֻלַמַאא׳) מעבדיו יראים את אללה; אללה עזוז וסולח. (28)

26:10וְעָשִׂיתָ חֲמִשִּׁים לֻלָאֹת עַל שְׂפַת הַיְרִיעָה הָאֶחָת הַקִּיצֹנָה בַּחֹבָרֶת וַחֲמִשִּׁים לֻלָאֹת עַל שְׂפַת הַיְרִיעָה הַחֹבֶרֶת הַשֵּׁנִית׃
وَتَصْنَعُ خَمْسِينَ عُرْوَةً عَلَى حَاشِيَةِ ٱلشُّقَّةِ ٱلْوَاحِدَةِ ٱلطَّرَفِيَّةِ مِنَ ٱلْمُوَصَّلِ ٱلْوَاحِدِ، وَخَمْسِينَ عُرْوَةً عَلَى حَاشِيَةِ ٱلشُّقَّةِ مِنَ ٱلْمُوَصَّلِ ٱلثَّانِي.
26:11וְעָשִׂיתָ קַרְסֵי נְחֹשֶׁת חֲמִשִּׁים וְהֵבֵאתָ אֶת־הַקְּרָסִים בַּלֻּלָאֹת וְחִבַּרְתָּ אֶת־הָאֹהֶל וְהָיָה אֶחָד׃
وَتَصْنَعُ خَمْسِينَ شِظَاظًا مِنْ نُحَاسٍ، وَتُدْخِلُ ٱلْأَشِظَّةَ فِي ٱلْعُرَى، وَتَصِلُ ٱلْخَيْمَةَ فَتَصِيرُ وَاحِدَةً.
26:12וְסֶרַח הָעֹדֵף בִּירִיעֹת הָאֹהֶל חֲצִי הַיְרִיעָה הָעֹדֶפֶת תִּסְרַח עַל אֲחֹרֵי הַמִּשְׁכָּן׃
وَأَمَّا ٱلْمُدَلَّى ٱلْفَاضِلُ مِنْ شُقَقِ ٱلْخَيْمَةِ، نِصْفُ ٱلشُّقَّةِ ٱلْمُوَصَّلَةِ ٱلْفَاضِلُ، فَيُدَلَّى عَلَى مُؤَخَّرِ ٱلْمَسْكَنِ.
26:13וְהָאַמָּה מִזֶּה וְהָאַמָּה מִזֶּה בָּעֹדֵף בְּאֹרֶךְ יְרִיעֹת הָאֹהֶל יִהְיֶה סָרוּחַ עַל־צִדֵּי הַמִּשְׁכָּן מִזֶּה וּמִזֶּה לְכַסֹּתוֹ׃
وَٱلذِّرَاعُ مِنْ هُنَا وَٱلذِّرَاعُ مِنْ هُنَاكَ، مِنَ ٱلْفَاضِلِ فِي طُولِ شُقَقِ ٱلْخَيْمَةِ، تَكُونَانِ مُدَلَّاتَيْنِ عَلَى جَانِبَيِ ٱلْمَسْكَنِ مِنْ هُنَا وَمِنْ هُنَاكَ لِتَغْطِيَتِهِ.
26:14וְעָשִׂיתָ מִכְסֶה לָאֹהֶל עֹרֹת אֵילִם מְאׇדָּמִים וּמִכְסֵה עֹרֹת תְּחָשִׁים מִלְמָעְלָה׃
وَتَصْنَعُ غِطَاءً لِلْخَيْمَةِ مِنْ جُلُودِ كِبَاشٍ مُحَمَّرَةٍ، وَغِطَاءً مِنْ جُلُودِ تُخَسٍ مِنْ فَوْقُ.
26:15וְעָשִׂיתָ אֶת־הַקְּרָשִׁים לַמִּשְׁכָּן עֲצֵי שִׁטִּים עֹמְדִים׃
«وَتَصْنَعُ ٱلْأَلْوَاحَ لِلْمَسْكَنِ مِنْ خَشَبِ ٱلسَّنْطِ قَائِمَةً.
26:16עֶשֶׂר אַמּוֹת אֹרֶךְ הַקָּרֶשׁ וְאַמָּה וַחֲצִי הָאַמָּה רֹחַב הַקֶּרֶשׁ הָאֶחָד׃
طُولُ ٱللَّوْحِ عَشَرُ أَذْرُعٍ، وَعَرْضُ ٱللَّوْحِ ٱلْوَاحِدِ ذِرَاعٌ وَنِصْفٌ.
26:17שְׁתֵּי יָדוֹת לַקֶּרֶשׁ הָאֶחָד מְשֻׁלָּבֹת אִשָּׁה אֶל־אֲחֹתָהּ כֵּן תַּעֲשֶׂה לְכֹל קַרְשֵׁי הַמִּשְׁכָּן׃
وَلِلَّوْحِ ٱلْوَاحِدِ رِجْلَانِ مَقْرُونَةٌ إِحْدَاهُمَا بِٱلْأُخْرَى. هَكَذَا تَصْنَعُ لِجَمِيعِ أَلْوَاحِ ٱلْمَسْكَنِ.
26:18וְעָשִׂיתָ אֶת־הַקְּרָשִׁים לַמִּשְׁכָּן עֶשְׂרִים קֶרֶשׁ לִפְאַת נֶגְבָּה תֵימָנָה׃
وَتَصْنَعُ ٱلْأَلْوَاحَ لِلْمَسْكَنِ عِشْرِينَ لَوْحًا إِلَى جِهَةِ ٱلْجَنُوبِ نَحْوَ ٱلتَّيْمَنِ.
26:19וְאַרְבָּעִים אַדְנֵי־כֶסֶף תַּעֲשֶׂה תַּחַת עֶשְׂרִים הַקָּרֶשׁ שְׁנֵי אֲדָנִים תַּחַת־הַקֶּרֶשׁ הָאֶחָד לִשְׁתֵּי יְדֹתָיו וּשְׁנֵי אֲדָנִים תַּחַת־הַקֶּרֶשׁ הָאֶחָד לִשְׁתֵּי יְדֹתָיו׃
وَتَصْنَعُ أَرْبَعِينَ قَاعِدَةً مِنْ فِضَّةٍ تَحْتَ ٱلْعِشْرِينَ لَوْحًا. تَحْتَ ٱللَّوْحِ ٱلْوَاحِدِ قَاعِدَتَانِ لِرِجْلَيْهِ، وَتَحْتَ ٱللَّوْحِ ٱلْوَاحِدِ قَاعِدَتَانِ لِرِجْلَيْهِ.
26:20וּלְצֶלַע הַמִּשְׁכָּן הַשֵּׁנִית לִפְאַת צָפוֹן עֶשְׂרִים קָרֶשׁ׃
وَلِجَانِبِ ٱلْمَسْكَنِ ٱلثَّانِي إِلَى جِهَةِ ٱلشِّمَالِ عِشْرِينَ لَوْحًا.
26:21וְאַרְבָּעִים אַדְנֵיהֶם כָּסֶף שְׁנֵי אֲדָנִים תַּחַת הַקֶּרֶשׁ הָאֶחָד וּשְׁנֵי אֲדָנִים תַּחַת הַקֶּרֶשׁ הָאֶחָד׃
وَأَرْبَعِينَ قَاعِدَةً لَهَا مِنْ فِضَّةٍ. تَحْتَ ٱللَّوْحِ ٱلْوَاحِدِ قَاعِدَتَانِ، وَتَحْتَ ٱللَّوْحِ ٱلْوَاحِدِ قَاعِدَتَانِ.
26:22וּלְיַרְכְּתֵי הַמִּשְׁכָּן יָמָּה תַּעֲשֶׂה שִׁשָּׁה קְרָשִׁים׃
وَلِمُؤَخَّرِ ٱلْمَسْكَنِ نَحْوَ ٱلْغَرْبِ تَصْنَعُ سِتَّةَ أَلْوَاحٍ.
26:23וּשְׁנֵי קְרָשִׁים תַּעֲשֶׂה לִמְקֻצְעֹת הַמִּשְׁכָּן בַּיַּרְכָתָיִם׃
وَتَصْنَعُ لَوْحَيْنِ لِزَاوِيَتَيِ ٱلْمَسْكَنِ فِي ٱلْمُؤَخَّرِ،
26:24וְיִהְיוּ תֹאֲמִם מִלְּמַטָּה וְיַחְדָּו יִהְיוּ תַמִּים עַל־רֹאשׁוֹ אֶל־הַטַּבַּעַת הָאֶחָת כֵּן יִהְיֶה לִשְׁנֵיהֶם לִשְׁנֵי הַמִּקְצֹעֹת יִהְיוּ׃
وَيَكُونَانِ مُزْدَوِجَيْنِ مِنْ أَسْفَلُ. وَعَلَى سَوَاءٍ يَكُونَانِ مُزْدَوِجَيْنِ إِلَى رَأْسِهِ إِلَى ٱلْحَلْقَةِ ٱلْوَاحِدَةِ. هَكَذَا يَكُونُ لِكِلَيْهِمَا. يَكُونَانِ لِلزَّاوِيَتَيْنِ.
26:25וְהָיוּ שְׁמֹנָה קְרָשִׁים וְאַדְנֵיהֶם כֶּסֶף שִׁשָּׁה עָשָׂר אֲדָנִים שְׁנֵי אֲדָנִים תַּחַת הַקֶּרֶשׁ הָאֶחָד וּשְׁנֵי אֲדָנִים תַּחַת הַקֶּרֶשׁ הָאֶחָד׃
فَتَكُونُ ثَمَانِيَةَ أَلْوَاحٍ، وَقَوَاعِدُهَا مِنْ فِضَّةٍ سِتَّ عَشْرَةَ قَاعِدَةً. تَحْتَ ٱللَّوْحِ ٱلْوَاحِدِ قَاعِدَتَانِ، وَتَحْتَ ٱللَّوْحِ ٱلْوَاحِدِ قَاعِدَتَانِ.
26:26וְעָשִׂיתָ בְרִיחִם עֲצֵי שִׁטִּים חֲמִשָּׁה לְקַרְשֵׁי צֶלַע־הַמִּשְׁכָּן הָאֶחָד׃
«وَتَصْنَعُ عَوَارِضَ مِنْ خَشَبِ ٱلسَّنْطِ، خَمْسًا لِأَلْوَاحِ جَانِبِ ٱلْمَسْكَنِ ٱلْوَاحِدِ،
26:27וַחֲמִשָּׁה בְרִיחִם לְקַרְשֵׁי צֶלַע־הַמִּשְׁכָּן הַשֵּׁנִית וַחֲמִשָּׁה בְרִיחִם לְקַרְשֵׁי צֶלַע הַמִּשְׁכָּן לַיַּרְכָתַיִם יָמָּה׃
وَخَمْسَ عَوَارِضَ لِأَلْوَاحِ جَانِبِ ٱلْمَسْكَنِ ٱلثَّانِي، وَخَمْسَ عَوَارِضَ لِأَلْوَاحِ جَانِبِ ٱلْمَسْكَنِ فِي ٱلْمُؤَخَّرِ نَحْوَ ٱلْغَرْبِ.
26:28וְהַבְּרִיחַ הַתִּיכֹן בְּתוֹךְ הַקְּרָשִׁים מַבְרִחַ מִן־הַקָּצֶה אֶל־הַקָּצֶה׃
وَٱلْعَارِضَةُ ٱلْوُسْطَى فِي وَسَطِ ٱلْأَلْوَاحِ تَنْفُذُ مِنَ ٱلطَّرَفِ إِلَى ٱلطَّرَفِ.
26:29וְאֶת־הַקְּרָשִׁים תְּצַפֶּה זָהָב וְאֶת־טַבְּעֹתֵיהֶם תַּעֲשֶׂה זָהָב בָּתִּים לַבְּרִיחִם וְצִפִּיתָ אֶת־הַבְּרִיחִם זָהָב׃
وَتُغَشِّي ٱلْأَلْوَاحَ بِذَهَبٍ، وَتَصْنَعُ حَلَقَاتِهَا مِنْ ذَهَبٍ بُيُوتًا لِلْعَوَارِضِ، وَتُغَشِّي ٱلْعَوَارِضَ بِذَهَبٍ.
26:30וַהֲקֵמֹתָ אֶת־הַמִּשְׁכָּן כְּמִשְׁפָּטוֹ אֲשֶׁר הׇרְאֵיתָ בָּהָר׃
وَتُقِيمُ ٱلْمَسْكَنَ كَرَسْمِهِ ٱلَّذِي أُظْهِرَ لَكَ فِي ٱلْجَبَلِ.
26:31וְעָשִׂיתָ פָרֹכֶת תְּכֵלֶת וְאַרְגָּמָן וְתוֹלַעַת שָׁנִי וְשֵׁשׁ מׇשְׁזָר מַעֲשֵׂה חֹשֵׁב יַעֲשֶׂה אֹתָהּ כְּרֻבִים׃
«وَتَصْنَعُ حِجَابًا مِنْ أَسْمَانْجُونِيٍّ وَأُرْجُوَانٍ وَقِرْمِزٍ وَبُوصٍ مَبْرُومٍ. صَنْعَةَ حَائِكٍ حَاذِقٍ يَصْنَعُهُ بِكَرُوبِيمَ.

שמות כו, לא–לג ⇄ קוראן 42:51 (״לא ניתן לאדם שידבר אליו אללה אלא בהתגלות או מאחורי מחיצה (חִג׳אבּ)…״)

42:51: ואין לבן אדם שאללה ידבר אליו אלא בהשראה, או מאחורי מסך, או שישלח שליח (מלאך) וזה ישרה ברשותו את אשר ירצה; הוא נעלה וחכם. (51)

۞وَمَا كَانَ لِبَشَرٍ أَن يُكَلِّمَهُ ٱللَّهُ إِلَّا وَحۡيًا أَوۡ مِن وَرَآيِٕ حِجَابٍ أَوۡ يُرۡسِلَ رَسُولٗا فَيُوحِيَ بِإِذۡنِهِۦ مَا يَشَآءُۚ إِنَّهُۥ عَلِيٌّ حَكِيمٞ (51)

״והבדילה הפרכת לכם בין הקדש ובין קדש הקדשים״ – מחיצה בין האדם לבין מקום השכינה. בקוראן האל מדבר אל האדם ״בהתגלות, או מאחורי מחיצה, או בשליח״.

الحجاب بين القدس وقدس الأقداس، مقابل «وما كان لبشر أن يكلمه الله إلا وحيًا أو من وراء حجاب».

שמות כז, ט–יט ⇄ בוח׳ארי 1587 (״העיר הזאת – אללה קידשה ביום שברא את השמים והארץ״)

בוח׳ארי 1587: ״העיר הזאת – אללה קידש אותה (ביום שברא את השמים והארץ)״.

صحيح البخاري 1587: إِنَّ هَذَا الْبَلَدَ حَرَّمَهُ اللَّهُ

חצר המשכן – תחום מוגדר סביב הבית. בחדית׳ על מכה: ״אללה קידש (חַרַّמַ) את העיר הזאת ביום שברא את השמים והארץ״ – ה״חַרַם״ כתחום קדושה סביב הבית.

حدود صحن المسكن حول البيت، مقابل حرم مكة: «إن هذا البلد حرّمه الله يوم خلق السماوات والأرض».

שמות כז, ט–יח ⇄ 2:125; 22:25

2:125: וכאשר עשינו את הבית מקום מועד לבני האדם ומבטחים: ״ועשו לכם ממקום עמידתו של אברהם מקום תפילה״. וציווינו את אברהם ואת ישמעאל: ״טהרו את ביתי לסובבים, למתבודדים, לכורעים ולמשתחווים״. (125) ✦ 22:25: אלה שכפרו ומרחיקים מדרך אללה ומן המסגד הקדוש, שעשינו אותו לבני האדם שווה – היושב בו והבא מן המדבר – ומי שמבקש בו סטייה בעוול, נטעים אותו מעונש כואב. (25)

وَإِذۡ جَعَلۡنَا ٱلۡبَيۡتَ مَثَابَةٗ لِّلنَّاسِ وَأَمۡنٗا وَٱتَّخِذُواْ مِن مَّقَامِ إِبۡرَٰهِـۧمَ مُصَلّٗىۖ وَعَهِدۡنَآ إِلَىٰٓ إِبۡرَٰهِـۧمَ وَإِسۡمَٰعِيلَ أَن طَهِّرَا بَيۡتِيَ لِلطَّآئِفِينَ وَٱلۡعَٰكِفِينَ وَٱلرُّكَّعِ ٱلسُّجُودِ (125) ✦ إِنَّ ٱلَّذِينَ كَفَرُواْ وَيَصُدُّونَ عَن سَبِيلِ ٱللَّهِ وَٱلۡمَسۡجِدِ ٱلۡحَرَامِ ٱلَّذِي جَعَلۡنَٰهُ لِلنَّاسِ سَوَآءً ٱلۡعَٰكِفُ فِيهِ وَٱلۡبَادِۚ وَمَن يُرِدۡ فِيهِ بِإِلۡحَادِۭ بِظُلۡمٖ نُّذِقۡهُ مِنۡ عَذَابٍ أَلِيمٖ (25)

חצר המשכן הפתוחה לעם כולו — כתיאור הבית המקודש בקוראן 22:25 הפתוח בשווה ״לתושב בו ולבא מן המדבר״, וכמקום ביטחון לבני אדם (2:125).

دار المسكن المفتوحة لعامة الشعب، مقابل جعل البيت الحرام مثابة للناس وأمنًا «سواء العاكف فيه والباد» (الحج 22:25 والبقرة 2:125).

27:10וְעַמֻּדָיו עֶשְׂרִים וְאַדְנֵיהֶם עֶשְׂרִים נְחֹשֶׁת וָוֵי הָעַמֻּדִים וַחֲשֻׁקֵיהֶם כָּסֶף׃
وَأَعْمِدَتُهَا عِشْرُونَ، وَقَوَاعِدُهَا عِشْرُونَ مِنْ نُحَاسٍ. رُزَزُ ٱلْأَعْمِدَةِ وَقُضْبَانُهَا مِنْ فِضَّةٍ.
27:11וְכֵן לִפְאַת צָפוֹן בָּאֹרֶךְ קְלָעִים מֵאָה אֹרֶךְ וְעַמֻּדָו עֶשְׂרִים וְאַדְנֵיהֶם עֶשְׂרִים נְחֹשֶׁת וָוֵי הָעַמֻּדִים וַחֲשֻׁקֵיהֶם כָּסֶף׃
وَكَذَلِكَ إِلَى جِهَةِ ٱلشِّمَالِ فِي ٱلطُّولِ أَسْتَارٌ مِئَةُ ذِرَاعٍ طُولًا. وَأَعْمِدَتُهَا عِشْرُونَ، وَقَوَاعِدُهَا عِشْرُونَ مِنْ نُحَاسٍ. رُزَزُ ٱلْأَعْمِدَةِ وَقُضْبَانُهَا مِنْ فِضَّةٍ.
27:12וְרֹחַב הֶחָצֵר לִפְאַת־יָם קְלָעִים חֲמִשִּׁים אַמָּה עַמֻּדֵיהֶם עֲשָׂרָה וְאַדְנֵיהֶם עֲשָׂרָה׃
وَفِي عَرْضِ ٱلدَّارِ إِلَى جِهَةِ ٱلْغَرْبِ أَسْتَارٌ خَمْسُونَ ذِرَاعًا. أَعْمِدَتُهَا عَشْرَةٌ، وَقَوَاعِدُهَا عَشْرٌ.
27:13וְרֹחַב הֶחָצֵר לִפְאַת קֵדְמָה מִזְרָחָה חֲמִשִּׁים אַמָּה׃
وَعَرْضُ ٱلدَّارِ إِلَى جِهَةِ ٱلشَّرْقِ نَحْوَ ٱلشُّرُوقِ خَمْسُونَ ذِرَاعًا.
27:14וַחֲמֵשׁ עֶשְׂרֵה אַמָּה קְלָעִים לַכָּתֵף עַמֻּדֵיהֶם שְׁלֹשָׁה וְאַדְנֵיהֶם שְׁלֹשָׁה׃
وَخَمْسَ عَشْرَةَ ذِرَاعًا مِنَ ٱلْأَسْتَارِ لِلْجَانِبِ ٱلْوَاحِدِ. أَعْمِدَتُهَا ثَلَاثَةٌ وَقَوَاعِدُهَا ثَلَاثٌ.
27:15וְלַכָּתֵף הַשֵּׁנִית חֲמֵשׁ עֶשְׂרֵה קְלָעִים עַמֻּדֵיהֶם שְׁלֹשָׁה וְאַדְנֵיהֶם שְׁלֹשָׁה׃
وَلِلْجَانِبِ ٱلثَّانِي خَمْسَ عَشْرَةَ ذِرَاعًا مِنَ ٱلْأَسْتَارِ. أَعْمِدَتُهَا ثَلَاثَةٌ وَقَوَاعِدُهَا ثَلَاثٌ.
27:16וּלְשַׁעַר הֶחָצֵר מָסָךְ עֶשְׂרִים אַמָּה תְּכֵלֶת וְאַרְגָּמָן וְתוֹלַעַת שָׁנִי וְשֵׁשׁ מׇשְׁזָר מַעֲשֵׂה רֹקֵם עַמֻּדֵיהֶם אַרְבָּעָה וְאַדְנֵיהֶם אַרְבָּעָה׃
وَلِبَابِ ٱلدَّارِ سَجْفٌ عِشْرُونَ ذِرَاعًا مِنْ أَسْمَانْجُونِيٍّ وَأُرْجُوَانٍ وَقِرْمِزٍ وَبُوصٍ مَبْرُومٍ صَنْعَةَ ٱلطَّرَّازِ. أَعْمِدَتُهُ أَرْبَعَةٌ، وَقَوَاعِدُهَا أَرْبَعٌ.
