4:21וַיְדַבֵּר יְהֹוָה אֶל־מֹשֶׁה לֵּאמֹר׃
وَكَلَّمَ ٱلرَّبُّ مُوسَى قَائِلًا:

במדבר ד, כא–מט ⇄ 4:58; 28:26

4:58: אללה מצווה עליכם להשיב את הפיקדונות לבעליהם, וכאשר תשפטו בין בני האדם – לשפוט בצדק. מה טוב הדבר שאללה מזהיר אתכם בו! אללה שומע ורואה. (58) ✦ 28:26: אמרה אחת מהן: ״אבי, שכור אותו; הטוב שבשכירים – החזק והנאמן״. (26)

۞إِنَّ ٱللَّهَ يَأۡمُرُكُمۡ أَن تُؤَدُّواْ ٱلۡأَمَٰنَٰتِ إِلَىٰٓ أَهۡلِهَا وَإِذَا حَكَمۡتُم بَيۡنَ ٱلنَّاسِ أَن تَحۡكُمُواْ بِٱلۡعَدۡلِۚ إِنَّ ٱللَّهَ نِعِمَّا يَعِظُكُم بِهِۦٓۗ إِنَّ ٱللَّهَ كَانَ سَمِيعَۢا بَصِيرٗا (58) ✦ قَالَتۡ إِحۡدَىٰهُمَا يَـٰٓأَبَتِ ٱسۡتَـٔۡجِرۡهُۖ إِنَّ خَيۡرَ مَنِ ٱسۡتَـٔۡجَرۡتَ ٱلۡقَوِيُّ ٱلۡأَمِينُ (26)

״ופקדתם עליהם בשמות את משמרת משאם״ — רישום מדויק ואישי של ציוד ציבורי המופקד בידי כל עובד (אחריות לפיקדונות, 4:58).

تعيين عهدة كل عامل باسمه بدقة متناهية صونًا للأمانات والممتلكات العامة (النساء 4:58).

4:22נָשֹׂא אֶת־רֹאשׁ בְּנֵי גֵרְשׁוֹן גַּם־הֵם לְבֵית אֲבֹתָם לְמִשְׁפְּחֹתָם׃
«خُذْ عَدَدَ بَنِي جَرْشُونَ أَيْضًا حَسَبَ بُيُوتِ آبَائِهِمْ وَعَشَائِرِهِمْ،
4:23מִבֶּן שְׁלֹשִׁים שָׁנָה וָמַעְלָה עַד בֶּן־חֲמִשִּׁים שָׁנָה תִּפְקֹד אוֹתָם כׇּל־הַבָּא לִצְבֹא צָבָא לַעֲבֹד עֲבֹדָה בְּאֹהֶל מוֹעֵד׃
مِنِ ٱبْنِ ثَلَاثِينَ سَنَةً فَصَاعِدًا إِلَى ٱبْنِ خَمْسِينَ سَنَةً تَعُدُّهُمْ، كُلَّ ٱلدَّاخِلِينَ لِيَتَجَنَّدُوا أَجْنَادًا، لِيَخْدِمُوا خِدْمَةً فِي خَيْمَةِ ٱلِٱجْتِمَاعِ.
4:24זֹאת עֲבֹדַת מִשְׁפְּחֹת הַגֵּרְשֻׁנִּי לַעֲבֹד וּלְמַשָּׂא׃
هَذِهِ خِدْمَةُ عَشَائِرِ ٱلْجَرْشُونِيِّينَ مِنَ ٱلْخِدْمَةِ وَٱلْحِمْلِ:
4:25וְנָשְׂאוּ אֶת־יְרִיעֹת הַמִּשְׁכָּן וְאֶת־אֹהֶל מוֹעֵד מִכְסֵהוּ וּמִכְסֵה הַתַּחַשׁ אֲשֶׁר־עָלָיו מִלְמָעְלָה וְאֶת־מָסַךְ פֶּתַח אֹהֶל מוֹעֵד׃
يَحْمِلُونَ شُقَقَ ٱلْمَسْكَنِ، وَخَيْمَةَ ٱلِٱجْتِمَاعِ وَغِطَاءَهَا، وَغِطَاءَ ٱلتُّخَسِ ٱلَّذِي عَلَيْهَا مِنْ فَوْقُ، وَسَجْفَ بَابِ خَيْمَةِ ٱلِٱجْتِمَاعِ،
4:26וְאֵת קַלְעֵי הֶחָצֵר וְאֶת־מָסַךְ פֶּתַח שַׁעַר הֶחָצֵר אֲשֶׁר עַל־הַמִּשְׁכָּן וְעַל־הַמִּזְבֵּחַ סָבִיב וְאֵת מֵיתְרֵיהֶם וְאֶת־כׇּל־כְּלֵי עֲבֹדָתָם וְאֵת כׇּל־אֲשֶׁר יֵעָשֶׂה לָהֶם וְעָבָדוּ׃
وَأَسْتَارَ ٱلدَّارِ وَسَجْفَ مَدْخَلِ بَابِ ٱلدَّارِ ٱللَّوَاتِي حَوْلَ ٱلْمَسْكَنِ وَحَوْلَ ٱلْمَذْبَحِ مُحِيطَةً، وَأَطْنَابَهُنَّ وَكُلَّ أَمْتِعَةِ خِدْمَتِهِنَّ. وَكُلُّ مَا يُعْمَلُ لَهُنَّ فَهُمْ يَصْنَعُونَهُ،
4:27עַל־פִּי אַהֲרֹן וּבָנָיו תִּהְיֶה כׇּל־עֲבֹדַת בְּנֵי הַגֵּרְשֻׁנִּי לְכׇל־מַשָּׂאָם וּלְכֹל עֲבֹדָתָם וּפְקַדְתֶּם עֲלֵהֶם בְּמִשְׁמֶרֶת אֵת כׇּל־מַשָּׂאָם׃
حَسَبَ قَوْلِ هَارُونَ وَبَنِيهِ تَكُونُ جَمِيعُ خِدْمَةِ بَنِي ٱلْجَرْشُونِيِّينَ مِنْ كُلِّ حَمْلِهِمْ وَمِنْ كُلِّ خِدْمَتِهِمْ. وَتُوَكِّلُهُمْ بِحِرَاسَةِ كُلِّ أَحْمَالِهِمْ.
4:28זֹאת עֲבֹדַת מִשְׁפְּחֹת בְּנֵי הַגֵּרְשֻׁנִּי בְּאֹהֶל מוֹעֵד וּמִשְׁמַרְתָּם בְּיַד אִיתָמָר בֶּן־אַהֲרֹן הַכֹּהֵן׃
هَذِهِ خِدْمَةُ عَشَائِرِ بَنِي ٱلْجَرْشُونِيِّينَ فِي خَيْمَةِ ٱلِٱجْتِمَاعِ، وَحِرَاسَتُهُمْ بِيَدِ إِيثَامَارَ بْنِ هَارُونَ ٱلكَاهِنِ.
4:29בְּנֵי מְרָרִי לְמִשְׁפְּחֹתָם לְבֵית־אֲבֹתָם תִּפְקֹד אֹתָם׃
«بَنُو مَرَارِي حَسَبَ عَشَائِرِهِمْ وَبُيُوتِ آبَائِهِمْ تَعُدُّهُمْ،
4:30מִבֶּן שְׁלֹשִׁים שָׁנָה וָמַעְלָה וְעַד בֶּן־חֲמִשִּׁים שָׁנָה תִּפְקְדֵם כׇּל־הַבָּא לַצָּבָא לַעֲבֹד אֶת־עֲבֹדַת אֹהֶל מוֹעֵד׃
مِنِ ٱبْنِ ثَلَاثِينَ سَنَةً فَصَاعِدًا إِلَى ٱبْنِ خَمْسِينَ سَنَةً تَعُدُّهُمْ، كُلَّ ٱلدَّاخِلِينَ فِي ٱلْجُنْدِ لِيَخْدِمُوا خِدْمَةَ خَيْمَةِ ٱلِٱجْتِمَاعِ.
4:31וְזֹאת מִשְׁמֶרֶת מַשָּׂאָם לְכׇל־עֲבֹדָתָם בְּאֹהֶל מוֹעֵד קַרְשֵׁי הַמִּשְׁכָּן וּבְרִיחָיו וְעַמּוּדָיו וַאֲדָנָיו׃
وَهَذِهِ حِرَاسَةُ حَمْلِهِمْ وَكُلُّ خِدْمَتِهِمْ فِي خَيْمَةِ ٱلِٱجْتِمَاعِ: أَلْوَاحُ ٱلْمَسْكَنِ وَعَوَارِضُهُ وَأَعْمِدَتُهُ وَفُرَضُهُ،
4:32וְעַמּוּדֵי הֶחָצֵר סָבִיב וְאַדְנֵיהֶם וִיתֵדֹתָם וּמֵיתְרֵיהֶם לְכׇל־כְּלֵיהֶם וּלְכֹל עֲבֹדָתָם וּבְשֵׁמֹת תִּפְקְדוּ אֶת־כְּלֵי מִשְׁמֶרֶת מַשָּׂאָם׃
وَأَعْمِدَةُ ٱلدَّارِ حَوَالَيْهَا وَفُرَضُهَا وَأَوْتَادُهَا وَأَطْنَابُهَا مَعَ كُلِّ أَمْتِعَتِهَا وَكُلِّ خِدْمَتِهَا. وَبِٱلْأَسْمَاءِ تَعُدُّونَ أَمْتِعَةَ حِرَاسَةِ حَمْلِهِمْ.
4:33זֹאת עֲבֹדַת מִשְׁפְּחֹת בְּנֵי מְרָרִי לְכׇל־עֲבֹדָתָם בְּאֹהֶל מוֹעֵד בְּיַד אִיתָמָר בֶּן־אַהֲרֹן הַכֹּהֵן׃
هَذِهِ خِدْمَةُ عَشَائِرِ بَنِي مَرَارِي. كُلُّ خِدْمَتِهِمْ فِي خَيْمَةِ ٱلِٱجْتِمَاعِ بِيَدِ إِيثَامَارَ بْنِ هَارُونَ ٱلْكَاهِنِ».
4:34וַיִּפְקֹד מֹשֶׁה וְאַהֲרֹן וּנְשִׂיאֵי הָעֵדָה אֶת־בְּנֵי הַקְּהָתִי לְמִשְׁפְּחֹתָם וּלְבֵית אֲבֹתָם׃
فَعَدَّ مُوسَى وَهَارُونُ وَرُؤَسَاءُ ٱلْجَمَاعَةِ بَنِي ٱلْقَهَاتِيِّينَ حَسَبَ عَشَائِرِهِمْ وَبُيُوتِ آبَائِهِمْ،
4:35מִבֶּן שְׁלֹשִׁים שָׁנָה וָמַעְלָה וְעַד בֶּן־חֲמִשִּׁים שָׁנָה כׇּל־הַבָּא לַצָּבָא לַעֲבֹדָה בְּאֹהֶל מוֹעֵד׃
مِنِ ٱبْنِ ثَلَاثِينَ سَنَةً فَصَاعِدًا إِلَى ٱبْنِ خَمْسِينَ سَنَةً، كُلَّ ٱلدَّاخِلِينَ فِي ٱلْجُنْدِ لِلْخِدْمَةِ فِي خَيْمَةِ ٱلِٱجْتِمَاعِ.
4:36וַיִּהְיוּ פְקֻדֵיהֶם לְמִשְׁפְּחֹתָם אַלְפַּיִם שְׁבַע מֵאוֹת וַחֲמִשִּׁים׃
فَكَانَ ٱلْمَعْدُودُونَ مِنْهُمْ حَسَبَ عَشَائِرِهِمْ أَلْفَيْنِ وَسَبْعَ مِئَةٍ وَخَمْسِينَ.
4:37אֵלֶּה פְקוּדֵי מִשְׁפְּחֹת הַקְּהָתִי כׇּל־הָעֹבֵד בְּאֹהֶל מוֹעֵד אֲשֶׁר פָּקַד מֹשֶׁה וְאַהֲרֹן עַל־פִּי יְהֹוָה בְּיַד־מֹשֶׁה׃
هَؤُلَاءِ هُمُ ٱلْمَعْدُودُونَ مِنْ عَشَائِرِ ٱلْقَهَاتِيِّينَ، كُلُّ ٱلْخَادِمِينَ فِي خَيْمَةِ ٱلِٱجْتِمَاعِ ٱلَّذِينَ عَدَّهُمْ مُوسَى وَهَارُونُ حَسَبَ قَوْلِ ٱلرَّبِّ عَنْ يَدِ مُوسَى.
4:38וּפְקוּדֵי בְּנֵי גֵרְשׁוֹן לְמִשְׁפְּחוֹתָם וּלְבֵית אֲבֹתָם׃
وَٱلْمَعْدُودُونَ مِنْ بَنِي جَرْشُونَ حَسَبَ عَشَائِرِهِمْ وَبُيُوتِ آبَائِهِمْ،
4:39מִבֶּן שְׁלֹשִׁים שָׁנָה וָמַעְלָה וְעַד בֶּן־חֲמִשִּׁים שָׁנָה כׇּל־הַבָּא לַצָּבָא לַעֲבֹדָה בְּאֹהֶל מוֹעֵד׃
مِنِ ٱبْنِ ثَلَاثِينَ سَنَةً فَصَاعِدًا إِلَى ٱبْنِ خَمْسِينَ سَنَةً، كُلُّ ٱلدَّاخِلِينَ فِي ٱلْجُنْدِ لِلْخِدْمَةِ فِي خَيْمَةِ ٱلِٱجْتِمَاعِ.
4:40וַיִּהְיוּ פְּקֻדֵיהֶם לְמִשְׁפְּחֹתָם לְבֵית אֲבֹתָם אַלְפַּיִם וְשֵׁשׁ מֵאוֹת וּשְׁלֹשִׁים׃
كَانَ ٱلْمَعْدُودُونَ مِنْهُمْ حَسَبَ عَشَائِرِهِمْ وَبُيُوتِ آبَائِهِمْ أَلْفَيْنِ وَسِتَّ مِئَةٍ وَثَلَاثِينَ.
4:41אֵלֶּה פְקוּדֵי מִשְׁפְּחֹת בְּנֵי גֵרְשׁוֹן כׇּל־הָעֹבֵד בְּאֹהֶל מוֹעֵד אֲשֶׁר פָּקַד מֹשֶׁה וְאַהֲרֹן עַל־פִּי יְהֹוָה׃
هَؤُلَاءِ هُمُ ٱلْمَعْدُودُونَ مِنْ عَشَائِرِ بَنِي جَرْشُونَ، كُلُّ ٱلْخَادِمِينَ فِي خَيْمَةِ ٱلِٱجْتِمَاعِ ٱلَّذِينَ عَدَّهُمْ مُوسَى وَهَارُونُ حَسَبَ قَوْلِ ٱلرَّبِّ.
4:42וּפְקוּדֵי מִשְׁפְּחֹת בְּנֵי מְרָרִי לְמִשְׁפְּחֹתָם לְבֵית אֲבֹתָם׃
وَٱلْمَعْدُودُونَ مِنْ عَشَائِرِ بَنِي مَرَارِي حَسَبَ عَشَائِرِهِمْ وَبُيُوتِ آبَائِهِمْ،

בוח׳ארי 4747: (בטקס הלִּעַאן) קמה (האישה) והעידה (ארבע עדויות). כשהגיעה לחמישית, עצרו אותה ואמרו: ״זו (העדות) המחייבת (את העונש אם תשקרי)״. אמר אבן עבאס: היססה ונסוגה עד שחשבנו שתחזור בה, ואחר כך אמרה: ״לא אבייש את בני עמי כל ימיהם״, והמשיכה (ונשבעה). אמר הנביא: ״הביטו בה: אם תלד אותו כחול עיניים, רחב ירכיים ועבה שוקיים – הוא של שריכ בן סחמאא״. וילדה אותו כך. אמר הנביא: ״לולא מה שכבר נקבע בספר אללה – היה לי ולה עניין (הייתי מעניש אותה)״. ✦ מוסלם 1492: מפי סַהְל בן סעד אלסאעדי: עֻוַיְמִר אלעַג׳לאני בא אל עאצם בן עדי אלאנצארי ואמר לו: ״מה דעתך, עאצם, לו אדם מצא עם אשתו איש – האם יהרוג אותו ואתם תהרגו אותו, או מה יעשה? שאל לי על כך, עאצם, את שליח אללה״. שאל עאצם, ושליח אללה מאס בשאלות וגינה אותן, עד שקשה היה לעאצם מה ששמע. כשחזר לביתו בא אליו עוימר ושאל מה אמר לו שליח אללה; אמר עאצם: ״לא הבאת לי טוב – שליח אללה מאס בשאלה״. אמר עוימר: ״באללה, לא אחדל עד שאשאל אותו״. בא עוימר אל שליח אללה בתוך הקהל ושאל, ואמר שליח אללה: ״כבר הורד (קוראן) עליך ועל אשתך; לך והבא אותה״. אמר סהל: והתקללו זה בזה (בשבועת הלִּעַאן), ואני עם הקהל אצל שליח אללה. וכשסיימו, אמר עוימר: ״שיקרתי עליה, שליח אללה, אם אחזיק בה״ – וגירש אותה שלוש (פעמים) לפני ששליח אללה ציווה עליו. אמר אבן שהאב: וזה היה הנוהג במתקללים.

במדבר ז, פח ⇄ קוראן 9:108 (״מסגד שנוסד על היראה מן היום הראשון״)

9:108: אל תעמוד בו לעולם. מסגד שנוסד על יראת שמיים מן היום הראשון – ראוי יותר שתעמוד בו; בו אנשים האוהבים להיטהר; ואללה אוהב את המיטהרים. (108)

لَا تَقُمۡ فِيهِ أَبَدٗاۚ لَّمَسۡجِدٌ أُسِّسَ عَلَى ٱلتَّقۡوَىٰ مِنۡ أَوَّلِ يَوۡمٍ أَحَقُّ أَن تَقُومَ فِيهِۚ فِيهِ رِجَالٞ يُحِبُّونَ أَن يَتَطَهَّرُواْۚ وَٱللَّهُ يُحِبُّ ٱلۡمُطَّهِّرِينَ (108)

״זאת חנכת המזבח אחרי המשח אתו״ – חנוכת בית התפילה. בקוראן: ״מסגד שנוסד על היראה מן היום הראשון – ראוי יותר שתעמוד בו״.

במדבר ז, פט ⇄ 2:248

ואמר להם נביאם: ״אות מלכותו – שיבוא אליכם הארון, ובו שכינה (סכינה) מאת ריבונכם ושריד ממה שהניחו בית משה ובית אהרן, והמלאכים נושאים אותו. בזאת אות לכם, אם מאמינים אתם״. (248)

وَقَالَ لَهُمۡ نَبِيُّهُمۡ إِنَّ ءَايَةَ مُلۡكِهِۦٓ أَن يَأۡتِيَكُمُ ٱلتَّابُوتُ فِيهِ سَكِينَةٞ مِّن رَّبِّكُمۡ وَبَقِيَّةٞ مِّمَّا تَرَكَ ءَالُ مُوسَىٰ وَءَالُ هَٰرُونَ تَحۡمِلُهُ ٱلۡمَلَـٰٓئِكَةُۚ إِنَّ فِي ذَٰلِكَ لَأٓيَةٗ لَّكُمۡ إِن كُنتُم مُّؤۡمِنِينَ (248)

ארון העדות והכרובים מול התאבות ובו ״סַכִּינַה״ (שכינה) ושריד ממה שהניחו בית משה ובית אהרן.

تابوت الشهادة والكروبيم مقابل التابوت فيه «سكينة» وبقية مما ترك آل موسى وآل هارون.

במדבר ז, פט ⇄ קוראן 4:164 (״ודיבר אללה עם משה דיבור״)

4:164: ושליחים אשר סיפרנו לך עליהם קודם לכן, ושליחים אשר לא סיפרנו לך עליהם; ואללה דיבר עם משה דיבור. (164)

وَرُسُلٗا قَدۡ قَصَصۡنَٰهُمۡ عَلَيۡكَ مِن قَبۡلُ وَرُسُلٗا لَّمۡ نَقۡصُصۡهُمۡ عَلَيۡكَۚ وَكَلَّمَ ٱللَّهُ مُوسَىٰ تَكۡلِيمٗا (164)

״ובבא משה אל אהל מועד לדבר אתו וישמע את הקול מדבר אליו מעל הכפרת״. בקוראן: ״ודיבר אללה עם משה דיבור (תַכְּלִימַא)״ – ומכאן כינויו של משה: ״כַּלִים אללה״.

במדבר ז, פט ⇄ 4:164; 42:51

4:164: ושליחים אשר סיפרנו לך עליהם קודם לכן, ושליחים אשר לא סיפרנו לך עליהם; ואללה דיבר עם משה דיבור. (164) ✦ 42:51: ואין לבן אדם שאללה ידבר אליו אלא בהשראה, או מאחורי מסך, או שישלח שליח (מלאך) וזה ישרה ברשותו את אשר ירצה; הוא נעלה וחכם. (51)

وَرُسُلٗا قَدۡ قَصَصۡنَٰهُمۡ عَلَيۡكَ مِن قَبۡلُ وَرُسُلٗا لَّمۡ نَقۡصُصۡهُمۡ عَلَيۡكَۚ وَكَلَّمَ ٱللَّهُ مُوسَىٰ تَكۡلِيمٗا (164) ✦ ۞وَمَا كَانَ لِبَشَرٍ أَن يُكَلِّمَهُ ٱللَّهُ إِلَّا وَحۡيًا أَوۡ مِن وَرَآيِٕ حِجَابٍ أَوۡ يُرۡسِلَ رَسُولٗا فَيُوحِيَ بِإِذۡنِهِۦ مَا يَشَآءُۚ إِنَّهُۥ عَلِيٌّ حَكِيمٞ (51)

״וישמע את הקול מדבר אליו מעל הכפורת... וידבר אליו״ — התגלות הדיבור הישיר למשה כ*כלים אללה* (4:164) מאחורי מחיצה (42:51).
