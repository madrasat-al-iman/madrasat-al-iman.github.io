וַיְקֹנֵן דָּוִד אֶת־הַקִּינָה הַזֹּאת עַל־שָׁאוּל וְעַל־יְהוֹנָתָן בְּנוֹ׃ (יז) וַיֹּאמֶר לְלַמֵּד בְּנֵי־יְהוּדָה קָשֶׁת הִנֵּה כְתוּבָה עַל־סֵפֶר הַיָּשָׁר׃ (יח) הַצְּבִי יִשְׂרָאֵל עַל־בָּמוֹתֶיךָ חָלָל אֵיךְ נָפְלוּ גִבּוֹרִים׃ (יט) אַל־תַּגִּידוּ בְגַת אַל־תְּבַשְּׂרוּ בְּחוּצֹת אַשְׁקְלוֹן פֶּן־תִּשְׂמַחְנָה בְּנוֹת פְּלִשְׁתִּים פֶּן־תַּעֲלֹזְנָה בְּנוֹת הָעֲרֵלִים׃ (כ) הָרֵי בַגִּלְבֹּעַ אַל־טַל וְאַל־מָטָר עֲלֵיכֶם וּשְׂדֵי תְרוּמֹת כִּי שָׁם נִגְעַל מָגֵן גִּבּוֹרִים מָגֵן שָׁאוּל בְּלִי מָשִׁיחַ בַּשָּׁמֶן׃ (כא) מִדַּם חֲלָלִים מֵחֵלֶב גִּבּוֹרִים קֶשֶׁת יְהוֹנָתָן לֹא נָשׂוֹג אָחוֹר וְחֶרֶב שָׁאוּל לֹא תָשׁוּב רֵיקָם׃ (כב) …

وَرَثَا دَاوُدُ بِهَذِهِ ٱلْمَرْثَاةِ شَاوُلَ وَيُونَاثَانَ ٱبْنَهُ، (17) وَقَالَ أَنْ يَتَعَلَّمَ بَنُو يَهُوذَا «نَشِيدَ ٱلْقَوْسِ». هُوَذَا ذَلِكَ مَكْتُوبٌ فِي سِفْرِ يَاشَرَ: (18) «اَلظَّبْيُ يَا إِسْرَائِيلُ مَقْتُولٌ عَلَى شَوَامِخِكَ. كَيْفَ سَقَطَ ٱلْجَبَابِرَةُ! (19) لَا تُخْبِرُوا فِي جَتَّ. لَا تُبَشِّرُوا فِي أَسْوَاقِ أَشْقَلُونَ، لِئَلَّا تَفْرَحَ بَنَاتُ ٱلْفِلِسْطِينِيِّينَ، لِئَلَّا تَشْمَتَ بَنَاتُ ٱلْغُلْفِ. (20) يَاجِبَالَ جِلْبُوعَ لَا يَكُنْ طَلٌّ وَلَا مَطَرٌ عَلَيْكُنَّ، وَلَا حُقُولُ تَقْدِمَاتٍ، لِأَنَّهُ هُنَاكَ طُرِحَ مِجَنُّ ٱلْجَبَابِرَةِ، مِجَنُّ شَاوُلَ بِلَا مَسْحٍ بِٱلدُّهْنِ. (21) مِنْ دَمِ ٱلْقَتْلَى، مِنْ شَحْمِ ٱلْجَبَابِرَةِ لَمْ تَرْجِعْ قَوْسُ يُونَاثَانَ إِلَى ٱلْوَرَاءِ، وَسَيْفُ شَاوُلَ لَمْ يَرْجِعْ خَائِبًا. (22) …

41:34: ולא שווים הטוב והרע. השב (לרע) במה שטוב יותר, והנה זה שביניך לבינו איבה – כאילו הוא ידיד קרוב. (34) ✦ בוח׳ארי 1312: (עבד אלרחמן בן אבי לילא סיפר:) סַהְל בן חֻנַיְף וקַיְס בן סַעְד ישבו באלקאדסיה, ועברה לפניהם הלוויה, וקמו. נאמר להם: ״היא מאנשי הארץ״ – כלומר, מבני החסות (שאינם מוסלמים). אמרו: ״הלוויה עברה לפני הנביא, וקם. נאמר לו: ׳היא הלוויה של יהודי׳. אמר: ׳הלא נפש היא?׳״.

وَلَا تَسۡتَوِي ٱلۡحَسَنَةُ وَلَا ٱلسَّيِّئَةُۚ ٱدۡفَعۡ بِٱلَّتِي هِيَ أَحۡسَنُ فَإِذَا ٱلَّذِي بَيۡنَكَ وَبَيۡنَهُۥ عَدَٰوَةٞ كَأَنَّهُۥ وَلِيٌّ حَمِيمٞ (34) ✦ صحيح البخاري 1312: كان سهل بن حنيف وقيس بن سعد قاعدين بالقادسية، فمروا عليهما بجنازة فقاما. فقيل لهما إنها من أهل الأرض، أى من أهل الذمة فقالا إن النبي صلى الله عليه وسلم مرت به جنازة فقام فقيل له إنها جنازة يهودي. فقال " أليست نفسا ".

שמואל ב ב, א; ה, יט ⇄ בוח׳ארי 1166 (צלאת אלאסתח׳ארה)

וַיְהִי אַחֲרֵי־כֵן וַיִּשְׁאַל דָּוִד בַּיהֹוָה לֵאמֹר הַאֶעֱלֶה בְּאַחַת עָרֵי יְהוּדָה וַיֹּאמֶר יְהֹוָה אֵלָיו עֲלֵה וַיֹּאמֶר דָּוִד אָנָה אֶעֱלֶה וַיֹּאמֶר חֶבְרֹנָה׃ (א) … וַיִּשְׁאַל דָּוִד בַּיהֹוָה לֵאמֹר הַאֶעֱלֶה אֶל־פְּלִשְׁתִּים הֲתִתְּנֵם בְּיָדִי וַיֹּאמֶר יְהֹוָה אֶל־דָּוִד עֲלֵה כִּי־נָתֹן אֶתֵּן אֶת־הַפְּלִשְׁתִּים בְּיָדֶךָ׃ (יט)

وَكَانَ بَعْدَ ذَلِكَ أَنَّ دَاوُدَ سَأَلَ ٱلرَّبَّ قَائِلًا: «أَأَصْعَدُ إِلَى إِحْدَى مَدَائِنِ يَهُوذَا؟» فَقَالَ لَهُ ٱلرَّبُّ: «ٱصْعَدْ». فَقَالَ دَاوُدُ: «إِلَى أَيْنَ أَصْعَدُ؟» فَقَالَ: «إِلَى حَبْرُونَ». (1) … وَسَأَلَ دَاوُدُ مِنَ ٱلرَّبِّ قَائِلًا: «أَأَصْعَدُ إِلَى ٱلْفِلِسْطِينِيِّينَ؟ أَتَدْفَعُهُمْ لِيَدِي؟» فَقَالَ ٱلرَّبُّ لِدَاوُدَ: «ٱصْعَدْ، لِأَنِّي دَفْعًا أَدْفَعُ ٱلْفِلِسْطِينِيِّينَ لِيَدِكَ». (19)

בוח׳ארי 1166: (ג׳אבר בן עבד אללה סיפר:) שליח אללה היה מלמד אותנו את הבקשה להכוונה (אִסְתִח׳ארה) בכל העניינים, כפי שהיה מלמד אותנו סורה מן הקוראן. היה אומר: ״כאשר אחד מכם מתכוון לעשות דבר – יכרע שתי רַכְעות שלא מן החובה, ואחר כך יאמר: אללה, אני מבקש את בחירתך בידיעתך ואת כוחך ביכולתך, ומבקש ממך מחסדך הגדול; כי אתה יכול ואני איני יכול, אתה יודע ואני איני יודע, ואתה יודע הנסתרות. אללה, אם ידוע לך שהדבר הזה טוב לי בדתי, בפרנסתי ובאחרית ענייני (או אמר: בענייני הקרובים והרחוקים) – גזור אותו לי והקל אותו לי, ואחר כך ברך לי בו; ואם ידוע לך שהדבר הזה רע לי בדתי, בפרנסתי ובאחרית ענייני (או אמר: בענייני הקרובים והרחוקים) – הרחק אותו ממני והרחק אותי ממנו, וגזור לי את הטוב באשר הוא, ואחר כך השׂבע אותי רצון בו״. ואמר: ״ויזכיר את בקשתו״.

صحيح البخاري 1166: كان رسول الله صلى الله عليه وسلم يعلمنا الاستخارة في الأمور كما يعلمنا السورة من القرآن يقول " إذا هم أحدكم بالأمر فليركع ركعتين من غير الفريضة ثم ليقل اللهم إني أستخيرك بعلمك وأستقدرك بقدرتك، وأسألك من فضلك العظيم، فإنك تقدر ولا أقدر وتعلم ولا أعلم وأنت علام الغيوب، اللهم إن كنت تعلم أن هذا الأمر خير لي في ديني ومعاشي وعاقبة أمري أو قال عاجل أمري وآجله فاقدره لي ويسره لي ثم بارك لي فيه، وإن كنت تعلم أن هذا الأمر شر لي في ديني ومعاشي وعاقبة أمري أو قال في عاجل أمري وآجله فاصرفه عني واصرفني عنه، واقدر لي الخير حيث كان ثم أرضني به قال ويسمي حاجته "

״וישאל דוד בה׳ לאמור האעלה באחת ערי יהודה?״ — פנייה מתמדת של המלך דוד להכוונה אלוהית לפני כל צעד מדיני וצבאי, כסונת אלאסתח׳ארה (בוח׳ארי 1162).

سؤال داود المستمر لله وطلب هدايته قبل كل قرار مصيري يوازي سنة صلاة الاستخارة (البخاري 1162).

וַיִּשְׁלַח יְהֹוָה אֶת־נָתָן אֶל־דָּוִד וַיָּבֹא אֵלָיו וַיֹּאמֶר לוֹ שְׁנֵי אֲנָשִׁים הָיוּ בְּעִיר אֶחָת אֶחָד עָשִׁיר וְאֶחָד רָאשׁ׃ (א) לְעָשִׁיר הָיָה צֹאן וּבָקָר הַרְבֵּה מְאֹד׃ (ב) וְלָרָשׁ אֵין־כֹּל כִּי אִם־כִּבְשָׂה אַחַת קְטַנָּה אֲשֶׁר קָנָה וַיְחַיֶּהָ וַתִּגְדַּל עִמּוֹ וְעִם־בָּנָיו יַחְדָּו מִפִּתּוֹ תֹאכַל וּמִכֹּסוֹ תִשְׁתֶּה וּבְחֵיקוֹ תִשְׁכָּב וַתְּהִי־לוֹ כְּבַת׃ (ג) וַיָּבֹא הֵלֶךְ לְאִישׁ הֶעָשִׁיר וַיַּחְמֹל לָקַחַת מִצֹּאנוֹ וּמִבְּקָרוֹ לַעֲשׂוֹת לָאֹרֵחַ הַבָּא־לוֹ וַיִּקַּח אֶת־כִּבְשַׂת הָאִישׁ הָרָאשׁ וַיַּעֲשֶׂהָ לָאִישׁ הַבָּא אֵלָיו׃ (ד) וַיִּחַר־אַף דָּוִד בָּאִישׁ מְאֹד וַיֹּאמֶר אֶל־נָתָן חַי־יְהֹוָה כִּי בֶן־מָוֶת הָאִישׁ הָעֹשֶׂה זֹאת׃ (ה) וְאֶת־הַכִּבְשָׂה יְשַׁלֵּם אַרְבַּעְתָּיִם עֵקֶב אֲשֶׁר עָשָׂה אֶת־הַדָּבָר הַזֶּה וְעַל אֲשֶׁר לֹא־חָמָל׃ (ו) וַיֹּאמֶר נָתָן אֶל־דָּוִד אַתָּה הָאִישׁ כֹּה־אָמַר יְהֹוָה אֱלֹהֵי יִשְׂרָאֵל אָנֹכִי מְשַׁחְתִּיךָ לְמֶלֶךְ עַל־יִשְׂרָאֵל וְאָנֹכִי הִצַּלְתִּיךָ מִיַּד שָׁאוּל׃ (ז)

فَأَرْسَلَ ٱلرَّبُّ نَاثَانَ إِلَى دَاوُدَ. فَجَاءَ إِلَيْهِ وَقَالَ لَهُ: «كَانَ رَجُلَانِ فِي مَدِينَةٍ وَاحِدَةٍ، وَاحِدٌ مِنْهُمَا غَنِيٌّ وَٱلْآخَرُ فَقِيرٌ. (1) وَكَانَ لِلْغَنِيِّ غَنَمٌ وَبَقَرٌ كَثِيرَةٌ جِدًّا. (2) وَأَمَّا ٱلْفَقِيرُ فَلَمْ يَكُنْ لَهُ شَيْءٌ إِلَا نَعْجَةٌ وَاحِدَةٌ صَغِيرَةٌ قَدِ ٱقْتَنَاهَا وَرَبَّاهَا وَكَبُرَتْ مَعَهُ وَمَعَ بَنِيهِ جَمِيعًا. تَأْكُلُ مِنْ لُقْمَتِهِ وَتَشْرَبُ مِنْ كَأْسِهِ وَتَنَامُ فِي حِضْنِهِ، وَكَانَتْ لَهُ كَٱبْنَةٍ. (3) فَجَاءَ ضَيْفٌ إِلَى ٱلرَّجُلِ ٱلْغَنِيِّ، فَعَفَا أَنْ يَأْخُذَ مِنْ غَنَمِهِ وَمِنْ بَقَرِهِ لِيُهَيِّئَ لِلضَّيْفِ ٱلَّذِي جَاءَ إِلَيْهِ، فَأَخَذَ نَعْجَةَ ٱلرَّجُلِ ٱلْفَقِيرِ وَهَيَّأَ لِلرَّجُلِ ٱلَّذِي جَاءَ إِلَيْهِ». (4) فَحَمِيَ غَضَبُ دَاوُدَ عَلَى ٱلرَّجُلِ جِدًّا، وَقَالَ لِنَاثَانَ: «حَيٌّ هُوَ ٱلرَّبُّ، إِنَّهُ يُقْتَلُ ٱلرَّجُلُ ٱلْفَاعِلُ ذَلِكَ، (5) وَيَرُدُّ ٱلنَّعْجَةَ أَرْبَعَةَ أَضْعَافٍ لِأَنَّهُ فَعَلَ هَذَا ٱلْأَمْرَ وَلِأَنَّهُ لَمْ يُشْفِقْ». (6) فَقَالَ نَاثَانُ لِدَاوُدَ: «أَنْتَ هُوَ ٱلرَّجُلُ! هَكَذَا قَالَ ٱلرَّبُّ إِلَهُ إِسْرَائِيلَ: أَنَا مَسَحْتُكَ مَلِكًا عَلَى إِسْرَائِيلَ وَأَنْقَذْتُكَ مِنْ يَدِ شَاوُلَ، (7)

והאם הגיע אליך סיפור בעלי הריב, כאשר טיפסו על חומת ההיכל? (21) כאשר נכנסו אל דוד ונבהל מהם. אמרו: ״אל תירא; שני בעלי דין אנו, אחד מאיתנו עשה עוול לחברו. שפוט בינינו באמת, ואל תטה, והדריכנו אל הדרך הישרה. (22) זה אחי – לו תשעים ותשע כבשות, ולי כבשה אחת; ואמר: ׳מסור אותה לי׳, וגבר עליי בדיבור״. (23) אמר: ״הוא עשה לך עוול בבקשו את כבשתך אל כבשותיו. ורבים מן השותפים עושים עוול זה לזה, מלבד המאמינים ועושי הטוב – ומעטים הם״. והבין דוד כי ניסינו אותו, וביקש סליחה מריבונו, ונפל כורע, ושב. (24)

۞وَهَلۡ أَتَىٰكَ نَبَؤُاْ ٱلۡخَصۡمِ إِذۡ تَسَوَّرُواْ ٱلۡمِحۡرَابَ (21) إِذۡ دَخَلُواْ عَلَىٰ دَاوُۥدَ فَفَزِعَ مِنۡهُمۡۖ قَالُواْ لَا تَخَفۡۖ خَصۡمَانِ بَغَىٰ بَعۡضُنَا عَلَىٰ بَعۡضٖ فَٱحۡكُم بَيۡنَنَا بِٱلۡحَقِّ وَلَا تُشۡطِطۡ وَٱهۡدِنَآ إِلَىٰ سَوَآءِ ٱلصِّرَٰطِ (22) إِنَّ هَٰذَآ أَخِي لَهُۥ تِسۡعٞ وَتِسۡعُونَ نَعۡجَةٗ وَلِيَ نَعۡجَةٞ وَٰحِدَةٞ فَقَالَ أَكۡفِلۡنِيهَا وَعَزَّنِي فِي ٱلۡخِطَابِ (23) قَالَ لَقَدۡ ظَلَمَكَ بِسُؤَالِ نَعۡجَتِكَ إِلَىٰ نِعَاجِهِۦۖ وَإِنَّ كَثِيرٗا مِّنَ ٱلۡخُلَطَآءِ لَيَبۡغِي بَعۡضُهُمۡ عَلَىٰ بَعۡضٍ إِلَّا ٱلَّذِينَ ءَامَنُواْ وَعَمِلُواْ ٱلصَّـٰلِحَٰتِ وَقَلِيلٞ مَّا هُمۡۗ وَظَنَّ دَاوُۥدُ أَنَّمَا فَتَنَّـٰهُ فَٱسۡتَغۡفَرَ رَبَّهُۥ وَخَرَّۤ رَاكِعٗاۤ وَأَنَابَ۩ (24)

משל כבשת הרש של נתן – ״לאחי זה תשעים ותשע כבשות ולי כבשה אחת, ואמר: מסור לי אותה״; דאוד מבין שהמשל עליו.

مثل نعجة الفقير عند ناثان – «إن هذا أخي له تسع وتسعون نعجة ولي نعجة واحدة فقال أكفلنيها»؛ وداود يفهم أن المثل عليه.

בוח׳ארי 660: מפי אבו הֻרַיְרַה: הנביא אמר: ״שבעה יצל עליהם אללה בצלו ביום שאין צל מלבד צלו: המנהיג הצודק; בחור שגדל בעבודת ריבונו; אדם שלבו קשור במסגדים; שני אנשים שאהבו זה את זה למען אללה – התאחדו על כך ונפרדו על כך; אדם שאישה בעלת מעמד ויופי חיזרה אחריו, ואמר: אני ירא את אללה; אדם שנתן צדקה והסתיר אותה עד שלא תדע שמאלו מה נותנת ימינו; ואדם שזכר את אללה בבדידות – וזלגו עיניו״. ✦ מוסלם 1827: אמר שליח אללה: ״השופטים בצדק יהיו אצל אללה על במות של אור, לימין הרחמן הנעלה והנשגב – ושתי ידיו ימין: אלה הנוהגים בצדק במשפטם, בבני ביתם ובכל אשר הופקדו עליו״.

صحيح البخاري 660: حَدَّثَنَا مُحَمَّدُ بْنُ بَشَّارٍ، قَالَ حَدَّثَنَا يَحْيَى، عَنْ عُبَيْدِ اللَّهِ، قَالَ حَدَّثَنِي خُبَيْبُ بْنُ عَبْدِ الرَّحْمَنِ، عَنْ حَفْصِ بْنِ عَاصِمٍ، عَنْ أَبِي هُرَيْرَةَ، عَنِ النَّبِيِّ صلى الله عليه وسلم قَالَ ‏ "‏ سَبْعَةٌ يُظِلُّهُمُ اللَّهُ فِي ظِلِّهِ يَوْمَ لاَ ظِلَّ إِلاَّ ظِلُّهُ الإِمَامُ الْعَادِلُ، وَشَابٌّ نَشَأَ فِي عِبَادَةِ رَبِّهِ، وَرَجُلٌ قَلْبُهُ مُعَلَّقٌ فِي الْمَسَاجِدِ، وَرَجُلاَنِ تَحَابَّا فِي اللَّهِ اجْتَمَعَا عَلَيْهِ وَتَفَرَّقَا عَلَيْهِ، وَرَجُلٌ طَلَبَتْهُ امْرَأَةٌ ذَاتُ مَنْصِبٍ وَجَمَالٍ فَقَالَ إِنِّي أَخَافُ اللَّهَ‏.‏ وَرَجُلٌ تَصَدَّقَ أَخْفَى حَتَّى لاَ تَعْلَمَ شِمَالُهُ مَا تُنْفِقُ يَمِينُهُ، وَرَجُلٌ ذَكَرَ اللَّهَ خَالِيًا فَفَاضَتْ عَيْنَاهُ ‏"‏‏.‏ ✦ صحيح مسلم 1827: قال رسول الله صلى الله عليه وسلم " إن المقسطين عند الله على منابر من نور عن يمين الرحمن عز وجل وكلتا يديه يمين الذين يعدلون في حكمهم وأهليهم وما ولوا " .

״מושל באדם צדיק, מושל יראת אלוהים, וכאור בוקר יזרח שמש״ — דמות המנהיג הצדיק וירא השמיים, ראשון השבעה החוסים בצל אללה ביום הדין (אמאם עאדל, בוח׳ארי 660; מוסלם 1827)!

«المتسلط على الناس بالعدل، المتسلط بخوف الله» يوازي أول السبعة الذين يظلهم الله في ظله يوم القيامة: «إمام عادل» (البخاري 660 ومسلم 1827).
