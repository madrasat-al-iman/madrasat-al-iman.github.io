וַיַּעֲשׂוּ בְנֵי־יִשְׂרָאֵל אֶת־הָרַע בְּעֵינֵי יְהֹוָה וַיַּעַבְדוּ אֶת־הַבְּעָלִים׃ (יא) וַיַּעַזְבוּ אֶת־יְהֹוָה אֱלֹהֵי אֲבוֹתָם הַמּוֹצִיא אוֹתָם מֵאֶרֶץ מִצְרַיִם וַיֵּלְכוּ אַחֲרֵי אֱלֹהִים אֲחֵרִים מֵאֱלֹהֵי הָעַמִּים אֲשֶׁר סְבִיבוֹתֵיהֶם וַיִּשְׁתַּחֲווּ לָהֶם וַיַּכְעִסוּ אֶת־יְהֹוָה׃ (יב) וַיַּעַזְבוּ אֶת־יְהֹוָה וַיַּעַבְדוּ לַבַּעַל וְלָעַשְׁתָּרוֹת׃ (יג) וַיִּחַר־אַף יְהֹוָה בְּיִשְׂרָאֵל וַיִּתְּנֵם בְּיַד־שֹׁסִים וַיָּשֹׁסּוּ אוֹתָם וַיִּמְכְּרֵם בְּיַד אוֹיְבֵיהֶם מִסָּבִיב וְלֹא־יָכְלוּ עוֹד לַעֲמֹד לִפְנֵי אוֹיְבֵיהֶם׃ (יד) בְּכֹל אֲשֶׁר יָצְאוּ יַד־יְהֹוָה הָיְתָה־בָּם לְרָעָה כַּאֲשֶׁר דִּבֶּר יְהֹוָה וְכַאֲשֶׁר נִשְׁבַּע יְהֹוָה לָהֶם וַיֵּצֶר לָהֶם מְאֹד׃ (טו) וַיָּקֶם יְהֹוָה שֹׁפְטִים וַיּוֹשִׁיעוּם מִיַּד שֹׁסֵיהֶם׃ (טז) …

وَفَعَلَ بَنُو إِسْرَائِيلَ ٱلشَّرَّ فِي عَيْنَيِ ٱلرَّبِّ وَعَبَدُوا ٱلْبَعْلِيمَ. (11) وَتَرَكُوا ٱلرَّبَّ إِلَهَ آبَائِهِمِ ٱلَّذِي أَخْرَجَهُمْ مِنْ أَرْضِ مِصْرَ، وَسَارُوا وَرَاءَ آلِهَةٍ أُخْرَى مِنْ آلِهَةِ ٱلشُّعُوبِ ٱلَّذِينَ حَوْلَهُمْ، وَسَجَدُوا لَهَا وَأَغَاظُوا ٱلرَّبَّ. (12) تَرَكُوا ٱلرَّبَّ وَعَبَدُوا ٱلْبَعْلَ وَعَشْتَارُوثَ. (13) فَحَمِيَ غَضَبُ ٱلرَّبِّ عَلَى إِسْرَائِيلَ، فَدَفَعَهُمْ بِأَيْدِي نَاهِبِينَ نَهَبُوهُمْ، وَبَاعَهُمْ بِيَدِ أَعْدَائِهِمْ حَوْلَهُمْ، وَلَمْ يَقْدِرُوا بَعْدُ عَلَى ٱلْوُقُوفِ أَمَامَ أَعْدَائِهِمْ. (14) حَيْثُمَا خَرَجُوا كَانَتْ يَدُ ٱلرَّبِّ عَلَيْهِمْ لِلشَّرِّ، كَمَا تَكَلَّمَ ٱلرَّبُّ وَكَمَا أَقْسَمَ ٱلرَّبُّ لَهُمْ. فَضَاقَ بِهِمُ ٱلْأَمْرُ جِدًّا. (15) وَأَقَامَ ٱلرَّبُّ قُضَاةً فَخَلَّصُوهُمْ مِنْ يَدِ نَاهِبِيهِمْ. (16) …

7:169: ואחריהם באו יורשים שירשו את הספר, לוקחים את רכוש העולם השפל הזה ואומרים: ״ייסלח לנו״; ואם יבוא אליהם רכוש כמוהו – ייקחוהו. הלא נכרתה עמם ברית הספר, שלא יאמרו על אללה אלא את האמת? והם למדו את אשר בו. והבית האחרון טוב יותר לאלה שיראים; הלא תבינו? (169) ✦ 19:59: ובא אחריהם דור (רע) אשר הזניחו את התפילה והלכו אחר התאוות; ועוד יפגשו תעייה (גַיּ – עונש). (59)

فَخَلَفَ مِنۢ بَعۡدِهِمۡ خَلۡفٞ وَرِثُواْ ٱلۡكِتَٰبَ يَأۡخُذُونَ عَرَضَ هَٰذَا ٱلۡأَدۡنَىٰ وَيَقُولُونَ سَيُغۡفَرُ لَنَا وَإِن يَأۡتِهِمۡ عَرَضٞ مِّثۡلُهُۥ يَأۡخُذُوهُۚ أَلَمۡ يُؤۡخَذۡ عَلَيۡهِم مِّيثَٰقُ ٱلۡكِتَٰبِ أَن لَّا يَقُولُواْ عَلَى ٱللَّهِ إِلَّا ٱلۡحَقَّ وَدَرَسُواْ مَا فِيهِۗ وَٱلدَّارُ ٱلۡأٓخِرَةُ خَيۡرٞ لِّلَّذِينَ يَتَّقُونَۚ أَفَلَا تَعۡقِلُونَ (169) ✦ ۞فَخَلَفَ مِنۢ بَعۡدِهِمۡ خَلۡفٌ أَضَاعُواْ ٱلصَّلَوٰةَ وَٱتَّبَعُواْ ٱلشَّهَوَٰتِۖ فَسَوۡفَ يَلۡقَوۡنَ غَيًّا (59)

״ויקם דור אחר אחריהם אשר לא ידעו את ה׳״ — מחזור התרחקות הדור השני ושובו בתשובה בעת צרה, מקביל ל־19:59 (״ובא אחריהם דור שאיבד את התפילה והלך אחר תאוות״).

«فقام بعدهم جيل آخر لم يعرف الرب» يوازي مريم 19:59: «فخلف من بعدهم خلف أضاعوا الصلاة واتبعوا الشهوات».

שופטים ב, טז–יט ⇄ קוראן 7:94–95; 10:12 (מחזוריות ההצלה בצרה והשכחה ברווחה)

וַיָּקֶם יְהֹוָה שֹׁפְטִים וַיּוֹשִׁיעוּם מִיַּד שֹׁסֵיהֶם׃ (טז) וְגַם אֶל־שֹׁפְטֵיהֶם לֹא שָׁמֵעוּ כִּי זָנוּ אַחֲרֵי אֱלֹהִים אֲחֵרִים וַיִּשְׁתַּחֲווּ לָהֶם סָרוּ מַהֵר מִן־הַדֶּרֶךְ אֲשֶׁר הָלְכוּ אֲבוֹתָם לִשְׁמֹעַ מִצְוֺת־יְהֹוָה לֹא־עָשׂוּ כֵן׃ (יז) וְכִי־הֵקִים יְהֹוָה לָהֶם שֹׁפְטִים וְהָיָה יְהֹוָה עִם־הַשֹּׁפֵט וְהוֹשִׁיעָם מִיַּד אֹיְבֵיהֶם כֹּל יְמֵי הַשּׁוֹפֵט כִּי־יִנָּחֵם יְהֹוָה מִנַּאֲקָתָם מִפְּנֵי לֹחֲצֵיהֶם וְדֹחֲקֵיהֶם׃ (יח) וְהָיָה בְּמוֹת הַשּׁוֹפֵט יָשֻׁבוּ וְהִשְׁחִיתוּ מֵאֲבוֹתָם לָלֶכֶת אַחֲרֵי אֱלֹהִים אֲחֵרִים לְעׇבְדָם וּלְהִשְׁתַּחֲוֺת לָהֶם לֹא הִפִּילוּ מִמַּעַלְלֵיהֶם וּמִדַּרְכָּם הַקָּשָׁה׃ (יט)

وَأَقَامَ ٱلرَّبُّ قُضَاةً فَخَلَّصُوهُمْ مِنْ يَدِ نَاهِبِيهِمْ. (16) وَلِقُضَاتِهِمْ أَيْضًا لَمْ يَسْمَعُوا، بَلْ زَنَوْا وَرَاءَ آلِهَةٍ أُخْرَى وَسَجَدُوا لَهَا. حَادُوا سَرِيعًا عَنِ ٱلطَّرِيقِ ٱلَّتِي سَارَ بِهَا آبَاؤُهُمْ لِسَمْعِ وَصَايَا ٱلرَّبِّ، لَمْ يَفْعَلُوا هَكَذَا. (17) وَحِينَمَا أَقَامَ ٱلرَّبُّ لَهُمْ قُضَاةً، كَانَ ٱلرَّبُّ مَعَ ٱلْقَاضِي، وَخَلَّصَهُمْ مِنْ يَدِ أَعْدَائِهِمْ كُلَّ أَيَّامِ ٱلْقَاضِي، لِأَنَّ ٱلرَّبَّ نَدِمَ مِنْ أَجْلِ أَنِينِهِمْ بِسَبَبِ مُضَايِقِيهِمْ وَزَاحِمِيهِمْ. (18) وَعِنْدَ مَوْتِ ٱلْقَاضِي كَانُوا يَرْجِعُونَ وَيَفْسُدُونَ أَكْثَرَ مِنْ آبَائِهِمْ، بِٱلذَّهَابِ وَرَاءَ آلِهَةٍ أُخْرَى لِيَعْبُدُوهَا وَيَسْجُدُوا لَهَا. لَمْ يَكُفُّوا عَنْ أَفْعَالِهِمْ وَطَرِيقِهِمْ ٱلْقَاسِيَةِ. (19)

7:94–95: ולא שלחנו נביא לשום עיר, שלא תפסנו את יושביה במצוקה ובצרה – אולי יתחננו. (94) ואחר כך החלפנו את הרעה בטובה, עד שהתרבו ואמרו: ״כבר פגעו באבותינו הצרה והשמחה״ – ותפסנום פתאום, והם אינם מרגישים. (95)

وَمَآ أَرۡسَلۡنَا فِي قَرۡيَةٖ مِّن نَّبِيٍّ إِلَّآ أَخَذۡنَآ أَهۡلَهَا بِٱلۡبَأۡسَآءِ وَٱلضَّرَّآءِ لَعَلَّهُمۡ يَضَّرَّعُونَ (94) ثُمَّ بَدَّلۡنَا مَكَانَ ٱلسَّيِّئَةِ ٱلۡحَسَنَةَ حَتَّىٰ عَفَواْ وَّقَالُواْ قَدۡ مَسَّ ءَابَآءَنَا ٱلضَّرَّآءُ وَٱلسَّرَّآءُ فَأَخَذۡنَٰهُم بَغۡتَةٗ وَهُمۡ لَا يَشۡعُرُونَ (95)

שופטים ד, ד–ה ⇄ 9:71; 27:23; 27:32–35

וּדְבוֹרָה אִשָּׁה נְבִיאָה אֵשֶׁת לַפִּידוֹת הִיא שֹׁפְטָה אֶת־יִשְׂרָאֵל בָּעֵת הַהִיא׃ (ד) וְהִיא יוֹשֶׁבֶת תַּחַת־תֹּמֶר דְּבוֹרָה בֵּין הָרָמָה וּבֵין בֵּית־אֵל בְּהַר אֶפְרָיִם וַיַּעֲלוּ אֵלֶיהָ בְּנֵי יִשְׂרָאֵל לַמִּשְׁפָּט׃ (ה)

وَدَبُورَةُ ٱمْرَأَةٌ نَبِيَّةٌ زَوْجَةُ لَفِيدُوتَ، هِيَ قَاضِيَةُ إِسْرَائِيلَ فِي ذَلِكَ ٱلْوَقْتِ. (4) وَهِيَ جَالِسَةٌ تَحْتَ نَخْلَةِ دَبُورَةَ بَيْنَ ٱلرَّامَةِ وَبَيْتِ إِيلَ فِي جَبَلِ أَفْرَايِمَ. وَكَانَ بَنُو إِسْرَائِيلَ يَصْعَدُونَ إِلَيْهَا لِلْقَضَاءِ. (5)

9:71: והמאמינים והמאמינות – בני ברית אלה לאלה: מצווים על הטוב ומונעים מן המגונה, ומקיימים את התפילה ונותנים את הזכאת, ומצייתים לאללה ולשליחו; אלה – ירחם עליהם אללה; אללה עזוז וחכם. (71) ✦ 27:23: מצאתי אישה המולכת עליהם, וניתן לה מכל דבר, ולה כס גדול. (23) ✦ 27:32–35: אמרה: ״הוי השרים, יעצוני בענייני; לא הכרעתי דבר עד שהייתם נוכחים״. (32) אמרו: ״אנחנו בעלי כוח ובעלי עוז רב, והעניין בידך; ראי מה תצווי״. (33) אמרה: ״המלכים, כאשר הם נכנסים לעיר, משחיתים אותה ועושים את נכבדי יושביה לשפלים; וכך הם עושים. (34) ואני שולחת אליהם מנחה, ואראה עם מה ישובו השליחים״. (35)

وَٱلۡمُؤۡمِنُونَ وَٱلۡمُؤۡمِنَٰتُ بَعۡضُهُمۡ أَوۡلِيَآءُ بَعۡضٖۚ يَأۡمُرُونَ بِٱلۡمَعۡرُوفِ وَيَنۡهَوۡنَ عَنِ ٱلۡمُنكَرِ وَيُقِيمُونَ ٱلصَّلَوٰةَ وَيُؤۡتُونَ ٱلزَّكَوٰةَ وَيُطِيعُونَ ٱللَّهَ وَرَسُولَهُۥٓۚ أُوْلَـٰٓئِكَ سَيَرۡحَمُهُمُ ٱللَّهُۗ إِنَّ ٱللَّهَ عَزِيزٌ حَكِيمٞ (71) ✦ إِنِّي وَجَدتُّ ٱمۡرَأَةٗ تَمۡلِكُهُمۡ وَأُوتِيَتۡ مِن كُلِّ شَيۡءٖ وَلَهَا عَرۡشٌ عَظِيمٞ (23) ✦ قَالَتۡ يَـٰٓأَيُّهَا ٱلۡمَلَؤُاْ أَفۡتُونِي فِيٓ أَمۡرِي مَا كُنتُ قَاطِعَةً أَمۡرًا حَتَّىٰ تَشۡهَدُونِ (32) قَالُواْ نَحۡنُ أُوْلُواْ قُوَّةٖ وَأُوْلُواْ بَأۡسٖ شَدِيدٖ وَٱلۡأَمۡرُ إِلَيۡكِ فَٱنظُرِي مَاذَا تَأۡمُرِينَ (33) قَالَتۡ إِنَّ ٱلۡمُلُوكَ إِذَا دَخَلُواْ قَرۡيَةً أَفۡسَدُوهَا وَجَعَلُوٓاْ أَعِزَّةَ أَهۡلِهَآ أَذِلَّةٗۚ وَكَذَٰلِكَ يَفۡعَلُونَ (34) وَإِنِّي مُرۡسِلَةٌ إِلَيۡهِم بِهَدِيَّةٖ فَنَاظِرَةُۢ بِمَ يَرۡجِعُ ٱلۡمُرۡسَلُونَ (35)
