יֶתֶר הַגָּזָם אָכַל הָאַרְבֶּה וְיֶתֶר הָאַרְבֶּה אָכַל הַיָּלֶק וְיֶתֶר הַיֶּלֶק אָכַל הֶחָסִיל׃ (ד) … יוֹם חֹשֶׁךְ וַאֲפֵלָה יוֹם עָנָן וַעֲרָפֶל כְּשַׁחַר פָּרֻשׂ עַל־הֶהָרִים עַם רַב וְעָצוּם כָּמֹהוּ לֹא נִהְיָה מִן־הָעוֹלָם וְאַחֲרָיו לֹא יוֹסֵף עַד־שְׁנֵי דּוֹר וָדוֹר׃ (ב) לְפָנָיו אָכְלָה אֵשׁ וְאַחֲרָיו תְּלַהֵט לֶהָבָה כְּגַן־עֵדֶן הָאָרֶץ לְפָנָיו וְאַחֲרָיו מִדְבַּר שְׁמָמָה וְגַם־פְּלֵיטָה לֹא־הָיְתָה לּוֹ׃ (ג) כְּמַרְאֵה סוּסִים מַרְאֵהוּ וּכְפָרָשִׁים כֵּן יְרוּצוּן׃ (ד) כְּקוֹל מַרְכָּבוֹת עַל־רָאשֵׁי הֶהָרִים יְרַקֵּדוּן כְּקוֹל לַהַב אֵשׁ אֹכְלָה קָשׁ כְּעַם עָצוּם עֱרוּךְ מִלְחָמָה׃ (ה) מִפָּנָיו יָחִילוּ עַמִּים כׇּל־פָּנִים קִבְּצוּ פָארוּר׃ (ו) כְּגִבּוֹרִים יְרֻצוּן כְּאַנְשֵׁי מִלְחָמָה יַעֲלוּ חוֹמָה וְאִישׁ בִּדְרָכָיו יֵלֵכוּן וְלֹא יְעַבְּטוּן אֹרְחוֹתָם׃ (ז) …

فَضْلَةُ ٱلْقَمَصِ أَكَلَهَا ٱلزَّحَّافُ، وَفَضْلَةُ ٱلزَّحَّافِ أَكَلَهَا ٱلْغَوْغَاءُ، وَفَضْلَةُ ٱلْغَوْغَاءِ أَكَلَهَا ٱلطَّيَّارُ. (4) … يَوْمُ ظَلَامٍ وَقَتَامٍ، يَوْمُ غَيْمٍ وَضَبَابٍ، مِثْلَ ٱلْفَجْرِ مُمْتَدًّا عَلَى ٱلْجِبَالِ. شَعْبٌ كَثِيرٌ وَقَوِيٌّ لَمْ يَكُنْ نَظِيرُهُ مُنْذُ ٱلْأَزَلِ، وَلَا يَكُونُ أَيْضًا بَعْدَهُ إِلَى سِنِي دَوْرٍ فَدَوْرٍ. (2) قُدَّامَهُ نَارٌ تَأْكُلُ، وَخَلْفَهُ لَهِيبٌ يَحْرِقُ. ٱلْأَرْضُ قُدَّامَهُ كَجَنَّةِ عَدْنٍ وَخَلْفَهُ قَفْرٌ خَرِبٌ، وَلَا تَكُونُ مِنْهُ نَجَاةٌ. (3) كَمَنْظَرِ ٱلْخَيْلِ مَنْظَرُهُ، وَمِثْلَ ٱلْأَفْرَاسِ يَرْكُضُونَ. (4) كَصَرِيفِ ٱلْمَرْكَبَاتِ عَلَى رُؤُوسِ ٱلْجِبَالِ يَثِبُونَ. كَزَفِيرِ لَهِيبِ نَارٍ تَأْكُلُ قَشًّا. كَقَوْمٍ أَقْوِيَاءَ مُصْطَفِّينَ لِلْقِتَالِ. (5) مِنْهُ تَرْتَعِدُ ٱلشُّعُوبِ. كُلُّ ٱلْوُجُوهِ تَجْمَعُ حُمْرَةً. (6) يَجْرُونَ كَأَبْطَالٍ. يَصْعَدُونَ ٱلسُّورَ كَرِجَالِ ٱلْحَرْبِ، وَيَمْشُونَ كُلُّ وَاحِدٍ فِي طَرِيقِهِ، وَلَا يُغَيِّرُونَ سُبُلَهُمْ. (7) …

קַדְּשׁוּ־צוֹם קִרְאוּ עֲצָרָה אִסְפוּ זְקֵנִים כֹּל יֹשְׁבֵי הָאָרֶץ בֵּית יְהֹוָה אֱלֹהֵיכֶם וְזַעֲקוּ אֶל־יְהֹוָה׃ (יד) … תִּקְעוּ שׁוֹפָר בְּצִיּוֹן קַדְּשׁוּ־צוֹם קִרְאוּ עֲצָרָה׃ (טו) אִסְפוּ־עָם קַדְּשׁוּ קָהָל קִבְצוּ זְקֵנִים אִסְפוּ עוֹלָלִים וְיֹנְקֵי שָׁדָיִם יֵצֵא חָתָן מֵחֶדְרוֹ וְכַלָּה מֵחֻפָּתָהּ׃ (טז) בֵּין הָאוּלָם וְלַמִּזְבֵּחַ יִבְכּוּ הַכֹּהֲנִים מְשָׁרְתֵי יְהֹוָה וְיֹאמְרוּ חוּסָה יְהֹוָה עַל־עַמֶּךָ וְאַל־תִּתֵּן נַחֲלָתְךָ לְחֶרְפָּה לִמְשׇׁל־בָּם גּוֹיִם לָמָּה יֹאמְרוּ בָעַמִּים אַיֵּה אֱלֹהֵיהֶם׃ (יז)

قَدِّسُوا صَوْمًا. نَادُوا بِٱعْتِكَافٍ. ٱجْمَعُوا ٱلشُّيُوخَ، جَمِيعَ سُكَّانِ ٱلْأَرْضِ إِلَى بَيْتِ ٱلرَّبِّ إِلَهِكُمْ وَٱصْرُخُوا إِلَى ٱلرَّبِّ. (17)

בוח׳ארי 1013: (אנס בן מאלכ סיפר) שאדם נכנס ביום השישי מפתח שהיה מול הדוכן, ושליח אללה עומד ודורש. עמד מול שליח אללה ואמר: ״שליח אללה, אבד המקנה ונחסמו הדרכים – התפלל לאללה שישלח לנו גשם״. הרים שליח אללה את ידיו ואמר: ״אללה, השקנו! אללה, השקנו! אללה, השקנו!״ אמר אנס: ״וחי אללה, לא ראינו בשמיים לא ענן ולא קרע ענן ולא דבר, ולא היה בינינו לבין (הר) סַלְע בית או חצר. ועלה מאחוריו ענן כמגן, וכשהגיע לאמצע השמיים – התפשט, ואחר כך המטיר. וחי אללה, לא ראינו את השמש שישה (ימים)״. אחר כך נכנס אדם מאותו פתח ביום השישי הבא, ושליח אללה עומד ודורש, ועמד מולו ואמר: ״שליח אללה, אבדו הנכסים ונחסמו הדרכים – התפלל לאללה שיעצור אותו״. הרים שליח אללה את ידיו ואמר: ״אללה, סביבנו ולא עלינו; אללה, על הגבעות וההרים, על החורשות, על התלים, על הוואדיות ועל מקומות צמיחת העצים״. ונפסק (הגשם), ויצאנו מהלכים בשמש. אמר שַׁרִיכּ (המוסר): שאלתי את אנס: ״האם הוא האדם הראשון?״ אמר: ״איני יודע״. ✦ בוח׳ארי 1021: מפי אנס: הנביא נאם ביום השישי, וקמו האנשים וצעקו: ״שליח אללה, הגשם נעצר, העצים האדימו והבהמות אבדו – התפלל לאללה שישקה אותנו״. אמר: ״אללה, השקנו״, פעמיים. ובאללה, לא ראינו בשמים פיסת ענן; ועלה ענן והמטיר. ירד מן הבמה והתפלל, וכשסיים – לא חדל (הגשם) לרדת עד יום השישי שאחריו. וכשקם הנביא לנאום צעקו אליו: ״הבתים נהרסו והדרכים נחסמו – התפלל לאללה שיעצור אותו מעלינו״. חייך הנביא ואמר: ״אללה, סביבנו ולא עלינו״. ונפתחה (השמים מעל) המדינה, והגשם ירד סביבה ולא ירדה טיפה במדינה; והבטתי במדינה – והיא כמו בתוך זר.

صحيح البخاري 1013: أن رجلا، دخل يوم الجمعة من باب كان وجاه المنبر، ورسول الله صلى الله عليه وسلم قائم يخطب فاستقبل رسول الله صلى الله عليه وسلم قائما فقال يا رسول الله، هلكت المواشي وانقطعت السبل، فادع الله يغيثنا. قال فرفع رسول الله صلى الله عليه وسلم يديه فقال " اللهم اسقنا، اللهم اسقنا، اللهم اسقنا ". قال أنس ولا والله ما نرى في السماء من سحاب ولا قزعة ولا شيئا، وما بيننا وبين سلع من بيت ولا دار، قال فطلعت من ورائه سحابة مثل الترس، فلما توسطت السماء انتشرت ثم أمطرت. قال والله ما رأينا الشمس ستا، ثم دخل رجل من ذلك الباب في الجمعة المقبلة، ورسول الله صلى الله عليه وسلم قائم يخطب، فاستقبله قائما فقال يا رسول الله، هلكت الأموال وانقطعت السبل، فادع الله يمسكها، قال فرفع رسول الله صلى الله عليه وسلم يديه ثم قال " اللهم حوالينا ولا علينا، اللهم على الآكام والجبال والآجام والظراب والأودية ومنابت الشجر ". قال فانقطعت وخرجنا نمشي في الشمس. قال شريك فسألت أنسا أهو الرجل الأول قال لا أدري. ✦ صحيح البخاري 1021: حَدَّثَنَا مُحَمَّدُ بْنُ أَبِي بَكْرٍ، حَدَّثَنَا مُعْتَمِرٌ، عَنْ عُبَيْدِ اللَّهِ، عَنْ ثَابِتٍ، عَنْ أَنَسٍ، قَالَ كَانَ النَّبِيُّ صلى الله عليه وسلم يَخْطُبُ يَوْمَ جُمُعَةٍ، فَقَامَ النَّاسُ فَصَاحُوا فَقَالُوا يَا رَسُولَ اللَّهِ، قَحَطَ الْمَطَرُ وَاحْمَرَّتِ الشَّجَرُ وَهَلَكَتِ الْبَهَائِمُ، فَادْعُ اللَّهَ يَسْقِينَا‏.‏ فَقَالَ ‏"‏ اللَّهُمَّ اسْقِنَا ‏"‏‏.‏ مَرَّتَيْنِ، وَايْمُ اللَّهِ مَا نَرَى فِي السَّمَاءِ قَزَعَةً مِنْ سَحَابٍ، فَنَشَأَتْ سَحَابَةٌ وَأَمْطَرَتْ، وَنَزَلَ عَنِ الْمِنْبَرِ فَصَلَّى، فَلَمَّا انْصَرَفَ لَمْ تَزَلْ تُمْطِرُ إِلَى الْجُمُعَةِ الَّتِي تَلِيهَا، فَلَمَّا قَامَ النَّبِيُّ صلى الله عليه وسلم يَخْطُبُ صَاحُوا إِلَيْهِ تَهَدَّمَتِ الْبُيُوتُ وَانْقَطَعَتِ السُّبُلُ، فَادْعُ اللَّهَ يَحْبِسُهَا عَنَّا‏.‏ فَتَبَسَّمَ النَّبِيُّ صلى الله عليه وسلم ثُمَّ قَالَ ‏"‏ اللَّهُمَّ حَوَالَيْنَا وَلاَ عَلَيْنَا ‏"‏‏.‏ فَكُشِطَتِ الْمَدِينَةُ، فَجَعَلَتْ تُمْطِرُ حَوْلَهَا وَلاَ تَمْطُرُ بِالْمَدِينَةِ قَطْرَةً، فَنَظَرْتُ إِلَى الْمَدِينَةِ وَإِنَّهَا لَفِي مِثْلِ الإِكْلِيلِ‏.‏

״קדשו צום, קראו עצרה, אספו זקנים כל יושבי הארץ... וזעקו אל ה׳״ — מקבילה פולחנית מדויקת ליציאת הקהל כולה לצום, תחנונים ותפילת הגשם (צלאת אלאסתסקאא) בשנות בצורת (בוח׳ארי 1013, 1021)!

تَتَحَوَّلُ ٱلشَّمْسُ إِلَى ظُلْمَةٍ، وَٱلْقَمَرُ إِلَى دَمٍ قَبْلَ أَنْ يَجِيءَ يَوْمُ ٱلرَّبِّ ٱلْعَظِيمُ ٱلْمَخُوفُ. (4)

כאשר השמש תיכרך, (1) וכאשר הכוכבים יועמו, (2) וכאשר ההרים יונעו, (3) וכאשר הנאקות המעוברות יוזנחו, (4) וכאשר חיות הבר ייאספו, (5) וכאשר הימים יבערו, (6) וכאשר הנפשות יצורפו (לגופן), (7) וכאשר הבת הקבורה בחיים תישאל: (8) על איזה חטא נהרגה? (9) וכאשר המגילות ייפרשו, (10) וכאשר השמים ייקלפו, (11) וכאשר התופת תוצת, (12) וכאשר הגן יוקרב – (13) תדע נפש מה הביאה. (14)

إِذَا ٱلشَّمۡسُ كُوِّرَتۡ (1) وَإِذَا ٱلنُّجُومُ ٱنكَدَرَتۡ (2) وَإِذَا ٱلۡجِبَالُ سُيِّرَتۡ (3) وَإِذَا ٱلۡعِشَارُ عُطِّلَتۡ (4) وَإِذَا ٱلۡوُحُوشُ حُشِرَتۡ (5) وَإِذَا ٱلۡبِحَارُ سُجِّرَتۡ (6) وَإِذَا ٱلنُّفُوسُ زُوِّجَتۡ (7) وَإِذَا ٱلۡمَوۡءُۥدَةُ سُئِلَتۡ (8) بِأَيِّ ذَنۢبٖ قُتِلَتۡ (9) وَإِذَا ٱلصُّحُفُ نُشِرَتۡ (10) وَإِذَا ٱلسَّمَآءُ كُشِطَتۡ (11) وَإِذَا ٱلۡجَحِيمُ سُعِّرَتۡ (12) وَإِذَا ٱلۡجَنَّةُ أُزۡلِفَتۡ (13) عَلِمَتۡ نَفۡسٞ مَّآ أَحۡضَرَتۡ (14)

השמש כורכת, הכוכבים נושרים.

الشمس كُوّرت، والنجوم انكدرت.

وَيَكُونُ أَنَّ كُلَّ مَنْ يَدْعُو بِٱسْمِ ٱلرَّبِّ يَنْجُو. لِأَنَّهُ فِي جَبَلِ صِهْيَوْنَ وَفِي أُورُشَلِيمَ تَكُونُ نَجَاةٌ، كَمَا قَالَ ٱلرَّبُّ. وَبَيْنَ ٱلْبَاقِينَ مَنْ يَدْعُوهُ ٱلرَّبُّ. (5)

״והיה כל אשר יקרא בשם ה׳ יימלט״ — שער התפילה וההצלה הפתוח לכל קורא באמת (2:186; 40:60).
