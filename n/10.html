עזרא א, א–ד ⇄ 18:83–98; 3:26

וּבִשְׁנַת אַחַת לְכוֹרֶשׁ מֶלֶךְ פָּרַס לִכְלוֹת דְּבַר־יְהֹוָה מִפִּי יִרְמְיָה הֵעִיר יְהֹוָה אֶת־רוּחַ כֹּרֶשׁ מֶלֶךְ־פָּרַס וַיַּעֲבֶר־קוֹל בְּכׇל־מַלְכוּתוֹ וְגַם־בְּמִכְתָּב לֵאמֹר׃ (א) כֹּה אָמַר כֹּרֶשׁ מֶלֶךְ פָּרַס כֹּל מַמְלְכוֹת הָאָרֶץ נָתַן לִי יְהֹוָה אֱלֹהֵי הַשָּׁמָיִם וְהוּא־פָקַד עָלַי לִבְנוֹת־לוֹ בַיִת בִּירוּשָׁלִַם אֲשֶׁר בִּיהוּדָה׃ (ב) מִי־בָכֶם מִכׇּל־עַמּוֹ יְהִי אֱלֹהָיו עִמּוֹ וְיַעַל לִירוּשָׁלִַם אֲשֶׁר בִּיהוּדָה וְיִבֶן אֶת־בֵּית יְהֹוָה אֱלֹהֵי יִשְׂרָאֵל הוּא הָאֱלֹהִים אֲשֶׁר בִּירוּשָׁלִָם׃ (ג) וְכׇל־הַנִּשְׁאָר מִכׇּל־הַמְּקֹמוֹת אֲשֶׁר הוּא גָר־שָׁם יְנַשְּׂאוּהוּ אַנְשֵׁי מְקֹמוֹ בְּכֶסֶף וּבְזָהָב וּבִרְכוּשׁ וּבִבְהֵמָה עִם־הַנְּדָבָה לְבֵית הָאֱלֹהִים אֲשֶׁר בִּירוּשָׁלִָם׃ (ד)

وَفِي ٱلسَّنَةِ ٱلْأُولَى لِكُورَشَ مَلِكِ فَارِسَ عِنْدَ تَمَامِ كَلَامِ ٱلرَّبِّ بِفَمِ إِرْمِيَا، نَبَّهَ ٱلرَّبُّ رُوحَ كُورَشَ مَلِكِ فَارِسَ فَأَطْلَقَ نِدَاءً فِي كُلِّ مَمْلَكَتِهِ وَبِالْكِتَابَةِ أَيْضًا قَائِلًا: (1) «هَكَذَا قَالَ كُورَشُ مَلِكُ فَارِسَ: جَمِيعُ مَمَالِكِ ٱلْأَرْضِ دَفَعَهَا لِي ٱلرَّبُّ إِلَهُ ٱلسَّمَاءِ، وَهُوَ أَوْصَانِي أَنْ أَبْنِيَ لَهُ بَيْتًا فِي أُورُشَلِيمَ ٱلَّتِي فِي يَهُوذَا. (2) مَنْ مِنْكُمْ مِنْ كُلِّ شَعْبِهِ، لِيَكُنْ إِلَهُهُ مَعَهُ، وَيَصْعَدْ إِلَى أُورُشَلِيمَ ٱلَّتِي فِي يَهُوذَا فَيَبْنِيَ بَيْتَ ٱلرَّبِّ إِلَهِ إِسْرَائِيلَ. هُوَ ٱلْإِلَهُ ٱلَّذِي فِي أُورُشَلِيمَ. (3) وَكُلُّ مَنْ بَقِيَ فِي أَحَدِ ٱلْأَمَاكِنِ حَيْثُ هُوَ مُتَغَرِّبٌ فَلْيُنْجِدْهُ أَهْلُ مَكَانِهِ بِفِضَّةٍ وَبِذَهَبٍ وَبِأَمْتِعَةٍ وَبِبَهَائِمَ مَعَ ٱلتَّبَرُّعِ لِبَيْتِ ٱلرَّبِّ ٱلَّذِي فِي أُورُشَلِيمَ». (4)

18:83–98: ושואלים אותך על בעל שתי הקרניים (ד׳ו אל־קרנין). אמור: ״אקרא לפניכם זכר ממנו״. (83) אנחנו ביססנו אותו בארץ, ונתנו לו דרך לכל דבר. (84) והלך בדרך, (85) עד אשר הגיע אל מקום שקיעת השמש, ומצאה שוקעת במעיין של טיט שחור, ומצא לידה עם. אמרנו: ״בעל שתי הקרניים, אם תעניש ואם תנהג בהם בטוב״. (86) אמר: ״מי שעשה עוול – נענישנו, ואחר כך יושב אל ריבונו ויענישנו עונש מר. (87) ומי שהאמין ועשה טוב – לו הגמול הטוב, ונאמר לו מצוותינו דברים קלים״. (88) אחר כך הלך בדרך, (89) עד אשר הגיע אל מקום זריחת השמש, ומצאה זורחת על עם אשר לא עשינו להם מחסה מפניה. (90) כך היה; וידענו היטב את כל אשר לו. (91) אחר כך הלך בדרך, (92) עד אשר הגיע בין שני הסכרים (ההרים), ומצא לפניהם עם אשר כמעט אינו מבין דיבור. (93) אמרו: ״בעל שתי הקרניים, יאג׳וג׳ ומאג׳וג׳ (גוג ומגוג) משחיתים בארץ. הנשלם לך מס על מנת שתעשה בינינו וביניהם סכר?״ (94) אמר: ״מה שנתן לי בו ריבוני – טוב (משכרכם). עזרו לי בכוח, ואעשה ביניכם וביניהם חומה. (95) הביאו לי גושי ברזל״. עד אשר השווה (את הגובה) בין שני צדי ההרים, אמר: ״נפחו!״ עד אשר עשהו אש, אמר: ״הביאו לי, ואצוק עליו נחושת מותכת״. (96) ולא יכלו לעלות עליו, ולא יכלו לנקבו. (97) אמר: ״זה רחמים מאת ריבוני. וכאשר תבוא הבטחת ריבוני – יעשהו עפר; והבטחת ריבוני אמת״. (98) ✦ 3:26: אמור: ״אללה, בעל המלכות, אתה נותן מלכות למי שתרצה ונוטל מלכות ממי שתרצה, מרומם את מי שתרצה ומשפיל את מי שתרצה. בידך הטוב, כי אתה כל יכול. (26)

وَيَسۡـَٔلُونَكَ عَن ذِي ٱلۡقَرۡنَيۡنِۖ قُلۡ سَأَتۡلُواْ عَلَيۡكُم مِّنۡهُ ذِكۡرًا (83) إِنَّا مَكَّنَّا لَهُۥ فِي ٱلۡأَرۡضِ وَءَاتَيۡنَٰهُ مِن كُلِّ شَيۡءٖ سَبَبٗا (84) فَأَتۡبَعَ سَبَبًا (85) حَتَّىٰٓ إِذَا بَلَغَ مَغۡرِبَ ٱلشَّمۡسِ وَجَدَهَا تَغۡرُبُ فِي عَيۡنٍ حَمِئَةٖ وَوَجَدَ عِندَهَا قَوۡمٗاۖ قُلۡنَا يَٰذَا ٱلۡقَرۡنَيۡنِ إِمَّآ أَن تُعَذِّبَ وَإِمَّآ أَن تَتَّخِذَ فِيهِمۡ حُسۡنٗا (86) قَالَ أَمَّا مَن ظَلَمَ فَسَوۡفَ نُعَذِّبُهُۥ ثُمَّ يُرَدُّ إِلَىٰ رَبِّهِۦ فَيُعَذِّبُهُۥ عَذَابٗا نُّكۡرٗا (87) وَأَمَّا مَنۡ ءَامَنَ وَعَمِلَ صَٰلِحٗا فَلَهُۥ جَزَآءً ٱلۡحُسۡنَىٰۖ وَسَنَقُولُ لَهُۥ مِنۡ أَمۡرِنَا يُسۡرٗا (88) ثُمَّ أَتۡبَعَ سَبَبًا (89) حَتَّىٰٓ إِذَا بَلَغَ مَطۡلِعَ ٱلشَّمۡسِ وَجَدَهَا تَطۡلُعُ عَلَىٰ قَوۡمٖ لَّمۡ نَجۡعَل لَّهُم مِّن دُونِهَا سِتۡرٗا (90) كَذَٰلِكَۖ وَقَدۡ أَحَطۡنَا بِمَا لَدَيۡهِ خُبۡرٗا (91) ثُمَّ أَتۡبَعَ سَبَبًا (92) حَتَّىٰٓ إِذَا بَلَغَ بَيۡنَ ٱلسَّدَّيۡنِ وَجَدَ مِن دُونِهِمَا قَوۡمٗا لَّا يَكَادُونَ يَفۡقَهُونَ قَوۡلٗا (93) قَالُواْ يَٰذَا ٱلۡقَرۡنَيۡنِ إِنَّ يَأۡجُوجَ وَمَأۡجُوجَ مُفۡسِدُونَ فِي ٱلۡأَرۡضِ فَهَلۡ نَجۡعَلُ لَكَ خَرۡجًا عَلَىٰٓ أَن تَجۡعَلَ بَيۡنَنَا وَبَيۡنَهُمۡ سَدّٗا (94) قَالَ مَا مَكَّنِّي فِيهِ رَبِّي خَيۡرٞ فَأَعِينُونِي بِقُوَّةٍ أَجۡعَلۡ بَيۡنَكُمۡ وَبَيۡنَهُمۡ رَدۡمًا (95) ءَاتُونِي زُبَرَ ٱلۡحَدِيدِۖ حَتَّىٰٓ إِذَا سَاوَىٰ بَيۡنَ ٱلصَّدَفَيۡنِ قَالَ ٱنفُخُواْۖ حَتَّىٰٓ إِذَا جَعَلَهُۥ نَارٗا قَالَ ءَاتُونِيٓ أُفۡرِغۡ عَلَيۡهِ قِطۡرٗا (96) فَمَا ٱسۡطَٰعُوٓاْ أَن يَظۡهَرُوهُ وَمَا ٱسۡتَطَٰعُواْ لَهُۥ نَقۡبٗا (97) قَالَ هَٰذَا رَحۡمَةٞ مِّن رَّبِّيۖ فَإِذَا جَآءَ وَعۡدُ رَبِّي جَعَلَهُۥ دَكَّآءَۖ وَكَانَ وَعۡدُ رَبِّي حَقّٗا (98) ✦ قُلِ ٱللَّهُمَّ مَٰلِكَ ٱلۡمُلۡكِ تُؤۡتِي ٱلۡمُلۡكَ مَن تَشَآءُ وَتَنزِعُ ٱلۡمُلۡكَ مِمَّن تَشَآءُ وَتُعِزُّ مَن تَشَآءُ وَتُذِلُّ مَن تَشَآءُۖ بِيَدِكَ ٱلۡخَيۡرُۖ إِنَّكَ عَلَىٰ كُلِّ شَيۡءٖ قَدِيرٞ (26)

הצהרת כורש המלך הפרסי המתיר שיבת ציון ובניין הבית (״העיר ה׳ את רוח כורש״) — בפרשנות מודרנית (אבו אלכלאם אזאד ואחרים) אף הוצע זיהוי בין כורש הצדיק לבין ד׳ו אלקרנין בסורה 18:83–98, וריבונות האל על מלכים (3:26).

וְכׇל־הַנִּשְׁאָר מִכׇּל־הַמְּקֹמוֹת אֲשֶׁר הוּא גָר־שָׁם יְנַשְּׂאוּהוּ אַנְשֵׁי מְקֹמוֹ בְּכֶסֶף וּבְזָהָב וּבִרְכוּשׁ וּבִבְהֵמָה עִם־הַנְּדָבָה לְבֵית הָאֱלֹהִים אֲשֶׁר בִּירוּשָׁלִָם׃ (ד) … וְכׇל־סְבִיבֹתֵיהֶם חִזְּקוּ בִידֵיהֶם בִּכְלֵי־כֶסֶף בַּזָּהָב בָּרְכוּשׁ וּבַבְּהֵמָה וּבַמִּגְדָּנוֹת לְבַד עַל־כׇּל־הִתְנַדֵּב׃ (ו)

وَكُلُّ مَنْ بَقِيَ فِي أَحَدِ ٱلْأَمَاكِنِ حَيْثُ هُوَ مُتَغَرِّبٌ فَلْيُنْجِدْهُ أَهْلُ مَكَانِهِ بِفِضَّةٍ وَبِذَهَبٍ وَبِأَمْتِعَةٍ وَبِبَهَائِمَ مَعَ ٱلتَّبَرُّعِ لِبَيْتِ ٱلرَّبِّ ٱلَّذِي فِي أُورُشَلِيمَ». (4) … وَكُلُّ ٱلَّذِينَ حَوْلَهُمْ أَعَانُوهُمْ بِآنِيَةِ فِضَّةٍ وَبِذَهَبٍ وَبِأَمْتِعَةٍ وَبِبَهَائِمَ وَبِتُحَفٍ، فَضْلًا عَنْ كُلِّ مَا تُبُرِّعَ بِهِ. (6)

2:272: לא עליך להדריכם, אלא אללה מדריך את מי שירצה. וכל טוב שתתנו – לנפשכם הוא; ואינכם נותנים אלא כדי לבקש את פני אללה. וכל טוב שתתנו – יושלם לכם, ולא ייעשה לכם עוול; (272) ✦ 5:2: הוי המאמינים, אל תחללו את סמלי אללה, ולא את החודש הקדוש, ולא את הקורבן, ולא את (הבהמות) העונדות קולרים, ולא את הבאים אל הבית הקדוש בבקשת חסד מאת ריבונם ורצונו. וכאשר תצאו מקדושת העלייה – צודו. ואל תביא אתכם שנאת אנשים, שהרחיקו אתכם מן המסגד הקדוש, להתעלל. ועזרו זה לזה בצדקה וביראה, ואל תעזרו זה לזה בחטא ובאיבה; ויראו את אללה, אללה קשה בעונש. (2)

۞لَّيۡسَ عَلَيۡكَ هُدَىٰهُمۡ وَلَٰكِنَّ ٱللَّهَ يَهۡدِي مَن يَشَآءُۗ وَمَا تُنفِقُواْ مِنۡ خَيۡرٖ فَلِأَنفُسِكُمۡۚ وَمَا تُنفِقُونَ إِلَّا ٱبۡتِغَآءَ وَجۡهِ ٱللَّهِۚ وَمَا تُنفِقُواْ مِنۡ خَيۡرٖ يُوَفَّ إِلَيۡكُمۡ وَأَنتُمۡ لَا تُظۡلَمُونَ (272) ✦ يَـٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُواْ لَا تُحِلُّواْ شَعَـٰٓئِرَ ٱللَّهِ وَلَا ٱلشَّهۡرَ ٱلۡحَرَامَ وَلَا ٱلۡهَدۡيَ وَلَا ٱلۡقَلَـٰٓئِدَ وَلَآ ءَآمِّينَ ٱلۡبَيۡتَ ٱلۡحَرَامَ يَبۡتَغُونَ فَضۡلٗا مِّن رَّبِّهِمۡ وَرِضۡوَٰنٗاۚ وَإِذَا حَلَلۡتُمۡ فَٱصۡطَادُواْۚ وَلَا يَجۡرِمَنَّكُمۡ شَنَـَٔانُ قَوۡمٍ أَن صَدُّوكُمۡ عَنِ ٱلۡمَسۡجِدِ ٱلۡحَرَامِ أَن تَعۡتَدُواْۘ وَتَعَاوَنُواْ عَلَى ٱلۡبِرِّ وَٱلتَّقۡوَىٰۖ وَلَا تَعَاوَنُواْ عَلَى ٱلۡإِثۡمِ وَٱلۡعُدۡوَٰنِۚ وَٱتَّقُواْ ٱللَّهَۖ إِنَّ ٱللَّهَ شَدِيدُ ٱلۡعِقَابِ (2)

וּכְהִתְפַּלֵּל עֶזְרָא וּכְהִתְוַדֹּתוֹ בֹּכֶה וּמִתְנַפֵּל לִפְנֵי בֵּית הָאֱלֹהִים נִקְבְּצוּ אֵלָיו מִיִּשְׂרָאֵל קָהָל רַב־מְאֹד אֲנָשִׁים וְנָשִׁים וִילָדִים כִּי־בָכוּ הָעָם הַרְבֵּה־בֶכֶה׃ (א) וַיַּעַן שְׁכַנְיָה בֶן־יְחִיאֵל מִבְּנֵי (עולם) [עֵילָם] וַיֹּאמֶר לְעֶזְרָא אֲנַחְנוּ מָעַלְנוּ בֵאלֹהֵינוּ וַנֹּשֶׁב נָשִׁים נׇכְרִיּוֹת מֵעַמֵּי הָאָרֶץ וְעַתָּה יֵשׁ־מִקְוֶה לְיִשְׂרָאֵל עַל־זֹאת׃ (ב) וְעַתָּה נִכְרׇת־בְּרִית לֵאלֹהֵינוּ לְהוֹצִיא כׇל־נָשִׁים וְהַנּוֹלָד מֵהֶם בַּעֲצַת אֲדֹנָי וְהַחֲרֵדִים בְּמִצְוַת אֱלֹהֵינוּ וְכַתּוֹרָה יֵעָשֶׂה׃ (ג) קוּם כִּי־עָלֶיךָ הַדָּבָר וַאֲנַחְנוּ עִמָּךְ חֲזַק וַעֲשֵׂה׃ (ד)

فَلَمَّا صَلَّى عَزْرَا وَٱعْتَرَفَ وَهُوَ بَاكٍ وَسَاقِطٌ أَمَامَ بَيْتِ ٱللهِ، ٱجْتَمَعَ إِلَيْهِ مِنْ إِسْرَائِيلَ جَمَاعَةٌ كَثِيرَةٌ جِدًّا مِنَ ٱلرِّجَالِ وَٱلنِّسَاءِ وَٱلْأَوْلَادِ، لِأَنَّ ٱلشَّعْبَ بَكَى بُكَاءً عَظِيمًا. (1) وَأَجَابَ شَكَنْيَا بْنُ يَحِيئِيلَ مِنْ بَنِي عِيلَامَ وَقَالَ لِعَزْرَا: «إِنَّنَا قَدْ خُنَّا إِلَهَنَا وَٱتَّخَذْنَا نِسَاءً غَرِيبَةً مِنْ شُعُوبِ ٱلْأَرْضِ. وَلَكِنِ ٱلْآنَ يُوجَدُ رَجَاءٌ لِإِسْرَائِيلَ فِي هَذَا. (2) فَلْنَقْطَعِ ٱلْآنَ عَهْدًا مَعَ إِلَهِنَا أَنْ نُخْرِجَ كُلَّ ٱلنِّسَاءِ وَٱلَّذِينَ وُلِدُوا مِنْهُنَّ، حَسَبَ مَشُورَةِ سَيِّدِي، وَٱلَّذِينَ يَخْشَوْنَ وَصِيَّةَ إِلَهِنَا، وَلْيُعْمَلْ حَسَبَ ٱلشَّرِيعَةِ. (3) قُمْ فَإِنَّ عَلَيْكَ ٱلْأَمْرَ وَنَحْنُ مَعَكَ. تَشَجَّعْ وَٱفْعَلْ». (4)

39:53: אמור: ״עבדיי, אשר הפריזו (בחטא) על נפשם, אל תתייאשו מרחמי אללה; אללה סולח לכל החטאים. הוא הסולח, הרחום״. (53) ✦ 42:25: והוא אשר מקבל את התשובה מעבדיו, ומוחל על הרעות, ויודע את אשר אתם עושים. (25)

۞قُلۡ يَٰعِبَادِيَ ٱلَّذِينَ أَسۡرَفُواْ عَلَىٰٓ أَنفُسِهِمۡ لَا تَقۡنَطُواْ مِن رَّحۡمَةِ ٱللَّهِۚ إِنَّ ٱللَّهَ يَغۡفِرُ ٱلذُّنُوبَ جَمِيعًاۚ إِنَّهُۥ هُوَ ٱلۡغَفُورُ ٱلرَّحِيمُ (53) ✦ وَهُوَ ٱلَّذِي يَقۡبَلُ ٱلتَّوۡبَةَ عَنۡ عِبَادِهِۦ وَيَعۡفُواْ عَنِ ٱلسَّيِّـَٔاتِ وَيَعۡلَمُ مَا تَفۡعَلُونَ (25)

״ועתה יש מקווה לישראל על זאת... קום כי עליך הדבר ואנחנו עמך, חזק ועשה!״ — דחיית הייאוש המוסרי ותמיכת הציבור במנהיג לתקן משבר (39:53).

«والآن يوجد رجاء لإسرائيل في هذا الأمر... قم لأن عليك الأمر ونحن معك» — تجديد الأمل والإصلاح الجماعي (الزمر 39:53).
