(9) فَأَعْطِنِي ٱلْآنَ حِكْمَةً وَمَعْرِفَةً لِأَخْرُجَ أَمَامَ هَذَا ٱلشَّعْبِ وَأَدْخُلَ، لِأَنَّهُ مَنْ يَقْدِرُ أَنْ يَحْكُمَ عَلَى شَعْبِكَ هَذَا ٱلْعَظِيمِ» (10) فَقَالَ ٱللهُ لِسُلَيْمَانَ: «مِنْ أَجْلِ أَنَّ هَذَا كَانَ فِي قَلْبِكَ، وَلَمْ تَسْأَلْ غِنًى وَلَا أَمْوَالًا وَلَا كَرَامَةً وَلَا أَنْفُسَ مُبْغِضِيكَ، وَلَا سَأَلْتَ أَيَّامًا كَثِيرَةً، بَلْ إِنَّمَا سَأَلْتَ لِنَفْسِكَ حِكْمَةً وَمَعْرِفَةً تَحْكُمُ بِهِمَا عَلَى شَعْبِي ٱلَّذِي مَلَّكْتُكَ عَلَيْهِ، (11) قَدْ أَعْطَيْتُكَ حِكْمَةً وَمَعْرِفَةً، وَأُعْطِيكَ غِنًى وَأَمْوَالًا وَكَرَامَةً لَمْ يَكُنْ مِثْلُهَا لِلْمُلُوكِ ٱلَّذِينَ قَبْلَكَ، وَلَا يَكُونُ مِثْلُهَا لِمَنْ بَعْدَكَ». وَأَدۡخِلۡنِي بِرَحۡمَتِكَ فِي عِبَادِكَ ٱلصَّـٰلِحِينَ (19) وَتَفَقَّدَ ٱلطَّيۡرَ فَقَالَ مَالِيَ لَآ أَرَى ٱلۡهُدۡهُدَ أَمۡ كَانَ مِنَ ٱلۡغَآئِبِينَ (20) لَأُعَذِّبَنَّهُۥ عَذَابٗا شَدِيدًا أَوۡ لَأَاْذۡبَحَنَّهُۥٓ أَوۡ لَيَأۡتِيَنِّي بِسُلۡطَٰنٖ مُّبِينٖ (21) فَمَكَثَ غَيۡرَ بَعِيدٖ فَقَالَ أَحَطتُ بِمَا لَمۡ تُحِطۡ بِهِۦ وَجِئۡتُكَ مِن سَبَإِۭ بِنَبَإٖ يَقِينٍ (22) إِنِّي وَجَدتُّ ٱمۡرَأَةٗ تَمۡلِكُهُمۡ وَأُوتِيَتۡ مِن كُلِّ شَيۡءٖ وَلَهَا عَرۡشٌ عَظِيمٞ (23) وَجَدتُّهَا وَقَوۡمَهَا يَسۡجُدُونَ لِلشَّمۡسِ مِن دُونِ ٱللَّهِ وَزَيَّنَ لَهُمُ ٱلشَّيۡطَٰنُ أَعۡمَٰلَهُمۡ فَصَدَّهُمۡ عَنِ ٱلسَّبِيلِ فَهُمۡ لَا يَهۡتَدُونَ (24) أَلَّاۤ يَسۡجُدُواْۤ لِلَّهِ ٱلَّذِي يُخۡرِجُ ٱلۡخَبۡءَ فِي ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ وَيَعۡلَمُ مَا تُخۡفُونَ وَمَا تُعۡلِنُونَ (25) ٱللَّهُ لَآ إِلَٰهَ إِلَّا هُوَ رَبُّ ٱلۡعَرۡشِ ٱلۡعَظِيمِ۩ (26) ۞قَالَ سَنَنظُرُ أَصَدَقۡتَ أَمۡ كُنتَ مِنَ ٱلۡكَٰذِبِينَ (27) ٱذۡهَب بِّكِتَٰبِي هَٰذَا فَأَلۡقِهۡ إِلَيۡهِمۡ ثُمَّ تَوَلَّ عَنۡهُمۡ فَٱنظُرۡ مَاذَا يَرۡجِعُونَ (28) قَالَتۡ يَـٰٓأَيُّهَا ٱلۡمَلَؤُاْ إِنِّيٓ أُلۡقِيَ إِلَيَّ كِتَٰبٞ كَرِيمٌ (29) إِنَّهُۥ مِن سُلَيۡمَٰنَ وَإِنَّهُۥ بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِيمِ (30) أَلَّا تَعۡلُواْ عَلَيَّ وَأۡتُونِي مُسۡلِمِينَ (31) قَالَتۡ يَـٰٓأَيُّهَا ٱلۡمَلَؤُاْ أَفۡتُونِي فِيٓ أَمۡرِي مَا كُنتُ قَاطِعَةً أَمۡرًا حَتَّىٰ تَشۡهَدُونِ (32) قَالُواْ نَحۡنُ أُوْلُواْ قُوَّةٖ وَأُوْلُواْ بَأۡسٖ شَدِيدٖ وَٱلۡأَمۡرُ إِلَيۡكِ فَٱنظُرِي مَاذَا تَأۡمُرِينَ (33) قَالَتۡ إِنَّ ٱلۡمُلُوكَ إِذَا دَخَلُواْ قَرۡيَةً أَفۡسَدُوهَا وَجَعَلُوٓاْ أَعِزَّةَ أَهۡلِهَآ أَذِلَّةٗۚ وَكَذَٰلِكَ يَفۡعَلُونَ (34) وَإِنِّي مُرۡسِلَةٌ إِلَيۡهِم بِهَدِيَّةٖ فَنَاظِرَةُۢ بِمَ يَرۡجِعُ ٱلۡمُرۡسَلُونَ (35) فَلَمَّا جَآءَ سُلَيۡمَٰنَ قَالَ أَتُمِدُّونَنِ بِمَالٖ فَمَآ ءَاتَىٰنِۦَ ٱللَّهُ خَيۡرٞ مِّمَّآ ءَاتَىٰكُمۚ بَلۡ أَنتُم بِهَدِيَّتِكُمۡ تَفۡرَحُونَ (36) ٱرۡجِعۡ إِلَيۡهِمۡ فَلَنَأۡتِيَنَّهُم بِجُنُودٖ لَّا قِبَلَ لَهُم بِهَا وَلَنُخۡرِجَنَّهُم مِّنۡهَآ أَذِلَّةٗ وَهُمۡ صَٰغِرُونَ (37) قَالَ يَـٰٓأَيُّهَا ٱلۡمَلَؤُاْ أَيُّكُمۡ يَأۡتِينِي بِعَرۡشِهَا قَبۡلَ أَن يَأۡتُونِي مُسۡلِمِينَ (38) قَالَ عِفۡرِيتٞ مِّنَ ٱلۡجِنِّ أَنَا۠ ءَاتِيكَ بِهِۦ قَبۡلَ أَن تَقُومَ مِن مَّقَامِكَۖ وَإِنِّي عَلَيۡهِ لَقَوِيٌّ أَمِينٞ (39) قَالَ ٱلَّذِي عِندَهُۥ عِلۡمٞ مِّنَ ٱلۡكِتَٰبِ أَنَا۠ ءَاتِيكَ بِهِۦ قَبۡلَ أَن يَرۡتَدَّ إِلَيۡكَ طَرۡفُكَۚ فَلَمَّا رَءَاهُ مُسۡتَقِرًّا عِندَهُۥ قَالَ هَٰذَا مِن فَضۡلِ رَبِّي لِيَبۡلُوَنِيٓ ءَأَشۡكُرُ أَمۡ أَكۡفُرُۖ وَمَن شَكَرَ فَإِنَّمَا يَشۡكُرُ لِنَفۡسِهِۦۖ وَمَن كَفَرَ فَإِنَّ رَبِّي غَنِيّٞ كَرِيمٞ (40) قَالَ نَكِّرُواْ لَهَا عَرۡشَهَا نَنظُرۡ أَتَهۡتَدِيٓ أَمۡ تَكُونُ مِنَ ٱلَّذِينَ لَا يَهۡتَدُونَ (41) فَلَمَّا جَآءَتۡ قِيلَ أَهَٰكَذَا عَرۡشُكِۖ قَالَتۡ كَأَنَّهُۥ هُوَۚ وَأُوتِينَا ٱلۡعِلۡمَ مِن قَبۡلِهَا وَكُنَّا مُسۡلِمِينَ (42) وَصَدَّهَا مَا كَانَت تَّعۡبُدُ مِن دُونِ ٱللَّهِۖ إِنَّهَا كَانَتۡ مِن قَوۡمٖ كَٰفِرِينَ (43) قِيلَ لَهَا ٱدۡخُلِي ٱلصَّرۡحَۖ فَلَمَّا رَأَتۡهُ حَسِبَتۡهُ لُجَّةٗ وَكَشَفَتۡ عَن سَاقَيۡهَاۚ قَالَ إِنَّهُۥ صَرۡحٞ مُّمَرَّدٞ مِّن قَوَارِيرَۗ قَالَتۡ رَبِّ إِنِّي ظَلَمۡتُ نَفۡسِي وَأَسۡلَمۡتُ مَعَ سُلَيۡمَٰنَ لِلَّهِ رَبِّ ٱلۡعَٰلَمِينَ (44)

דברי הימים ב ט, א–יב ⇄ 27:20–44

וּמַלְכַּת־שְׁבָא שָׁמְעָה אֶת־שֵׁמַע שְׁלֹמֹה וַתָּבוֹא לְנַסּוֹת אֶת־שְׁלֹמֹה בְחִידוֹת בִּירוּשָׁלִַם בְּחַיִל כָּבֵד מְאֹד וּגְמַלִּים נֹשְׂאִים בְּשָׂמִים וְזָהָב לָרֹב וְאֶבֶן יְקָרָה וַתָּבוֹא אֶל־שְׁלֹמֹה וַתְּדַבֵּר עִמּוֹ אֵת כׇּל־אֲשֶׁר הָיָה עִם־לְבָבָהּ׃ (א) וַיַּגֶּד־לָהּ שְׁלֹמֹה אֶת־כׇּל־דְּבָרֶיהָ וְלֹא־נֶעְלַם דָּבָר מִשְּׁלֹמֹה אֲשֶׁר לֹא הִגִּיד לָהּ׃ (ב) וַתֵּרֶא מַלְכַּת־שְׁבָא אֵת חׇכְמַת שְׁלֹמֹה וְהַבַּיִת אֲשֶׁר בָּנָה׃ (ג) וּמַאֲכַל שֻׁלְחָנוֹ וּמוֹשַׁב עֲבָדָיו וּמַעֲמַד מְשָׁרְתָיו וּמַלְבּוּשֵׁיהֶם וּמַשְׁקָיו וּמַלְבּוּשֵׁיהֶם וַעֲלִיָּתוֹ אֲשֶׁר יַעֲלֶה בֵּית יְהֹוָה וְלֹא־הָיָה עוֹד בָּהּ רוּחַ׃ (ד) וַתֹּאמֶר אֶל־הַמֶּלֶךְ אֱמֶת הַדָּבָר אֲשֶׁר שָׁמַעְתִּי בְּאַרְצִי עַל־דְּבָרֶיךָ וְעַל־חׇכְמָתֶךָ׃ (ה) וְלֹא־הֶאֱמַנְתִּי לְדִבְרֵיהֶם עַד אֲשֶׁר־בָּאתִי וַתִּרְאֶינָה עֵינַי וְהִנֵּה לֹא הֻגַּד־לִי חֲצִי מַרְבִּית חׇכְמָתֶךָ יָסַפְתָּ עַל־הַשְּׁמוּעָה אֲשֶׁר שָׁמָעְתִּי׃ (ו) אַשְׁרֵי אֲנָשֶׁיךָ וְאַשְׁרֵי עֲבָדֶיךָ אֵלֶּה הָעֹמְדִים לְפָנֶיךָ תָּמִיד וְשֹׁמְעִים אֶת־חׇכְמָתֶךָ׃ (ז) יְהִי יְהֹוָה אֱלֹהֶיךָ בָּרוּךְ אֲשֶׁר חָפֵץ בְּךָ לְתִתְּךָ עַל־כִּסְאוֹ לְמֶלֶךְ לַיהֹוָה אֱלֹהֶיךָ בְּאַהֲבַת אֱלֹהֶיךָ אֶת־יִשְׂרָאֵל לְהַעֲמִידוֹ לְעוֹלָם וַיִּתֶּנְךָ עֲלֵיהֶם לְמֶלֶךְ לַעֲשׂוֹת מִשְׁפָּט וּצְדָקָה׃ (ח) וַתִּתֵּן לַמֶּלֶךְ מֵאָה וְעֶשְׂרִים כִּכַּר זָהָב וּבְשָׂמִים לָרֹב מְאֹד וְאֶבֶן יְקָרָה וְלֹא הָיָה כַּבֹּשֶׂם הַהוּא אֲשֶׁר־נָתְנָה מַלְכַּת־שְׁבָא לַמֶּלֶךְ שְׁלֹמֹה׃ (ט) וְגַם־עַבְדֵי (חירם) [חוּרָם] וְעַבְדֵי שְׁלֹמֹה אֲשֶׁר־הֵבִיאוּ זָ …

وَسَمِعَتْ مَلِكَةُ سَبَا بِخَبَرِ سُلَيْمَانَ، فَأَتَتْ لِتَمْتَحِنَ سُلَيْمَانَ بِمَسَائِلَ إِلَى أُورُشَلِيمَ، بِمَوْكِبٍ عَظِيمٍ جِدًّا، وَجِمَالٍ حَامِلَةٍ أَطْيَابًا وَذَهَبًا بِكَثْرَةٍ وَحِجَارَةً كَرِيمَةً، فَأَتَتْ إِلَى سُلَيْمَانَ وَكَلَّمَتْهُ عَنْ كُلِّ مَا فِي قَلْبِهَا. (1) فَأَخْبَرَهَا سُلَيْمَانُ بِكُلِّ كَلَامِهَا. وَلَمْ يُخْفَ عَنْ سُلَيْمَانَ أَمْرٌ إِلَّا وَأَخْبَرَهَا بِهِ. (2) فَلَمَّا رَأَتْ مَلِكَةُ سَبَا حِكْمَةَ سُلَيْمَانَ وَٱلْبَيْتَ ٱلَّذِي بَنَاهُ، (3) وَطَعَامَ مَائِدَتِهِ، وَمَجْلِسَ عَبِيدِهِ، وَمَوْقِفَ خُدَّامِهِ وَمَلَابِسَهُمْ، وَسُقَاتَهُ وَمَلَابِسَهُمْ، وَمُحْرَقَاتِهِ ٱلَّتِي كَانَ يُصْعِدُهَا فِي بَيْتِ ٱلرَّبِّ، لَمْ تَبْقَ فِيهَا رُوحٌ بَعْدُ. (4) فَقَالَتْ لِلْمَلِكِ: «صَحِيحٌ ٱلْخَبَرُ ٱلَّذِي سَمِعْتُهُ فِي أَرْضِي عَنْ أُمُورِكَ وَعَنْ حِكْمَتِكَ. (5) وَلَمْ أُصَدِّقْ كَلَامَهُمْ حَتَّى جِئْتُ وَأَبْصَرَتْ عَيْنَايَ، فَهُوَذَا لَمْ أُخْبَرْ بِنِصْفِ كَثْرَةِ حِكْمَتِكَ. زِدْتَ عَلَى ٱلْخَبَرِ ٱلَّذِي سَمِعْتُهُ. (6) فَطُوبَى لِرِجَالِكَ وَطُوبَى لِعَبِيدِكَ هَؤُلَاءِ ٱلْوَاقِفِينَ أَمَامَكَ دَائِمًا وَٱلسَّامِعِينَ حِكْمَتَكَ. (7) لِيَكُنْ مُبَارَكًا ٱلرَّبُّ إِلَهُكَ ٱلَّذِي سُرَّ بِكَ وَجَعَلَكَ عَلَى كُرْسِيِّهِ مَلِكًا لِلرَّبِّ إِلَهِكَ. لِأَنَّ إِلَهَكَ أَحَبَّ إِسْرَائِيلَ لِيُثْبِتَهُ إِلَى ٱلْأَبَدِ، قَدْ جَعَلَكَ عَلَيْهِمْ مَلِكًا، لِتُجْرِيَ حُكْمًا وَعَدْلًا». (8) وَأَهْدَتْ لِلْمَلِكِ مِئَةً وَعِشْرِينَ وَزْنَةَ ذَهَبٍ وَأَطْيَابًا كَثِيرَةً جِدًّا وَحِجَارَةً كَرِيمَةً، وَلَمْ يَكُنْ مِثْلُ ذَلِكَ ٱلطِّيبِ ٱلّ …

ופקד את העופות, ואמר: ״מה לי שאינני רואה את הדוכיפת? האם היא מן הנעדרים? (20) אענישנה עונש קשה, או אשחטנה, או שתביא לי ראיה ברורה״. (21) ושהתה זמן לא רב, ואמרה: ״השגתי את אשר לא השגת, ובאתי אליך משבא עם בשורה ודאית. (22) מצאתי אישה המולכת עליהם, וניתן לה מכל דבר, ולה כס גדול. (23) מצאתיה ואת עמה משתחווים לשמש מלבד אללה; והשטן יפה להם את מעשיהם והרחיקם מן הדרך, והם אינם מודרכים – (24) שלא ישתחוו לאללה, המוציא את הצפון בשמים ובארץ, ויודע את אשר תסתירו ואת אשר תגלו. (25) אללה – אין אלוה מלבדו, ריבון הכס הגדול״. (26) אמר: ״נראה אם אמת דיברת, או מן השקרנים אתה. (27) לכי עם מכתבי זה והשליכיהו אליהם, ואחר כך פני מהם וראי מה ישיבו״. (28) אמרה: ״הוי השרים, הושלך אליי מכתב נכבד. (29) הוא משלמה, והוא: בשם אללה הרחמן הרחום. (30) אל תתנשאו עליי, ובואו אליי מתמסרים״. (31) אמרה: ״הוי השרים, יעצוני בענייני; לא הכרעתי דבר עד שהייתם נוכחים״. (32) אמרו: ״אנחנו בעלי כוח ובעלי עוז רב, והעניין בידך; ראי מה תצווי״. (33) אמרה: ״המלכים, כאשר הם נכנסים לעיר, משחיתים אותה ועושים את נכבדי יושביה לשפלים; וכך הם עושים. (34) ואני שולחת אליהם מנחה, ואראה עם מה ישובו השליחים״. (35) וכאשר בא (השליח) אל שלמה, אמר: ״התתמכו בי בהון? מה שנתן לי אללה טוב ממה שנתן לכם. לא, אתם במנחתכם שמחים. (36) שוב אליהם; ונבוא אליהם עם צבאות שאין להם כוח לעמוד מולם, ונוציאם ממנה שפלים, והם מושפלים״. (37) אמר: ״הוי השרים, מי מכם יביא לי את כסאה לפני שיבואו אליי מתמסרים?״ (38) אמר עִפְרִית מן הג׳ן: ״אני אביאו אליך לפני שתקום ממקומך; ואני חזק ונאמן לכך״. (39) אמר זה אשר אצלו דעת מן הספר: ״אני אביאו אליך לפני שתשוב אליך עינך״. וכאשר ראהו עומד אצלו, אמר: ״זה מחסד ריבוני, לנסותני האודה או אכפור. ומי שמודה – לטובת עצמו מודה, ומי שכופר – ריבוני אינו נצרך, נדיב״. (40) אמר: ״שנו לה את כסאה; נראה אם תכיר, או תהיה מאלה שאינם מכירים״. (41) וכאשר באה, נאמר: ״האם כזה כסאך?״ אמרה: ״כאילו הוא״. (ואמר שלמה:) ״וניתנה לנו הדעת לפניה, והיינו מתמסרים״. (42) והרחיק אותה מה שעבדה מלבד אללה, כי הייתה מעם כופר. (43) נאמר לה: ״היכנסי לארמון״. וכאשר ראתה אותו, חשבה אותו למים עמוקים, וגילתה את שוקיה. אמר: ״זה ארמון מרוצף זכוכית״. אמרה: ״ריבוני, חטאתי לנפשי, ואני מתמסרת עם שלמה לאללה, ריבון העולמים״. (44)

وَتَفَقَّدَ ٱلطَّيۡرَ فَقَالَ مَالِيَ لَآ أَرَى ٱلۡهُدۡهُدَ أَمۡ كَانَ مِنَ ٱلۡغَآئِبِينَ (20) لَأُعَذِّبَنَّهُۥ عَذَابٗا شَدِيدًا أَوۡ لَأَاْذۡبَحَنَّهُۥٓ أَوۡ لَيَأۡتِيَنِّي بِسُلۡطَٰنٖ مُّبِينٖ (21) فَمَكَثَ غَيۡرَ بَعِيدٖ فَقَالَ أَحَطتُ بِمَا لَمۡ تُحِطۡ بِهِۦ وَجِئۡتُكَ مِن سَبَإِۭ بِنَبَإٖ يَقِينٍ (22) إِنِّي وَجَدتُّ ٱمۡرَأَةٗ تَمۡلِكُهُمۡ وَأُوتِيَتۡ مِن كُلِّ شَيۡءٖ وَلَهَا عَرۡشٌ عَظِيمٞ (23) وَجَدتُّهَا وَقَوۡمَهَا يَسۡجُدُونَ لِلشَّمۡسِ مِن دُونِ ٱللَّهِ وَزَيَّنَ لَهُمُ ٱلشَّيۡطَٰنُ أَعۡمَٰلَهُمۡ فَصَدَّهُمۡ عَنِ ٱلسَّبِيلِ فَهُمۡ لَا يَهۡتَدُونَ (24) أَلَّاۤ يَسۡجُدُواْۤ لِلَّهِ ٱلَّذِي يُخۡرِجُ ٱلۡخَبۡءَ فِي ٱلسَّمَٰوَٰتِ وَٱلۡأَرۡضِ وَيَعۡلَمُ مَا تُخۡفُونَ وَمَا تُعۡلِنُونَ (25) ٱللَّهُ لَآ إِلَٰهَ إِلَّا هُوَ رَبُّ ٱلۡعَرۡشِ ٱلۡعَظِيمِ۩ (26) ۞قَالَ سَنَنظُرُ أَصَدَقۡتَ أَمۡ كُنتَ مِنَ ٱلۡكَٰذِبِينَ (27) ٱذۡهَب بِّكِتَٰبِي هَٰذَا فَأَلۡقِهۡ إِلَيۡهِمۡ ثُمَّ تَوَلَّ عَنۡهُمۡ فَٱنظُرۡ مَاذَا يَرۡجِعُونَ (28) قَالَتۡ يَـٰٓأَيُّهَا ٱلۡمَلَؤُاْ إِنِّيٓ أُلۡقِيَ إِلَيَّ كِتَٰبٞ كَرِيمٌ (29) إِنَّهُۥ مِن سُلَيۡمَٰنَ وَإِنَّهُۥ بِسۡمِ ٱللَّهِ ٱلرَّحۡمَٰنِ ٱلرَّحِيمِ (30) أَلَّا تَعۡلُواْ عَلَيَّ وَأۡتُونِي مُسۡلِمِينَ (31) قَالَتۡ يَـٰٓأَيُّهَا ٱلۡمَلَؤُاْ أَفۡتُونِي فِيٓ أَمۡرِي مَا كُنتُ قَاطِعَةً أَمۡرًا حَتَّىٰ تَشۡهَدُونِ (32) قَالُواْ نَحۡنُ أُوْلُواْ قُوَّةٖ وَأُوْلُواْ بَأۡسٖ شَدِيدٖ وَٱلۡأَمۡرُ إِلَيۡكِ فَٱنظُرِي مَاذَا تَأۡمُرِينَ (33) قَالَتۡ إِنَّ ٱلۡمُلُوكَ إِذَا دَخَلُواْ قَرۡيَةً أَفۡسَدُوهَا وَجَعَلُوٓاْ أَعِزَّةَ أَهۡلِهَآ أَذِلَّةٗۚ وَكَذَٰلِكَ يَفۡعَلُونَ (34) وَإِنِّي مُرۡسِلَةٌ إِلَيۡهِم بِهَدِيَّةٖ فَنَاظِرَةُۢ بِمَ يَرۡجِعُ ٱلۡمُرۡسَلُونَ (35) فَلَمَّا جَآءَ سُلَيۡمَٰنَ قَالَ أَتُمِدُّونَنِ بِمَالٖ فَمَآ ءَاتَىٰنِۦَ ٱللَّهُ خَيۡرٞ مِّمَّآ ءَاتَىٰكُمۚ بَلۡ أَنتُم بِهَدِيَّتِكُمۡ تَفۡرَحُونَ (36) ٱرۡجِعۡ إِلَيۡهِمۡ فَلَنَأۡتِيَنَّهُم بِجُنُودٖ لَّا قِبَلَ لَهُم بِهَا وَلَنُخۡرِجَنَّهُم مِّنۡهَآ أَذِلَّةٗ وَهُمۡ صَٰغِرُونَ (37) قَالَ يَـٰٓأَيُّهَا ٱلۡمَلَؤُاْ أَيُّكُمۡ يَأۡتِينِي بِعَرۡشِهَا قَبۡلَ أَن يَأۡتُونِي مُسۡلِمِينَ (38) قَالَ عِفۡرِيتٞ مِّنَ ٱلۡجِنِّ أَنَا۠ ءَاتِيكَ بِهِۦ قَبۡلَ أَن تَقُومَ مِن مَّقَامِكَۖ وَإِنِّي عَلَيۡهِ لَقَوِيٌّ أَمِينٞ (39) قَالَ ٱلَّذِي عِندَهُۥ عِلۡمٞ مِّنَ ٱلۡكِتَٰبِ أَنَا۠ ءَاتِيكَ بِهِۦ قَبۡلَ أَن يَرۡتَدَّ إِلَيۡكَ طَرۡفُكَۚ فَلَمَّا رَءَاهُ مُسۡتَقِرًّا عِندَهُۥ قَالَ هَٰذَا مِن فَضۡلِ رَبِّي لِيَبۡلُوَنِيٓ ءَأَشۡكُرُ أَمۡ أَكۡفُرُۖ وَمَن شَكَرَ فَإِنَّمَا يَشۡكُرُ لِنَفۡسِهِۦۖ وَمَن كَفَرَ فَإِنَّ رَبِّي غَنِيّٞ كَرِيمٞ (40) قَالَ نَكِّرُواْ لَهَا عَرۡشَهَا نَنظُرۡ أَتَهۡتَدِيٓ أَمۡ تَكُونُ مِنَ ٱلَّذِينَ لَا يَهۡتَدُونَ (41) فَلَمَّا جَآءَتۡ قِيلَ أَهَٰكَذَا عَرۡشُكِۖ قَالَتۡ كَأَنَّهُۥ هُوَۚ وَأُوتِينَا ٱلۡعِلۡمَ مِن قَبۡلِهَا وَكُنَّا مُسۡلِمِينَ (42) وَصَدَّهَا مَا كَانَت تَّعۡبُدُ مِن دُونِ ٱللَّهِۖ إِنَّهَا كَانَتۡ مِن قَوۡمٖ كَٰفِرِينَ (43) قِيلَ لَهَا ٱدۡخُلِي ٱلصَّرۡحَۖ فَلَمَّا رَأَتۡهُ حَسِبَتۡهُ لُجَّةٗ وَكَشَفَتۡ عَن سَاقَيۡهَاۚ قَالَ إِنَّهُۥ صَرۡحٞ مُّمَرَّدٞ مِّن قَوَارِيرَۗ قَالَتۡ رَبِّ إِنِّي ظَلَمۡتُ نَفۡسِي وَأَسۡلَمۡتُ مَعَ سُلَيۡمَٰنَ لِلَّهِ رَبِّ ٱلۡعَٰلَمِينَ (44)

וַיְדַבְּרוּ אֵלָיו לֵאמֹר אִם־תִּהְיֶה לְטוֹב לְהָעָם הַזֶּה וּרְצִיתָם וְדִבַּרְתָּ אֲלֵהֶם דְּבָרִים טוֹבִים וְהָיוּ לְךָ עֲבָדִים כׇּל־הַיָּמִים׃ (ז) וַיַּעֲזֹב אֶת־עֲצַת הַזְּקֵנִים אֲשֶׁר יְעָצֻהוּ וַיִּוָּעַץ אֶת־הַיְלָדִים אֲשֶׁר גָּדְלוּ אִתּוֹ הָעֹמְדִים לְפָנָיו׃ (ח) וַיֹּאמֶר אֲלֵהֶם מָה אַתֶּם נוֹעָצִים וְנָשִׁיב דָּבָר אֶת־הָעָם הַזֶּה אֲשֶׁר דִּבְּרוּ אֵלַי לֵאמֹר הָקֵל מִן־הָעֹל אֲשֶׁר־נָתַן אָבִיךָ עָלֵינוּ׃ (ט) וַיְדַבְּרוּ אִתּוֹ הַיְלָדִים אֲשֶׁר גָּדְלוּ אִתּוֹ לֵאמֹר כֹּה־תֹאמַר לָעָם אֲשֶׁר־דִּבְּרוּ אֵלֶיךָ לֵאמֹר אָבִיךָ הִכְבִּיד אֶת־עֻלֵּנוּ וְאַתָּה הָקֵל מֵעָלֵינוּ כֹּה תֹּאמַר אֲלֵהֶם קָטׇנִּי עָבָה מִמׇּתְנֵי אָבִי׃ (י) וְעַתָּה אָבִי הֶעְמִיס עֲלֵיכֶם עֹל כָּבֵד וַאֲנִי אֹסִיף עַל־עֻלְּכֶם אָבִי יִסַּר אֶתְכֶם בַּשּׁוֹטִים וַאֲנִי בָּעַקְרַבִּים׃ (יא)

(10) وَٱلْآنَ أَبِي حَمَّلَكُمْ نِيرًا ثَقِيلًا وَأَنَا أَزِيدُ عَلَى نِيرِكُمْ. أَبِي أَدَّبَكُمْ بِٱلسِّيَاطِ وَأَمَّا أَنَا فَبِٱلْعَقَارِبِ». (11)

וּכְהָצֵר לוֹ חִלָּה אֶת־פְּנֵי יְהֹוָה אֱלֹהָיו וַיִּכָּנַע מְאֹד מִלִּפְנֵי אֱלֹהֵי אֲבֹתָיו׃ (יב) וַיִּתְפַּלֵּל אֵלָיו וַיֵּעָתֶר לוֹ וַיִּשְׁמַע תְּחִנָּתוֹ וַיְשִׁיבֵהוּ יְרוּשָׁלִַם לְמַלְכוּתוֹ וַיֵּדַע מְנַשֶּׁה כִּי יְהֹוָה הוּא הָאֱלֹהִים׃ (יג)

وَلَمَّا تَضَايَقَ طَلَبَ وَجْهَ ٱلرَّبِّ إِلَهِهِ، وَتَوَاضَعَ جِدًّا أَمَامَ إِلَهِ آبَائِهِ، (12) وَصَلَّى إِلَيْهِ فَٱسْتَجَابَ لَهُ وَسَمِعَ تَضَرُّعَهُ، وَرَدَّهُ إِلَى أُورُشَلِيمَ إِلَى مَمْلَكَتِهِ. فَعَلِمَ مَنَسَّى أَنَّ ٱلرَّبَّ هُوَ ٱللهُ. (13)

39:53: אמור: ״עבדיי, אשר הפריזו (בחטא) על נפשם, אל תתייאשו מרחמי אללה; אללה סולח לכל החטאים. הוא הסולח, הרחום״. (53) ✦ 25:70: מלבד מי שחזר בתשובה, האמין ועשה מעשה טוב – לאלה יחליף אללה את מעשיהם הרעים במעשים טובים; ואללה סולח, רחום. (70) ✦ 4:110: ומי שעושה רע או עושה עוול לנפשו, ואחר כך מבקש מחילה מאללה – ימצא את אללה סולח ורחום. (110)

۞قُلۡ يَٰعِبَادِيَ ٱلَّذِينَ أَسۡرَفُواْ عَلَىٰٓ أَنفُسِهِمۡ لَا تَقۡنَطُواْ مِن رَّحۡمَةِ ٱللَّهِۚ إِنَّ ٱللَّهَ يَغۡفِرُ ٱلذُّنُوبَ جَمِيعًاۚ إِنَّهُۥ هُوَ ٱلۡغَفُورُ ٱلرَّحِيمُ (53) ✦ إِلَّا مَن تَابَ وَءَامَنَ وَعَمِلَ عَمَلٗا صَٰلِحٗا فَأُوْلَـٰٓئِكَ يُبَدِّلُ ٱللَّهُ سَيِّـَٔاتِهِمۡ حَسَنَٰتٖۗ وَكَانَ ٱللَّهُ غَفُورٗا رَّحِيمٗا (70) ✦ وَمَن يَعۡمَلۡ سُوٓءًا أَوۡ يَظۡلِمۡ نَفۡسَهُۥ ثُمَّ يَسۡتَغۡفِرِ ٱللَّهَ يَجِدِ ٱللَّهَ غَفُورٗا رَّحِيمٗا (110)

תשובת מנשה המלך (המלך שחטא יותר מכל מלכי יהודה, נשבה בשלשלאות לבבל, נכנע והתפלל והאל קיבל את תשובתו והשיבו!) — המחשה קיצונית לשער התשובה הפתוח גם לחוטא הגדול ביותר (39:53: ״אל תתייאשו מרחמי אללה, אללה סולח לכל העוונות כולם!״).

توبة الملك منسى رغم جسامة ذنوبه وقبول الله تضرعه في الأسر يطابق تمامًا الزمر 39:53: «لا تقنطوا من رحمة الله إن الله يغفر الذنوب جميعًا»!

דברי הימים ב לו, טו–טז ⇄ 36:30

וַיִּשְׁלַח יְהֹוָה אֱלֹהֵי אֲבוֹתֵיהֶם עֲלֵיהֶם בְּיַד מַלְאָכָיו הַשְׁכֵּם וְשָׁלוֹחַ כִּי־חָמַל עַל־עַמּוֹ וְעַל־מְעוֹנוֹ׃ (טו) וַיִּהְיוּ מַלְעִבִים בְּמַלְאֲכֵי הָאֱלֹהִים וּבוֹזִים דְּבָרָיו וּמִתַּעְתְּעִים בִּנְבִאָיו עַד עֲלוֹת חֲמַת־יְהֹוָה בְּעַמּוֹ עַד־לְאֵין מַרְפֵּא׃ (טז)

فَأَرْسَلَ ٱلرَّبُّ إِلَهُ آبَائِهِمْ إِلَيْهِمْ عَنْ يَدِ رُسُلِهِ مُبَكِّرًا وَمُرْسِلًا لِأَنَّهُ شَفِقَ عَلَى شَعْبِهِ وَعَلَى مَسْكَنِهِ، (15) فَكَانُوا يَهْزَأُونَ بِرُسُلِ ٱللهِ، وَرَذَلُوا كَلَامَهُ وَتَهَاوَنُوا بِأَنْبِيَائِهِ حَتَّى ثَارَ غَضَبُ ٱلرَّبِّ عَلَى شَعْبِهِ حَتَّى لَمْ يَكُنْ شِفَاءٌ. (16)

אוי לעבדים! לא בא אליהם שליח אלא לעגו לו. (30)

يَٰحَسۡرَةً عَلَى ٱلۡعِبَادِۚ مَا يَأۡتِيهِم مِّن رَّسُولٍ إِلَّا كَانُواْ بِهِۦ يَسۡتَهۡزِءُونَ (30)

״ויהיו מלעבים במלאכי האלהים ובוזים דבריו ומתעתעים בנביאיו״ – ״אוי לעבדים! לא בא אליהם שליח שלא לעגו לו״.
