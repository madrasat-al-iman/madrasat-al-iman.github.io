(95) הביאו לי גושי ברזל״. עד אשר השווה (את הגובה) בין שני צדי ההרים, אמר: ״נפחו!״ עד אשר עשהו אש, אמר: ״הביאו לי, ואצוק עליו נחושת מותכת״. (96) ולא יכלו לעלות עליו, ולא יכלו לנקבו. (97) אמר: ״זה רחמים מאת ריבוני. וכאשר תבוא הבטחת ריבוני – יעשהו עפר; והבטחת ריבוני אמת״. (98)

قَالُواْ يَٰذَا ٱلۡقَرۡنَيۡنِ إِنَّ يَأۡجُوجَ وَمَأۡجُوجَ مُفۡسِدُونَ فِي ٱلۡأَرۡضِ فَهَلۡ نَجۡعَلُ لَكَ خَرۡجًا عَلَىٰٓ أَن تَجۡعَلَ بَيۡنَنَا وَبَيۡنَهُمۡ سَدّٗا (94) قَالَ مَا مَكَّنِّي فِيهِ رَبِّي خَيۡرٞ فَأَعِينُونِي بِقُوَّةٍ أَجۡعَلۡ بَيۡنَكُمۡ وَبَيۡنَهُمۡ رَدۡمًا (95) ءَاتُونِي زُبَرَ ٱلۡحَدِيدِۖ حَتَّىٰٓ إِذَا سَاوَىٰ بَيۡنَ ٱلصَّدَفَيۡنِ قَالَ ٱنفُخُواْۖ حَتَّىٰٓ إِذَا جَعَلَهُۥ نَارٗا قَالَ ءَاتُونِيٓ أُفۡرِغۡ عَلَيۡهِ قِطۡرٗا (96) فَمَا ٱسۡطَٰعُوٓاْ أَن يَظۡهَرُوهُ وَمَا ٱسۡتَطَٰعُواْ لَهُۥ نَقۡبٗا (97) قَالَ هَٰذَا رَحۡمَةٞ مِّن رَّبِّيۖ فَإِذَا جَآءَ وَعۡدُ رَبِّي جَعَلَهُۥ دَكَّآءَۖ وَكَانَ وَعۡدُ رَبِّي حَقّٗا (98)

וַיְהִי דְבַר־יְהֹוָה אֵלַי לֵאמֹר׃ (א) בֶּן־אָדָם שִׂים פָּנֶיךָ אֶל־גּוֹג אֶרֶץ הַמָּגוֹג נְשִׂיא רֹאשׁ מֶשֶׁךְ וְתֻבָל וְהִנָּבֵא עָלָיו׃ (ב) וְאָמַרְתָּ כֹּה אָמַר אֲדֹנָי יֱהֹוִה הִנְנִי אֵלֶיךָ גּוֹג נְשִׂיא רֹאשׁ מֶשֶׁךְ וְתֻבָל׃ (ג) וְשׁוֹבַבְתִּיךָ וְנָתַתִּי חַחִים בִּלְחָיֶיךָ וְהוֹצֵאתִי אוֹתְךָ וְאֶת־כׇּל־חֵילֶךָ סוּסִים וּפָרָשִׁים לְבֻשֵׁי מִכְלוֹל כֻּלָּם קָהָל רָב צִנָּה וּמָגֵן תֹּפְשֵׂי חֲרָבוֹת כֻּלָּם׃ (ד) פָּרַס כּוּשׁ וּפוּט אִתָּם כֻּלָּם מָגֵן וְכוֹבָע׃ (ה) גֹּמֶר וְכׇל־אֲגַפֶּיהָ בֵּית תּוֹגַרְמָה יַרְכְּתֵי צָפוֹן וְאֶת־כׇּל־אֲגַפָּיו עַמִּים רַבִּים אִתָּךְ׃ (ו) הִכֹּן וְהָכֵן לְךָ אַתָּה וְכׇל־קְהָלֶךָ הַנִּקְהָלִים עָלֶיךָ וְהָיִיתָ לָהֶם לְמִשְׁמָר׃ (ז) מִיָּמִים רַבִּים תִּפָּקֵד בְּאַחֲרִית הַשָּׁנִים תָּבוֹא אֶל־אֶרֶץ מְשׁוֹבֶבֶת מֵחֶרֶב מְקֻבֶּצֶת מֵעַמִּים רַבִּים עַל הָרֵי יִשְׂרָאֵל אֲשֶׁר־הָיוּ לְחׇרְבָּה תָּמִיד וְהִיא מֵעַמִּים הוּצָאָה וְיָשְׁבוּ לָבֶטַח כֻּלָּם׃ (ח) וְעָלִיתָ כַּשֹּׁאָה תָבוֹא כֶּעָנָן לְכַסּוֹת הָאָרֶץ תִּהְיֶה אַתָּה וְכׇל־אֲגַפֶּיךָ וְעַמִּים רַבִּים אוֹתָךְ׃ (ט) כֹּה אָמַר אֲדֹנָי יֱהֹוִה וְהָיָה בַּיּוֹם הַהוּא יַעֲלוּ דְבָרִים עַל־לְבָבֶךָ וְחָשַׁבְתָּ מַחֲשֶׁבֶת רָעָה׃ (י) וְאָמַרְתָּ אֶעֱלֶה עַל־אֶרֶץ פְּרָזוֹת אָבוֹא הַשֹּׁקְטִים יֹשְׁבֵי לָבֶטַח כֻּלָּם יֹשְׁבִים בְּאֵין חוֹמָה וּבְרִיחַ וּדְלָתַיִם אֵין לָהֶם׃ (יא) לִשְׁלֹל שָׁלָל וְלָבֹז בַּז לְהָשִׁיב יָדְךָ עַל־חֳרָבוֹת נוֹשָׁבוֹת וְאֶל־עַם מְאֻסָּף מִגּוֹיִם עֹשֶׂה מִקְנֶה וְקִנְיָן יֹשְׁבֵי עַל־טַבּוּר הָאָרֶץ׃ (יב) שְׁבָא וּדְדָן וְסֹחֲרֵי תַרְשִׁישׁ …

وَكَانَ إِلَيَّ كَلَامُ ٱلرَّبِّ قَائِلًا: (1) «يَا ٱبْنَ آدَمَ، ٱجْعَلْ وَجْهَكَ عَلَى جُوجٍ، أَرْضِ مَاجُوجَ رَئِيسِ رُوشٍ مَاشِكَ وَتُوبَالَ، وَتَنَبَّأْ عَلَيْهِ (2) وَقُلْ: هَكَذَا قَالَ ٱلسَّيِّدُ ٱلرَّبُّ: هَأَنَذَا عَلَيْكَ يَا جُوجُ رَئِيسُ رُوشٍ مَاشِكَ وَتُوبَالَ. (3) وَأُرْجِعُكَ، وَأَضَعُ شَكَائِمَ فِي فَكَّيْكَ، وَأُخْرِجُكَ أَنْتَ وَكُلَّ جَيْشِكَ خَيْلًا وَفُرْسَانًا كُلَّهُمْ لَابِسِينَ أَفْخَرَ لِبَاسٍ، جَمَاعَةً عَظِيمَةً مَعَ أَتْرَاسٍ وَمَجَانَّ، كُلَّهُمْ مُمْسِكِينَ ٱلسُّيُوفَ. (4) فَارِسَ وَكُوشَ وَفُوطَ مَعَهُمْ، كُلَّهُمْ بِمِجَنٍّ وَخُوذَةٍ، (5) وَجُومَرَ وَكُلَّ جُيُوشِهِ، وَبَيْتَ تُوجَرْمَةَ مِنْ أَقَاصِي ٱلشِّمَالِ مَعَ كُلِّ جَيْشِهِ، شُعُوبًا كَثِيرِينَ مَعَكَ. (6) اِسْتَعِدَّ وَهَيِّئْ لِنَفْسِكَ أَنْتَ وَكُلُّ جَمَاعَاتِكَ ٱلْمُجْتَمِعَةِ إِلَيْكَ، فَصِرْتَ لَهُمْ مُوَقَّرًا. (7) بَعْدَ أَيَّامٍ كَثِيرَةٍ تُفْتَقَدُ. فِي ٱلسِّنِينَ ٱلْأَخِيرَةِ تَأْتِي إِلَى ٱلْأَرْضِ ٱلْمُسْتَرَدَّةِ مِنَ ٱلسَّيْفِ ٱلْمَجْمُوعَةِ مِنْ شُعُوبٍ كَثِيرَةٍ عَلَى جِبَالِ إِسْرَائِيلَ ٱلَّتِي كَانَتْ دَائِمَةً خَرِبَةً، لِلَّذِينَ أُخْرِجُوا مِنَ ٱلشُّعُوبِ وَسَكَنُوا آمِنِينَ كُلُّهُمْ. (8) وَتَصْعَدُ وَتَأْتِي كَزَوْبَعَةٍ، وَتَكُونُ كَسَحَابَةٍ تُغَشِّي ٱلْأَرْضَ أَنْتَ وَكُلُّ جُيُوشِكَ وَشُعُوبٌ كَثِيرُونَ مَعَكَ. (9) هَكَذَا قَالَ ٱلسَّيِّدُ ٱلرَّبُّ: وَيَكُونُ فِي ذَلِكَ ٱلْيَوْمِ أَنَّ أُمُورًا تَخْطُرُ بِبَالِكَ فَتُفَكِّرُ فِكْرًا رَدِيئًا، (10) وَتَقُولُ: إِنِّي أَصْعَدُ عَلَى أَرْضٍ أَعْرَاءٍ. آتِي ٱلْهَادِئِينَ ٱلسَّاكِنِينَ فِي أَمْنٍ، كُلُّهُمْ سَا …

אמרו: ״בעל שתי הקרניים, יאג׳וג׳ ומאג׳וג׳ (גוג ומגוג) משחיתים בארץ. הנשלם לך מס על מנת שתעשה בינינו וביניהם סכר?״ (94) אמר: ״מה שנתן לי בו ריבוני – טוב (משכרכם). עזרו לי בכוח, ואעשה ביניכם וביניהם חומה. (95) הביאו לי גושי ברזל״. עד אשר השווה (את הגובה) בין שני צדי ההרים, אמר: ״נפחו!״ עד אשר עשהו אש, אמר: ״הביאו לי, ואצוק עליו נחושת מותכת״. (96) ולא יכלו לעלות עליו, ולא יכלו לנקבו. (97) אמר: ״זה רחמים מאת ריבוני. וכאשר תבוא הבטחת ריבוני – יעשהו עפר; והבטחת ריבוני אמת״. (98) ונניח ביום ההוא לאלה מהם להתנחשל אלה באלה, ויתקע בשופר, ונאסוף אותם יחד. (99)

قَالُواْ يَٰذَا ٱلۡقَرۡنَيۡنِ إِنَّ يَأۡجُوجَ وَمَأۡجُوجَ مُفۡسِدُونَ فِي ٱلۡأَرۡضِ فَهَلۡ نَجۡعَلُ لَكَ خَرۡجًا عَلَىٰٓ أَن تَجۡعَلَ بَيۡنَنَا وَبَيۡنَهُمۡ سَدّٗا (94) قَالَ مَا مَكَّنِّي فِيهِ رَبِّي خَيۡرٞ فَأَعِينُونِي بِقُوَّةٍ أَجۡعَلۡ بَيۡنَكُمۡ وَبَيۡنَهُمۡ رَدۡمًا (95) ءَاتُونِي زُبَرَ ٱلۡحَدِيدِۖ حَتَّىٰٓ إِذَا سَاوَىٰ بَيۡنَ ٱلصَّدَفَيۡنِ قَالَ ٱنفُخُواْۖ حَتَّىٰٓ إِذَا جَعَلَهُۥ نَارٗا قَالَ ءَاتُونِيٓ أُفۡرِغۡ عَلَيۡهِ قِطۡرٗا (96) فَمَا ٱسۡطَٰعُوٓاْ أَن يَظۡهَرُوهُ وَمَا ٱسۡتَطَٰعُواْ لَهُۥ نَقۡبٗا (97) قَالَ هَٰذَا رَحۡمَةٞ مِّن رَّبِّيۖ فَإِذَا جَآءَ وَعۡدُ رَبِّي جَعَلَهُۥ دَكَّآءَۖ وَكَانَ وَعۡدُ رَبِّي حَقّٗا (98) ۞وَتَرَكۡنَا بَعۡضَهُمۡ يَوۡمَئِذٖ يَمُوجُ فِي بَعۡضٖۖ وَنُفِخَ فِي ٱلصُّورِ فَجَمَعۡنَٰهُمۡ جَمۡعٗا (99)

וַיְהִי דְבַר־יְהֹוָה אֵלַי לֵאמֹר׃ (א) בֶּן־אָדָם שִׂים פָּנֶיךָ אֶל־גּוֹג אֶרֶץ הַמָּגוֹג נְשִׂיא רֹאשׁ מֶשֶׁךְ וְתֻבָל וְהִנָּבֵא עָלָיו׃ (ב) וְאָמַרְתָּ כֹּה אָמַר אֲדֹנָי יֱהֹוִה הִנְנִי אֵלֶיךָ גּוֹג נְשִׂיא רֹאשׁ מֶשֶׁךְ וְתֻבָל׃ (ג) וְשׁוֹבַבְתִּיךָ וְנָתַתִּי חַחִים בִּלְחָיֶיךָ וְהוֹצֵאתִי אוֹתְךָ וְאֶת־כׇּל־חֵילֶךָ סוּסִים וּפָרָשִׁים לְבֻשֵׁי מִכְלוֹל כֻּלָּם קָהָל רָב צִנָּה וּמָגֵן תֹּפְשֵׂי חֲרָבוֹת כֻּלָּם׃ (ד) פָּרַס כּוּשׁ וּפוּט אִתָּם כֻּלָּם מָגֵן וְכוֹבָע׃ (ה) גֹּמֶר וְכׇל־אֲגַפֶּיהָ בֵּית תּוֹגַרְמָה יַרְכְּתֵי צָפוֹן וְאֶת־כׇּל־אֲגַפָּיו עַמִּים רַבִּים אִתָּךְ׃ (ו) הִכֹּן וְהָכֵן לְךָ אַתָּה וְכׇל־קְהָלֶךָ הַנִּקְהָלִים עָלֶיךָ וְהָיִיתָ לָהֶם לְמִשְׁמָר׃ (ז) מִיָּמִים רַבִּים תִּפָּקֵד בְּאַחֲרִית הַשָּׁנִים תָּבוֹא אֶל־אֶרֶץ מְשׁוֹבֶבֶת מֵחֶרֶב מְקֻבֶּצֶת מֵעַמִּים רַבִּים עַל הָרֵי יִשְׂרָאֵל אֲשֶׁר־הָיוּ לְחׇרְבָּה תָּמִיד וְהִיא מֵעַמִּים הוּצָאָה וְיָשְׁבוּ לָבֶטַח כֻּלָּם׃ (ח) וְעָלִיתָ כַּשֹּׁאָה תָבוֹא כֶּעָנָן לְכַסּוֹת הָאָרֶץ תִּהְיֶה אַתָּה וְכׇל־אֲגַפֶּיךָ וְעַמִּים רַבִּים אוֹתָךְ׃ (ט) כֹּה אָמַר אֲדֹנָי יֱהֹוִה וְהָיָה בַּיּוֹם הַהוּא יַעֲלוּ דְבָרִים עַל־לְבָבֶךָ וְחָשַׁבְתָּ מַחֲשֶׁבֶת רָעָה׃ (י) וְאָמַרְתָּ אֶעֱלֶה עַל־אֶרֶץ פְּרָזוֹת אָבוֹא הַשֹּׁקְטִים יֹשְׁבֵי לָבֶטַח כֻּלָּם יֹשְׁבִים בְּאֵין חוֹמָה וּבְרִיחַ וּדְלָתַיִם אֵין לָהֶם׃ (יא) לִשְׁלֹל שָׁלָל וְלָבֹז בַּז לְהָשִׁיב יָדְךָ עַל־חֳרָבוֹת נוֹשָׁבוֹת וְאֶל־עַם מְאֻסָּף מִגּוֹיִם עֹשֶׂה מִקְנֶה וְקִנְיָן יֹשְׁבֵי עַל־טַבּוּר הָאָרֶץ׃ (יב) שְׁבָא וּדְדָן וְסֹחֲרֵי תַרְשִׁישׁ …

وَكَانَ إِلَيَّ كَلَامُ ٱلرَّبِّ قَائِلًا: (1) «يَا ٱبْنَ آدَمَ، ٱجْعَلْ وَجْهَكَ عَلَى جُوجٍ، أَرْضِ مَاجُوجَ رَئِيسِ رُوشٍ مَاشِكَ وَتُوبَالَ، وَتَنَبَّأْ عَلَيْهِ (2) وَقُلْ: هَكَذَا قَالَ ٱلسَّيِّدُ ٱلرَّبُّ: هَأَنَذَا عَلَيْكَ يَا جُوجُ رَئِيسُ رُوشٍ مَاشِكَ وَتُوبَالَ. (3) وَأُرْجِعُكَ، وَأَضَعُ شَكَائِمَ فِي فَكَّيْكَ، وَأُخْرِجُكَ أَنْتَ وَكُلَّ جَيْشِكَ خَيْلًا وَفُرْسَانًا كُلَّهُمْ لَابِسِينَ أَفْخَرَ لِبَاسٍ، جَمَاعَةً عَظِيمَةً مَعَ أَتْرَاسٍ وَمَجَانَّ، كُلَّهُمْ مُمْسِكِينَ ٱلسُّيُوفَ. (4) فَارِسَ وَكُوشَ وَفُوطَ مَعَهُمْ، كُلَّهُمْ بِمِجَنٍّ وَخُوذَةٍ، (5) وَجُومَرَ وَكُلَّ جُيُوشِهِ، وَبَيْتَ تُوجَرْمَةَ مِنْ أَقَاصِي ٱلشِّمَالِ مَعَ كُلِّ جَيْشِهِ، شُعُوبًا كَثِيرِينَ مَعَكَ. (6) اِسْتَعِدَّ وَهَيِّئْ لِنَفْسِكَ أَنْتَ وَكُلُّ جَمَاعَاتِكَ ٱلْمُجْتَمِعَةِ إِلَيْكَ، فَصِرْتَ لَهُمْ مُوَقَّرًا. (7) بَعْدَ أَيَّامٍ كَثِيرَةٍ تُفْتَقَدُ. فِي ٱلسِّنِينَ ٱلْأَخِيرَةِ تَأْتِي إِلَى ٱلْأَرْضِ ٱلْمُسْتَرَدَّةِ مِنَ ٱلسَّيْفِ ٱلْمَجْمُوعَةِ مِنْ شُعُوبٍ كَثِيرَةٍ عَلَى جِبَالِ إِسْرَائِيلَ ٱلَّتِي كَانَتْ دَائِمَةً خَرِبَةً، لِلَّذِينَ أُخْرِجُوا مِنَ ٱلشُّعُوبِ وَسَكَنُوا آمِنِينَ كُلُّهُمْ. (8) وَتَصْعَدُ وَتَأْتِي كَزَوْبَعَةٍ، وَتَكُونُ كَسَحَابَةٍ تُغَشِّي ٱلْأَرْضَ أَنْتَ وَكُلُّ جُيُوشِكَ وَشُعُوبٌ كَثِيرُونَ مَعَكَ. (9) هَكَذَا قَالَ ٱلسَّيِّدُ ٱلرَّبُّ: وَيَكُونُ فِي ذَلِكَ ٱلْيَوْمِ أَنَّ أُمُورًا تَخْطُرُ بِبَالِكَ فَتُفَكِّرُ فِكْرًا رَدِيئًا، (10) وَتَقُولُ: إِنِّي أَصْعَدُ عَلَى أَرْضٍ أَعْرَاءٍ. آتِي ٱلْهَادِئِينَ ٱلسَّاكِنِينَ فِي أَمْنٍ، كُلُّهُمْ سَا …

עד אשר ייפתחו יאג׳וג׳ ומאג׳וג׳, והם ממהרים מכל גבעה. (96)

יחזקאל לח, יד–טז ⇄ 18:94–98

לָכֵן הִנָּבֵא בֶן־אָדָם וְאָמַרְתָּ לְגוֹג כֹּה אָמַר אֲדֹנָי יֱהֹוִה הֲלוֹא בַּיּוֹם הַהוּא בְּשֶׁבֶת עַמִּי יִשְׂרָאֵל לָבֶטַח תֵּדָע׃ (יד) וּבָאתָ מִמְּקוֹמְךָ מִיַּרְכְּתֵי צָפוֹן אַתָּה וְעַמִּים רַבִּים אִתָּךְ רֹכְבֵי סוּסִים כֻּלָּם קָהָל גָּדוֹל וְחַיִל רָב׃ (טו) וְעָלִיתָ עַל־עַמִּי יִשְׂרָאֵל כֶּעָנָן לְכַסּוֹת הָאָרֶץ בְּאַחֲרִית הַיָּמִים תִּהְיֶה וַהֲבִאוֹתִיךָ עַל־אַרְצִי לְמַעַן דַּעַת הַגּוֹיִם אֹתִי בְּהִקָּדְשִׁי בְךָ לְעֵינֵיהֶם גּוֹג׃ (טז)

«لِذَلِكَ تَنَبَّأْ يَا ٱبْنَ آدَمَ، وَقُلْ لِجُوجٍ: هَكَذَا قَالَ ٱلسَّيِّدُ ٱلرَّبُّ: فِي ذَلِكَ ٱلْيَوْمِ عِنْدَ سُكْنَى شَعْبِي إِسْرَائِيلَ آمِنِينَ، أَفَلَا تَعْلَمُ؟ (14) وَتَأْتِي مِنْ مَوْضِعِكَ مِنْ أَقَاصِي ٱلشِّمَالِ أَنْتَ وَشُعُوبٌ كَثِيرُونَ مَعَكَ، كُلُّهُمْ رَاكِبُونَ خَيْلًا، جَمَاعَةٌ عَظِيمَةٌ وَجَيْشٌ كَثِيرٌ. (15) وَتَصْعَدُ عَلَى شَعْبِي إِسْرَائِيلَ كَسَحَابَةٍ تُغَشِّي ٱلْأَرْضَ. فِي ٱلْأَيَّامِ ٱلْأَخِيرَةِ يَكُونُ. وَآتِي بِكَ عَلَى أَرْضِي لِكَيْ تَعْرِفَنِي ٱلْأُمَمُ، حِينَ أَتَقَدَّسُ فِيكَ أَمَامَ أَعْيُنِهِمْ يَاجُوجُ. (16)

אמרו: ״בעל שתי הקרניים, יאג׳וג׳ ומאג׳וג׳ (גוג ומגוג) משחיתים בארץ. הנשלם לך מס על מנת שתעשה בינינו וביניהם סכר?״ (94) אמר: ״מה שנתן לי בו ריבוני – טוב (משכרכם). עזרו לי בכוח, ואעשה ביניכם וביניהם חומה. (95) הביאו לי גושי ברזל״. עד אשר השווה (את הגובה) בין שני צדי ההרים, אמר: ״נפחו!״ עד אשר עשהו אש, אמר: ״הביאו לי, ואצוק עליו נחושת מותכת״. (96) ולא יכלו לעלות עליו, ולא יכלו לנקבו. (97) אמר: ״זה רחמים מאת ריבוני. וכאשר תבוא הבטחת ריבוני – יעשהו עפר; והבטחת ריבוני אמת״. (98)

قَالُواْ يَٰذَا ٱلۡقَرۡنَيۡنِ إِنَّ يَأۡجُوجَ وَمَأۡجُوجَ مُفۡسِدُونَ فِي ٱلۡأَرۡضِ فَهَلۡ نَجۡعَلُ لَكَ خَرۡجًا عَلَىٰٓ أَن تَجۡعَلَ بَيۡنَنَا وَبَيۡنَهُمۡ سَدّٗا (94) قَالَ مَا مَكَّنِّي فِيهِ رَبِّي خَيۡرٞ فَأَعِينُونِي بِقُوَّةٍ أَجۡعَلۡ بَيۡنَكُمۡ وَبَيۡنَهُمۡ رَدۡمًا (95) ءَاتُونِي زُبَرَ ٱلۡحَدِيدِۖ حَتَّىٰٓ إِذَا سَاوَىٰ بَيۡنَ ٱلصَّدَفَيۡنِ قَالَ ٱنفُخُواْۖ حَتَّىٰٓ إِذَا جَعَلَهُۥ نَارٗا قَالَ ءَاتُونِيٓ أُفۡرِغۡ عَلَيۡهِ قِطۡرٗا (96) فَمَا ٱسۡطَٰعُوٓاْ أَن يَظۡهَرُوهُ وَمَا ٱسۡتَطَٰعُواْ لَهُۥ نَقۡبٗا (97) قَالَ هَٰذَا رَحۡمَةٞ مِّن رَّبِّيۖ فَإِذَا جَآءَ وَعۡدُ رَبِّي جَعَلَهُۥ دَكَّآءَۖ وَكَانَ وَعۡدُ رَبِّي حَقّٗا (98)

וְהִנֵּה כְּבוֹד אֱלֹהֵי יִשְׂרָאֵל בָּא מִדֶּרֶךְ הַקָּדִים וְקוֹלוֹ כְּקוֹל מַיִם רַבִּים וְהָאָרֶץ הֵאִירָה מִכְּבֹדוֹ׃ (ב)

وَإِذَا بِمَجْدِ إِلَهِ إِسْرَائِيلَ جَاءَ مِنْ طَرِيقِ ٱلشَّرْقِ وَصَوْتُهُ كَصَوْتِ مِيَاهٍ كَثِيرَةٍ، وَٱلْأَرْضُ أَضَاءَتْ مِنْ مَجْدِهِ. (2)

ותאיר הארץ באור ריבונה, ויונח הספר, ויובאו הנביאים והעדים, וייחרץ ביניהם באמת, ולא ייעשה להם עוול. (69)

وَأَشۡرَقَتِ ٱلۡأَرۡضُ بِنُورِ رَبِّهَا وَوُضِعَ ٱلۡكِتَٰبُ وَجِاْيٓءَ بِٱلنَّبِيِّـۧنَ وَٱلشُّهَدَآءِ وَقُضِيَ بَيۡنَهُم بِٱلۡحَقِّ وَهُمۡ لَا يُظۡلَمُونَ (69)

״והארץ האירה באור ריבונה״ מול ״והארץ האירה מכבדו״.
