וַיְהִי בְּהַעֲלוֹת יְהֹוָה אֶת־אֵלִיָּהוּ בַּסְעָרָה הַשָּׁמָיִם וַיֵּלֶךְ אֵלִיָּהוּ וֶאֱלִישָׁע מִן־הַגִּלְגָּל׃ (א) וַיֹּאמֶר אֵלִיָּהוּ אֶל־אֱלִישָׁע שֵׁב־נָא פֹה כִּי יְהֹוָה שְׁלָחַנִי עַד־בֵּית־אֵל וַיֹּאמֶר אֱלִישָׁע חַי־יְהֹוָה וְחֵי־נַפְשְׁךָ אִם־אֶעֶזְבֶךָּ וַיֵּרְדוּ בֵּית־אֵל׃ (ב) וַיֵּצְאוּ בְנֵי־הַנְּבִיאִים אֲשֶׁר־בֵּית־אֵל אֶל־אֱלִישָׁע וַיֹּאמְרוּ אֵלָיו הֲיָדַעְתָּ כִּי הַיּוֹם יְהֹוָה לֹקֵחַ אֶת־אֲדֹנֶיךָ מֵעַל רֹאשֶׁךָ וַיֹּאמֶר גַּם־אֲנִי יָדַעְתִּי הֶחֱשׁוּ׃ (ג) וַיֹּאמֶר לוֹ אֵלִיָּהוּ אֱלִישָׁע שֵׁב־נָא פֹה כִּי יְהֹוָה שְׁלָחַנִי יְרִיחוֹ וַיֹּאמֶר חַי־יְהֹוָה וְחֵי־נַפְשְׁךָ אִם־אֶעֶזְבֶךָּ וַיָּבֹאוּ יְרִיחוֹ׃ (ד) וַיִּגְּשׁוּ בְנֵי־הַנְּבִיאִים אֲשֶׁר־בִּירִיחוֹ אֶל־אֱלִישָׁע וַיֹּאמְרוּ אֵלָיו הֲיָדַעְתָּ כִּי הַיּוֹם יְהֹוָה לֹקֵחַ אֶת־אֲדֹנֶיךָ מֵעַל רֹאשֶׁךָ וַיֹּאמֶר גַּם־אֲנִי יָדַעְתִּי הֶחֱשׁוּ׃ (ה) וַיֹּאמֶר לוֹ אֵלִיָּהוּ שֵׁב־נָא פֹה כִּי יְהֹוָה שְׁלָחַנִי הַיַּרְדֵּנָה וַיֹּאמֶר חַי־יְהֹוָה וְחֵי־נַפְשְׁךָ אִם־אֶעֶזְבֶךָּ וַיֵּלְכוּ שְׁנֵיהֶם׃ (ו) וַחֲמִשִּׁים אִישׁ מִבְּנֵי הַנְּבִיאִים הָלְכוּ וַיַּעַמְדוּ מִנֶּגֶד מֵרָחוֹק וּשְׁנֵיהֶם עָמְדוּ עַל־הַיַּרְדֵּן׃ (ז) וַיִּקַּח אֵלִיָּהוּ אֶת־אַדַּרְתּוֹ וַיִּגְלֹם וַיַּכֶּה אֶת־הַמַּיִם וַיֵּחָצוּ הֵנָּה וָהֵנָּה וַיַּעַבְרוּ שְׁנֵיהֶם בֶּחָרָבָה׃ (ח) וַיְהִי כְעׇבְרָם וְאֵלִיָּהוּ אָמַר אֶל־אֱלִישָׁע שְׁאַל מָה אֶעֱשֶׂה־לָּךְ בְּטֶרֶם אֶלָּקַח מֵעִמָּךְ וַיֹּאמֶר אֱלִישָׁע וִיהִי נָא פִּי־שְׁנַיִם בְּרוּחֲךָ אֵלָי׃ (ט) וַיֹּאמֶר הִקְשִׁיתָ לִשְׁאוֹל אִם־תִּרְאֶה אֹתִי לֻקָּח מֵאִתָּך …

وَكَانَ عِنْدَ إِصْعَادِ ٱلرَّبِّ إِيلِيَّا فِي ٱلْعَاصِفَةِ إِلَى ٱلسَّمَاءِ، أَنَّ إِيلِيَّا وَأَلِيشَعَ ذَهَبَا مِنَ ٱلْجِلْجَالِ. (1) فَقَالَ إِيلِيَّا لِأَلِيشَعَ: «ٱمْكُثْ هُنَا لِأَنَّ ٱلرَّبَّ قَدْ أَرْسَلَنِي إِلَى بَيْتِ إِيلَ». فَقَالَ أَلِيشَعُ: «حَيٌّ هُوَ ٱلرَّبُّ، وَحَيَّةٌ هِيَ نَفْسُكَ، إِنِّي لَا أَتْرُكُكَ». وَنَزَلَا إِلَى بَيْتِ إِيلَ. (2) فَخَرَجَ بَنُو ٱلْأَنْبِيَاءِ ٱلَّذِينَ فِي بَيْتِ إِيلَ إِلَى أَلِيشَعَ وَقَالُوا لَهُ: «أَتَعْلَمُ أَنَّهُ ٱلْيَوْمَ يَأْخُذُ ٱلرَّبُّ سَيِّدَكَ مِنْ عَلَى رَأْسِكَ؟» فَقَالَ: «نَعَمْ، إِنِّي أَعْلَمُ فَٱصْمُتُوا». (3) ثُمَّ قَالَ لَهُ إِيلِيَّا: «يَا أَلِيشَعُ، ٱمْكُثْ هُنَا لِأَنَّ ٱلرَّبَّ قَدْ أَرْسَلَنِي إِلَى أَرِيحَا». فَقَالَ: «حَيٌّ هُوَ ٱلرَّبُّ، وَحَيَّةٌ هِيَ نَفْسُكَ، إِنِّي لَا أَتْرُكُكَ». وَأَتَيَا إِلَى أَرِيحَا. (4) فَتَقَدَّمَ بَنُو ٱلْأَنْبِيَاءِ ٱلَّذِينَ فِي أَرِيحَا إِلَى أَلِيشَعَ وَقَالُوا لَهُ: «أَتَعْلَمُ أَنَّهُ ٱلْيَوْمَ يَأْخُذُ ٱلرَّبُّ سَيِّدَكَ مِنْ عَلَى رَأْسِكَ؟» فَقَالَ: «نَعَمْ، إِنِّي أَعْلَمُ فَٱصْمُتُوا». (5) ثُمَّ قَالَ لَهُ إِيلِيَّا: «أُمْكُثْ هُنَا لِأَنَّ ٱلرَّبَّ قَدْ أَرْسَلَنِي إِلَى ٱلْأُرْدُنِّ». فَقَالَ: «حَيٌّ هُوَ ٱلرَّبُّ، وَحَيَّةٌ هِيَ نَفْسُكَ، إِنِّي لَا أَتْرُكُكَ». وَٱنْطَلَقَا كِلَاهُمَا. (6) فَذَهَبَ خَمْسُونَ رَجُلًا مِنْ بَنِي ٱلْأَنْبِيَاءِ وَوَقَفُوا قُبَالَتَهُمَا مِنْ بَعِيدٍ. وَوَقَفَ كِلَاهُمَا بِجَانِبِ ٱلْأُرْدُنِّ. (7) وَأَخَذَ إِيلِيَّا رِدَاءَهُ وَلَفَّهُ وَضَرَبَ ٱلْمَاءَ، فَٱنْفَلَقَ إِلَى هُنَا وَهُنَاكَ، فَعَبَرَا كِلَاهُمَا فِي ٱلْيَبَسِ. …

ואת ישמעאל ואת אלישע ואת יונה ואת לוט – ואת כולם העדפנו על העולמים. (86)

وَإِسۡمَٰعِيلَ وَٱلۡيَسَعَ وَيُونُسَ وَلُوطٗاۚ وَكُلّٗا فَضَّلۡنَا عَلَى ٱلۡعَٰلَمِينَ (86)

(2) فَخَرَجَ بَنُو ٱلْأَنْبِيَاءِ ٱلَّذِينَ فِي بَيْتِ إِيلَ إِلَى أَلِيشَعَ وَقَالُوا لَهُ: «أَتَعْلَمُ أَنَّهُ ٱلْيَوْمَ يَأْخُذُ ٱلرَّبُّ سَيِّدَكَ مِنْ عَلَى رَأْسِكَ؟» فَقَالَ: «نَعَمْ، إِنِّي أَعْلَمُ فَٱصْمُتُوا». (3) ثُمَّ قَالَ لَهُ إِيلِيَّا: «يَا أَلِيشَعُ، ٱمْكُثْ هُنَا لِأَنَّ ٱلرَّبَّ قَدْ أَرْسَلَنِي إِلَى أَرِيحَا». فَقَالَ: «حَيٌّ هُوَ ٱلرَّبُّ، وَحَيَّةٌ هِيَ نَفْسُكَ، إِنِّي لَا أَتْرُكُكَ». وَأَتَيَا إِلَى أَرِيحَا. (4) فَتَقَدَّمَ بَنُو ٱلْأَنْبِيَاءِ ٱلَّذِينَ فِي أَرِيحَا إِلَى أَلِيشَعَ وَقَالُوا لَهُ: «أَتَعْلَمُ أَنَّهُ ٱلْيَوْمَ يَأْخُذُ ٱلرَّبُّ سَيِّدَكَ مِنْ عَلَى رَأْسِكَ؟» فَقَالَ: «نَعَمْ، إِنِّي أَعْلَمُ فَٱصْمُتُوا». (5) ثُمَّ قَالَ لَهُ إِيلِيَّا: «أُمْكُثْ هُنَا لِأَنَّ ٱلرَّبَّ قَدْ أَرْسَلَنِي إِلَى ٱلْأُرْدُنِّ». فَقَالَ: «حَيٌّ هُوَ ٱلرَّبُّ، وَحَيَّةٌ هِيَ نَفْسُكَ، إِنِّي لَا أَتْرُكُكَ». وَٱنْطَلَقَا كِلَاهُمَا. בַּיָּמִים הָהֵם חָלָה חִזְקִיָּהוּ לָמוּת וַיָּבֹא אֵלָיו יְשַׁעְיָהוּ בֶן־אָמוֹץ הַנָּבִיא וַיֹּאמֶר אֵלָיו כֹּה־אָמַר יְהֹוָה צַו לְבֵיתֶךָ כִּי מֵת אַתָּה וְלֹא תִחְיֶה׃ (א) וַיַּסֵּב אֶת־פָּנָיו אֶל־הַקִּיר וַיִּתְפַּלֵּל אֶל־יְהֹוָה לֵאמֹר׃ (ב) אָנָּה יְהֹוָה זְכׇר־נָא אֵת אֲשֶׁר הִתְהַלַּכְתִּי לְפָנֶיךָ בֶּאֱמֶת וּבְלֵבָב שָׁלֵם וְהַטּוֹב בְּעֵינֶיךָ עָשִׂיתִי וַיֵּבְךְּ חִזְקִיָּהוּ בְּכִי גָדוֹל׃ (ג) וַיְהִי יְשַׁעְיָהוּ לֹא יָצָא (העיר) [חָצֵר] הַתִּיכֹנָה וּדְבַר־יְהֹוָה הָיָה אֵלָיו לֵאמֹר׃ (ד) שׁוּב וְאָמַרְתָּ אֶל־חִזְקִיָּהוּ נְגִיד־עַמִּי כֹּה־אָמַר יְהֹוָה אֱלֹהֵי דָּוִד אָבִיךָ שָׁמַעְתִּי אֶת־תְּפִלָּתֶךָ רָאִיתִי אֶת־דִּמְעָתֶךָ הִנְנִי רֹפֶא לָךְ בַּיּוֹם הַשְּׁלִישִׁי תַּעֲלֶה בֵּית יְהֹוָה׃ (ה) וְהֹסַפְתִּי עַל־יָמֶיךָ חֲמֵשׁ עֶשְׂרֵה שָׁנָה וּמִכַּף מֶלֶךְ־אַשּׁוּר אַצִּילְךָ וְאֵת הָעִיר הַזֹּאת וְגַנּוֹתִי עַל־הָעִיר הַזֹּאת לְמַעֲנִי וּלְמַעַן דָּוִד עַבְדִּי׃ (ו)

فِي تِلْكَ ٱلْأَيَّامِ مَرِضَ حَزَقِيَّا لِلْمَوْتِ، فَجَاءَ إِلَيْهِ إِشَعْيَا بْنُ آمُوصَ ٱلنَّبِيُّ وَقَالَ لَهُ: «هَكَذَا قَالَ ٱلرَّبُّ: أَوْصِ بَيْتَكَ لِأَنَّكَ تَمُوتُ وَلَا تَعِيشُ». (1) فَوَجَّهَ وَجْهَهُ إِلَى ٱلْحَائِطِ وَصَلَّى إِلَى ٱلرَّبِّ قَائِلًا: (2) «آهِ يَارَبُّ، ٱذْكُرْ كَيْفَ سِرْتُ أَمَامَكَ بِٱلْأَمَانَةِ وَبِقَلْبٍ سَلِيمٍ، وَفَعَلْتُ ٱلْحَسَنَ فِي عَيْنَيْكَ». وَبَكَى حَزَقِيَّا بُكَاءً عَظِيمًا. (3) وَلَمْ يَخْرُجْ إِشَعْيَا إِلَى ٱلْمَدِينَةِ ٱلْوُسْطَى حَتَّى كَانَ كَلَامُ ٱلرَّبِّ إِلَيْهِ قَائِلًا: (4) «ٱرْجِعْ وَقُلْ لِحَزَقِيَّا رَئِيسِ شَعْبِي: هَكَذَا قَالَ ٱلرَّبُّ إِلَهُ دَاوُدَ أَبِيكَ: قَدْ سَمِعْتُ صَلَاتَكَ. قَدْ رَأَيْتُ دُمُوعَكَ. هَأَنَذَا أَشْفِيكَ. فِي ٱلْيَوْمِ ٱلثَّالِثِ تَصْعَدُ إِلَى بَيْتِ ٱلرَّبِّ. (5) وَأَزِيدُ عَلَى أَيَّامِكَ خَمْسَ عَشْرَةَ سَنَةً، وَأُنْقِذُكَ مِنْ يَدِ مَلِكِ أَشُّورَ مَعَ هَذِهِ ٱلْمَدِينَةِ، وَأُحَامِي عَنْ هَذِهِ ٱلْمَدِينَةِ مِنْ أَجْلِ نَفْسِي، وَمِنْ أَجْلِ دَاوُدَ عَبْدِي». (6)

13:39: אללה מוחה את אשר ירצה ומקיים, ואצלו אם הספר. (39) ✦ תרמד׳י 2139: אמר שליח אללה: ״אין משיב את הגזירה אלא התפילה, ואין מוסיף לחיים אלא הצדקה והטוב״.

يَمۡحُواْ ٱللَّهُ مَا يَشَآءُ وَيُثۡبِتُۖ وَعِندَهُۥٓ أُمُّ ٱلۡكِتَٰبِ (39) ✦ سنن الترمذي 2139: قال رسول الله صلى الله عليه وسلم " لا يرد القضاء إلا الدعاء ولا يزيد في العمر إلا البر " .

תפילת חזקיהו החולה המשנה את גזירת מותו ומוסיפה לו 15 שנות חיים (״שמעתי את תפילתך ראיתי את דמעתך הנני רופא לך״) — מקבילה מדויקת ל־13:39 (״אללה מוחק ומקיים מה שירצה״) ולחדית׳ ״אין משיב את הגזירה אלא התפילה!״ (תרמד׳י 2139).

استجابة دعاء حزقيا المريض وزيادة عمره خمس عشرة سنة يوازي الرعد 13:39 وحديث «لا يردّ القضاء إلا الدعاء» (الترمذي 2139)!

מלכים ב כב, ח–יג ⇄ 5:44; 7:169

וַיֹּאמֶר חִלְקִיָּהוּ הַכֹּהֵן הַגָּדוֹל עַל־שָׁפָן הַסֹּפֵר סֵפֶר הַתּוֹרָה מָצָאתִי בְּבֵית יְהֹוָה וַיִּתֵּן חִלְקִיָּה אֶת־הַסֵּפֶר אֶל־שָׁפָן וַיִּקְרָאֵהוּ׃ (ח) וַיָּבֹא שָׁפָן הַסֹּפֵר אֶל־הַמֶּלֶךְ וַיָּשֶׁב אֶת־הַמֶּלֶךְ דָּבָר וַיֹּאמֶר הִתִּיכוּ עֲבָדֶיךָ אֶת־הַכֶּסֶף הַנִּמְצָא בַבַּיִת וַיִּתְּנֻהוּ עַל־יַד עֹשֵׂי הַמְּלָאכָה הַמֻּפְקָדִים בֵּית יְהֹוָה׃ (ט) וַיַּגֵּד שָׁפָן הַסֹּפֵר לַמֶּלֶךְ לֵאמֹר סֵפֶר נָתַן לִי חִלְקִיָּה הַכֹּהֵן וַיִּקְרָאֵהוּ שָׁפָן לִפְנֵי הַמֶּלֶךְ׃ (י) וַיְהִי כִּשְׁמֹעַ הַמֶּלֶךְ אֶת־דִּבְרֵי סֵפֶר הַתּוֹרָה וַיִּקְרַע אֶת־בְּגָדָיו׃ (יא) וַיְצַו הַמֶּלֶךְ אֶת־חִלְקִיָּה הַכֹּהֵן וְאֶת־אֲחִיקָם בֶּן־שָׁפָן וְאֶת־עַכְבּוֹר בֶּן־מִיכָיָה וְאֵת שָׁפָן הַסֹּפֵר וְאֵת עֲשָׂיָה עֶבֶד־הַמֶּלֶךְ לֵאמֹר׃ (יב) לְכוּ דִרְשׁוּ אֶת־יְהֹוָה בַּעֲדִי וּבְעַד־הָעָם וּבְעַד כׇּל־יְהוּדָה עַל־דִּבְרֵי הַסֵּפֶר הַנִּמְצָא הַזֶּה כִּי־גְדוֹלָה חֲמַת יְהֹוָה אֲשֶׁר־הִיא נִצְּתָה בָנוּ עַל אֲשֶׁר לֹא־שָׁמְעוּ אֲבֹתֵינוּ עַל־דִּבְרֵי הַסֵּפֶר הַזֶּה לַעֲשׂוֹת כְּכׇל־הַכָּתוּב עָלֵינוּ׃ (יג)

فَقَالَ حِلْقِيَّا ٱلْكَاهِنُ ٱلْعَظِيمُ لِشَافَانَ ٱلْكَاتِبِ: «قَدْ وَجَدْتُ سِفْرَ ٱلشَّرِيعَةِ فِي بَيْتِ ٱلرَّبِّ». وَسَلَّمَ حِلْقِيَّا ٱلسِّفْرَ لِشَافَانَ فَقَرَأَهُ. (8) وَجَاءَ شَافَانُ ٱلْكَاتِبُ إِلَى ٱلْمَلِكِ وَرَدَّ عَلَى ٱلْمَلِكِ جَوَابًا وَقَالَ: «قَدْ أَفْرَغَ عَبِيدُكَ ٱلْفِضَّةَ ٱلْمَوْجُودَةَ فِي ٱلْبَيْتِ وَدَفَعُوهَا إِلَى يَدِ عَامِلِي ٱلشُّغْلِ وُكَلَاءِ بَيْتِ ٱلرَّبِّ». (9) وَأَخْبَرَ شَافَانُ ٱلْكَاتِبُ ٱلْمَلِكَ قَائِلًا: «قَدْ أَعْطَانِي حِلْقِيَّا ٱلْكَاهِنُ سِفْرًا». وَقَرَأَهُ شَافَانُ أَمَامَ ٱلْمَلِكِ. (10) فَلَمَّا سَمِعَ ٱلْمَلِكُ كَلَامَ سِفْرِ ٱلشَّرِيعَةِ مَزَّقَ ثِيَابَهُ. (11) وَأَمَرَ ٱلْمَلِكُ حِلْقِيَّا ٱلْكَاهِنَ وَأَخِيقَامَ بْنَ شَافَانَ وَعَكْبُورَ بْنَ مِيخَا وَشَافَانَ ٱلْكَاتِبَ وَعَسَايَا عَبْدَ ٱلْمَلِكِ قَائِلًا: (12) «ٱذْهَبُوا ٱسْأَلُوا ٱلرَّبَّ لِأَجْلِي وَلِأَجْلِ ٱلشَّعْبِ وَلِأَجْلِ كُلِّ يَهُوذَا مِنْ جِهَةِ كَلَامِ هَذَا ٱلسِّفْرِ ٱلَّذِي وُجِدَ، لِأَنَّهُ عَظِيمٌ هُوَ غَضَبُ ٱلرَّبِّ ٱلَّذِي ٱشْتَعَلَ عَلَيْنَا، مِنْ أَجْلِ أَنَّ آبَاءَنَا لَمْ يَسْمَعُوا لِكَلَامِ هَذَا ٱلسِّفْرِ لِيَعْمَلُوا حَسَبَ كُلِّ مَا هُوَ مَكْتُوبٌ عَلَيْنَا». (13)

5:44: אנחנו הורדנו את התורה, בה הדרכה ואור; שופטים בה הנביאים המתמסרים את היהודים, וכן הרבנים והחכמים (האחבאר), כפי שהופקדו על ספר אללה והיו עליו עדים. לכן אל תיראו מבני האדם, אלא ממני, ואל תמכרו את אותותיי במחיר מועט. ומי שאינו שופט לפי מה שהוריד אללה – הם הכופרים. (44) ✦ 7:169: ואחריהם באו יורשים שירשו את הספר, לוקחים את רכוש העולם השפל הזה ואומרים: ״ייסלח לנו״; ואם יבוא אליהם רכוש כמוהו – ייקחוהו. הלא נכרתה עמם ברית הספר, שלא יאמרו על אללה אלא את האמת? והם למדו את אשר בו. והבית האחרון טוב יותר לאלה שיראים; הלא תבינו? (169)

إِنَّآ أَنزَلۡنَا ٱلتَّوۡرَىٰةَ فِيهَا هُدٗى وَنُورٞۚ يَحۡكُمُ بِهَا ٱلنَّبِيُّونَ ٱلَّذِينَ أَسۡلَمُواْ لِلَّذِينَ هَادُواْ وَٱلرَّبَّـٰنِيُّونَ وَٱلۡأَحۡبَارُ بِمَا ٱسۡتُحۡفِظُواْ مِن كِتَٰبِ ٱللَّهِ وَكَانُواْ عَلَيۡهِ شُهَدَآءَۚ فَلَا تَخۡشَوُاْ ٱلنَّاسَ وَٱخۡشَوۡنِ وَلَا تَشۡتَرُواْ بِـَٔايَٰتِي ثَمَنٗا قَلِيلٗاۚ وَمَن لَّمۡ يَحۡكُم بِمَآ أَنزَلَ ٱللَّهُ فَأُوْلَـٰٓئِكَ هُمُ ٱلۡكَٰفِرُونَ (44) ✦ فَخَلَفَ مِنۢ بَعۡدِهِمۡ خَلۡفٞ وَرِثُواْ ٱلۡكِتَٰبَ يَأۡخُذُونَ عَرَضَ هَٰذَا ٱلۡأَدۡنَىٰ وَيَقُولُونَ سَيُغۡفَرُ لَنَا وَإِن يَأۡتِهِمۡ عَرَضٞ مِّثۡلُهُۥ يَأۡخُذُوهُۚ أَلَمۡ يُؤۡخَذۡ عَلَيۡهِم مِّيثَٰقُ ٱلۡكِتَٰبِ أَن لَّا يَقُولُواْ عَلَى ٱللَّهِ إِلَّا ٱلۡحَقَّ وَدَرَسُواْ مَا فِيهِۗ وَٱلدَّارُ ٱلۡأٓخِرَةُ خَيۡرٞ لِّلَّذِينَ يَتَّقُونَۚ أَفَلَا تَعۡقِلُونَ (169)

מציאת ספר התורה שנשכח בבית המקדש בימי יאשיהו ותנועת התשובה העממית — התחדשות דתית מתוך שיבה אל הכתוב המקורי (5:44).

العثور على سفر التوراة في الهيكل زمن يوشيا وتجديد التوبة بالعودة إلى الكتاب الأصلي (المائدة 5:44).

מלכים ב כג, כה ⇄ 50:33; 39:54

וְכָמֹהוּ לֹא־הָיָה לְפָנָיו מֶלֶךְ אֲשֶׁר־שָׁב אֶל־יְהֹוָה בְּכׇל־לְבָבוֹ וּבְכׇל־נַפְשׁוֹ וּבְכׇל־מְאֹדוֹ כְּכֹל תּוֹרַת מֹשֶׁה וְאַחֲרָיו לֹא־קָם כָּמֹהוּ׃ (כה)

وَلَمْ يَكُنْ قَبْلَهُ مَلِكٌ مِثْلُهُ قَدْ رَجَعَ إِلَى ٱلرَّبِّ بِكُلِّ قَلْبِهِ وَكُلِّ نَفْسِهِ وَكُلِّ قُوَّتِهِ حَسَبَ كُلِّ شَرِيعَةِ مُوسَى، وَبَعْدَهُ لَمْ يَقُمْ مِثْلُهُ. (25)

50:33: אשר ירא את הרחמן בנסתר, ובא בלב שב: (33) ✦ 39:54: ושובו אל ריבונכם והתמסרו לו, לפני שיבוא אליכם העונש, ואחר כך לא תיעזרו. (54)

مَّنۡ خَشِيَ ٱلرَّحۡمَٰنَ بِٱلۡغَيۡبِ وَجَآءَ بِقَلۡبٖ مُّنِيبٍ (33) ✦ وَأَنِيبُوٓاْ إِلَىٰ رَبِّكُمۡ وَأَسۡلِمُواْ لَهُۥ مِن قَبۡلِ أَن يَأۡتِيَكُمُ ٱلۡعَذَابُ ثُمَّ لَا تُنصَرُونَ (54)

שבחו של המלך יאשיהו (״וכמוהו לא היה לפניו מלך אשר שב אל ה׳ בכל לבבו ובכל נפשו״) — מודל המנהיג החוזר אל האל בלב שלם (מניב, 50:33).

וגזרנו לבני ישראל בספר: ״פעמיים תשחיתו בארץ, ותתנשאו התנשאות גדולה״. (4) וכאשר באה הבטחת הראשונה מהן, שלחנו עליכם עבדים לנו בעלי עוז קשה, ושוטטו בין הבתים; והייתה זו הבטחה שנעשתה. (7) אולי ירחמכם ריבונכם; ואם תשובו – נשוב; ועשינו את גיהינום לכופרים בית כלא. (8)

וּבַחֹדֶשׁ הַחֲמִישִׁי בְּשִׁבְעָה לַחֹדֶשׁ הִיא שְׁנַת תְּשַׁע־עֶשְׂרֵה שָׁנָה לַמֶּלֶךְ נְבֻכַדְנֶאצַּר מֶלֶךְ־בָּבֶל בָּא נְבוּזַרְאֲדָן רַב־טַבָּחִים עֶבֶד מֶלֶךְ־בָּבֶל יְרוּשָׁלִָם׃ (ח) וַיִּשְׂרֹף אֶת־בֵּית־יְהֹוָה וְאֶת־בֵּית הַמֶּלֶךְ וְאֵת כׇּל־בָּתֵּי יְרוּשָׁלִַם וְאֶת־כׇּל־בֵּית גָּדוֹל שָׂרַף בָּאֵשׁ׃ (ט) וְאֶת־חוֹמֹת יְרוּשָׁלִַם סָבִיב נָתְצוּ כׇּל־חֵיל כַּשְׂדִּים אֲשֶׁר רַב־טַבָּחִים׃ (י) וְאֵת יֶתֶר הָעָם הַנִּשְׁאָרִים בָּעִיר וְאֶת־הַנֹּפְלִים אֲשֶׁר נָפְלוּ עַל־הַמֶּלֶךְ בָּבֶל וְאֵת יֶתֶר הֶהָמוֹן הֶגְלָה נְבוּזַרְאֲדָן רַב־טַבָּחִים׃ (יא) וּמִדַּלַּת הָאָרֶץ הִשְׁאִיר רַב־טַבָּחִים לְכֹרְמִים וּלְיֹגְבִים׃ (יב)

وَفِي ٱلشَّهْرِ ٱلْخَامِسِ، فِي سَابِعِ ٱلشَّهْرِ، وَهِيَ ٱلسَّنَةُ ٱلتَّاسِعَةَ عَشْرَةَ لِلْمَلِكِ نَبُوخَذْنَاصَّرَ مَلِكِ بَابِلَ، جَاءَ نَبُوزَرَادَانُ رَئِيسُ ٱلشُّرَطِ عَبْدُ مَلِكِ بَابِلَ إِلَى أُورُشَلِيمَ، (8) وَأَحْرَقَ بَيْتَ ٱلرَّبِّ وَبَيْتَ ٱلْمَلِكِ، وَكُلَّ بُيُوتِ أُورُشَلِيمَ، وَكُلَّ بُيُوتِ ٱلْعُظَمَاءِ أَحْرَقَهَا بِٱلنَّارِ. (9) وَجَمِيعُ أَسْوَارِ أُورُشَلِيمَ مُسْتَدِيرًا هَدَمَهَا كُلُّ جُيُوشِ ٱلْكِلْدَانِيِّينَ ٱلَّذِينَ مَعَ رَئِيسِ ٱلشُّرَطِ. (10) وَبَقِيَّةُ ٱلشَّعْبِ ٱلَّذِينَ بَقُوا فِي ٱلْمَدِينَةِ، وَٱلْهَارِبُونَ ٱلَّذِينَ هَرَبُوا إِلَى مَلِكِ بَابِلَ، وَبَقِيَّةُ ٱلْجُمْهُورِ سَبَاهُمْ نَبُوزَرَادَانُ رَئِيسُ ٱلشُّرَطِ. (11) وَلَكِنَّ رَئِيسَ ٱلشُّرَطِ أَبْقَى مِنْ مَسَاكِينِ ٱلْأَرْضِ كَرَّامِينَ وَفَلَّاحِينَ. (12)

וגזרנו לבני ישראל בספר: ״פעמיים תשחיתו בארץ, ותתנשאו התנשאות גדולה״. (4) וכאשר באה הבטחת הראשונה מהן, שלחנו עליכם עבדים לנו בעלי עוז קשה, ושוטטו בין הבתים; והייתה זו הבטחה שנעשתה. (5) אחר כך השבנו לכם את הגבורה עליהם, ותמכנו בכם בהון ובבנים, ועשינו אתכם רבים יותר. (6) אם תיטיבו – לעצמכם תיטיבו, ואם תרעו – לה (לנפשכם). וכאשר באה הבטחת האחרונה – (שלחנום) להשחית את פניכם ולהיכנס אל המסגד כפי שנכנסו אליו בפעם הראשונה, ולהרוס כליל את אשר כבשו. (7)

وَقَضَيۡنَآ إِلَىٰ بَنِيٓ إِسۡرَـٰٓءِيلَ فِي ٱلۡكِتَٰبِ لَتُفۡسِدُنَّ فِي ٱلۡأَرۡضِ مَرَّتَيۡنِ وَلَتَعۡلُنَّ عُلُوّٗا كَبِيرٗا (4) فَإِذَا جَآءَ وَعۡدُ أُولَىٰهُمَا بَعَثۡنَا عَلَيۡكُمۡ عِبَادٗا لَّنَآ أُوْلِي بَأۡسٖ شَدِيدٖ فَجَاسُواْ خِلَٰلَ ٱلدِّيَارِۚ وَكَانَ وَعۡدٗا مَّفۡعُولٗا (5) ثُمَّ رَدَدۡنَا لَكُمُ ٱلۡكَرَّةَ عَلَيۡهِمۡ وَأَمۡدَدۡنَٰكُم بِأَمۡوَٰلٖ وَبَنِينَ وَجَعَلۡنَٰكُمۡ أَكۡثَرَ نَفِيرًا (6) إِنۡ أَحۡسَنتُمۡ أَحۡسَنتُمۡ لِأَنفُسِكُمۡۖ وَإِنۡ أَسَأۡتُمۡ فَلَهَاۚ فَإِذَا جَآءَ وَعۡدُ ٱلۡأٓخِرَةِ لِيَسُـُٔواْ وُجُوهَكُمۡ وَلِيَدۡخُلُواْ ٱلۡمَسۡجِدَ كَمَا دَخَلُوهُ أَوَّلَ مَرَّةٖ وَلِيُتَبِّرُواْ مَا عَلَوۡاْ تَتۡبِيرًا (7)

חורבן הבית הראשון בידי נבוזראדן – ״ושלחנו עליכם עבדים לנו בעלי כוח… ונכנסו למסגד כפי שנכנסו בפעם הראשונה״.

خراب البيت الأول على يد نبوزرادان – «بعثنا عليكم عبادا لنا أولي بأس شديد… وليدخلوا المسجد كما دخلوه أول مرة».
