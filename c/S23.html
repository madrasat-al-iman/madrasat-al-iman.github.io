המפרשים המוסלמים מזהים את העובר ליד העיר ההרוסה כעֻזַיר (עזרא) או ירמיהו; העצמות המתכסות בשר לנגד עיניו (2:259) הן מיחזקאל לז, והשינה הארוכה – מחוני המעגל ומאבימלך.

حزقيال 37: 1–14 יחזקאל לז, א–יד

הָיְתָה עָלַי יַד־יְהֹוָה וַיּוֹצִאֵנִי בְרוּחַ יְהֹוָה וַיְנִיחֵנִי בְּתוֹךְ הַבִּקְעָה וְהִיא מְלֵאָה עֲצָמוֹת׃ (א) וְהֶעֱבִירַנִי עֲלֵיהֶם סָבִיב סָבִיב וְהִנֵּה רַבּוֹת מְאֹד עַל־פְּנֵי הַבִּקְעָה וְהִנֵּה יְבֵשׁוֹת מְאֹד׃ (ב) וַיֹּאמֶר אֵלַי בֶּן־אָדָם הֲתִחְיֶינָה הָעֲצָמוֹת הָאֵלֶּה וָאֹמַר אֲדֹנָי יֱהֹוִה אַתָּה יָדָעְתָּ׃ (ג) וַיֹּאמֶר אֵלַי הִנָּבֵא עַל־הָעֲצָמוֹת הָאֵלֶּה וְאָמַרְתָּ אֲלֵיהֶם הָעֲצָמוֹת הַיְבֵשׁוֹת שִׁמְעוּ דְּבַר־יְהֹוָה׃ (ד) כֹּה אָמַר אֲדֹנָי יֱהֹוִה לָעֲצָמוֹת הָאֵלֶּה הִנֵּה אֲנִי מֵבִיא בָכֶם רוּחַ וִחְיִיתֶם׃ (ה) וְנָתַתִּי עֲלֵיכֶם גִּידִים וְהַעֲלֵתִי עֲלֵיכֶם בָּשָׂר וְקָרַמְתִּי עֲלֵיכֶם עוֹר וְנָתַתִּי בָכֶם רוּחַ וִחְיִיתֶם וִידַעְתֶּם כִּי־אֲנִי יְהֹוָה׃ (ו) וְנִבֵּאתִי כַּאֲשֶׁר צֻוֵּיתִי וַיְהִי־קוֹל כְּהִנָּבְאִי וְהִנֵּה־רַעַשׁ וַתִּקְרְבוּ עֲצָמוֹת עֶצֶם אֶל־עַצְמוֹ׃ (ז) וְרָאִיתִי וְהִנֵּה־עֲלֵיהֶם גִּדִים וּבָשָׂר עָלָה וַיִּקְרַם עֲלֵיהֶם עוֹר מִלְמָעְלָה וְרוּחַ אֵין בָּהֶם׃ (ח) וַיֹּאמֶר אֵלַי הִנָּבֵא אֶל־הָרוּחַ הִנָּבֵא בֶן־אָדָם וְאָמַרְתָּ אֶל־הָרוּחַ כֹּה־אָמַר אֲדֹנָי יֱהֹוִה מֵאַרְבַּע רוּחוֹת בֹּאִי הָרוּחַ וּפְחִי בַּהֲרוּגִים הָאֵלֶּה וְיִחְיוּ׃ (ט) וְהִנַּבֵּאתִי כַּאֲשֶׁר צִוָּנִי וַתָּבוֹא בָהֶם הָרוּחַ וַיִּחְיוּ וַיַּעַמְדוּ עַל־רַגְלֵיהֶם חַיִל גָּדוֹל מְאֹד־מְאֹד׃ (י) וַיֹּאמֶר אֵלַי בֶּן־אָדָם הָעֲצָמוֹת הָאֵלֶּה כׇּל־בֵּית יִשְׂרָאֵל הֵמָּה הִנֵּה אֹמְרִים יָבְשׁוּ עַצְמוֹתֵינוּ וְאָבְדָה תִקְוָתֵנוּ נִגְזַרְנוּ לָנוּ׃ (יא) לָכֵן הִנָּבֵא וְאָמַרְתָּ אֲלֵיהֶם כֹּה־אָמַר אֲדֹנָי יֱהֹוִה הִנֵּה אֲנִי פֹתֵחַ אֶת־קִבְרוֹתֵיכֶם וְהַעֲלֵיתִי אֶתְכֶם מִקִּבְרוֹתֵיכֶם עַמִּי וְהֵבֵאתִי אֶתְכֶם אֶל־אַדְמַת יִשְׂרָאֵל׃ (יב) וִידַעְתֶּם כִּי־אֲנִי יְהֹוָה בְּפִתְחִי אֶת־קִבְרוֹתֵיכֶם וּבְהַעֲלוֹתִי אֶתְכֶם מִקִּבְרוֹתֵיכֶם עַמִּי׃ (יג) וְנָתַתִּי רוּחִי בָכֶם וִחְיִיתֶם וְהִנַּחְתִּי אֶתְכֶם עַל־אַדְמַתְכֶם וִידַעְתֶּם כִּי־אֲנִי יְהֹוָה דִּבַּרְתִּי וְעָשִׂיתִי נְאֻם־יְהֹוָה׃ (יד)

كَانَتْ عَلَيَّ يَدُ ٱلرَّبِّ، فَأَخْرَجَني بِرُوحِ ٱلرَّبِّ وَأَنْزَلَنِي فِي وَسْطِ ٱلْبُقْعَةِ وَهِيَ مَلآنَةٌ عِظَامًا، (1) وَأَمَرَّنِي عَلَيْهَا مِنْ حَوْلِهَا وَإِذَا هِيَ كَثِيرَةٌ جِدًّا عَلَى وَجْهِ ٱلْبُقْعَةِ، وَإِذَا هِيَ يَابِسَةٌ جِدًّا. (2) فَقَالَ لِي: «يَا ٱبْنَ آدَمَ، أَتَحْيَا هَذِهِ ٱلْعِظَامُ؟» فَقُلْتُ: «يَا سَيِّدُ ٱلرَّبُّ أَنْتَ تَعْلَمُ». (3) فَقَالَ لِي: «تَنَبَّأْ عَلَى هَذِهِ ٱلْعِظَامِ وَقُلْ لَهَا: أَيَّتُهَا ٱلْعِظَامُ ٱلْيَابِسَةُ، ٱسْمَعِي كَلِمَةَ ٱلرَّبِّ: (4) هَكذَا قَالَ ٱلسَّيِّدُ ٱلرَّبُّ لِهَذِهِ ٱلْعِظَامِ: هَأَنَذَا أُدْخِلُ فِيكُمْ رُوحًا فَتَحْيَوْنَ. (5) وَأَضَعُ عَلَيْكُمْ عَصَبًا وأَكْسِيكُمْ لَحْمًا وَأَبْسُطُ عَلَيْكُمْ جِلْدًا وَأَجْعَلُ فِيكُمْ رُوحًا، فَتَحْيَوْنَ وَتَعْلَمُونَ أَنِّي أَنَا ٱلرَّبُّ». (6) فَتَنَبَّأْتُ كمَا أُمِرتُ. وَبَيْنَمَا أَنَا أَتنَبَّأُ كَانَ صَوْتٌ، وَإِذَا رَعْشٌ، فَتَقَارَبَتِ ٱلْعِظَامُ كُلُّ عَظْمٍ إِلَى عَظْمِهِ. (7) ونَظَرْتُ وَإِذَا بِٱلْعَصَبِ وَٱللَّحْمِ كَسَاهَا، وبُسِطَ ٱلْجِلْدُ علَيْهَا مِنْ فَوْقُ، وَلَيْسَ فِيهَا رُوحٌ. (8) فَقَالَ لِي: «تَنَبَّأْ لِلرُّوحِ، تَنَبَّأْ يَاٱبْنَ آدَمَ، وَقُلْ لِلرُّوحِ: هَكذَا قَالَ ٱلسَّيِّدُ ٱلرَّبُّ: هَلُمَّ يَارُوحُ مِنَ ٱلرِّيَاحِ ٱلْأَرْبَعِ وَهُبَّ عَلَى هَؤُلَاءِ ٱلْقَتْلَى لِيَحْيَوْا». (9) فَتَنَبَّأْتُ كَمَا أَمَرَني، فَدَخَلَ فِيهِمِ ٱلرُّوحُ، فَحَيُوا وَقَامُوا عَلَى أَقدَامِهِمْ جَيْشٌ عَظيمٌ جِدًّا جِدًّا. (10) ثُمَّ قَالَ لِي: «يَا ٱبْنَ آدَمَ، هَذِهِ ٱلعِظَامُ هِيَ كُلُّ بَيتِ إِسْرَائِيلَ. هَا هُمْ يَقُولُونَ: يَبِسَتْ عِظَامُنَا وَهَلَكَ رَجَاؤُنَا. قَدِ ٱنْقَطَعْنَا. (11) لِذَلِكَ تَنَبَّأْ وَقُلْ لَهُمْ: هَكذَا قَالَ ٱلسَّيِّدُ ٱلرَّبُّ: هَأَنَذَا أَفتَحُ قُبُورَكُمْ وأُصْعِدُكُمْ مِنْ قُبُورِكُمْ يَا شَعْبِي، وَآتِي بِكُمْ إِلَى أَرْضِ إِسْرَائِيلَ. (12) فَتَعْلَمُونَ أَنِّي أَنَا ٱلرَّبُّ عِنْدَ فَتْحِي قُبُورَكُمْ وَإِصْعَادِي إِيَّاكُمْ مِنْ قُبُورِكُمْ يَاشَعْبِي. (13) وأَجْعَلُ رُوحِي فِيكُمْ فتَحْيَوْنَ، وَأَجْعَلُكُمْ فِي أَرْضِكُمْ، فَتَعْلَمُونَ أَنِّي أنَا ٱلرَّبُّ تَكَلَّمْتُ وَأَفْعَلُ، يَقُولُ ٱلرَّبُّ». (14)

וָאָבוֹא אֶל־יְרוּשָׁלִָם וָאֱהִי־שָׁם יָמִים שְׁלֹשָׁה׃ (יא) וָאָקוּם לַיְלָה אֲנִי וַאֲנָשִׁים מְעַט עִמִּי וְלֹא־הִגַּדְתִּי לְאָדָם מָה אֱלֹהַי נֹתֵן אֶל־לִבִּי לַעֲשׂוֹת לִירוּשָׁלִָם וּבְהֵמָה אֵין עִמִּי כִּי אִם־הַבְּהֵמָה אֲשֶׁר אֲנִי רֹכֵב בָּהּ׃ (יב) וָאֵצְאָה בְשַׁעַר־הַגַּיְא לַיְלָה וְאֶל־פְּנֵי עֵין הַתַּנִּין וְאֶל־שַׁעַר הָאַשְׁפֹּת וָאֱהִי שֹׂבֵר בְּחוֹמֹת יְרוּשָׁלִַם אֲשֶׁר־[הֵם פְּרוּצִים] (המפרוצים) וּשְׁעָרֶיהָ אֻכְּלוּ בָאֵשׁ׃ (יג) וָאֶעֱבֹר אֶל־שַׁעַר הָעַיִן וְאֶל־בְּרֵכַת הַמֶּלֶךְ וְאֵין־מָקוֹם לַבְּהֵמָה לַעֲבֹר תַּחְתָּי׃ (יד) וָאֱהִי עֹלֶה בַנַּחַל לַיְלָה וָאֱהִי שֹׂבֵר בַּחוֹמָה וָאָשׁוּב וָאָבוֹא בְּשַׁעַר הַגַּיְא וָאָשׁוּב׃ (טו) וְהַסְּגָנִים לֹא יָדְעוּ אָנָה הָלַכְתִּי וּמָה אֲנִי עֹשֶׂה וְלַיְּהוּדִים וְלַכֹּהֲנִים וְלַחֹרִים וְלַסְּגָנִים וּלְיֶתֶר עֹשֵׂה הַמְּלָאכָה עַד־כֵּן לֹא הִגַּדְתִּי׃ (טז) וָאוֹמַר אֲלֵהֶם אַתֶּם רֹאִים הָרָעָה אֲשֶׁר אֲנַחְנוּ בָהּ אֲשֶׁר יְרוּשָׁלִַם חֲרֵבָה וּשְׁעָרֶיהָ נִצְּתוּ בָאֵשׁ לְכוּ וְנִבְנֶה אֶת־חוֹמַת יְרוּשָׁלִַם וְלֹא־נִהְיֶה עוֹד חֶרְפָּה׃ (יז)

فَجِئْتُ إِلَى أُورُشَلِيمَ وَكُنْتُ هُنَاكَ ثَلَاثَةَ أَيَّامٍ. (11) ثُمَّ قُمْتُ لَيْلًا أَنَا وَرِجَالٌ قَلِيلُونَ مَعِي، وَلَمْ أُخْبِرْ أَحَدًا بِمَا جَعَلَهُ إِلَهِي فِي قَلْبِي لِأَعْمَلَهُ فِي أُورُشَلِيمَ. وَلَمْ يَكُنْ مَعِي بَهِيمَةٌ إِلَّا ٱلْبَهِيمَةُ ٱلَّتِي كُنْتُ رَاكِبَهَا. (12) وَخَرَجْتُ مِنْ بَابِ ٱلْوَادِي لَيْلًا أَمَامَ عَيْنِ ٱلتِّنِّينِ إِلَى بَابِ ٱلدِّمْنِ، وَصِرْتُ أَتَفَرَّسُ فِي أَسْوَارِ أُورُشَلِيمَ ٱلْمُنْهَدِمَةِ وَأَبْوَابِهَا ٱلَّتِي أَكَلَتْهَا ٱلنَّارُ. (13) وَعَبَرْتُ إِلَى بَابِ ٱلْعَيْنِ وَإِلَى بِرْكَةِ ٱلْمَلِكِ، وَلَمْ يَكُنْ مَكَانٌ لِعُبُورِ ٱلْبَهِيمَةِ ٱلَّتِي تَحْتِي. (14) فَصَعِدْتُ فِي ٱلْوَادِي لَيْلًا وَكُنْتُ أَتَفَرَّسُ فِي ٱلسُّورِ، ثُمَّ عُدْتُ فَدَخَلْتُ مِنْ بَابِ ٱلْوَادِي رَاجِعًا. (15) وَلَمْ يَعْرِفِ ٱلْوُلَاةُ إِلَى أَيْنَ ذَهَبْتُ، وَلَا مَا أَنَا عَامِلٌ، وَلَمْ أُخْبِرْ إِلَى ذَلِكَ ٱلْوَقْتِ ٱلْيَهُودَ وَٱلْكَهَنَةَ وَٱلْأَشْرَافَ وَٱلْوُلَاةَ وَبَاقِي عَامِلِي ٱلْعَمَلِ. (16) ثُمَّ قُلْتُ لَهُمْ: «أَنْتُمْ تَرَوْنَ ٱلشَّرَّ ٱلَّذِي نَحْنُ فِيهِ، كَيْفَ أَنَّ أُورُشَلِيمَ خَرِبَةٌ، وَأَبْوَابَهَا قَدْ أُحْرِقَتْ بِٱلنَّارِ. هَلُمَّ فَنَبْنِيَ سُورَ أُورُشَلِيمَ وَلَا نَكُونُ بَعْدُ عَارًا». (17)
