סיפור שמי משותף של היבריס טכנולוגי־אדריכלי: ניסיון לבנות מגדל שיגיע אל השמיים כדי לקרוא תיגר על האל, המביא לקריסת הבניין ולפיזור האנושות לשפות ולעמים שונים (בראשית יא מול קוראן 16:26 ומסורת התפסיר על בבל).

כבר זממו אלה שלפניהם, ובא אללה אל בניינם מן היסודות, ונפלה עליהם התקרה מעליהם, ובא אליהם העונש ממקום שלא הרגישו. (26)

قَدۡ مَكَرَ ٱلَّذِينَ مِن قَبۡلِهِمۡ فَأَتَى ٱللَّهُ بُنۡيَٰنَهُم مِّنَ ٱلۡقَوَاعِدِ فَخَرَّ عَلَيۡهِمُ ٱلسَّقۡفُ مِن فَوۡقِهِمۡ وَأَتَىٰهُمُ ٱلۡعَذَابُ مِنۡ حَيۡثُ لَا يَشۡعُرُونَ (26)

البقرة 2: 258 קוראן 2:258

הלא ראית את זה אשר התווכח עם אברהם על ריבונו, משום שנתן לו אללה מלוכה? כאשר אמר אברהם: ״ריבוני הוא המחיה והממית״, אמר: ״אני מחיה וממית״. אמר אברהם: ״אללה מביא את השמש מן המזרח, הבא אתה אותה מן המערב!״ ונדהם הכופר. ואללה אינו מדריך את העם עושה העוול. (258)

أَلَمۡ تَرَ إِلَى ٱلَّذِي حَآجَّ إِبۡرَٰهِـۧمَ فِي رَبِّهِۦٓ أَنۡ ءَاتَىٰهُ ٱللَّهُ ٱلۡمُلۡكَ إِذۡ قَالَ إِبۡرَٰهِـۧمُ رَبِّيَ ٱلَّذِي يُحۡيِۦ وَيُمِيتُ قَالَ أَنَا۠ أُحۡيِۦ وَأُمِيتُۖ قَالَ إِبۡرَٰهِـۧمُ فَإِنَّ ٱللَّهَ يَأۡتِي بِٱلشَّمۡسِ مِنَ ٱلۡمَشۡرِقِ فَأۡتِ بِهَا مِنَ ٱلۡمَغۡرِبِ فَبُهِتَ ٱلَّذِي كَفَرَۗ وَٱللَّهُ لَا يَهۡدِي ٱلۡقَوۡمَ ٱلظَّـٰلِمِينَ (258)

التكوين 10: 8–10 בראשית י, ח–י

וְכוּשׁ יָלַד אֶת־נִמְרֹד הוּא הֵחֵל לִהְיוֹת גִּבֹּר בָּאָרֶץ׃ (ח) הוּא־הָיָה גִבֹּר־צַיִד לִפְנֵי יְהֹוָה עַל־כֵּן יֵאָמַר כְּנִמְרֹד גִּבּוֹר צַיִד לִפְנֵי יְהֹוָה׃ (ט) וַתְּהִי רֵאשִׁית מַמְלַכְתּוֹ בָּבֶל וְאֶרֶךְ וְאַכַּד וְכַלְנֵה בְּאֶרֶץ שִׁנְעָר׃ (י)

وَكُوشُ وَلَدَ نِمْرُودَ ٱلَّذِي ٱبْتَدَأَ يَكُونُ جَبَّارًا فِي ٱلْأَرْضِ، (8) ٱلَّذِي كَانَ جَبَّارَ صَيْدٍ أَمَامَ ٱلرَّبِّ. لِذَلِكَ يُقَالُ: «كَنِمْرُودَ جَبَّارُ صَيْدٍ أَمَامَ ٱلرَّبِّ». (9) وَكَانَ ٱبْتِدَاءُ مَمْلَكَتِهِ بَابِلَ وَأَرَكَ وَأَكَّدَ وَكَلْنَةَ، فِي أَرْضِ شِنْعَارَ. (10)

التكوين 11: 1–9 בראשית יא, א–ט

וַיְהִי כׇל־הָאָרֶץ שָׂפָה אֶחָת וּדְבָרִים אֲחָדִים׃ (א) וַיְהִי בְּנׇסְעָם מִקֶּדֶם וַיִּמְצְאוּ בִקְעָה בְּאֶרֶץ שִׁנְעָר וַיֵּשְׁבוּ שָׁם׃ (ב) וַיֹּאמְרוּ אִישׁ אֶל־רֵעֵהוּ הָבָה נִלְבְּנָה לְבֵנִים וְנִשְׂרְפָה לִשְׂרֵפָה וַתְּהִי לָהֶם הַלְּבֵנָה לְאָבֶן וְהַחֵמָר הָיָה לָהֶם לַחֹמֶר׃ (ג) וַיֹּאמְרוּ הָבָה נִבְנֶה־לָּנוּ עִיר וּמִגְדָּל וְרֹאשׁוֹ בַשָּׁמַיִם וְנַעֲשֶׂה־לָּנוּ שֵׁם פֶּן־נָפוּץ עַל־פְּנֵי כׇל־הָאָרֶץ׃ (ד) וַיֵּרֶד יְהֹוָה לִרְאֹת אֶת־הָעִיר וְאֶת־הַמִּגְדָּל אֲשֶׁר בָּנוּ בְּנֵי הָאָדָם׃ (ה) וַיֹּאמֶר יְהֹוָה הֵן עַם אֶחָד וְשָׂפָה אַחַת לְכֻלָּם וְזֶה הַחִלָּם לַעֲשׂוֹת וְעַתָּה לֹא־יִבָּצֵר מֵהֶם כֹּל אֲשֶׁר יָזְמוּ לַעֲשׂוֹת׃ (ו) הָבָה נֵרְדָה וְנָבְלָה שָׁם שְׂפָתָם אֲשֶׁר לֹא יִשְׁמְעוּ אִישׁ שְׂפַת רֵעֵהוּ׃ (ז) וַיָּפֶץ יְהֹוָה אֹתָם מִשָּׁם עַל־פְּנֵי כׇל־הָאָרֶץ וַיַּחְדְּלוּ לִבְנֹת הָעִיר׃ (ח) עַל־כֵּן קָרָא שְׁמָהּ בָּבֶל כִּי־שָׁם בָּלַל יְהֹוָה שְׂפַת כׇּל־הָאָרֶץ וּמִשָּׁם הֱפִיצָם יְהֹוָה עַל־פְּנֵי כׇּל־הָאָרֶץ׃ (ט)

وَكَانَتِ ٱلْأَرْضُ كُلُّهَا لِسَانًا وَاحِدًا وَلُغَةً وَاحِدَةً. (1) وَحَدَثَ فِي ٱرْتِحَالِهِمْ شَرْقًا أَنَّهُمْ وَجَدُوا بُقْعَةً فِي أَرْضِ شِنْعَارَ وَسَكَنُوا هُنَاكَ. (2) وَقَالَ بَعْضُهُمْ لِبَعْضٍ: «هَلُمَّ نَصْنَعُ لِبْنًا وَنَشْوِيهِ شَيًّا». فَكَانَ لَهُمُ ٱللِّبْنُ مَكَانَ ٱلْحَجَرِ، وَكَانَ لَهُمُ ٱلْحُمَرُ مَكَانَ ٱلطِّينِ. (3) وَقَالُوا: «هَلُمَّ نَبْنِ لِأَنْفُسِنَا مَدِينَةً وَبُرْجًا رَأْسُهُ بِٱلسَّمَاءِ. وَنَصْنَعُ لِأَنْفُسِنَا ٱسْمًا لِئَلَّا نَتَبَدَّدَ عَلَى وَجْهِ كُلِّ ٱلْأَرْضِ». (4) فَنَزَلَ ٱلرَّبُّ لِيَنْظُرَ ٱلْمَدِينَةَ وَٱلْبُرْجَ ٱللَّذَيْنِ كَانَ بَنُو آدَمَ يَبْنُونَهُمَا. (5) وَقَالَ ٱلرَّبُّ: «هُوَذَا شَعْبٌ وَاحِدٌ وَلِسَانٌ وَاحِدٌ لِجَمِيعِهِمْ، وَهَذَا ٱبْتِدَاؤُهُمْ بِٱلْعَمَلِ. وَٱلْآنَ لَا يَمْتَنِعُ عَلَيْهِمْ كُلُّ مَا يَنْوُونَ أَنْ يَعْمَلُوهُ. (6) هَلُمَّ نَنْزِلْ وَنُبَلْبِلْ هُنَاكَ لِسَانَهُمْ حَتَّى لَا يَسْمَعَ بَعْضُهُمْ لِسَانَ بَعْضٍ». (7) فَبَدَّدَهُمُ ٱلرَّبُّ مِنْ هُنَاكَ عَلَى وَجْهِ كُلِّ ٱلْأَرْضِ، فَكَفُّوا عَنْ بُنْيَانِ ٱلْمَدِينَةِ، (8) لِذَلِكَ دُعِيَ ٱسْمُهَا «بَابِلَ» لِأَنَّ ٱلرَّبَّ هُنَاكَ بَلْبَلَ لِسَانَ كُلِّ ٱلْأَرْضِ. وَمِنْ هُنَاكَ بَدَّدَهُمُ ٱلرَّبُّ عَلَى وَجْهِ كُلِّ ٱلْأَرْضِ. (9)

إشعياء 14: 13–15 ישעיהו יד, יג–טו

וְאַתָּה אָמַרְתָּ בִלְבָבְךָ הַשָּׁמַיִם אֶעֱלֶה מִמַּעַל לְכוֹכְבֵי־אֵל אָרִים כִּסְאִי וְאֵשֵׁב בְּהַר־מוֹעֵד בְּיַרְכְּתֵי צָפוֹן׃ (יג) אֶעֱלֶה עַל־בָּמֳתֵי עָב אֶדַּמֶּה לְעֶלְיוֹן׃ (יד) אַךְ אֶל־שְׁאוֹל תּוּרָד אֶל־יַרְכְּתֵי־בוֹר׃ (טו)

وَأَنْتَ قُلْتَ فِي قَلْبِكَ: أَصْعَدُ إِلَى ٱلسَّمَاوَاتِ. أَرْفَعُ كُرْسِيِّي فَوْقَ كَوَاكِبِ ٱللهِ، وَأَجْلِسُ عَلَى جَبَلِ ٱلِٱجْتِمَاعِ فِي أَقَاصِي ٱلشَّمَالِ. (13) أَصْعَدُ فَوْقَ مُرْتَفَعَاتِ ٱلسَّحَابِ. أَصِيرُ مِثْلَ ٱلْعَلِيِّ. (14) لَكِنَّكَ ٱنْحَدَرْتَ إِلَى ٱلْهَاوِيَةِ، إِلَى أَسَافِلِ ٱلْجُبِّ. (15)
